balkfm_bosnia.005 = { # Disruptive Influence
	type = country_event
	placement = root

	title = balkfm_bosnia.005.t
	desc = balkfm_bosnia.005.d
	flavor = balkfm_bosnia.005.f

	event_image = {
		video = "southamerica_aristocrats"
	}

	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		has_global_variable = balkfm_humac_happened
	}

	immediate = {
		remove_global_variable = balkfm_humac_happened
		balkfm_hersek_scopes_effect = yes
		c:MON = {
			save_scope_as = montenegro_scope
			ruler = {
				save_scope_as = montenegro_ruler_scope
			}
		}
		s:STATE_WESTERN_SERBIA = {
			random_scope_state = {
				save_scope_as = northern_serbia_state_scope
			}
		}
	}

	option = { # stir the pot
		name = balkfm_bosnia.005.a
		default_option = yes
		change_relations = {
			country = c:MON
			value = 20
		}
		set_global_variable = balkfm_stabby_stabby # triggers balkfm_montenegro.032 (Knives in the Dark)
		ai_chance = {
			base = 10
			modifier = {
				trigger = {
					c:MON ?= {
						relations:root < relations_threshold:neutral
					}
				}
				add = -10
			}
		}
	}

	option = { # do nothing
		name = balkfm_bosnia.005.b
		custom_tooltip = gbbf_nothing_happens
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					c:MON ?= {
						relations:root < relations_threshold:neutral
					}
				}
				add = 10
			}
		}
	}

}
