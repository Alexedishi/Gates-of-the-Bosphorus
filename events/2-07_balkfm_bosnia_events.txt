balkfm_bosnia.004 = {
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	duration = 3
	dlc = dlc_grefm
	
	title = balkfm_bosnia.004.t
	desc = balkfm_bosnia.004.d
	flavor = balkfm_bosnia.004.f
	
	gui_window = event_window_com_base
	
	event_image = {
		video = "votp_hostile_court"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		# triggered by eqfm_karadag.105
	}
	
	immediate = {
		eqfm_save_ali_pasha_scopes_effect = yes
		c:TUR = {
			random_scope_character = {
				limit = {
					has_modifier = modifier_turkfm_arrested_traitor
				}
				save_scope_as = balkfm_omer_scope
			}
		}
		c:BOS = {
			if = {
				limit = {
					ruler = {
						OR = {
							has_template = balkfm_rizvanbegovic_ali
							has_template = balkfm_rizvanbegovic_mehmed
						}
					}
				}
				ruler = {
					set_variable = balkfm_ali_pasha_var
					place_character_in_void = 12
				}
				scope:balkfm_omer_scope = {
					transfer_character = c:BOS
					set_character_as_ruler = yes
					free_character_from_void = yes
				}
				heir ?= {
					disinherit_character = yes
				}
			}
		}
		random_character_in_void = {
			limit = {
				has_variable = balkfm_ali_pasha_var
			}
			save_scope_as = balkfm_ali_pasha_scope
		}
	}
	
	option = {
		name = balkfm_bosnia.004.a
		default_option = yes
		trigger = {
			c:TUR ?= this
		}
		show_as_tooltip = {
			scope:balkfm_ali_pasha_scope = {
				kill_character = yes
			}
		}
		remove_modifier = modifier_turkfm_imperial_challenger
		show_as_tooltip = {
			scope:balkfm_omer_scope = {
				transfer_character = c:BOS
				set_character_as_ruler = yes
			}
			c:BOS = {
				add_modifier = {
					name = modifier_eqfm_military_governor
					months = very_long_modifier_time
					is_decaying = yes
				}
			}
		}
	}
	
	option = {
		name = balkfm_bosnia.004.b
		trigger = {
			c:BOS ?= this
		}
		show_as_tooltip = {
			scope:balkfm_ali_pasha_scope = {
				kill_character = yes
			}
			scope:balkfm_omer_scope = {
				transfer_character = c:BOS
				set_character_as_ruler = yes
			}
		}
		add_modifier = {
			name = modifier_eqfm_military_governor
			months = very_long_modifier_time
			is_decaying = yes
		}
	}
	
	option = {
		name = balkfm_bosnia.004.c
		trigger = {
			c:MON ?= this
		}
		show_as_tooltip = {
			scope:balkfm_ali_pasha_scope = {
				kill_character = yes
			}
			scope:balkfm_omer_scope = {
				transfer_character = c:BOS
				set_character_as_ruler = yes
			}
			add_modifier = {
				name = modifier_eqfm_military_governor
				months = very_long_modifier_time
				is_decaying = yes
			}
		}
	}
}
