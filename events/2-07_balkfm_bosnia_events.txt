balkfm_bosnia.003 = { # The Reordering Begins
	type = country_event
	placement = root

	title = balkfm_bosnia.003.t
	desc = balkfm_bosnia.003.d
	flavor = balkfm_bosnia.003.f

	event_image = {
		video = "unspecific_ruler_speaking_to_people"
	}

	icon = "gfx/interface/icons/event_icons/event_fez.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		NOT = {
			has_variable = balkfm_tanizmat_considered
		}
		has_global_variable = turkfm_tanzimat_started
	}

	immediate = {
		set_variable = balkfm_tanizmat_considered
		c:TUR = {
			save_scope_as = ottoman_scope
		}
	}

	option = {
		name = balkfm_bosnia.003.a
		default_option = yes
		add_journal_entry = {
			type = je_balkfm_bos_tanzimat
		}
		change_relations = {
			country = c:TUR
			value = 25
		}
		create_diplomatic_catalyst = {
			type = catalyst_relations_level_increased
			target = c:TUR
		}
		c:TUR = {
			set_secret_goal = {
				country = root
				secret_goal = protect
			}
		}
	}

	option = {
		name = balkfm_bosnia.003.b
		change_relations = {
			country = c:TUR
			value = -25
		}
		create_diplomatic_catalyst = {
			type = catalyst_diplomatic_incident
			target = c:TUR
		}
		create_diplomatic_catalyst = {
			type = catalyst_relations_level_decreased
			target = c:TUR
		}
	}

}
