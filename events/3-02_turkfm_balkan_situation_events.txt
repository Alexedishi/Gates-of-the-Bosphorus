turkfm_balkan_situation.005 = { # Riots Break Out in State
	type = country_event
	placement = scope:turkfm_riot_state_scope

	title = turkfm_balkan_situation.005.t
	desc = turkfm_balkan_situation.005.d
	flavor = turkfm_balkan_situation.005.f

	event_image = {
		video = "votp_agitator_speaking"
	}

	icon = "gfx/interface/icons/event_icons/bicorne_hat.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		turkfm_hostility_events_trigger = yes
		var:turkfm_riot_current_tally_var < turkfm_riot_potential_value
		NOT = {
			has_variable = turkfm_riot_cooldown_var
		}
		any_scope_state = {
			region = sr:region_balkans
			NOT = {
				has_modifier = modifier_turkfm_race_riots
			}
		}
	}

	immediate = {
		set_variable = {
			name = turkfm_riot_cooldown_var
			years = 1
		}
		turkfm_tension_cooldown_effect = yes
		random_scope_state = {
			limit = {
				region = sr:region_balkans
				NOT = {
					has_modifier = modifier_turkfm_race_riots
				}
			}
			save_scope_as = turkfm_riot_state_scope
		}
		scope:turkfm_riot_state_scope = {
			add_modifier = modifier_turkfm_race_riots
		}
		change_global_variable = {
			name = turkfm_riot_counter_var
			add = 1
		}
		change_variable = {
			name = turkfm_riot_current_tally_var
			add = 1
		}
		if = {
			limit = {
				NOT = {
					has_journal_entry = je_turkfm_balkan_situation_riots
				}
			}
			add_journal_entry = {
				type = je_turkfm_balkan_situation_riots
			}
			every_country = {
				limit = {
					has_journal_entry = je_turkfm_eastern_question
				}
				post_notification = notification_turkfm_race_riots
			}
		}
		else = {
			every_country = {
				limit = {
					has_journal_entry = je_turkfm_eastern_question
				}
				post_notification = notification_turkfm_race_riots_worsen
			}
		}
		turkfm_update_riot_variables_effect = yes
	}

	option = {
		name = turkfm_balkan_situation.005.a
		default_option = yes
		custom_tooltip = turkfm_balkan_situation.005.tt
		show_as_tooltip = {
			scope:turkfm_riot_state_scope = {
				add_modifier = modifier_turkfm_race_riots
			}
		}
	}

}

turkfm_balkan_situation.006 = { # Balkan Country Rebellion
	type = country_event
	placement = scope:turkfm_rebellion_scope

	title = turkfm_balkan_situation.006.t
	desc = turkfm_balkan_situation.006.d
	flavor = turkfm_balkan_situation.006.f

	event_image = {
		video = "votp_agitator_speaking"
	}

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 1

	dlc = dlc_grefm

	trigger = {
		# triggered by turkfm_secession_start on_action
	}

	immediate = {
		set_variable = {
			name = turkfm_recent_revolt_var
			months = 3
		}
		random_country = {
			limit = {
				has_variable = turkfm_rebellion_tag_tracker
			}
			save_scope_as = turkfm_rebellion_scope
			remove_variable = turkfm_rebellion_tag_tracker
		}
		every_country = {
			limit = {
				has_diplomatic_relevance = root
				NOT = {
					this = root
				}
			}
			create_bidirectional_truce = {
				country = scope:turkfm_rebellion_scope
				months = 36
			}
		}
		set_relations = {
			country = scope:turkfm_rebellion_scope
			value = -50
		}
		random_diplomatic_play = {
			limit = {
				initiator = root
				target = scope:turkfm_rebellion_scope
			}
			save_scope_as = turkfm_rebellion_play_scope
		}
	}

	option = { # Assert Our Dominion
		name = turkfm_balkan_situation.006.a
		default_option = yes
		custom_tooltip = turkfm_play_continues
		every_country = {
			limit = {
				has_journal_entry = je_turkfm_eastern_question
			}
			post_notification = notification_turkfm_revolt_seccession
		}
	}

	option = { # Let Them Go
		name = turkfm_balkan_situation.006.b
		scope:turkfm_rebellion_play_scope = {
			end_play = yes
		}
		create_diplomatic_pact = {
			country = scope:turkfm_rebellion_scope
			type = protectorate
		}
		create_bidirectional_truce = {
			country = scope:turkfm_rebellion_scope
			months = 60
		}
		add_radicals = {
			culture = cu:turkish
			value = medium_radicals
		}
		add_modifier = {
			name = modifier_turkfm_caved_to_separatists
			days = normal_modifier_time
		}
		hidden_effect = {
			every_country = {
				limit = {
					has_journal_entry = je_turkfm_eastern_question
				}
				post_notification = notification_turkfm_revolt_protectorate
			}
		}
	}

}
