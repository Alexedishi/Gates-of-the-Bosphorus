mgrbfm_egypt.003 = {
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	duration = 3
	dlc = dlc_grefm

	title = mgrbfm_egypt.003.t
	desc = mgrbfm_egypt.003.d
	flavor = {
		triggered_desc = { 
			desc = mgrbfm_egypt.003.f1
			trigger = { 
				heir = {
					has_template = template_mgrbfm_alawi_abbas
				}
			}
		}
		triggered_desc = {
			desc = mgrbfm_egypt.003.f2
			trigger = {
				heir = {
					has_template = template_mgrbfm_alawi_ibrahim
				}
			}
		}
	}

	event_image = {
		video = "votp_monarch_holding_court"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		ruler = {
			has_template = template_mgrbfm_alawi_muhammad
			has_trait = psychological_affliction
			NOR = {
				has_variable = mgrbfm_old_spider_var
				has_trait = gbbf_trait_old_spider
			}
		}
	}

	cancellation_trigger = {
		ruler = {
			NOT = {
				has_template = template_mgrbfm_alawi_muhammad
			}
		}
	}

	immediate = {
		ruler = {
			save_scope_as = mgrbfm_ruler_scope
		}
		if = {
			limit = {
				NOT = {
					exists = heir
				}
			}
			if = {
				limit = {
					ruler = {
						has_template = template_mgrbfm_alawi_muhammad
					}
				}
				create_character = {
					template = template_mgrbfm_alawi_abbas
				}
			}
		}
		create_character = {
			template = template_mgrbfm_alawi_abbas
			on_created = {
				place_character_in_void = 12
				save_scope_as = mgrbfm_abbas_scope
			}
		}
		create_character = {
			template = template_mgrbfm_alawi_said
			on_created = {
				place_character_in_void = 12
				save_scope_as = mgrbfm_said_scope
			}
		}
		heir = {
			save_scope_as = mgrbfm_heir_scope
		}
	}

	option = {
		name = mgrbfm_egypt.003.a
		default_option = yes
		ruler = {
			kill_character = yes
		}
		heir ?= {
			set_character_as_ruler = yes
		}
		if = {
			limit = {
				scope:mgrbfm_heir_scope = {
					has_template = template_mgrbfm_alawi_ibrahim
				}
			}
			set_heir = scope:mgrbfm_abbas_scope
		}
		else_if = {
			limit = {
				scope:mgrbfm_heir_scope = {
					has_template = template_mgrbfm_alawi_abbas
				}
			}
			set_heir = scope:mgrbfm_said_scope
		}
	}

	after = {
		scope:mgrbfm_abbas_scope = {
			gbbf_free_from_void = yes
		}
		scope:mgrbfm_said_scope = {
			gbbf_free_from_void = yes
		}
	}
}

mgrbfm_egypt.110 = {
	type = country_event
	placement = scope:mgrbfm_flood_scope
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 3
	dlc = dlc_grefm

	title = mgrbfm_egypt.110.t
	desc = mgrbfm_egypt.110.d
	flavor = mgrbfm_egypt.110.f

	event_image = {
		video = "unspecific_world_fair"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		# triggered by mgrbfm_nile_inundation_effect
	}

	immediate = {
		random_state_in_lower_nile = {
			limit = {
				state_region = {
					has_variable = mgrbfm_flood_state_var
				}
			}
			save_scope_as = mgrbfm_flood_scope
		}
	}

	option = {
		name = mgrbfm_egypt.110.a
		default_option = yes
		trigger = {
			NOT = {
				any_scope_state = {
					is_in_geographic_region = geographic_region_lower_nile
					has_building = gbbf_building_nile_barrage
				}
			}
		}
		random_scope_state = {
			limit = {
				scope:mgrbfm_flood_scope = this
			}
			add_modifier = {
				name = modifier_mgrbfm_nile_flood_epicenter
				months = short_modifier_time
				multiplier = global_var:mgrbfm_nile_flood_var
				is_decaying = yes
			}
		}
		every_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_lower_nile
				scope:mgrbfm_flood_scope != this
			}
			add_modifier = {
				name = modifier_mgrbfm_widespread_nile_floods
				months = short_modifier_time
				multiplier = global_var:mgrbfm_nile_flood_var
				is_decaying = yes
			}
		}
	}

	option = {
		name = mgrbfm_egypt.110.b
		trigger = {
			any_scope_state = {
				is_in_geographic_region = geographic_region_lower_nile
				has_building = gbbf_building_nile_barrage
				NOR = {
					is_production_method_active = {
						building_type = gbbf_building_nile_barrage
						production_method = pm_gbbf_adequate_reservior
					}
					is_production_method_active = {
						building_type = gbbf_building_nile_barrage
						production_method = pm_gbbf_massive_reservior
					}
				}
			}
		}
		random_scope_state = {
			limit = {
				scope:mgrbfm_flood_scope = this
			}
			add_modifier = {
				name = modifier_mgrbfm_nile_flood_epicenter
				months = short_modifier_time
				multiplier = {
					add = global_var:mgrbfm_nile_flood_var
					scope:mgrbfm_flood_scope = {
						if = {
							limit = {
								is_production_method_active = {
									building_type = gbbf_building_nile_barrage
									production_method = pm_gbbf_adequate_reservior
								}
							}
							divide = 2
						}
						else_if = {
							limit = {
								is_production_method_active = {
									building_type = gbbf_building_nile_barrage
									production_method = pm_gbbf_massive_reservior
								}
							}
							divide = 4
						}
					}
				}
				is_decaying = yes
			}
		}
		every_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_lower_nile
				scope:mgrbfm_flood_scope != this
				NOR = {
					is_production_method_active = {
						building_type = gbbf_building_nile_barrage
						production_method = pm_gbbf_adequate_reservior
					}
					is_production_method_active = {
						building_type = gbbf_building_nile_barrage
						production_method = pm_gbbf_massive_reservior
					}
				}
			}
			add_modifier = {
				name = modifier_mgrbfm_widespread_nile_floods
				months = short_modifier_time
				multiplier = global_var:mgrbfm_nile_flood_var
				is_decaying = yes
			}
		}
		every_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_lower_nile
				scope:mgrbfm_flood_scope != this
				is_production_method_active = {
					building_type = gbbf_building_nile_barrage
					production_method = pm_gbbf_adequate_reservior
				}
			}
			add_modifier = {
				name = modifier_mgrbfm_localized_nile_floods
				months = short_modifier_time
				multiplier = global_var:mgrbfm_nile_flood_var
				is_decaying = yes
			}
		}
		every_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_lower_nile
				scope:mgrbfm_flood_scope != this
				is_production_method_active = {
					building_type = gbbf_building_nile_barrage
					production_method = pm_gbbf_massive_reservior
				}
			}
			add_modifier = {
				name = modifier_mgrbfm_mitigated_nile_floods
				months = short_modifier_time
			}
		}
	}

	option = {
		name = mgrbfm_egypt.110.c
		trigger = {
			any_scope_state = {
				filter = {
					is_in_geographic_region = geographic_region_lower_nile
				}
				count = all
				OR = {
					is_production_method_active = {
						building_type = gbbf_building_nile_barrage
						production_method = pm_gbbf_adequate_reservior
					}
					is_production_method_active = {
						building_type = gbbf_building_nile_barrage
						production_method = pm_gbbf_massive_reservior
					}
				}
			}
		}
		every_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_lower_nile
			}
			add_modifier = {
				name = modifier_mgrbfm_mitigated_nile_floods
				months = short_modifier_time
			}
		}
	}
}

	icon = "gfx/interface/icons/event_icons/event_ruined_building.dds"
	duration = 3
	dlc = dlc_grefm

	title = mgrbfm_egypt.112.t
	desc = mgrbfm_egypt.112.d
	flavor = mgrbfm_egypt.112.f

	event_image = {
		video = "unspecific_devastation"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		is_involved_in_journal_entry = je_mgrbfm_egypt_nile_river
		any_scope_state = {
			is_in_geographic_region = geographic_region_lower_nile
			has_building = building_railway
			OR = {
				has_modifier = modifier_mgrbfm_nile_flood_epicenter
				has_modifier = modifier_mgrbfm_widespread_nile_floods
			}
		}
	}

	cancellation_trigger = {
		NOT = {
			is_involved_in_journal_entry = je_mgrbfm_egypt_nile_river
		}
	}

	cooldown = {
		years = 5
	}

	immediate = {
		random_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_lower_nile
				has_building = building_railway
			}
			save_scope_as = mgrbfm_nile_flood_scope
		}
	}

	option = {
		name = mgrbfm_egypt.112.a
		default_option = yes
		scope:mgrbfm_nile_flood_scope = {
			add_modifier = {
				name = modifier_mgrbfm_flooded_railways
				months = gbbf_short_modifier_time
				multiplier = global_var:mgrbfm_nile_flood_var
			}
		}
	}
}
