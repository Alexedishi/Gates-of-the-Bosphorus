mgrbfm_egypt.003 = {
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	duration = 3
	dlc = dlc_grefm

	title = mgrbfm_egypt.003.t
	desc = mgrbfm_egypt.003.d
	flavor = {
		triggered_desc = { 
			desc = mgrbfm_egypt.003.f1
			trigger = { 
				heir = {
					has_template = mgrbfm_alawi_abbas
				}
			}
		}
		triggered_desc = {
			desc = mgrbfm_egypt.003.f2
			trigger = {
				heir = {
					has_template = mgrbfm_alawi_ibrahim
				}
			}
		}
	}

	event_image = {
		video = "votp_monarch_holding_court"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		ruler = {
			has_template = mgrbfm_alawi_muhammad
			has_trait = psychological_affliction
			NOR = {
				has_variable = mgrbfm_old_spider_var
				has_trait = gbbf_trait_old_spider
			}
		}
	}

	cancellation_trigger = {
		ruler = {
			NOT = {
				has_template = mgrbfm_alawi_muhammad
			}
		}
	}

	immediate = {
		ruler = {
			save_scope_as = mgrbfm_ruler_scope
		}
		if = {
			limit = {
				NOT = {
					exists = heir
				}
			}
			if = {
				limit = {
					ruler = {
						has_template = mgrbfm_alawi_muhammad
					}
				}
				create_character = {
					template = mgrbfm_alawi_abbas
				}
			}
		}
		create_character = {
			template = mgrbfm_alawi_abbas
			on_created = {
				place_character_in_void = 12
				save_scope_as = mgrbfm_abbas_scope
			}
		}
		create_character = {
			template = mgrbfm_alawi_said
			on_created = {
				place_character_in_void = 12
				save_scope_as = mgrbfm_said_scope
			}
		}
		heir = {
			save_scope_as = mgrbfm_heir_scope
		}
	}

	option = {
		name = mgrbfm_egypt.003.a
		default_option = yes
		ruler = {
			kill_character = yes
		}
		heir ?= {
			set_character_as_ruler = yes
		}
		if = {
			limit = {
				scope:mgrbfm_heir_scope = {
					has_template = mgrbfm_alawi_ibrahim
				}
			}
			set_heir = scope:mgrbfm_abbas_scope
		}
		else_if = {
			limit = {
				scope:mgrbfm_heir_scope = {
					has_template = mgrbfm_alawi_abbas
				}
			}
			set_heir = scope:mgrbfm_said_scope
		}
	}

	after = {
		scope:mgrbfm_abbas_scope = {
			gbbf_free_from_void = yes
		}
		scope:mgrbfm_said_scope = {
			gbbf_free_from_void = yes
		}
	}
}

mgrbfm_egypt.110 = {
	type = country_event
	placement = scope:mgrbfm_nile_flood_state_scope
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 2
	dlc = dlc_grefm

	title = mgrbfm_egypt.110.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = mgrbfm_egypt.110.d1
				trigger = {
					NOT = {
						any_scope_state = {
							is_in_geographic_region = geographic_region_lower_nile
							has_building = gbbf_building_nile_barrage
						}
					}
				}
			}
			triggered_desc = {
				desc = mgrbfm_egypt.110.d2
				trigger = {
					any_scope_state = {
						is_in_geographic_region = geographic_region_lower_nile
						is_production_method_active = {
							building_type = gbbf_building_nile_barrage
							production_method = pm_gbbf_no_reservior
						}
					}
				}
			}
			triggered_desc = {
				desc = mgrbfm_egypt.110.d3
				trigger = {
					any_scope_state = {
						filter = {
							is_in_geographic_region = geographic_region_lower_nile
						}
						count = all
						OR = {
							is_production_method_active = {
								building_type = gbbf_building_nile_barrage
								production_method = pm_gbbf_adequate_reservior
							}
							is_production_method_active = {
								building_type = gbbf_building_nile_barrage
								production_method = pm_gbbf_massive_reservior
							}
						}
					}
				}
			}
		}
	}
	flavor = mgrbfm_egypt.110.f

	event_image = {
		video = "gfx/event_pictures/unspecific_devastation.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		# triggered by mgrbfm_nile_inundation_effect
	}

	immediate = {
		random_state_in_lower_nile = {
			save_scope_as = mgrbfm_nile_flood_state_scope
		}
	}

	option = {
		name = mgrbfm_egypt.110.a
		default_option = yes
		trigger = {
			any_scope_state = {
				filter = {
					is_in_geographic_region = geographic_region_lower_nile
				}
				count = all
				NOT = {
					has_building = gbbf_building_nile_barrage
				}
			}
		}
		every_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_lower_nile
			}
			add_modifier = {
				name = modifier_mgrbfm_widespread_nile_floods
				months = gbbf_short_modifier_time
				multiplier = global_var:je_mgrbfm_egypt_nile_river.var:mgrbfm_nile_flood_var
				is_decaying = yes
			}
		}
	}

	option = {
		name = mgrbfm_egypt.110.b
		trigger = {
			any_scope_state = {
				is_in_geographic_region = geographic_region_lower_nile
				is_production_method_active = {
					building_type = gbbf_building_nile_barrage
					production_method = pm_gbbf_no_reservior
				}
			}
		}
		every_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_lower_nile
				NOR = {
					is_production_method_active = {
						building_type = gbbf_building_nile_barrage
						production_method = pm_gbbf_adequate_reservior
					}
					is_production_method_active = {
						building_type = gbbf_building_nile_barrage
						production_method = pm_gbbf_massive_reservior
					}
				}
			}
			add_modifier = {
				name = modifier_mgrbfm_widespread_nile_floods
				months = gbbf_short_modifier_time
				multiplier = global_var:je_mgrbfm_egypt_nile_river.var:mgrbfm_nile_flood_var
				is_decaying = yes
			}
		}
		every_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_lower_nile
				is_production_method_active = {
					building_type = gbbf_building_nile_barrage
					production_method = pm_gbbf_adequate_reservior
				}
			}
			add_modifier = {
				name = modifier_mgrbfm_localized_nile_floods
				months = gbbf_short_modifier_time
				multiplier = global_var:je_mgrbfm_egypt_nile_river.var:mgrbfm_nile_flood_var
				is_decaying = yes
			}
		}
		every_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_lower_nile
				is_production_method_active = {
					building_type = gbbf_building_nile_barrage
					production_method = pm_gbbf_massive_reservior
				}
			}
			add_modifier = {
				name = modifier_mgrbfm_mitigated_nile_floods
				months = gbbf_short_modifier_time
			}
		}
	}

	option = {
		name = mgrbfm_egypt.110.c
		trigger = {
			any_scope_state = {
				filter = {
					is_in_geographic_region = geographic_region_lower_nile
				}
				count = all
				OR = {
					is_production_method_active = {
						building_type = gbbf_building_nile_barrage
						production_method = pm_gbbf_adequate_reservior
					}
					is_production_method_active = {
						building_type = gbbf_building_nile_barrage
						production_method = pm_gbbf_massive_reservior
					}
				}
			}
		}
		every_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_lower_nile
			}
			add_modifier = {
				name = modifier_mgrbfm_mitigated_nile_floods
				months = gbbf_short_modifier_time
			}
		}
	}
	
	after = {
		hidden_effect = {
			if = {
				limit = {
					is_involved_in_journal_entry = je_mgrbfm_egypt_nile_river
					any_scope_state = {
						is_in_geographic_region = geographic_region_lower_nile
						OR = {
							has_modifier = modifier_mgrbfm_widespread_nile_floods
							has_modifier = modifier_mgrbfm_mitigated_nile_floods
						}
					}
				}
				random_scope_state = {
					limit = {
						is_in_geographic_region = geographic_region_lower_nile
						OR = {
							has_modifier = modifier_mgrbfm_widespread_nile_floods
							has_modifier = modifier_mgrbfm_mitigated_nile_floods
						}
					}
					save_scope_as = mgrbfm_flood_state_scope
				}
				random_list = {
					90 = {}
					5 = {
						trigger_event = mgrbfm_egypt.112
						trigger = {
							scope:mgrbfm_flood_state_scope = {
								has_modifier = modifier_mgrbfm_nilometer_system
								OR = {
									has_modifier = modifier_mgrbfm_localized_nile_floods
									has_modifier = modifier_mgrbfm_widespread_nile_floods
								}
							}
						}
					}
					5 = {
