turkfm_milletindir.101 = {
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	duration = 3
	dlc = dlc_grefm

	title = turkfm_milletindir.101.t
	desc = turkfm_milletindir.101.d
	flavor = turkfm_milletindir.101.f

	event_image = {
		video = "unspecific_devastation"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		# triggered by je_turkfm_armenian_expulsions
	}

	immediate = {
		if = {
			limit = {
				has_modifier = modifier_turkfm_cooking_the_books
			}
			remove_modifier = modifier_turkfm_cooking_the_books
		}
		if = {
			limit = {
				has_modifier = modifier_turkfm_shooting_the_press
			}
			remove_modifier = modifier_turkfm_shooting_the_press
		}
	}

	option = {
		name = turkfm_milletindir.101.a
		default_option = yes
		every_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_greater_armenia
			}
			add_modifier = {
				name = modifier_turkfm_anatolian_frontier
				months = gbbf_long_modifier_time
			}
			state_region = {
				if = {
					limit = {
						NOT = {
							is_homeland = turkish
						}
					}
					add_homeland = cu:turkish
				}
			}
		}
	}
}

turkfm_milletindir.103 = {
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	duration = 3
	dlc = dlc_grefm

	title = turkfm_milletindir.103.t
	desc = turkfm_milletindir.103.d
	flavor = turkfm_milletindir.103.f

	event_image = {
		video = "ip3_police_raid"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		# triggered by turkfm_armenian_expulsion_effect via je_turkfm_armenian_expulsions on_monthly_pulse
	}

	immediate = {
		set_variable = turkfm_armenian_intervention_considered
		c:TUR ?= {
			save_scope_as = ottoman_scope
		}
		root = {
			save_scope_as = protester_scope
		}
	}

	option = {
		name = turkfm_milletindir.103.a
		default_option = yes
		change_relations = {
			country = scope:russia_scope
			value = -30
		}
		c:TUR ?= {
			post_notification = notification_turkfm_expulsions_protested
			create_diplomatic_catalyst = {
				type = catalyst_event_negative
				target = root
			}
		}
	}

	option = {
		name = turkfm_milletindir.103.b
		if = {
			limit = {
				any_scope_pop = {
					has_pop_culture = armenian
				}
			}
			add_radicals = {
				value = very_large_radicals
				culture = cu:circassian
			}
		}
		else = {
			custom_tooltip = gbbf_nothing_happens
		}
	}

	option = { # Fuck it, we ball
		name = turkfm_milletindir.103.c
		create_diplomatic_play = {
			type = dp_gbbf_halt_expulsions
			target_country = scope:ottoman_scope
			add_war_goal = {
				holder = root
				type = gbbf_halt_expulsions
				target_country = c:TUR
				primary_demand = yes
			}
		}
		create_incident = {
			country = scope:ottoman_scope
			value = 10
		}
		change_relations = {
			country = scope:ottoman_scope
			value = -50
		}
	}
}
