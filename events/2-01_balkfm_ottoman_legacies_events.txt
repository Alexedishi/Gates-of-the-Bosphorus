## Ottoman Legacies Mechanical Events ##
balkfm_legacies.010 = { # The Brigands Raid Again!
	type = country_event
	placement = scope:brigandage_state

	title = balkfm_legacies.010.t
	desc = balkfm_legacies.010.d
	flavor = balkfm_legacies.010.f

	event_image = {
		video = "rp1_social_banditry"
	}

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		any_scope_state = {
			NOT = { has_modifier = modifier_balkfm_brigandage }
		}
	}

	cancellation_trigger = {
		NOT = {
			has_journal_entry = je_balkfm_brigandage
		}
	}

	cooldown = {
		years = 3
	}

	immediate = {
		random_scope_state = {
			limit = {
				NOT = { has_modifier = modifier_balkfm_brigandage }
			}
			save_scope_as = brigandage_state
		}
	}

	option = {
		name = balkfm_legacies.010.a
		default_option = yes
		scope:brigandage_state = {
			add_modifier = {
				name = modifier_balkfm_brigandage
				months = gbbf_normal_modifier_time
			}
			add_radicals_in_state = {
				value = small_radicals
			}
		}
		if = {
			limit = {
				has_journal_entry = je_grefm_bavarocracy_main
			}
			custom_tooltip = {
				text = je_grefm_bavarocracy_minus_50
				change_variable = {
					name = grefm_bavarocracy_var
					add = -50
				}
			}
		}
	}

}
