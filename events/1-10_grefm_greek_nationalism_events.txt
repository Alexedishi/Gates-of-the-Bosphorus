grefm_greek_nationalism.003 = { # option to switch capital to Constantinople
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	title = grefm_greek_nationalism.003.t
	desc = grefm_greek_nationalism.003.d
	flavor = grefm_greek_nationalism.003.f

	event_image = {
		video = "unspecific_world_fair"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		owns_entire_state_region = STATE_EASTERN_THRACE
		has_journal_entry = je_grefm_megali_idea
		s:STATE_EASTERN_THRACE = {
			any_scope_state = {
				is_capital = no
			}
		}
		NOT = { has_variable = grefm_capital_considered }
	}

	immediate = {
		s:STATE_EASTERN_THRACE = {
			random_scope_state = {
				save_scope_as = eastern_thrace_scope
			}
		}
		s:STATE_ATTICA = {
			random_scope_state = {
				save_scope_as = attica_scope
			}
		}
		set_variable = grefm_capital_considered
	}

	cancellation_trigger = {
		s:STATE_EASTERN_THRACE = {
			any_scope_state = {
				is_capital = yes
			}
		}
	}

	option = { # yes please
		name = grefm_greek_nationalism.003.a
		default_option = yes
		set_capital = STATE_EASTERN_THRACE
		scope:eastern_thrace_scope = {
			add_loyalists_in_state = {
				value = medium_radicals
				culture = cu:greek
			}
		}
	}

	option = { # no I'm a poor boy
		name = grefm_greek_nationalism.003.b
		scope:attica_scope = {
			add_modifier = {
				name = modifier_grefm_development_focus
				days = normal_modifier_time
			}
			add_loyalists_in_state = {
				value = small_radicals
				culture = cu:greek
			}
		}
	}

}

grefm_greek_nationalism.009 = { # The Greek Ethnogenesis
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	title = grefm_greek_nationalism.009.t
	desc = grefm_greek_nationalism.009.d
	flavor = grefm_greek_nationalism.009.f

	event_image = {
		video = "unspecific_world_fair"
	}

	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		# triggered by scripted buttons in je_grefm_greek_nationalism
	}

	immediate = {
		if = {
			limit = {
				var:grefm_culture_var = 1 # Hellene
			}
			cu:greek = {
				save_scope_as = grefm_national_culture_scope
			}
		}
		else_if = {
			limit = {
				var:grefm_culture_var = 2 # Rhomaios
			}
			cu:rhomaios = {
				save_scope_as = grefm_national_culture_scope
			}
		}
		else = {
			cu:greek = {
				save_scope_as = grefm_national_culture_scope
			}
		}
	}

	option = { # Hellene
		name = grefm_greek_nationalism.009.a
		default_option = yes
		trigger = {
			var:grefm_culture_var = 1
		}
		add_modifier = {
			name = modifier_grefm_greek_imperialism
			days = long_modifier_time
		}
		add_loyalists = {
			culture = cu:greek
			value = medium_radicals
		}
		add_radicals = {
			culture = cu:rhomaios
			value = medium_radicals
		}
		if = {
			limit = {
				NOT = {
					country_has_primary_culture = cu:greek
				}
			}
			add_primary_culture = cu:greek
		}
		if = {
			limit = {
				country_has_primary_culture = cu:rhomaios
			}
			remove_primary_culture = cu:rhomaios
		}
		set_variable = grefm_nationalism_complete
	}

	option = { # Rhomaios
		name = grefm_greek_nationalism.009.a
		default_option = yes
		trigger = {
			var:grefm_culture_var = 2
		}
		add_modifier = {
			name = modifier_grefm_greek_imperialism
			days = long_modifier_time
		}
		add_radicals = {
			culture = cu:greek
			value = medium_radicals
		}
		add_loyalists = {
			culture = cu:rhomaios
			value = medium_radicals
		}
		if = {
			limit = {
				NOT = {
					country_has_primary_culture = cu:rhomaios
				}
			}
			add_primary_culture = cu:rhomaios
		}
		if = {
			limit = {
				country_has_primary_culture = cu:greek
			}
			remove_primary_culture = cu:greek
		}
		set_variable = grefm_nationalism_complete
	}

	option = { # Hybrid
		name = grefm_greek_nationalism.009.a
		default_option = yes
		trigger = {
			var:grefm_culture_var = 3
		}
		add_modifier = {
			name = modifier_grefm_greek_imperialism
			days = long_modifier_time
		}
		add_loyalists = {
			culture = cu:greek
			value = medium_radicals
		}
		add_loyalists = {
			culture = cu:rhomaios
			value = medium_radicals
		}
		if = {
			limit = {
				NOT = {
					country_has_primary_culture = cu:greek
				}
			}
			add_primary_culture = cu:greek
		}
		if = {
			limit = {
				NOT = {
					country_has_primary_culture = cu:rhomaios
				}
			}
			add_primary_culture = cu:rhomaios
		}

	}
