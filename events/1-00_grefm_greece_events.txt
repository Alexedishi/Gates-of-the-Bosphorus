grefm_greece.003 = {
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_otto.dds"

	title = grefm_greece.003.t
	desc = grefm_greece.003.d
	flavor = grefm_greece.003.f

	event_image = {
		video = "europenorthamerica_political_extremism"
	}

	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		NOR = {
			has_variable = grefm_1843_revolution_happened
			has_modifier = modifier_grefm_armansperg_regency
		}
		var:grefm_bavarocracy_instability_var >= 200
	}

	immediate = {
		ruler = {
			save_scope_as = king_otto_scope
		}
		ig:ig_armed_forces = {
			leader = {
				save_scope_as = ig_armed_forces_scope
			}
		}
		set_variable = grefm_1843_revolution_happened
	}

	option = {
		# Accept the New Constitution
		name = grefm_greece.003.a
		default_option = yes
		if = {
			limit = {
				OR = {
					has_law_or_variant = law_type:law_hereditary_bureaucrats
				}
			}
			activate_law = law_type:law_appointed_bureaucrats
		}
		if = {
			limit = {
				AND = {
					OR = {
						has_law_or_variant = law_type:law_autocracy
						has_law_or_variant = law_type:law_oligarchy
						has_law_or_variant = law_type:law_landed_voting
						has_law_or_variant = law_type:law_wealth_voting
						has_law_or_variant = law_type:law_census_voting
					}
					has_technology_researched = egalitarianism
				}
			}
			activate_law = law_type:law_universal_suffrage
		}
		else_if = {
			limit = {
				OR = {
					has_law_or_variant = law_type:law_autocracy
					has_law_or_variant = law_type:law_oligarchy
					has_law_or_variant = law_type:law_landed_voting
					has_law_or_variant = law_type:law_wealth_voting
				}
			}
			activate_law = law_type:law_census_voting
		}
		every_interest_group = {
			limit = {
				law_stance = {
					law = law_type:law_autocracy
					value > neutral
				}
			}
			add_modifier = {
				name = modifier_grefm_constitution_enforced
				months = gbbf_normal_modifier_time
				is_decaying = yes
			}
		}
		add_radicals = {
			strata = upper
			value = medium_radicals
		}
		add_loyalists = {
			strata = middle
			value = small_radicals
		}
		add_loyalists = {
			strata = lower
			value = small_radicals
		}
		call_election = {
			months = 1
		}
		grefm_bavarocracy_medium_support_effect = yes
		set_variable = {
			name = grefm_bavarocracy_instability_var
			value = 0
		}
	}

	option = {
		# Othonian Absolutism Time
		name = grefm_greece.003.b
		ruler = {
			add_modifier = {
				name = modifier_grefm_tyrant_king
				months = gbbf_long_modifier_time
			}
		}
		every_interest_group = {
			limit = {
				law_stance = {
					law = law_type:law_autocracy
					value < neutral
				}
			}
			add_modifier = {
				name = modifier_grefm_constitution_refused
				months = gbbf_normal_modifier_time
				is_decaying = yes
			}
		}
		add_loyalists = {
			strata = upper
			value = medium_radicals
		}
		add_radicals = {
			strata = middle
			value = medium_radicals
		}
		add_radicals = {
			strata = lower
			value = medium_radicals
		}
		scope:king_otto_scope = {
			set_as_interest_group_leader = yes # Otto discharged von Rudhart in 1837 to directly chair his privy council; this is represented here for gameplay reasons
		}
		if = {
			limit = {
				has_law_or_variant = law_type:law_oligarchy
			}
			activate_law = law_type:law_autocracy
		}
		else_if = {
			limit = {
				NOT = {
					has_law_or_variant = law_type:law_oligarchy
				}
			}
			activate_law = law_type:law_autocracy
		}
		set_variable = {
			name = grefm_tyranny_var
			value = 0
		}
		set_variable = {
			name = grefm_bavarocracy_instability_var
			value = 0
		}
	}

	option = {
		# Find a Greek King instead
		name = grefm_greece.003.c
		trigger = {
			is_player = yes
		}
		create_incident = {
			country = c:GBR
			value = 15
		}
		create_incident = {
			country = c:FRA
			value = 15
		}
		create_incident = {
			country = c:RUS
			value = 15
		}
		ruler = {
			exile_character = yes
		}
		heir ?= {
			exile_character = yes
		}
		scope:ig_armed_forces_scope = {
			set_character_as_ruler = yes
		}
		if = {
			limit = {
				OR = {
					has_law_or_variant = law_type:law_hereditary_bureaucrats
				}
			}
			activate_law = law_type:law_appointed_bureaucrats
		}
		if = {
			limit = {
				AND = {
					OR = {
						has_law_or_variant = law_type:law_autocracy
						has_law_or_variant = law_type:law_oligarchy
						has_law_or_variant = law_type:law_landed_voting
						has_law_or_variant = law_type:law_wealth_voting
						has_law_or_variant = law_type:law_census_voting
					}
					has_technology_researched = egalitarianism
				}
			}
			activate_law = law_type:law_universal_suffrage
		}
		else_if = {
			limit = {
				OR = {
					has_law_or_variant = law_type:law_autocracy
					has_law_or_variant = law_type:law_oligarchy
					has_law_or_variant = law_type:law_landed_voting
					has_law_or_variant = law_type:law_wealth_voting
				}
			}
			activate_law = law_type:law_census_voting
		}
		every_interest_group = {
			limit = {
				law_stance = {
					law = law_type:law_autocracy
					value > neutral
				}
			}
			add_modifier = {
				name = modifier_grefm_constitution_enforced
				months = gbbf_normal_modifier_time
				is_decaying = yes
			}
		}
		add_radicals = {
			pop_type = aristocrats
			value = medium_radicals
		}
		add_loyalists = {
			strata = middle
			value = small_radicals
		}
		add_loyalists = {
			strata = lower
			value = small_radicals
		}
		trigger_event = {
			id = grefm_greece.021 # The Search for a King
			days = 30
		}

	}

	trigger = {
		has_variable = grefm_venizelism_military_agenda
		## Some Notes on the completion requirements (drawn from Greek-language sources):
		# The Greek Army during the First Balkan War consisted of roughly 130k men with no corps-level or higher structure in place until war was declared.
		# The primary force was was the "Army of Thessaly", with roughly 100k men, of which 80k were organized into frontline units. The "Concise History of the Balkan Wars" states:
		# 59 Infantry with 4 Evzones battalions
		# 8 Cavalry Companies
		# 32 Artillery Companies
		# The Secondary force was of the "Army of Epirus" that varied from 10k to 13k men with various forms of organization.
		# A third force the "Interior Army" is referenced in some sources with between 10k to 20k men, but doesn't seem to have been intended as a frontline formation.
		# The Hellenic navy had around a dozen ships during the Balkan Wars with a rapid expansion of the Navy occurring after the outbreak of hostilities; They joined WWI with 22 surface combatants.
		# I have contradicting sources about when exactly conscription was adopted and in which forms, but it was definitely in place as general conscription by 1917 when Greece joined WWI.
		has_law_or_variant = law_type:law_mass_conscription
		any_military_formation = {
			is_army = yes
			any_combat_unit = {
				count >= 80 # This is based on the frontline forces of the Army of Thessaly; see above.
				OR = {
					has_unit_type = unit_type:combat_unit_type_trench_infantry
					has_unit_type = unit_type:combat_unit_type_squad_infantry
					has_unit_type = unit_type:combat_unit_type_mechanized_infantry
					has_unit_type = unit_type:combat_unit_type_shrapnel_artillery
					has_unit_type = unit_type:combat_unit_type_siege_artillery
					has_unit_type = unit_type:combat_unit_type_heavy_tank
				}
			}
		}
		any_military_formation = {
			is_fleet = yes
			any_scope_ship = {
				count >= 20 # This is a round number based on surface combatants at the start of WWI; see above.
				OR = {
					is_ship_type = ship_type:ship_type_early_ironclad
					is_ship_type = ship_type:ship_type_modern_ironclad
					is_ship_type = ship_type:ship_type_pre_dreadnought
					is_ship_type = ship_type:ship_type_dreadnought
					is_ship_type = ship_type:ship_type_super_dreadnought
					
					is_ship_type = ship_type:ship_type_light_cruiser
					is_ship_type = ship_type:ship_type_protected_cruiser
					is_ship_type = ship_type:ship_type_armored_cruiser
					
					is_ship_type = ship_type:ship_type_submarine
					is_ship_type = ship_type:ship_type_destroyer
				}
			}
		}
	}
