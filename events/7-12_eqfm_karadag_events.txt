eqfm_karadag.010 = {
	# Negotiations On Head-taking (Receiver)
	type = country_event
	placement = scope:montenegro_scope
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 1
	dlc = dlc_grefm
	
	title = eqfm_karadag.010.t
	desc = eqfm_karadag.010.d
	flavor = eqfm_karadag.010.f
	
	event_image = {
		video = "asia_politician_parliament_motion"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		# triggered by eqfm_renounce_headtaking_button
	}
	
	immediate = {
		random_country = {
			limit = {
				has_variable = turkfm_negociated_headtaking
			}
			save_scope_as = turkfm_headtaking_negotiator
		}
		set_variable = {
			name = turkfm_headtaking_recipient
			months = 3
		}
	}
	
	option = {
		# Reject headtaking offer
		name = eqfm_karadag.010.a
		default_option = yes
		add_modifier = {
			name = modifier_eqfm_retained_headtaking
			months = gbbf_long_modifier_time
		}
		scope:turkfm_headtaking_negotiator = {
			trigger_event = {
				id = eqfm_karadag.011 # Negotiations On Head-taking
			}
			set_variable = turkfm_headtaking_rejected
		}
	}
	
	option = {
		# Accept headtaking offer
		name = eqfm_karadag.010.b
		if = {
			limit = {
				has_modifier = modifier_eqfm_practicing_headtaking_mon
			}
			remove_modifier = modifier_eqfm_practicing_headtaking_mon
		}
		if = {
			limit = {
				has_modifier = modifier_eqfm_practicing_headtaking_tur
			}
			remove_modifier = modifier_eqfm_practicing_headtaking_tur
		}
		add_modifier = {
			name = modifier_eqfm_abolished_headtaking
			months = gbbf_normal_modifier_time
		}
		scope:turkfm_headtaking_negotiator = {
			trigger_event = {
				id = eqfm_karadag.011 # Negotiations On Head-taking
			}
		}
	}
}

	# Negotiations on Head-taking (MON)
	type = country_event

		}
		triggered_desc = {
			desc = eqfm_karadag.011.d2
			trigger = {

			}

	}

		random_country = {
			limit = {
				has_variable = turkfm_headtaking_recipient
			}

		add_modifier = {
			name = modifier_eqfm_unilaterally_abolished_headtaking
			months = gbbf_long_modifier_time
		}
		remove_variable = turkfm_headtaking_rejected
	}
	
	option = {
		# Head-taking offer accepted
		name = eqfm_karadag.011.c
		trigger = {
			NOT = {
				has_variable = turkfm_headtaking_rejected
			}
		}
		if = {
			limit = {
				has_modifier = modifier_eqfm_practicing_headtaking_mon
			}
			remove_modifier = modifier_eqfm_practicing_headtaking_mon
		}

				}

			}

	immediate = {
		c:MON = {
			save_scope_as = montenegro_scope

		}
	}
	
	option = {
		# Accept the Deal
		name = eqfm_karadag.012.a
		default_option = yes
		show_as_tooltip = {
			s:STATE_MONTENEGRO = {
				set_owner_of_provinces = {
					country = scope:montenegro_scope
					provinces = {
						x90A0E0 # Kotor
					}
				}
			}
			c:MON = {
				set_owes_obligation_to = {
					country = root
					setting = yes
				}
			}
		}
		create_diplomatic_catalyst = {
			type = catalyst_relations_level_increased
			target = scope:montenegro_scope
		}
		change_relations = {
			country = c:MON
			value = 25
		}
	}
	
	option = {
		# Reject the Deal
		name = eqfm_karadag.012.b
		change_relations = {
			country = c:MON
			value = -25
		}
		create_diplomatic_catalyst = {
			type = catalyst_relations_level_decreased
			target = c:MON
		}
	}
	
	after = {
		trigger_event = {
			id = eqfm_karadag.013
			days = { 5 15 }
		}
	}
