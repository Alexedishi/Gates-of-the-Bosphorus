eqfm_black_mountain.010 = {
	# Negotiations On Head-taking (Receiver)
	type = country_event
	placement = scope:montenegro_scope
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 1
	dlc = dlc_grefm
	
	title = eqfm_black_mountain.010.t
	desc = eqfm_black_mountain.010.d
	flavor = eqfm_black_mountain.010.f
	
	event_image = {
		video = "asia_politician_parliament_motion"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		# triggered by eqfm_renounce_headtaking_button
	}
	
	immediate = {
		random_country = {
			limit = {
				has_variable = turkfm_negociated_headtaking
			}
			save_scope_as = turkfm_headtaking_negotiator
		}
		set_variable = {
			name = turkfm_headtaking_recipient
			months = 3
		}
	}
	
	option = {
		# Reject headtaking offer
		name = eqfm_black_mountain.010.a
		default_option = yes
		add_modifier = {
			name = modifier_eqfm_retained_headtaking
			months = gbbf_10_years
		}
		scope:turkfm_headtaking_negotiator = {
			trigger_event = {
				id = eqfm_black_mountain.011 # Negotiations On Head-taking
			}
			set_variable = turkfm_headtaking_rejected
		}
	}
	
	option = {
		# Accept headtaking offer
		name = eqfm_black_mountain.010.b
		if = {
			limit = {
				has_modifier = modifier_eqfm_practicing_headtaking_mon
			}
			remove_modifier = modifier_eqfm_practicing_headtaking_mon
		}
		if = {
			limit = {
				has_modifier = modifier_eqfm_practicing_headtaking_tur
			}
			remove_modifier = modifier_eqfm_practicing_headtaking_tur
		}
		add_modifier = {
			name = modifier_eqfm_abolished_headtaking
			months = gbbf_5_years
		}
		scope:turkfm_headtaking_negotiator = {
			trigger_event = {
				id = eqfm_black_mountain.011 # Negotiations On Head-taking
			}
		}
	}
}
