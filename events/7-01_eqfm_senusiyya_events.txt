eqfm_senusiyya.001 = {
	type = country_event
	placement = scope:eqfm_senusiyya_state_scope
	icon = "gfx/interface/icons/generic_icons/gbbf_runestone_moon.dds"
	duration = 3
	dlc = dlc_grefm

	title = eqfm_senusiyya.001.t
	desc = eqfm_senusiyya.001.d
	flavor = eqfm_senusiyya.001.f

	gui_window = event_window_1char_adventure
	left_icon = scope:eqfm_senussi_character_scope

	event_image = {
		video = "middleeast_islam"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	trigger = {
		# triggered by je_eqfm_senusiyya
	}

	immediate = {
		eqfm_save_senussi_scope_effect = yes
		set_variable = eqfm_senusiyya_started
		if = {
			limit = {
				any_state_in_senusiyya = {
					has_variable = eqfm_senusiyya_entrenchment_var
				}
			}
			random_state_in_senusiyya = {
				limit = {
					has_variable = eqfm_senusiyya_entrenchment_var
				}
				save_scope_as = eqfm_senusiyya_state_scope
			}
		}
		else = {
			s:STATE_LIBYA = {
				random_scope_state = {
					save_scope_as = eqfm_senusiyya_state_scope
				}
			}
		}
	}

	option = {
		name = eqfm_senusiyya.001.a
		default_option = yes
		custom_tooltip = eqfm_senusiyya.001.a.tt
	}
}

		}
		custom_tooltip = {
			text = eqfm_starting_senusiyya_entrenchment
			s:STATE_LIBYA = {
				set_variable = {
					name = eqfm_senusiyya_entrenchment_var
					value = 25
				}
			}
		}
		every_country = {
			limit = {
				is_involved_in_journal_entry = je_eqfm_senusiyya

			}
			trigger_event = {
				id = eqfm_senusiyya.003

	# Senusiyya Spreads to State Region (Headless)

		has_global_variable = eqfm_intimidated_sensussi

			# Evaluate STATE_LIBYA
			AND = {
				s:STATE_LIBYA = {
					has_variable = eqfm_senusiyya_entrenchment_var

				}

					s:STATE_TRIPOLI = {
