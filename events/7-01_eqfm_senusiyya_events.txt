eqfm_senusiyya.003 = {
	type = country_event
	placement = scope:kufra_scope
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3
	dlc = dlc_grefm
	
	title = eqfm_senusiyya.003.t
	desc = eqfm_senusiyya.003.d
	flavor = eqfm_senusiyya.003.f
	
	gui_window = event_window_1char_adventure
	left_icon = scope:eqfm_senussi_character_scope
	
	event_image = {
		video = "africa_desert_expedition"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		NOT = {
			has_variable = eqfm_senussi_ran_away
		}
		exists = c:ZWY
		OR = {
			has_state_in_state_region = STATE_LIBYAN_DESERT
			has_state_in_state_region = STATE_LIBYA
		}
		OR = {
			exists = c:TRI
			exists = c:FZN
		}
		OR = {
			has_global_variable = eqfm_intimidated_sensussi
			global_var:je_eqfm_senusiyya.var:eqfm_senusiyya_var >= 50
		}
	}
	
	cancellation_trigger = {
		NOT = {
			exists = c:ZWY
		}
		NAND = {
			exists = c:TRI
			exists = c:FZN
		}
	}
	
	cooldown = {
		years = 5
	}
	
	immediate = {
		eqfm_update_senussi_character_effect = yes
		c:TRI = {
			save_scope_as = libya_scope
		}
		c:ZWY = {
			save_scope_as = kufra_scope
		}
		set_variable = {
			name = eqfm_senussi_ran_away
			months = 60
		}
		set_global_variable = eqfm_intimidated_sensussi
	}
	
	option = {
		name = eqfm_senusiyya.003.a
		default_option = yes
	}
}

		video = "middleeast_courtroom_upheaval"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		# triggered by eqfm_armed_delegation_button
	}
	
	cancellation_trigger = {
		eqfm_has_desert_expedition_trigger = no
	}
	
	immediate = {
		eqfm_update_senussi_character_effect = yes
		random_scope_general = {
			limit = {
				has_modifier = modifier_eqfm_leading_desert_expedition

			save_scope_as = eqfm_intimidation_scope
		}
		c:ZWY = {
			capital = {
				save_scope_as = kufra_capital_scope
			}
		}
	}
	
	option = {
		name = eqfm_senusiyya.004.a
		default_option = yes
		eqfm_add_senusiyya_pressure_effect = {
			value = -20
		}
	}
}
