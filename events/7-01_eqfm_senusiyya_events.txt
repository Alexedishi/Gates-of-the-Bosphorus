eqfm_senusiyya.003 = { # Senussi Withdraws to the Desert
	type = country_event
	placement = scope:kufra_scope
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	
	title = eqfm_senusiyya.003.t
	desc = eqfm_senusiyya.003.d
	flavor = eqfm_senusiyya.003.f
	
	gui_window = event_window_1char_adventure
	
	left_icon = scope:eqfm_senussi_character_scope
	
	event_image = {
		video = "africa_desert_expedition"
	}
	
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	duration = 3
	
	dlc = dlc_grefm
	
	trigger = {
		NOT = {
			has_variable = eqfm_senussi_ran_away
		}
		has_global_variable = mgrbfm_intimidated_sensussi
		var:eqfm_senusiyya_var <= 50
	}
	
	immediate = {
		set_variable = eqfm_senussi_ran_away
		eqfm_update_senussi_effect = yes
		c:ZWY = {
			save_scope_as = kufra_scope
		}
	}
	
	option = {
		name = eqfm_senusiyya.003.a
		default_option = yes
	}
	
}

eqfm_senusiyya.004 = { # Armed Diplomacy
	type = country_event
	placement = scope:kufra_capital_scope
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	
	title = eqfm_senusiyya.004.t
	desc = eqfm_senusiyya.004.d
	flavor = eqfm_senusiyya.004.f
	
	gui_window = event_window_2char_discussion
	
	left_icon = scope:eqfm_senussi_character_scope
	right_icon = scope:turkfm_intimidation_scope
	
	event_image = {
		video = "middleeast_courtroom_upheaval"
	}
	
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	duration = 3
	
	dlc = dlc_grefm
	
	trigger = {
		# triggered by eqfm_armed_delegation_button
	}
	
	cancellation_trigger = {
		eqfm_has_desert_expedition_trigger = no
	}
	
	immediate = {
		eqfm_update_senussi_effect = yes
		random_scope_general = {
			limit = {
				has_modifier = modifier_eqfm_leading_desert_expedition
			}
			save_scope_as = turkfm_intimidation_scope
		}
		c:ZWY = {
			capital = {
				save_scope_as = kufra_capital_scope
			}
		}
	}
	
	option = {
		name = eqfm_senusiyya.004.a
		default_option = yes
		custom_tooltip = {
			text = eqfm_senussi_lose_10
			change_variable = {
				name = eqfm_senusiyya_var
				subtract = 10
			}
			clamp_variable = {
				name = eqfm_senusiyya_var
				max = 100
				min = 0
			}
		}
	}
	
}
