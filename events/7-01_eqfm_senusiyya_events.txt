eqfm_sanusiyyah.001 = {
	type = country_event
	placement = scope:eqfm_sanusiyyah_state_scope
	icon = "gfx/interface/icons/generic_icons/gbbf_runestone_moon.dds"
	duration = 3
	dlc = dlc_grefm

	title = eqfm_sanusiyyah.001.t
	desc = eqfm_sanusiyyah.001.d
	flavor = eqfm_sanusiyyah.001.f

	gui_window = event_window_1char_adventure
	left_icon = scope:eqfm_senussi_character_scope

	event_image = {
		video = "middleeast_islam"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	trigger = {
		# triggered by je_eqfm_sanusiyyah
	}

	immediate = {
		eqfm_save_senussi_scope_effect = yes
		set_variable = eqfm_sanusiyyah_started
		if = {
			limit = {
				any_state_in_sanusiyyah = {
					has_variable = eqfm_sanusiyyah_entrenchment_var
				}
			}
			random_state_in_sanusiyyah = {
				limit = {
					has_variable = eqfm_sanusiyyah_entrenchment_var
				}
				save_scope_as = eqfm_sanusiyyah_state_scope
			}
		}
		else = {
			s:STATE_LIBYA = {
				random_scope_state = {
					save_scope_as = eqfm_sanusiyyah_state_scope
				}
			}
		}
	}

	option = {
		name = eqfm_sanusiyyah.001.a
		default_option = yes
		custom_tooltip = eqfm_sanusiyyah.001.a.tt
	}
}

eqfm_sanusiyyah.200 = {
	# Senusiyya Spreads to State Region (Headless)
	type = country_event
	hidden = yes

		has_global_variable = eqfm_intimidated_sensussi
		NOT = {
			has_global_variable = eqfm_sanusiyyah_contagion_cooldown
		}

			global_var:je_eqfm_sanusiyyah.var:eqfm_sanusiyyah_phase_var ?= flag:eqfm_sanusiyyah_unease
			global_var:je_eqfm_sanusiyyah.var:eqfm_sanusiyyah_phase_var ?= flag:eqfm_sanusiyyah_belligerency
		}
		OR = {
			# Evaluate STATE_LIBYA
			AND = {
				s:STATE_LIBYA = {
					has_variable = eqfm_sanusiyyah_entrenchment_var

				}

					s:STATE_TRIPOLI = {

		video = "africa_construction_colony"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		# triggered by Senusiyya Expedition buttons
	}
	
	cancellation_trigger = {
		OR = {
			NOT = {
				any_scope_general = {
					is_busy = no
					NOT = {
						has_role_of_type = ruler
					}
				}
			}
			eqfm_has_sanusiyyah_expedition_trigger = no
		}
	}
	
	immediate = {
		set_variable = {
			name = eqfm_sanusiyyah_expedition_transit_var
			months = 3
		}
		random_scope_character = {
			limit = {
				has_role_of_type = general
				is_busy = no
				NOT = {
					has_role_of_type = ruler
				}
			}
			save_scope_as = eqfm_sanusiyyah_leader_1
		}
		if = {
			limit = {
				any_scope_character = {
					has_role_of_type = general
					is_busy = no
					NOT = {
						has_role_of_type = ruler
					}
					OR = {
						has_trait = surveyor
						has_trait = plains_commander
					}
					NOT = {
						scope:eqfm_sanusiyyah_leader_1 ?= this
					}
				}
			}
			random_scope_character = {
				limit = {
					has_role_of_type = general
					is_busy = no
					NOT = {
						has_role_of_type = ruler
					}
					OR = {
						has_trait = surveyor
						has_trait = plains_commander
					}
					NOT = {
						scope:eqfm_sanusiyyah_leader_1 ?= this
					}
				}
				save_scope_as = eqfm_sanusiyyah_leader_2
			}
		}
		else = {
			random_scope_character = {
				limit = {
					has_role_of_type = general
					is_busy = no
					NOR = {
						has_role_of_type = ruler
					}
					NOT = {
						scope:eqfm_sanusiyyah_leader_1 ?= this
					}
				}
				save_scope_as = eqfm_sanusiyyah_leader_2
			}
		}
		random_scope_character = {
			limit = {
				has_role_of_type = general
				is_busy = no
				NOT = {
					has_role_of_type = ruler
				}
				NOR = {
					scope:eqfm_sanusiyyah_leader_1 ?= this
					scope:eqfm_sanusiyyah_leader_2 ?= this
				}
			}
			save_scope_as = eqfm_sanusiyyah_leader_3
		}
	}
	
	option = {
		name = eqfm_sanusiyyah.400.a
		default_option = yes
		trigger = {
			exists = scope:eqfm_sanusiyyah_leader_1
		}
		scope:eqfm_sanusiyyah_leader_1 = {
			add_modifier = modifier_eqfm_leading_sanusiyyah_expedition
		}
		add_journal_entry = {
			type = je_eqfm_sanusiyyah_expedition
		}
		set_variable = {
			name = eqfm_sanusiyyah_expedition_transit_var
			months = 3
		}
	}
	
	option = {
		name = eqfm_sanusiyyah.400.b
		trigger = {
			exists = scope:eqfm_sanusiyyah_leader_2
		}
		scope:eqfm_sanusiyyah_leader_2 = {
			add_modifier = modifier_eqfm_leading_sanusiyyah_expedition
		}
		add_journal_entry = {
			type = je_eqfm_sanusiyyah_expedition
		}
		set_variable = {
			name = eqfm_sanusiyyah_expedition_transit_var
			months = 3
		}
	}
	
	option = {
		name = eqfm_sanusiyyah.400.c
		trigger = {
			exists = scope:eqfm_sanusiyyah_leader_3
		}
		scope:eqfm_sanusiyyah_leader_3 = {
			add_modifier = modifier_eqfm_leading_sanusiyyah_expedition
		}
		add_journal_entry = {
			type = je_eqfm_sanusiyyah_expedition
		}
		set_variable = {
			name = eqfm_sanusiyyah_expedition_transit_var
			months = 3
		}
	}
	
	option = {
		name = eqfm_sanusiyyah.400.e
		custom_tooltip = {
			text = gbbf_nothing_happens
			eqfm_clear_sanusiyyah_expedition_variables_effect = yes
		}
	}
}
