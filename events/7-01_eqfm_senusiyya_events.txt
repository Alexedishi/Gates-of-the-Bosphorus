eqfm_senusiyya.010 = {
	# Senusiyya Spreads to Country
	type = country_event
	placement = root
	icon = "gfx/interface/icons/generic_icons/gbbf_runestone_moon.dds"
	duration = 3
	dlc = dlc_grefm
	
	title = eqfm_senusiyya.010.t
	desc = eqfm_senusiyya.010.d
	flavor = eqfm_senusiyya.010.f
	
	event_image = {
		video = "middleeast_islam"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		has_global_variable = eqfm_intimidated_sensussi
		NOT = {
			has_variable = eqfm_senusiyya_influence_var
		}
		gbbf_is_muslim_trigger = yes
		any_country_in_senusiyya = {
			is_adjacent_to_country = root
			has_variable = eqfm_senusiyya_influence_var
			var:eqfm_senusiyya_influence_var >= 50
		}
		OR = {
			AND = {
				has_state_in_state_region = STATE_LIBYA
				s:STATE_LIBYA = {
					any_scope_state = {
						has_variable = eqfm_senusiyya_entrenchment_var
					}
					eqfm_state_region_bedouin_radical_fraction_value >= 0.25
				}
			}
			AND = {
				has_state_in_state_region = STATE_LIBYAN_DESERT
				s:STATE_LIBYAN_DESERT = {
					any_scope_state = {
						has_variable = eqfm_senusiyya_entrenchment_var
					}
					eqfm_state_region_bedouin_radical_fraction_value >= 0.25
				}
			}
			AND = {
				has_state_in_state_region = STATE_TRIPOLI
				s:STATE_TRIPOLI = {
					any_scope_state = {
						has_variable = eqfm_senusiyya_entrenchment_var
					}
					eqfm_state_region_bedouin_radical_fraction_value >= 0.25
				}
			}
			AND = {
				has_state_in_state_region = STATE_MATRUH
				s:STATE_MATRUH = {
					any_scope_state = {
						has_variable = eqfm_senusiyya_entrenchment_var
					}
					eqfm_state_region_bedouin_radical_fraction_value >= 0.25
				}
			}
			AND = {
				has_state_in_state_region = STATE_EGYPTIAN_DESERT
				s:STATE_EGYPTIAN_DESERT = {
					any_scope_state = {
						has_variable = eqfm_senusiyya_entrenchment_var
					}
					eqfm_state_region_bedouin_radical_fraction_value >= 0.25
				}
			}
		}
	}
	
	cancellation_trigger = {
		NOT = {
			has_global_variable = je_eqfm_senusiyya
		}
	}
	
	cooldown = {
		years = 1
	}
	
	immediate = {
		random_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_senusiyya
			}
			save_scope_as = eqfm_senusiyya_target_state
			state_region = {
				if = {
					limit = {
						NOT = {
							has_variable = eqfm_senusiyya_entrenchment_var
						}
					}
					set_variable = {
						name = eqfm_senusiyya_entrenchment_var
						value = 25
					}
				}
			}
		}
		save_scope_as = eqfm_senusiyya_target_country
		every_country = {
			limit = {
				OR = {
					has_interest_marker_in_region = region_north_africa
					has_interest_marker_in_region = region_nile_basin
				}
			}
			post_notification = notification_eqfm_senusiyya_spreads_to_country
		}
		eqfm_save_senussi_scope_effect = yes
	}
	
	option = {
		name = eqfm_senusiyya.010.a
		default_option = yes
		custom_tooltip = {
			text = eqfm_normal_starting_senusiyya_influence
			set_variable = {
				name = eqfm_senusiyya_influence_var
				value = 25
			}
		}
	}
	
	option = {
		name = eqfm_senusiyya.010.b
		custom_tooltip = {
			text = eqfm_reduced_starting_senusiyya_influence
			set_variable = {
				name = eqfm_senusiyya_influence_var
				value = 15
			}
		}
		add_modifier = {
			name = modifier_eqfm_senusiyya_unwelcome
			months = gbbf_normal_modifier_time
		}
		add_radicals = {
			culture = cu:bedouin
			value = medium_radicals
		}
	}
	
	option = {
		name = eqfm_senusiyya.010.c
		custom_tooltip = {
			text = eqfm_increased_starting_senusiyya_influence
			set_variable = {
				name = eqfm_senusiyya_influence_var
				value = 35
			}
		}
		scope:eqfm_senusiyya_target_state = {
			eqfm_add_senusiyya_entrenchment_effect = {
				value = 5
			}
		}
		add_loyalists = {
			culture = cu:bedouin
			value = small_radicals
		}
		ig:ig_devout = {
			add_modifier = {
				name = modifier_eqfm_senusiyya_welcomed
				months = gbbf_normal_modifier_time
			}
		}
	}
}

eqfm_senusiyya.011 = {
	# Senusiyya Spreads to State Region (Headless)
	type = country_event
	hidden = yes
	
	trigger = {
		has_global_variable = eqfm_intimidated_sensussi
		NOT = {
			has_global_variable = eqfm_senusiyya_contagion_cooldown
		}
		s:STATE_LIBYA = {
			has_variable = eqfm_senusiyya_entrenchment_var
			var:eqfm_senusiyya_entrenchment_var >= 50
		}
		OR = {
			s:STATE_TRIPOLI = {
				NOT = {
					has_variable = eqfm_senusiyya_entrenchment_var
				}
			}
			s:STATE_LIBYAN_DESERT = {
				NOT = {
					has_variable = eqfm_senusiyya_entrenchment_var
				}
			}
			AND = {
				s:STATE_TRIPOLI = {
					has_variable = eqfm_senusiyya_entrenchment_var
					var:eqfm_senusiyya_entrenchment_var >= 50
				}
				s:STATE_LIBYAN_DESERT = {
					has_variable = eqfm_senusiyya_entrenchment_var
					var:eqfm_senusiyya_entrenchment_var >= 50
				}
				OR = {
					s:STATE_EGYPTIAN_DESERT = {
						NOT = {
							has_variable = eqfm_senusiyya_entrenchment_var
						}
					}
					s:STATE_MATRUH = {
						NOT = {
							has_variable = eqfm_senusiyya_entrenchment_var
						}
					}
				}
			}
			AND = { # From Tripolitania into Chad
				s:STATE_LIBYA = {
					has_variable = eqfm_senusiyya_entrenchment_var
					var:eqfm_senusiyya_entrenchment_var >= 50
				}
				s:STATE_CHAD = {
					NOT = {
						has_variable = eqfm_senusiyya_entrenchment_var
					}
				}
			}
		}
	}

	cancellation_trigger = {
		has_global_variable = eqfm_seusiyya_spread_null
	}
	
	immediate = {
		set_global_variable = {
			name = eqfm_senusiyya_contagion_cooldown
			years = 2
		}
		if = { # Attempt Selection within Libya
			limit = {
				s:STATE_LIBYA = {
					has_variable = eqfm_senusiyya_entrenchment_var
					var:eqfm_senusiyya_entrenchment_var >= 50
				}
				OR = {
					s:STATE_LIBYAN_DESERT = {
						NOT = {
							has_variable = eqfm_senusiyya_entrenchment_var
						}
					}
					s:STATE_TRIPOLI = {
						NOT = {
							has_variable = eqfm_senusiyya_entrenchment_var
						}
					}
				}
			}
			if = {
				limit = {
					s:STATE_LIBYAN_DESERT = {
						NOT = {
							has_variable = eqfm_senusiyya_entrenchment_var
						}
					}
				}
				s:STATE_LIBYAN_DESERT = {
					save_scope_as = eqfm_senusiyya_target_state_region
				}
			}
			else_if = {
				limit = {
					s:STATE_TRIPOLI = {
						NOT = {
							has_variable = eqfm_senusiyya_entrenchment_var
						}
					}
				}
				s:STATE_TRIPOLI = {
					save_scope_as = eqfm_senusiyya_target_state_region
				}
			}
			else = {
				error_log = "Event eqfm_senusiyya.011 was triggered and attempted to select a state region within Tripolitania, but no valid state_region was available."
				set_global_variable = {
					name = eqfm_seusiyya_spread_null
					months = 1
				}
			}
		}
		else_if = { # Attempt selection within Egypt
			limit = {
				s:STATE_LIBYA = {
					has_variable = eqfm_senusiyya_entrenchment_var
					var:eqfm_senusiyya_entrenchment_var >= 50
				}
				s:STATE_LIBYAN_DESERT = {
					has_variable = eqfm_senusiyya_entrenchment_var
					var:eqfm_senusiyya_entrenchment_var >= 50
				}
				OR = {
					s:STATE_EGYPTIAN_DESERT = {
						NOT = {
							has_variable = eqfm_senusiyya_entrenchment_var
						}
					}
					s:STATE_MATRUH = {
						NOT = {
							has_variable = eqfm_senusiyya_entrenchment_var
						}
					}
				}
			}
			if = {
				limit = {
					s:STATE_EGYPTIAN_DESERT = {
						NOT = {
							has_variable = eqfm_senusiyya_entrenchment_var
						}
					}
				}
				s:STATE_EGYPTIAN_DESERT = {
					save_scope_as = eqfm_senusiyya_target_state_region
				}
			}
			else_if = {
				limit = {
					s:STATE_MATRUH = {
						NOT = {
							has_variable = eqfm_senusiyya_entrenchment_var
						}
					}
				}
				s:STATE_MATRUH = {
					save_scope_as = eqfm_senusiyya_target_state_region
				}
			}
			else = {
				error_log = "Event eqfm_senusiyya.011 was triggered and attempted to select a state region within Egypt, but no valid state_region was available."
				set_global_variable = {
					name = eqfm_seusiyya_spread_null
					months = 1
				}
			}
		}
		else_if = { # Attempt selection within Chad
			limit = {
				s:STATE_LIBYA = {
					has_variable = eqfm_senusiyya_entrenchment_var
					var:eqfm_senusiyya_entrenchment_var >= 50
				}
				s:STATE_CHAD = {
					NOT = {
						has_variable = eqfm_senusiyya_entrenchment_var
					}
				}
			}
			if = {
				limit = {
					s:STATE_CHAD = {
						NOT = {
							has_variable = eqfm_senusiyya_entrenchment_var
						}
					}
				}
				s:STATE_CHAD = {
					save_scope_as = eqfm_senusiyya_target_state_region
				}
			}
			else = {
				error_log = "Event eqfm_senusiyya.011 was triggered and attempted to select a state region within Chad, but no valid state_region was available."
				set_global_variable = {
					name = eqfm_seusiyya_spread_null
					months = 1
				}
			}
		}
		else = {
			error_log = "Event eqfm_senusiyya.011 was triggered, but no valid state_region was available. Event has been cancelled."
			set_global_variable = {
				name = eqfm_seusiyya_spread_null
				months = 1
			}
		}
		if = {
			limit = {
				exists = scope:eqfm_senusiyya_target_state_region
			}
			scope:eqfm_senusiyya_target_state_region = {
				set_variable = {
					name = eqfm_senusiyya_contagion_victim
					months = 3
				}
				set_variable = {
					name = eqfm_senusiyya_entrenchment_var
					value = 25
				}
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_senusiyya
				}
				if = {
					limit = {
						gbbf_is_muslim_trigger = yes
						any_scope_state = {
							is_in_geographic_region = geographic_region_senusiyya
						}
					}
					trigger_event = {
						id = eqfm_senusiyya.012 # Senusiyya Spreads to State Region
					}
				}
				else = {
					post_notification = notification_eqfm_senusiyya_spreads_to_state_region
				}
			}
		}
		else = {
			error_log = "Event eqfm_senusiyya.011 made it past target selection, but the saved scope was not found."
		}
	}
}
