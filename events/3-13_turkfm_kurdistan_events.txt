turkfm_kurdistan.005 = { # The Eyâlet-i Kurdistan (KUR)
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_kurdish.dds"

	title = turkfm_kurdistan.005.t
	desc = turkfm_kurdistan.005.d
	flavor = turkfm_kurdistan.005.f

	event_image = {
		video = "ep1_transfer_of_authority"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		# triggered by turkfm_kurdistan.004
	}

	immediate = {
		c:TUR = {
			save_scope_as = ottoman_scope
		}
	}

	option = {
		name = turkfm_kurdistan.005.a
		default_option = yes
		add_modifier = {
			name = modifier_turkfm_autonomy_granted
			months = 120
		}
		if = {
			limit = {
				OR = {
					army_size < 20
					has_law_or_variant = law_type:law_hereditary_bureaucrats
					has_law_or_variant = law_type:law_peasant_levies
					has_law_or_variant = law_type:law_no_police
					NOT = {
						any_scope_building = {
							is_building_type = building_university
						}
					}
					NOT = {
						any_scope_building = {
							is_building_type = building_government_administration
						}
					}
					bureaucracy < 0
					ig:ig_intelligentsia = {
						is_marginal = yes
					}

				}
			}
			add_journal_entry = {
				type = je_turkfm_kur_tribes
			}
			add_modifier = modifier_turkfm_tribal_tradition
		}
		else = {
			add_modifier = {
				name = modifier_turkfm_centralized_tribe
				days = long_modifier_time
			}
			add_loyalists = {
				value = medium_radicals
			}
		}
		turkfm_add_kurdistan_claims_effect = yes
	}
}
