## The Senusiyya Movement Mechanical Events ##
turkfm_libya.110 = { # Senusiyya Spreads to Country
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	title = turkfm_libya.110.t
	desc = turkfm_libya.110.d
	flavor = turkfm_libya.110.f

	event_image = {
		video = "middleeast_islam"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		var:turkfm_senussi_var >= 100
		any_country ={
			is_adjacent_to_country = root
			country_has_state_religion = rel:sunni
			NOR = {
				has_journal_entry = je_turkfm_senusiyya_sub
				has_variable = turkfm_senussi_spanked
			}
			any_scope_state = {
				gbbf_senussi_geography_trigger = yes
			}
		}
	}

	cancellation_trigger = {
		has_variable = turkfm_senussi_spanked
	}

	cooldown = {
		years = 1
	}

	immediate = {
		turkfm_update_senussi_effect = yes
		random_country = {
			limit = {
				is_adjacent_to_country = root
				country_has_state_religion = rel:sunni
				NOR = {
					has_journal_entry = je_turkfm_senusiyya_sub
					has_variable = turkfm_senussi_spanked
				}
				any_scope_state = {
					gbbf_senussi_geography_trigger = yes
				}
			}
			save_scope_as = turkfm_senussi_target
		}
		scope:turkfm_senussi_target = {
			turkfm_initialize_senussi_effect = yes
			turkfm_senussi_arrival_effect = yes
		}
	}

	option = {
		name = turkfm_libya.110.a
		default_option = yes
		scope:turkfm_senussi_target = {
			if = {
				limit = {
					has_variable = turkfm_senussi_has_landed
				}
				show_as_tooltip = {
					trigger_event = {
						id = turkfm_libya.111 # Senusiyya Arrives in Country
					}
				}
				hidden_effect = {
					add_journal_entry = {
						type = je_turkfm_senusiyya_sub
					}
				}
				post_notification = notification_turkfm_senussi_arrival
			}
			else = {
				trigger_event = {
					id = turkfm_libya.111 # Senusiyya Arrives in Country
				}
			}
		}
	}

}

turkfm_libya.112 = { # Senusiyya Takes Root in State
	type = country_event
	placement = scope:turkfm_senussi_state_scope
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	title = turkfm_libya.112.t
	desc = turkfm_libya.112.d
	flavor = turkfm_libya.112.f

	gui_window = event_window_1char_adventure

	left_icon = scope:turkfm_senussi_scope

	event_image = {
		video = "middleeast_islam"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		var:turkfm_senussi_var >= 70
		NOT = {
			has_modifier = modifier_turkfm_desert_raids
		}
		any_scope_state = {
			gbbf_senussi_geography_trigger = yes
			NOT = {
				has_modifier = modifier_turkfm_senussi_stronghold
			}
		}
	}

	cooldown = {
		years = 4
	}

	immediate = {
		turkfm_update_senussi_effect = yes
		random_scope_state = {
			limit = {
				gbbf_senussi_geography_trigger = yes
				NOT = {
					has_modifier = modifier_turkfm_senussi_stronghold
				}
			}
			save_scope_as = turkfm_senussi_state_scope
		}
	}

	option = { # Allow them to Spread
		name = turkfm_libya.112.a
		default_option = yes
		scope:turkfm_senussi_state_scope = {
			add_modifier = modifier_turkfm_senussi_stronghold
		}
		custom_tooltip = {
			text = turkfm_senussi_lose_20
			change_variable = {
				name = turkfm_senussi_var
				subtract = 20
			}
		}
	}

	option = { # Attempt to Suppress Them
		name = turkfm_libya.112.b
		default_option = yes
		trigger = {
			has_technology_researched = centralization
			has_global_variable = turkfm_intimidated_sensussi
			authority >= 200
		}
		add_modifier = modifier_turkfm_desert_raids
		custom_tooltip = {
			text = turkfm_supress_senusiyya_effect
			set_variable = {
				name = turkfm_desert_raids_var
				years = 2
			}
		}
	}

}

	type = country_event

		video = "africa_desert_expedition"
	}

	trigger = {

		}
		custom_tooltip = {
			text = turkfm_senussi_gain_20
			change_variable = {
				name = turkfm_senussi_var
				add = 20
			}
		}
	}
