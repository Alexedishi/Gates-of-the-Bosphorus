eqfm_sanusiyyah.400 = {
	type = country_event
	placement = root
	icon = "gfx/interface/icons/invention_icons/colonization.dds"
	duration = 2
	dlc = dlc_grefm
	
	title = eqfm_sanusiyyah.400.t
	desc = eqfm_sanusiyyah.400.d
	flavor = eqfm_sanusiyyah.400.f
	
	event_image = {
		video = "africa_construction_colony"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		# triggered by Sanusiyyah Expedition buttons
	}
	
	cancellation_trigger = {
		OR = {
			NOT = {
				any_scope_general = {
					is_busy = no
					NOT = {
						has_role_of_type = ruler
					}
				}
			}
			eqfm_has_sanusiyyah_expedition_trigger = no
		}
	}
	
	immediate = {
		set_variable = {
			name = eqfm_sanusiyyah_expedition_transit_var
			months = 3
		}
		random_scope_character = {
			limit = {
				has_role_of_type = general
				is_busy = no
				NOT = {
					has_role_of_type = ruler
				}
			}
			save_scope_as = eqfm_sanusiyyah_leader_1
		}
		if = {
			limit = {
				any_scope_character = {
					has_role_of_type = general
					is_busy = no
					NOT = {
						has_role_of_type = ruler
					}
					OR = {
						has_trait = surveyor
						has_trait = plains_commander
					}
					NOT = {
						scope:eqfm_sanusiyyah_leader_1 ?= this
					}
				}
			}
			random_scope_character = {
				limit = {
					has_role_of_type = general
					is_busy = no
					NOT = {
						has_role_of_type = ruler
					}
					OR = {
						has_trait = surveyor
						has_trait = plains_commander
					}
					NOT = {
						scope:eqfm_sanusiyyah_leader_1 ?= this
					}
				}
				save_scope_as = eqfm_sanusiyyah_leader_2
			}
		}
		else = {
			random_scope_character = {
				limit = {
					has_role_of_type = general
					is_busy = no
					NOR = {
						has_role_of_type = ruler
					}
					NOT = {
						scope:eqfm_sanusiyyah_leader_1 ?= this
					}
				}
				save_scope_as = eqfm_sanusiyyah_leader_2
			}
		}
		random_scope_character = {
			limit = {
				has_role_of_type = general
				is_busy = no
				NOT = {
					has_role_of_type = ruler
				}
				NOR = {
					scope:eqfm_sanusiyyah_leader_1 ?= this
					scope:eqfm_sanusiyyah_leader_2 ?= this
				}
			}
			save_scope_as = eqfm_sanusiyyah_leader_3
		}
	}
	
	option = {
		name = eqfm_sanusiyyah.400.a
		default_option = yes
		trigger = {
			exists = scope:eqfm_sanusiyyah_leader_1
		}
		scope:eqfm_sanusiyyah_leader_1 = {
			add_modifier = modifier_eqfm_leading_sanusiyyah_expedition
		}
		add_journal_entry = {
			type = je_eqfm_sanusiyyah_expedition
		}
		set_variable = {
			name = eqfm_sanusiyyah_expedition_transit_var
			months = 3
		}
	}
	
	option = {
		name = eqfm_sanusiyyah.400.b
		trigger = {
			exists = scope:eqfm_sanusiyyah_leader_2
		}
		scope:eqfm_sanusiyyah_leader_2 = {
			add_modifier = modifier_eqfm_leading_sanusiyyah_expedition
		}
		add_journal_entry = {
			type = je_eqfm_sanusiyyah_expedition
		}
		set_variable = {
			name = eqfm_sanusiyyah_expedition_transit_var
			months = 3
		}
	}
	
	option = {
		name = eqfm_sanusiyyah.400.c
		trigger = {
			exists = scope:eqfm_sanusiyyah_leader_3
		}
		scope:eqfm_sanusiyyah_leader_3 = {
			add_modifier = modifier_eqfm_leading_sanusiyyah_expedition
		}
		add_journal_entry = {
			type = je_eqfm_sanusiyyah_expedition
		}
		set_variable = {
			name = eqfm_sanusiyyah_expedition_transit_var
			months = 3
		}
	}
	
	option = {
		name = eqfm_sanusiyyah.400.e
		custom_tooltip = {
			text = gbbf_nothing_happens
			eqfm_clear_sanusiyyah_expedition_variables_effect = yes
		}
	}
}
