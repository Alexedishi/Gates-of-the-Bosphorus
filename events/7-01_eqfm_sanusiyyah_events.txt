eqfm_sanusiyyah.001 = {
	type = country_event
	placement = scope:eqfm_sanusiyyah_state_scope

	duration = 3
	dlc = dlc_grefm

	title = eqfm_sanusiyyah.001.t
	desc = eqfm_sanusiyyah.001.d
	flavor = eqfm_sanusiyyah.001.f

	gui_window = event_window_1char_adventure
	left_icon = scope:eqfm_senussi_character_scope

	event_image = {
		video = "middleeast_islam"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	trigger = {
		# triggered by je_eqfm_sanusiyyah
	}

	immediate = {
		eqfm_save_senussi_scope_effect = yes
		set_variable = eqfm_sanusiyyah_started
		if = {
			limit = {
				any_state_in_sanusiyyah = {
					has_variable = eqfm_sanusiyyah_entrenchment_var
				}
			}
			random_state_in_sanusiyyah = {
				limit = {
					has_variable = eqfm_sanusiyyah_entrenchment_var
				}
				save_scope_as = eqfm_sanusiyyah_state_scope
			}
		}
		else = {
			s:STATE_LIBYA = {
				random_scope_state = {
					save_scope_as = eqfm_sanusiyyah_state_scope
				}
			}
		}
	}

	option = {
		name = eqfm_sanusiyyah.001.a
		default_option = yes
		custom_tooltip = eqfm_sanusiyyah.001.a.tt
	}
}

eqfm_sanusiyyah.200 = {
	# Sanusiyyah Spreads to State Region (Headless)
	type = country_event
	hidden = yes

	trigger = {
		has_global_variable = eqfm_intimidated_sensussi
		NOT = {
			has_global_variable = eqfm_sanusiyyah_contagion_cooldown
		}
		OR = {
			global_var:je_eqfm_sanusiyyah.var:eqfm_sanusiyyah_phase_var ?= flag:eqfm_sanusiyyah_unease

		}
		OR = {
			# Evaluate STATE_LIBYA
			AND = {
				s:STATE_LIBYA = {
					has_variable = eqfm_sanusiyyah_entrenchment_var

				}

					s:STATE_TRIPOLI = {
