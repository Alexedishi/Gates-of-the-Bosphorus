eqfm_sanusiyyah.100 = {
	#
	type = country_event
	placement = scope:eqfm_senussi_marabout_state
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3
	dlc = dlc_grefm
	
	title = eqfm_sanusiyyah.100.t
	desc = eqfm_sanusiyyah.100.d
	flavor = eqfm_sanusiyyah.100.f
	
	event_image = {
		video = "africa_city_center"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		NOT = {
			has_variable = eqfm_desert_event_cooldown_var
		}
		has_variable = eqfm_sanusiyyah_influence_var
		var:eqfm_sanusiyyah_influence_var < 75
		s:STATE_LIBYAN_DESERT = {
			any_scope_state = {
				owner ?= {
					is_country_type = decentralized
				}
			}
		}
	}
	
	cancellation_trigger = {
		NOR = {
			is_involved_in_journal_entry = je_eqfm_sanusiyyah
			has_variable = eqfm_sanusiyyah_influence_var
		}
		var:eqfm_sanusiyyah_influence_var >= 90
	}
	
	cooldown = {
		years = 8
	}
	
	immediate = {
		random_scope_state = {
			limit = {
				NOT = {
					has_modifier = modifier_eqfm_dissident_teachings
				}
			}
			save_scope_as = eqfm_senussi_marabout_state
		}
		set_variable = {
			name = eqfm_desert_event_cooldown_var
			months = 6
		}
		if = {
			limit = {
				exists = c:TUR
				is_subject_of = c:TUR
			}
			c:TUR = {
				save_scope_as = eqfm_ottoman_scope
			}
		}
	}
	
	option = {
		name = eqfm_sanusiyyah.100.a
		default_option = yes
		eqfm_add_sanusiyyah_influence_effect = {
			value = 10
		}
		if = {
			limit = {
				exists = scope:eqfm_ottoman_scope
			}
			change_relations = {
				country = c:TUR
				value = -15
			}
			c:TUR = {
				create_diplomatic_catalyst = {
					type = catalyst_event_negative
					target = root
				}
			}
		}
	}
	
	option = {
		name = eqfm_sanusiyyah.100.b
		default_option = yes
		trigger = {
			OR = {
				has_law_or_variant = law_type:law_censorship
				has_law_or_variant = law_type:law_outlawed_dissent
			}
		}
		show_as_unavailable = { always = yes }
		highlighted_option = yes
		eqfm_add_sanusiyyah_influence_effect = {
			value = -20
		}
		scope:eqfm_senussi_marabout_state = {
			add_radicals_in_state = {
				pop_type = clergymen
				value = medium_radicals
			}
		}
		if = {
			limit = {
				exists = scope:eqfm_ottoman_scope
			}
			change_relations = {
				country = c:TUR
				value = 15
			}
			c:TUR = {
				create_diplomatic_catalyst = {
					type = catalyst_event_positive
					target = root
				}
			}
		}
	}
	
}
