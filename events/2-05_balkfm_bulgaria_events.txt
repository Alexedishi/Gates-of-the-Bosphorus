		name = balkfm_bulgaria.001.a

			type = je_balkfm_bul_battenburg_monarchy

			name = modifier_balkfm_national_awakening

		}
		hidden_effect = {
			trigger_event = {
				id = balkfm_bulgaria.012
				days = 180
			}
		}
		hidden_effect = {
			if = {
				limit = {
					NOT = {
						has_variable = eocfm_is_autocephalous
					}
				}
				trigger_event = {
					id = balkfm_bulgaria.160
					days = 240
				}
			}
		}
	}

}

balkfm_bulgaria.002 = { # Democratic Obstructions
	type = country_event
	placement = root

	title = balkfm_bulgaria.002.t
	desc = balkfm_bulgaria.002.d
	flavor = balkfm_bulgaria.002.f

	event_image = {
		video = "unspecific_politicians_arguing"
	}

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		NOT = {
			has_variable = balkfm_bulgaria_consitution_1
		}
		var:balkfm_coup_var <= 104 # Second Session of the National Assembly, relative to Alexander I's ascent to the throne
		NOT = {
			has_law_or_variant = law_type:law_autocracy
		}
	}

	immediate = {
		set_variable = balkfm_bulgaria_consitution_1
	}

	cancellation_trigger = {
		has_law_or_variant = law_type:law_oligarchy
	}

	option = { # yeet the law
		name = balkfm_bulgaria.002.a
		default_option = yes
		if = {
			limit = {
				NOR = {
					has_law_or_variant = law_type:law_autocracy
					has_law_or_variant = law_type:law_oligarchy
				}
			}
			activate_law = law_type:law_oligarchy
		}
		else_if = {
			limit = {
				has_law_or_variant = law_type:law_oligarchy
			}
			activate_law = law_type:law_autocracy
		}
		add_radicals = {
			pop_type = academics
			value = very_large_radicals
		}
		add_radicals = {
			pop_type = officers
			value = very_large_radicals
		}
		add_loyalists = {
			strata = lower
			value = medium_radicals
		}
		change_relations = {
			country = c:RUS
			value = 20
		}
		custom_tooltip = balkfm_countercoup_reduce_2
	}

	option = { # protect the constitution
		name = balkfm_bulgaria.002.b
		add_modifier = {
			name = modifier_balkfm_honorable_regime
			months = gbbf_long_modifier_time
		}
		add_radicals = {
			strata = upper
			value = large_radicals
		}
		change_relations = {
			country = c:RUS
			value = -20
		}
	}

}
