			has_global_variable = je_eqfm_balkan_powder_keg
		}

			name = eqfm_riot_cooldown_var
			years = 1
		}
		eqfm_balkan_powder_keg_event_cooldown_effect = yes
		random_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_rumelia
				NOT = {
					has_modifier = modifier_eqfm_race_riots
				}
			}
			save_scope_as = eqfm_riot_state_scope
		}
		scope:eqfm_riot_state_scope = {
			add_modifier = {
				name = modifier_eqfm_race_riots
				months = 24
			}
		}
		if = {
			limit = {
				"global_variable_map(eqfm_balkan_powder_keg_varmap|flag:eqfm_balkan_powder_keg_riot_counter_var)" = 1
			}
			eqfm_post_balkan_powder_keg_notification_effect = {
				notification = eqfm_race_riots
			}
		}
		else = {
			eqfm_post_balkan_powder_keg_notification_effect = {
				notification = eqfm_race_riots_worsen
			}
		}
		eqfm_calculate_balkan_powder_keg_variables_effect = yes
	}
	
	option = {
		name = eqfm_balkans.006.a
		default_option = yes
		custom_tooltip = eqfm_balkans.006.tt
