eqfm_balkans.002 = { # The Rise of Nationalism
	type = country_event
	placement = root

	title = eqfm_balkans.002.t
	desc = eqfm_balkans.002.d
	flavor = eqfm_balkans.002.f

	event_image = {
		video = "middleeast_middleclass_cafe"
	}

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		NOT = {
			has_variable = turkfm_nationalism_wave_happened
		}
		OR = {
			has_technology_researched = nationalism
			any_country = {
				is_subject_of = root
				is_in_geographic_region = geographic_region_rumelia
				has_technology_researched = nationalism
			}
		}
	}

	immediate = {
		set_variable = turkfm_nationalism_wave_happened

			limit = {

		}
		else = {
			random_country = {
				limit =	{
					is_subject_of = root
					is_in_geographic_region = geographic_region_rumelia
					has_technology_researched = nationalism
				}
				save_scope_as = eqfm_nationalism_scope
			}
		}
	}

	option = {
		name = eqfm_balkans.002.a
		default_option = yes
		custom_tooltip = eqfm_nationalism_warning
	}

}

eqfm_balkans.005 = { # Riots Break Out in State
	type = country_event
	placement = scope:eqfm_riot_state_scope

	title = eqfm_balkans.005.t
	desc = eqfm_balkans.005.d
	flavor = eqfm_balkans.005.f

	event_image = {
		video = "votp_agitator_speaking"
	}

	icon = "gfx/interface/icons/event_icons/bicorne_hat.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		eqfm_hostility_events_trigger = yes
		global_var:eqfm_riot_counter_var < eqfm_riot_potential_value
		NOT = {
			has_variable = eqfm_riot_cooldown_var
		}
		any_scope_state = {
			is_in_geographic_region = geographic_region_rumelia
			NOT = {
				has_modifier = modifier_eqfm_race_riots
			}
		}
	}

	immediate = {
		set_variable = {
			name = eqfm_riot_cooldown_var
			years = 1
		}
		eqfm_tension_cooldown_effect = yes
		random_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_rumelia
				NOT = {
					has_modifier = modifier_eqfm_race_riots
				}
			}
			save_scope_as = eqfm_riot_state_scope
		}
		scope:eqfm_riot_state_scope = {
			add_modifier = {
				name = modifier_eqfm_race_riots
				months = 24
			}
		}
		set_global_variable = {
			name = eqfm_riot_counter_var
			value = eqfm_riot_counter_value
		}
		if = {
			limit = {
				global_var:eqfm_riot_counter_var = 1
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_balkan_situation
				}
				post_notification = eqfm_race_riots
			}
		}
		else = {
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_balkan_situation
				}
				post_notification = eqfm_race_riots_worsen
			}
		}
	}

	option = {
		name = eqfm_balkans.005.a
		default_option = yes
		custom_tooltip = eqfm_balkans.005.tt
		show_as_tooltip = {
			scope:eqfm_riot_state_scope = {
				add_modifier = {
					name = modifier_eqfm_race_riots
					months = 24
				}
			}
		}
	}

}

eqfm_balkans.006 = { # Balkan Country Rebellion
	type = country_event
	placement = scope:eqfm_rebellion_scope

	title = eqfm_balkans.006.t
	desc = eqfm_balkans.006.d
	flavor = eqfm_balkans.006.f

	event_image = {
		video = "votp_agitator_speaking"
	}

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 1

	dlc = dlc_grefm

	trigger = {
		# triggered by eqfm_secession_start on_action
	}

	immediate = {
		random_country = {
			limit = {
				has_variable = eqfm_rebellion_tag_tracker
			}
			save_scope_as = eqfm_rebellion_scope
			remove_variable = eqfm_rebellion_tag_tracker
		}
		every_country = {
			limit = {
				has_diplomatic_relevance = root
				NOT = {
					this = root
				}
			}
			create_bidirectional_truce = {
				country = scope:eqfm_rebellion_scope
				months = 36
			}
		}
		random_diplomatic_play = {
			limit = {
				initiator = root
				target = scope:eqfm_rebellion_scope
			}
			save_scope_as = eqfm_rebellion_play_scope
		}
	}

	option = { # Assert Our Dominion
		name = eqfm_balkans.006.a
		default_option = yes
		custom_tooltip = turkfm_play_continues
		every_country = {
			limit = {
				is_involved_in_journal_entry = je_eqfm_balkan_situation
			}
			post_notification = eqfm_revolt_seccession
		}
	}

	option = { # Let Them Go
		name = eqfm_balkans.006.b
		scope:eqfm_rebellion_play_scope = {
			end_play = yes
		}
		create_diplomatic_pact = {
			country = scope:eqfm_rebellion_scope
			type = protectorate
		}
		create_bidirectional_truce = {
			country = scope:eqfm_rebellion_scope
			months = 60
		}
		add_radicals = {
			culture = cu:turkish
			value = medium_radicals
		}
		add_modifier = {
			name = modifier_eqfm_caved_to_separatists
			months = gbbf_normal_modifier_time
		}
		hidden_effect = {
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_balkan_situation
				}
				post_notification = eqfm_revolt_protectorate
			}
		}
	}

}
