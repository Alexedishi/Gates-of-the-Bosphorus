eqfm_pirates.101 = { # Pacifying the Adriatic
	type = country_event
	placement = root

	title = eqfm_pirates.101.t
	desc = eqfm_pirates.101.d
	flavor = eqfm_pirates.101.f

	event_image = {
		video = "africa_construction_colony"
	}

	icon = "gfx/interface/icons/event_icons/event_piracy.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		# triggered by turkfm_deploy_expedition_button
	}

	cancellation_trigger = {
		has_global_variable = eqfm_jack_sparrow_whacked
	}

	immediate = {
		random_scope_character = {
			limit = {
				is_busy = no
				has_role = admiral
				is_ruler = no
			}
			save_scope_as = expedition_leader_1
		}
		if = {
			limit = {
				any_scope_character = {
					is_busy = no
					is_ruler = no
					has_role = admiral
					OR = {
						has_trait = basic_naval_commander
						has_trait = experienced_naval_commander
						has_trait = expert_naval_commander
						has_trait = convoy_raider_commander
						has_trait = experienced_convoy_raider
						has_trait = expert_convoy_raider
					}
					NOT = {
						scope:expedition_leader_1 ?= this
					}
				}
			}
			random_scope_character = {
				limit = {
					is_busy = no
					is_ruler = no
					has_role = admiral
					OR = {
						has_trait = basic_naval_commander
						has_trait = experienced_naval_commander
						has_trait = expert_naval_commander
						has_trait = convoy_raider_commander
						has_trait = experienced_convoy_raider
						has_trait = expert_convoy_raider
					}
					NOT = {
						scope:expedition_leader_1 ?= this
					}
				}
				save_scope_as = expedition_leader_2
			}
		}
		else = {
			random_scope_character = {
				limit = {
					is_busy = no
					is_ruler = no
					has_role = admiral
					NOT = {
						scope:expedition_leader_1 ?= this
					}
				}
				save_scope_as = expedition_leader_2
			}
		}
		random_scope_character = {
			limit = {
				is_busy = no
				is_ruler = no
				has_role = admiral
				NOR = {
					scope:expedition_leader_1 ?= this
					scope:expedition_leader_2 ?= this
				}
			}
			save_scope_as = expedition_leader_3
		}
		random_scope_character = {
			limit = {
				is_busy = no
				is_ruler = no
				has_role = admiral
				NOR = {
					scope:expedition_leader_1 ?= this
					scope:expedition_leader_2 ?= this
					scope:expedition_leader_3 ?= this
				}
			}
			save_scope_as = expedition_leader_4
		}
	}

	option = {
		name = eqfm_pirates.101.a
		default_option = yes
		trigger = {
			exists = scope:expedition_leader_1
		}
		scope:expedition_leader_1 ?= {
			set_character_busy_and_immortal = yes
			add_modifier = modifier_eqfm_leading_adriatic_expedition
		}
		eqfm_jack_sparrow_effect = yes
	}

	option = {
		name = eqfm_pirates.101.b
		trigger = {
			exists = scope:expedition_leader_2
		}
		scope:expedition_leader_2 ?= {
			set_character_busy_and_immortal = yes
			add_modifier = modifier_eqfm_leading_adriatic_expedition
		}
		eqfm_jack_sparrow_effect = yes
	}

	option = {
		name = eqfm_pirates.101.c
		trigger = {
			exists = scope:expedition_leader_3
		}
		scope:expedition_leader_3 ?= {
			set_character_busy_and_immortal = yes
			add_modifier = modifier_eqfm_leading_adriatic_expedition
		}
		eqfm_jack_sparrow_effect = yes
	}

	option = {
		name = eqfm_pirates.101.e
		trigger = {
			exists = scope:expedition_leader_4
		}
		scope:expedition_leader_4 ?= {
			set_character_busy_and_immortal = yes
			add_modifier = modifier_eqfm_leading_adriatic_expedition
		}
		eqfm_jack_sparrow_effect = yes
	}

	after = {
		scope:eqfm_cutler_beckett_scope = {
			set_variable = {
				name = com_opinion
				value = flag:eqfm_cutler_beckett_opinion
			}
		}
		trigger_event = {
			id = eqfm_pirates.102 # Cutler Beckett Arrives in the Adriatic
			days = { 15 45 }
		}
		set_variable = eqfm_expedition_piracy_transit_var
	}

}
