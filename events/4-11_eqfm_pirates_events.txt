eqfm_pirates.002 = { # Pacifying the Adriatic
	type = country_event
	placement = root

	title = eqfm_pirates.002.t
	desc = eqfm_pirates.002.d
	flavor = eqfm_pirates.002.f

	event_image = {
		video = "africa_construction_colony"
	}

	icon = "gfx/interface/icons/event_icons/event_piracy.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		# triggered by turkfm_deploy_expedition_button
	}

	cancellation_trigger = {
		has_global_variable = eqfm_jack_sparrow_whacked
	}

	immediate = {
		random_scope_character = {
			limit = {
				is_busy = no
				has_role = admiral
				is_ruler = no
			}
			save_scope_as = expedition_leader_1
		}
		if = {
			limit = {
				any_scope_character = {
					is_busy = no
					is_ruler = no
					has_role = admiral
					OR = {
						has_trait = basic_naval_commander
						has_trait = experienced_naval_commander
						has_trait = expert_naval_commander
						has_trait = convoy_raider_commander
						has_trait = experienced_convoy_raider
						has_trait = expert_convoy_raider
					}
					NOT = {
						scope:expedition_leader_1 ?= this
					}
				}
			}
			random_scope_character = {
				limit = {
					is_busy = no
					is_ruler = no
					has_role = admiral
					OR = {
						has_trait = basic_naval_commander
						has_trait = experienced_naval_commander
						has_trait = expert_naval_commander
						has_trait = convoy_raider_commander
						has_trait = experienced_convoy_raider
						has_trait = expert_convoy_raider
					}
					NOT = {
						scope:expedition_leader_1 ?= this
					}
				}
				save_scope_as = expedition_leader_2
			}
		}
		else = {
			random_scope_character = {
				limit = {
					is_busy = no
					is_ruler = no
					has_role = admiral
					NOT = {
						scope:expedition_leader_1 ?= this
					}
				}
				save_scope_as = expedition_leader_2
			}
		}
		random_scope_character = {
			limit = {
				is_busy = no
				is_ruler = no
				has_role = admiral
				NOR = {
					scope:expedition_leader_1 ?= this
					scope:expedition_leader_2 ?= this
				}
			}
			save_scope_as = expedition_leader_3
		}
		random_scope_character = {
			limit = {
				is_busy = no
				is_ruler = no
				has_role = admiral
				NOR = {
					scope:expedition_leader_1 ?= this
					scope:expedition_leader_2 ?= this
					scope:expedition_leader_3 ?= this
				}
			}
			save_scope_as = expedition_leader_4
		}
	}

	option = {
		name = eqfm_pirates.002.a
		default_option = yes
		trigger = {
			exists = scope:expedition_leader_1
		}
		scope:expedition_leader_1 ?= {
			set_character_busy_and_immortal = yes
			add_modifier = modifier_eqfm_leading_adriatic_expedition
		}
		eqfm_jack_sparrow_effect = yes
	}

	option = {
		name = eqfm_pirates.002.b
		trigger = {
			exists = scope:expedition_leader_2
		}
		scope:expedition_leader_2 ?= {
			set_character_busy_and_immortal = yes
			add_modifier = modifier_eqfm_leading_adriatic_expedition
		}
		eqfm_jack_sparrow_effect = yes
	}

	option = {
		name = eqfm_pirates.002.c
		trigger = {
			exists = scope:expedition_leader_3
		}
		scope:expedition_leader_3 ?= {
			set_character_busy_and_immortal = yes
			add_modifier = modifier_eqfm_leading_adriatic_expedition
		}
		eqfm_jack_sparrow_effect = yes
	}

	option = {
		name = eqfm_pirates.002.e
		trigger = {
			exists = scope:expedition_leader_4
		}
		scope:expedition_leader_4 ?= {
			set_character_busy_and_immortal = yes
			add_modifier = modifier_eqfm_leading_adriatic_expedition
		}
		eqfm_jack_sparrow_effect = yes
	}

	after = {
		trigger_event = {
			id = eqfm_pirates.003 # Cutler Beckett Arrives in the Adriatic
			days = { 15 45 }
		}
		set_variable = eqfm_expedition_piracy_transit_var
	}

}

eqfm_pirates.010 = { # Coastal Village Raided!
	type = country_event
	placement = scope:eqfm_piracy_state_scope

	title = eqfm_pirates.010.t
	desc = eqfm_pirates.010.d
	flavor = eqfm_pirates.010.f

	gui_window = event_window_1char_adventure

	left_icon = scope:eqfm_jack_sparrow_scope

	event_image = {
		video = "southamerica_war_civilians"
	}

	icon = "gfx/interface/icons/event_icons/event_piracy.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		NOT = {
			has_variable = eqfm_piracy_cooldown_var
		}
		OR = {
			eqfm_has_adriatic_expedition_trigger = yes
			any_scope_state = {
				gbbf_adriatic_geography_trigger = yes
				NOR = {
					has_modifier = modifier_eqfm_coasts_raided
					has_modifier = modifier_eqfm_pirate_nursing_home
				}
			}
		}
	}

	cancellation_trigger = {
		has_global_variable = eqfm_jack_sparrow_whacked
	}

	cooldown = {
		years = 3
	}

	immediate = {
		set_variable = {
			name = eqfm_piracy_cooldown_var
			months = 6
		}
		eqfm_piracy_scopes_effect = yes
		if = {
			limit = {
				eqfm_has_adriatic_expedition_trigger = yes
			}
			random_state = {
				limit = {
					gbbf_adriatic_geography_trigger = yes
					NOT = {
						has_modifier = modifier_eqfm_coasts_raided
					}
				}
				save_scope_as = eqfm_piracy_state_scope
			}
		}
		else = {
			random_scope_state = {
				limit = {
					gbbf_adriatic_geography_trigger = yes
					NOT = {
						has_modifier = modifier_eqfm_coasts_raided
					}
				}
				save_scope_as = eqfm_piracy_state_scope
			}
		}
	}

	option = { # Maybe they shouldn't live on the coast
		name = eqfm_pirates.010.a
		default_option = yes
		scope:eqfm_piracy_state_scope = {
			add_modifier = {
				name = modifier_eqfm_coasts_raided
				months = gbbf_duration_2_plus_half_years
			}
			add_radicals_in_state = {
				strata = lower
				value = small_radicals
			}
		}
	}

	option = { # Provide Relief
		name = eqfm_pirates.010.b
		scope:eqfm_piracy_state_scope = {
			add_modifier = {
				name = modifier_eqfm_coasts_raided
				months = gbbf_duration_2_plus_half_years
			}
			add_modifier = {
				name = modifier_eqfm_aid_distributed_state
				months = gbbf_duration_2_plus_half_years
			}
			add_loyalists_in_state = {
				strata = lower
				value = small_radicals
			}
