balkfm_croatia.200 = { # Illyrian Sentiment
	type = country_event
	placement = root

	title = balkfm_croatia.200.t
	desc = balkfm_croatia.200.d
	flavor = balkfm_croatia.200.f

	gui_window = event_window_1char_tabloid

	left_icon = scope:balkfm_draskovic_scope

	icon = "gfx/interface/icons/event_icons/event_croatia.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		# triggered from GLOBAL
	}

	cancellation_trigger = {
		NOT = {
			has_journal_entry = je_balkfm_cro_croation_question
		}
	}

	immediate = {
		random_character_in_void = {
			limit = {
				has_template = template_balkfm_draskovic_janko
			}
			save_scope_as = balkfm_draskovic_scope
		}
		ig:ig_landowners = {
			leader = {
				save_scope_as = balkfm_metternich_scope
			}
		}
		c:TUR = {
			save_scope_as = ottoman_scope
		}
	}

	option = {
		name = balkfm_croatia.200.a
		default_option = yes
		every_scope_state = {
			limit = {
				state_region = {
					is_homeland = cu:croat
				}
			}
			add_modifier = modifier_balkfm_croatian_revivalism
		}
		add_radicals = {
			culture = cu:croat
			value = medium_radicals
		}
	}
	
	orphan = yes

}

balkfm_croatia.201 = { # An Austro-Slavic Proposal
	type = country_event
	placement = root

	title = balkfm_croatia.201.t
	desc = balkfm_croatia.201.d
	flavor = balkfm_croatia.201.f

	gui_window = event_window_1char_tabloid

	left_icon = scope:balkfm_borovsky_scope

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		NOT = {
			has_variable = balkfm_austro_slavic_proposal_var
		}
		has_technology_researched = nationalism
	}

	cancellation_trigger = {
		NOT = {
			has_journal_entry = je_balkfm_cro_croation_question
		}
	}

	immediate = {
		set_variable = balkfm_austro_slavic_proposal_var
		create_character = {
			template = template_balkfm_borovsky_karel
			on_created = {
				place_character_in_void = 12
			}
		}
		random_character_in_void = {
			limit = {
				has_template = template_balkfm_borovsky_karel
			}
			save_scope_as = balkfm_borovsky_scope
		}
	}

	option = { # Support the Proposal
		name = balkfm_croatia.201.a
		default_option = yes
		add_radicals = {
			culture = cu:hungarian
			value = medium_radicals
		}
		balkfm_add_yugoslav_loyalists_or_radicals_effect = {
			type = loyalists
			value = medium_radicals
		}
	}

	option = { # Oppose the Proposal
		name = balkfm_croatia.201.b
		add_loyalists = {
			culture = cu:hungarian
			value = medium_radicals
		}
		balkfm_add_yugoslav_loyalists_or_radicals_effect = {
			type = radicals
			value = medium_radicals
		}
	}

}

	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

		# triggered by je_balkfm_cro_triune_restoration
	}

	immediate = {
		c:AUS = {
			save_scope_as = austria_scope
		}
		c:SER = {
			save_scope_as = serbia_scope
		}
	}

	option = {
		name = balkfm_croatia.300.a
		default_option = yes
		add_journal_entry = {
			type = je_balkfm_cro_triune_restoration
		}
		if = {
			limit = {
				NOR = {
					has_claim = s:STATE_SLAVONIA
					owns_entire_state_region = STATE_SLAVONIA
				}
			}
			s:STATE_SLAVONIA = {
				add_claim = ROOT
			}
		}
		if = {
			limit = {
				NOR = {
					has_claim = s:STATE_CROATIA
					owns_entire_state_region = STATE_CROATIA
				}
			}
			s:STATE_CROATIA = {
				add_claim = ROOT
			}
		}
		if = {
			limit = {
				NOR = {
					has_claim = s:STATE_DALMATIA
					owns_entire_state_region = STATE_DALMATIA
				}
			}
			s:STATE_DALMATIA = {
				add_claim = ROOT
			}
		}
	}

}
