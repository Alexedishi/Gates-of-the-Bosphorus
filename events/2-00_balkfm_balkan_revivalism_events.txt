balkfm_revivalism.002 = { # The Bulgarian Question
	type = country_event
	placement = root

	title = balkfm_revivalism.002.t
	desc = balkfm_revivalism.002.d
	flavor = balkfm_revivalism.002.f

	event_image = {
		video = "unspecific_politicians_arguing"
	}

	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		c:BUL ?= this
		NOT = {
			has_variable = balkfm_national_question_happened
		}
		has_technology_researched = egalitarianism
	}

	immediate = {
		set_variable = balkfm_national_question_happened
	}

	option = { # Focus on Nationalism
		name = balkfm_revivalism.002.a
		default_option = yes
		if = {
			limit = {
				NOR = {
					has_claim = s:STATE_NORTHERN_THRACE
					owns_entire_state_region = STATE_NORTHERN_THRACE
				}
			}
			s:STATE_NORTHERN_THRACE = {
				add_claim = ROOT
			}
		}
		if = {
			limit = {
				NOR = {
					has_claim = s:STATE_MACEDONIA
					owns_entire_state_region = STATE_MACEDONIA
				}
			}
			s:STATE_MACEDONIA = {
				add_claim = ROOT
			}
		}
		create_incident = {
			country = c:TUR
			value = 15
		}
		balkfm_add_nationalism_progress_effect = yes
	}

	option = { # Just Coexist
		name = balkfm_revivalism.002.b
		balkfm_add_nationalism_progress_effect = yes
	}
}

balkfm_revivalism.009 = { # The Birth of the Bulgarian State
	type = country_event
	placement = root

	title = balkfm_revivalism.009.t
	desc = {
		triggered_desc {
			desc = balkfm_revivalism.009.d1
			trigger = {
				is_subject = yes
			}
		}
		triggered_desc {
			desc = balkfm_revivalism.009.d2
			trigger = {
				is_subject = no
			}
		}
	}
	flavor = {
		triggered_desc {
			desc = balkfm_revivalism.009.f1
			trigger = {
				is_subject = yes
			}
		}
		triggered_desc {
			desc = balkfm_revivalism.009.f2
			trigger = {
				is_subject = no
			}
		}
	}

	event_image = {
		video = "ep1_transfer_of_authority"
	}

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		# triggered by je_balkfm_bul_millet
	}

	immediate = {
		c:TUR = {
			save_scope_as = ottoman_scope
		}
		overlord ?= {
			save_scope_as = balkfm_overlord_scope
		}
		random_country = {
			limit = {
				has_variable = balkfm_backing_bulgaria
			}
			save_scope_as = balkfm_support_scope
		}
	}

	option = {
		name = balkfm_revivalism.009.a
		default_option = yes
		if = {
			limit = {
				country_has_primary_culture = cu:turkish
			}
			remove_primary_culture = cu:turkish
		}
		if = {
			limit = {
				NOT = {
					country_has_state_religion = rel:orthodox
				}
			}
			set_state_religion = rel:orthodox
			ig:ig_devout = {
				set_interest_group_name = ig_orthodox_church
				remove_ideology = ideology_patriarchal
				add_ideology = ideology_orthodox_patriarch
			}
		}
		if = {
			limit = {
				country_has_primary_culture = cu:bosniak
			}
			trigger_event = {
				id = balkfm_bosnia.001 # The Small Land of the Serbs
				days = 30
			}
		}
		if = {
			limit = {
				country_has_primary_culture = cu:albanian
			}
			trigger_event = {
				id = balkfm_albania.001 # The Children of the Eagles
				days = 30
			}
		}
	}

}
