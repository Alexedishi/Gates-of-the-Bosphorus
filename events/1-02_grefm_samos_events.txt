			value = medium_radicals
		}
	}

}

grefm_samos.005 = { # Stefanos Desposed
	type = country_event
	placement = root

	title = grefm_samos.005.t
	desc = grefm_samos.005.d
	flavor = grefm_samos.005.f

	event_image = {
		video = "europenorthamerica_political_extremism"
	}

	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 1

	dlc = dlc_grefm

	trigger = {
		NOT = {
			has_variable = grefm_stefanos_var
		}
		var:grefm_samos_support_var <= 0
		has_law_or_variant = law_type:law_monarchy
	}

	immediate = {
		grefm_generate_samos_ruler_effect = yes
		random_character_in_void = {
			limit = {
				has_variable = grefm_samos_ruler_var
			}
			save_scope_as = samos_next_ruler_scope
		}
	}

	option = {
		name = grefm_samos.005.a
		default_option = yes
		ruler = {
			exile_character = yes
		}
		heir ?= {
			disinherit_character = yes
		}
		scope:samos_next_ruler_scope = {
			set_character_as_ruler = yes
			remove_variable = grefm_samos_ruler_var
		}
		add_modifier = {
			name = modifier_grefm_ruler_ejected
			months = gbbf_short_modifier_time
			is_decaying = yes
		}
	}

}
