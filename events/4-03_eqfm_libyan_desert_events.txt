eqfm_libyan_desert.001 = { # 
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3
	dlc = dlc_grefm

	title = eqfm_libyan_desert.001.t
	desc = eqfm_libyan_desert.001.d
	flavor = eqfm_libyan_desert.001.f

	event_image = {
		video = "africa_desert_expedition"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		# triggered by eqfm_desert_expedition_button
	}

	cancellation_trigger = {
		NOT = {
			has_journal_entry = je_eqfm_libyan_desert
		}
	}

	immediate = {
		set_variable = {
			name = eqfm_desert_provinces_var
			value = eqfm_desert_provinces_value
		}
	}

	option = {
		name = eqfm_libyan_desert.001.a
		default_option = yes
		custom_tooltip = turkfm_no_effects
	}

}

	flavor = eqfm_libyan_desert.009.f

	event_image = {
		video = "africa_desert_expedition"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {

	}

	immediate = {
		s:STATE_LIBYAN_DESERT = {
			random_scope_state = {
				save_scope_as = turkfm_desert_scope
			}
		}
	}

	option = {
		name = eqfm_libyan_desert.009.a
		default_option = yes
		scope:turkfm_desert_scope = {
			add_modifier = {
				name = modifier_eqfm_tamed_wilderness
				days = long_modifier_time
			}
		}
	}

}
