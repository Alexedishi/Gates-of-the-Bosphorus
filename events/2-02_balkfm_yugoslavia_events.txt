	type = country_event

			}
			triggered_desc = {
				desc = balkfm_yugoslavia.001.f3

				desc = balkfm_yugoslavia.001.f4

				desc = balkfm_yugoslavia.001.f1

	}

	event_image = {
		video = "ip3_agitator_reading_manifesto"
	}

	icon = "gfx/interface/icons/event_icons/event_flag_yug.dds"

	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		# triggered by je_balkfm_nacertanije
	}

	immediate = {
		# I'm not harmonizing that ruler immortality nonsense
		save_scope_as = yugoslavia_seeker_scope
		if = {
			limit = {
				NOT = {
					any_character_in_void = {
						has_template = template_balkfm_garasanin_ilija
					}
				}
			}
			create_character = {
				template = template_balkfm_garasanin_ilija
				on_created = {
					place_character_in_void = 720 # 60 year lease on life
				}
			}
		}
		random_character_in_void = {
			limit = {
				has_template = template_balkfm_garasanin_ilija
			}
			save_scope_as = garasanin_scope
		}
		c:TUR = {
			save_scope_as = ottoman_scope
		}
	}

	option = {
		name = balkfm_yugoslavia.001.a
		default_option = yes
		hidden_effect = {
			every_country = {
				limit = {
					NOT = { THIS = ROOT }
					OR = {
						country_has_primary_culture = cu:croat
						country_has_primary_culture = cu:serb
						country_has_primary_culture = cu:slovene
						country_has_primary_culture = cu:bosniak
					}
				}
				post_notification = yugoslavia_seeking_warning
			}
		}
		ordered_diplomatically_relevant_country = {
			limit = {
				NOT = { THIS = ROOT }
				owner = {
					OR = {
						country_has_primary_culture = cu:croat
						country_has_primary_culture = cu:serb
						country_has_primary_culture = cu:slovene
						country_has_primary_culture = cu:bosniak
					}
					relations:root < relations_threshold:cordial
				}
			}
			order_by = country_rank
			ROOT = {
				create_incident = {
					country = PREV
					value = 10
				}
			}
		}
		add_modifier = {
			name = seeking_yugoslavia
			months = gbbf_duration_15_years
		}
		custom_tooltip = balkfm_yugoslavism_tt
	}

	option = {
		name = balkfm_yugoslavia.001.b
		every_country = {
			limit = {
				NOT = { THIS = ROOT }
				OR = {
					country_has_primary_culture = cu:croat
					country_has_primary_culture = cu:serb
					country_has_primary_culture = cu:slovene
					country_has_primary_culture = cu:bosniak
				}
			}
			change_relations = { country = ROOT value = 10 }
		}
	}

}

## Yugoslavian Unification Mechanical Events ##
balkfm_yugoslavia.015 = { # Yugoslav Country Seeks Union (Same Culture)
	type = country_event
	placement = root

	title = balkfm_yugoslavia.015.t
	desc = balkfm_yugoslavia.015.d
	flavor = balkfm_yugoslavia.015.f

	event_image = {
		video = "unspecific_signed_contract"
	}

	icon = "gfx/interface/icons/event_icons/event_flag_yug.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		global_var:balkfm_yugoslav_leader_var ?= this
		NOT = {
			has_modifier = modifier_balkfm_established_union
		}
		any_country = {
			has_journal_entry = je_balkfm_nacertanije
			is_adjacent_to_country = root
			balkfm_shares_slav_culture_trigger = yes
			is_player = no
			is_diplomatic_play_committed_participant = no
			is_subject = no
			relations:root >= relations_threshold:amicable
		}
	}

	cancellation_trigger = {
		NOT = {
			exists = scope:balkfm_yugoslav_union_scope
		}
	}

	immediate = {
		random_country = {
			limit = {
				has_journal_entry = je_balkfm_nacertanije
				is_adjacent_to_country = root
				balkfm_shares_slav_culture_trigger = yes
				is_player = no
				is_diplomatic_play_committed_participant = no
				is_subject = no
				relations:root >= relations_threshold:amicable
			}
			save_scope_as = balkfm_yugoslav_union_scope
		}
	}

	option = {
		name = balkfm_yugoslavia.015.a
		default_option = yes
		create_diplomatic_pact = {
			country = scope:balkfm_yugoslav_union_scope
			type = protectorate
		}
		add_modifier = {
			name = modifier_balkfm_established_union
			months = gbbf_duration_2_plus_half_years
		}
		ai_chance = {
			base = 100
		}
	}

	option = {
		name = balkfm_yugoslavia.015.b
		change_relations = {
			country = scope:balkfm_yugoslav_union_scope
			value = -50
		}
		balkfm_sad_yugoslavs_effect = yes
		ai_chance = {
			base = -100
		}
	}

}
