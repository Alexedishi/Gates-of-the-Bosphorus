grefm_byzantium.020 = { # Chariot Racing Technical Event
	type = country_event
	placement = root
	hidden = yes

	trigger = {
		has_variable = grefm_chariot_racing_revived
		month = 7
		OR = {
			any_scope_state = {
				is_production_method_active = {
					building_type = gbbf_building_hippodrome
					production_method = pm_gbbf_original_roster
				}
			}
			any_scope_state = {
				is_production_method_active = {
					building_type = gbbf_building_hippodrome
					production_method = pm_gbbf_expanded_roster
				}
			}
		}
	}

	cancellation_trigger = {
		NOR = {
			any_scope_state = {
				is_production_method_active = {
					building_type = gbbf_building_hippodrome
					production_method = pm_gbbf_original_roster
				}
			}
			any_scope_state = {
				is_production_method_active = {
					building_type = gbbf_building_hippodrome
					production_method = pm_gbbf_expanded_roster
				}
			}
		}
	}

	cooldown = {
		months = 40
	}

	immediate = {
		if = {
			limit = {
				any_scope_state = {
					is_production_method_active = {
						building_type = gbbf_building_hippodrome
						production_method = pm_gbbf_original_roster
					}
				}
			}
			random_list = {
				40 = {
					set_variable = grefm_chariot_reds
					trigger_event = { id = grefm_byzantium.023 } # A Day at the Races
				}
				40 = {
					set_variable = grefm_chariot_whites
					trigger_event = { id = grefm_byzantium.023 } # A Day at the Races
				}
				40 = {
					set_variable = grefm_chariot_greens
					trigger_event = { id = grefm_byzantium.023 } # A Day at the Races
				}
				40 = {
					set_variable = grefm_chariot_blues
					trigger_event = { id = grefm_byzantium.023 } # A Day at the Races
				}
				1 = {
					trigger_event = { id = grefm_byzantium.024 } # Inconclusive Race Results
				}
				1 = {
					trigger_event = { id = grefm_byzantium.025 } # Sporting Riot
				}
				1 = {
					trigger_event = { id = grefm_byzantium.026 } # Ruler Attempts to Join Race
				}
				1 = {
					trigger_event = { id = grefm_byzantium.027 } # Team Caught Cheating
				}
			}
		}
		else_if = {
			limit = {
				any_scope_state = {
					is_production_method_active = {
						building_type = gbbf_building_hippodrome
						production_method = pm_gbbf_expanded_roster
					}
				}
			}
			random_list = {
				25 = {
					set_variable = grefm_chariot_reds
					trigger_event = { id = grefm_byzantium.023 } # A Day at the Races
				}
				25 = {
					set_variable = grefm_chariot_whites
					trigger_event = { id = grefm_byzantium.023 } # A Day at the Races
				}
				25 = {
					set_variable = grefm_chariot_greens
					trigger_event = { id = grefm_byzantium.023 } # A Day at the Races
				}
				25 = {
					set_variable = grefm_chariot_blues
					trigger_event = { id = grefm_byzantium.023 } # A Day at the Races
				}
				25 = {
					set_variable = grefm_chariot_purples
					trigger_event = { id = grefm_byzantium.023 } # A Day at the Races
				}
				25 = {
					set_variable = grefm_chariot_golds
					trigger_event = { id = grefm_byzantium.023 } # A Day at the Races
				}
				1 = {
					trigger_event = { id = grefm_byzantium.024 } # Inconclusive Race Results
				}
				1 = {
					trigger_event = { id = grefm_byzantium.025 } # Sporting Riot
				}
				1 = {
					trigger_event = { id = grefm_byzantium.026 } # Ruler Attempts to Join Race
				}

grefm_byzantium.023 = { # A Day at the Races
	type = country_event

	title = grefm_byzantium.023.t
	desc = grefm_byzantium.023.d
	flavor = grefm_byzantium.023.f

	placement = scope:byzantium_capital_state

	event_image = {
		video = "byzfm_chariot_race"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		# triggered by grefm_byzantium.020
	}

	immediate = {
		s:STATE_EASTERN_THRACE = {
			random_scope_state = {
				save_scope_as = byzantium_capital_state
			}
		}
		if = {
			limit = {
				has_variable = grefm_chariot_reds
			}
			ig:ig_industrialists = {
				save_scope_as = grefm_chariot_winner_scope
			}
		}
		if = {
			limit = {
				has_variable = grefm_chariot_whites
			}
			ig:ig_intelligentsia = {
				save_scope_as = grefm_chariot_winner_scope
			}
		}
		if = {
			limit = {
				has_variable = grefm_chariot_greens
			}
			ig:ig_rural_folk = {
				save_scope_as = grefm_chariot_winner_scope
			}
		}
		if = {
			limit = {
				has_variable = grefm_chariot_blues
			}
			ig:ig_petty_bourgeoisie = {
				save_scope_as = grefm_chariot_winner_scope
			}
		}
		if = {
			limit = {
				has_variable = grefm_chariot_purples
			}
			ig:ig_landowners = {
				save_scope_as = grefm_chariot_winner_scope
			}
		}
		if = {
			limit = {
				has_variable = grefm_chariot_golds
			}
			ig:ig_trade_unions = {
				save_scope_as = grefm_chariot_winner_scope
			}
		}
	}

	option = {
		name = grefm_byzantium.023.a
		default_option = yes
		scope:byzantium_capital_state = {
			add_modifier = {
				name = modifier_grefm_held_chariot_races
				months = 48
			}
		}
		scope:grefm_chariot_winner_scope = {
			add_modifier = {
				name = modifier_grefm_ig_won_races
				months = 48
			}
		}
		grefm_cleanup_quadrennial_race_effect = yes
	}

}
