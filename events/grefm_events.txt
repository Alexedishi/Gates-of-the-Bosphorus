grefm.830 = { # Chariot Racing Technical Event
	type = country_event
	placement = root
	hidden = yes
	
	trigger = {
		has_variable = grefm_chariot_racing_revived
		month = 7
		OR = {
			any_scope_state = {
				is_production_method_active = {
					building_type = gbbf_building_hippodrome
					production_method = pm_gbbf_original_roster
				}
			}
			any_scope_state = {
				is_production_method_active = {
					building_type = gbbf_building_hippodrome
					production_method = pm_gbbf_expanded_roster
				}
			}
		}
	}
	
	cancellation_trigger = {
		NOR = {
			any_scope_state = {
				is_production_method_active = {
					building_type = gbbf_building_hippodrome
					production_method = pm_gbbf_original_roster
				}
			}
			any_scope_state = {
				is_production_method_active = {
					building_type = gbbf_building_hippodrome
					production_method = pm_gbbf_expanded_roster
				}
			}
		}
	}
	
	cooldown = {
		months = 40
	}

	immediate = {
		if = {
			limit = {
				any_scope_state = {
					is_production_method_active = {
						building_type = gbbf_building_hippodrome
						production_method = pm_gbbf_original_roster
					}
				}
			}
			random_list = {
				40 = {
					set_variable = grefm_chariot_reds
					trigger_event = { id = grefm.833 } # A Day at the Races
				}
				40 = {
					set_variable = grefm_chariot_whites
					trigger_event = { id = grefm.833 } # A Day at the Races
				}
				40 = {
					set_variable = grefm_chariot_greens
					trigger_event = { id = grefm.833 } # A Day at the Races
				}
				40 = {
					set_variable = grefm_chariot_blues
					trigger_event = { id = grefm.833 } # A Day at the Races
				}
				1 = {
					trigger_event = { id = grefm.834 } # Inconclusive Race Results
				}
				1 = {
					trigger_event = { id = grefm.835 } # Sporting Riot
				}
				1 = {
					trigger_event = { id = grefm.836 } # Ruler Attempts to Join Race
				}
				1 = {
					trigger_event = { id = grefm.837 } # Team Caught Cheating
				}
			}
		}
		else_if = {
			limit = {
				any_scope_state = {
					is_production_method_active = {
						building_type = gbbf_building_hippodrome
						production_method = pm_gbbf_expanded_roster
					}
				}
			}
			random_list = {
				25 = {
					set_variable = grefm_chariot_reds
					trigger_event = { id = grefm.833 } # A Day at the Races
				}
				25 = {
					set_variable = grefm_chariot_whites
					trigger_event = { id = grefm.833 } # A Day at the Races
				}
				25 = {
					set_variable = grefm_chariot_greens
					trigger_event = { id = grefm.833 } # A Day at the Races
				}
				25 = {
					set_variable = grefm_chariot_blues
					trigger_event = { id = grefm.833 } # A Day at the Races
				}
				25 = {
					set_variable = grefm_chariot_purples
					trigger_event = { id = grefm.833 } # A Day at the Races
				}
				25 = {
					set_variable = grefm_chariot_golds
					trigger_event = { id = grefm.833 } # A Day at the Races
				}
				1 = {
					trigger_event = { id = grefm.834 } # Inconclusive Race Results
				}
				1 = {
