grefm_roman_empire.001 = { # The Purple Phoenix
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_aquila.dds"

	title = grefm_roman_empire.001.t
	desc = grefm_roman_empire.001.d
	flavor = grefm_roman_empire.001.f

	gui_window = event_window_widescreen_classic

	event_image = {
		texture = "gfx/event_pictures/gbbf_backgrounds/votp_monarch_holding_court.dds"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/ti_ipermajo_kontakion"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		# triggered by je_grefm_purple_phoenix
	}

	immediate = {
		random_scope_building = {
			limit = {
				is_building_type = building_hagia_sophia
			}
			save_scope_as = hagia_sofia_scope
		}
		save_scope_as = formation_country
		every_country = {
			limit = {
				has_diplomatic_relevance = ROOT
			}
			post_notification = unification_notification_byz
		}
	}

	option = {
		name = grefm_roman_empire.001.a
		default_option = yes
		trigger_event = {
			id = grefm_roman_empire.002 # Nova Roma Constantinopolitana
			days = 30
		}
		add_modifier = {
			name = modifier_grefm_rebirth
			days = very_long_modifier_time
		}
		if = {
			limit = {
				NOT = {
					country_has_primary_culture = cu:rhomaios
				}
			}
			add_primary_culture = cu:rhomaios
		}
	}

}
