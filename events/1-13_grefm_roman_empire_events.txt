grefm_roman_empire.001 = { # The New Greece Rises
	type = country_event
	placement = ROOT
	icon = "gfx/interface/icons/event_icons/event_hellenic.dds"

	title = grefm_roman_empire.001.t
	desc = grefm_roman_empire.001.d
	flavor = grefm_roman_empire.001.f

	gui_window = event_window_widescreen_classic

	event_image = {
		texture = "gfx/event_pictures/gbbf_backgrounds/votp_monarch_holding_court.dds"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/epitaph_of_seikilos"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		# triggered by je_grefm_byzantine_restoration
	}

	immediate = {
		random_scope_building = {
			limit = {
				is_building_type = building_hagia_sophia
			}
			save_scope_as = hagia_sofia_scope
		}
		save_scope_as = formation_country
		every_country = {
			limit = {
				has_diplomatic_relevance = ROOT
			}
			post_notification = unification_notification_byz
		}
	}

	option = {
		name = grefm_roman_empire.001.a
		default_option = yes
		if = {
			limit = {
				capital = {
					state_region != s:STATE_EASTERN_THRACE
				}
			}
			set_capital = STATE_EASTERN_THRACE
		}
		every_state_region = {
			limit = {
				OR = {
					region = sr:region_anatolia
					region = sr:region_balkans
				}
				any_scope_state = {
					NOT = { owner = ROOT }
					NOT = { has_claim_by = ROOT }
				}
			}
			add_claim = ROOT
		}
		grefm_add_levant_claims_effect = yes
	}

	option = {
		custom_tooltip = {
			text = je_grefm_byzantine_restoration_rejected_effect
			set_variable = grefm_byzantine_restoration_rejected
		}
		add_modifier = {
			name = modifier_grefm_rebirth
			months = gbbf_duration_20_years
		}
	}

}

grefm_roman_empire.002 = { # The Eternal City
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_hellenic.dds"

	title = grefm_roman_empire.002.t
	desc = grefm_roman_empire.002.d
	flavor = grefm_roman_empire.002.f

	event_image = {
		video = "gbbf_georgios_averof"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		OR = {
			has_modifier = modifier_grefm_reclaimed_balkans
			has_modifier = modifier_grefm_reclaimed_anatolia
			has_modifier = modifier_grefm_reclaimed_levant
			has_modifier = modifier_grefm_reclaimed_egypt
		}
		NOR = {
			has_variable = grefm_embraced_roman_reconquest
			has_variable = grefm_rejected_roman_reconquest
		}
	}

	immediate = {
		random_country = {
			limit = {
				capital = {
					region = sr:region_italy
				}
			}
			save_scope_as = italy_incident_scope
		}
	}

	option = {
		name = grefm_roman_empire.002.a
		default_option = yes
		create_incident = {
			country = scope:italy_incident_scope
			value = 10
		}
		every_state_region = {
			limit = {
				OR = {
					region = sr:region_italy
				}
				any_scope_state = {
					NOT = { owner = ROOT }
					NOT = { has_claim_by = ROOT }
				}
			}
			add_claim = ROOT
		}
		custom_tooltip = {
			text = grefm_embraced_roman_reconquest
			set_variable = grefm_embraced_roman_reconquest
		}
	}

	option = {
		name = grefm_roman_empire.002.b
		custom_tooltip = {
			text = grefm_rejected_roman_reconquest
			set_variable = grefm_rejected_roman_reconquest
		}
	}

}
