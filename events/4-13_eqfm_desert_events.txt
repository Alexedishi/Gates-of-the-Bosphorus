eqfm_libyan_desert.005 = { # Securing the Desert
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3
	dlc = dlc_grefm

	title = eqfm_libyan_desert.005.t
	desc = eqfm_libyan_desert.005.d
	flavor = eqfm_libyan_desert.005.f

	event_image = {
		video = "africa_desert_expedition"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		# triggered by eqfm_deploy_desert_expedition_button
	}

	cancellation_trigger = {
		NOT ={
			has_journal_entry = je_eqfm_libyan_desert
		}
	}

	immediate = {
		set_variable = {
			name = eqfm_desert_expedition_cooldown_var
			months = 4
		}
		random_scope_character = {
			limit = {
				has_role = general
				is_busy = no
				is_ruler = no
			}
			save_scope_as = expedition_leader_1
		}
		if = {
			limit = {
				any_scope_character = {
					has_role = general
					is_busy = no
					is_ruler = no
					OR = {
						has_trait = surveyor
						has_trait = plains_commander
					}
					NOT = {
						scope:expedition_leader_1 ?= this
					}
				}
			}
			random_scope_character = {
				limit = {
					has_role = general
					is_busy = no
					is_ruler = no
					OR = {
						has_trait = surveyor
						has_trait = plains_commander
					}
					NOT = {
						scope:expedition_leader_1 ?= this
					}
				}
				save_scope_as = expedition_leader_2
			}
		}
		else = {
			random_scope_character = {
				limit = {
					has_role = general
					is_busy = no
					is_ruler = no
					NOT = {
						scope:expedition_leader_1 ?= this
					}
				}
				save_scope_as = expedition_leader_2
			}
		}
		random_scope_character = {
			limit = {
				has_role = general
				is_busy = no
				is_ruler = no
				NOR = {
					scope:expedition_leader_1 ?= this
					scope:expedition_leader_2 ?= this
				}
			}
			save_scope_as = expedition_leader_3
		}
	}

	option = {
		name = eqfm_libyan_desert.005.a
		default_option = yes
		trigger = {
			NOR = {
				exists = scope:expedition_leader_1
				exists = scope:expedition_leader_2
				exists = scope:expedition_leader_3
			}
		}
		custom_tooltip = {
			text = gbbf_nothing_happens
			remove_variable = eqfm_desert_expedition_var
		}
	}

	option = {
		name = eqfm_libyan_desert.005.b
		trigger = {
			exists = scope:expedition_leader_1
		}
		scope:expedition_leader_1 ?= {
			set_character_busy_and_immortal = yes
			add_modifier = modifier_eqfm_leading_desert_expedition
		}
	}

	option = {
		name = eqfm_libyan_desert.005.c
		trigger = {
			exists = scope:expedition_leader_2
		}
		scope:expedition_leader_2 ?= {
			set_character_busy_and_immortal = yes
			add_modifier = modifier_eqfm_leading_desert_expedition
		}
	}

	option = {
		name = eqfm_libyan_desert.005.e
		trigger = {
			exists = scope:expedition_leader_3
		}
		scope:expedition_leader_3 ?= {
			set_character_busy_and_immortal = yes
			add_modifier = modifier_eqfm_leading_desert_expedition
		}
	}

	after = {
		if = {
			limit = {
				OR = {
					exists = scope:expedition_leader_1
					exists = scope:expedition_leader_2
					exists = scope:expedition_leader_3
				}
			}
			trigger_event = {
				id = eqfm_libyan_desert.006 # Expedition Departs into the Desert
				days = { 15 30 }
			}
			set_variable = eqfm_expedition_desert_transit_var
		}
		eqfm_update_desert_variables_effect = yes
	}

}

eqfm_libyan_desert.010 = { # 
	type = country_event
	placement = scope:eqfm_senussi_marabout_state
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	duration = 3
	dlc = dlc_grefm

	title = eqfm_libyan_desert.010.t
	desc = eqfm_libyan_desert.010.d
	flavor = eqfm_libyan_desert.010.f

	event_image = {
		video = "africa_city_center"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		has_journal_entry = je_turkfm_senusiyya_main
		s:STATE_LIBYAN_DESERT = {
			any_scope_state = {
				owner ?= {
					is_country_type = decentralized
				}
			}
		}
	}

	cancellation_trigger = {
		NOT = {
			has_journal_entry = je_eqfm_libyan_desert
		}
	}

	cooldown = {
		years = 8
	}

	immediate = {
		random_scope_state = { 
			limit = {
				NOT = {
					has_modifier = modifier_eqfm_dissident_teachings
				}
			}
			save_scope_as = eqfm_senussi_marabout_state
		}
	}

	option = {
		name = eqfm_libyan_desert.010.a
		default_option = yes
	}

	option = {
		name = eqfm_libyan_desert.010.b
		default_option = yes
		trigger = {
			OR = {
				has_law_or_variant = law_type:law_censorship
				has_law_or_variant = law_type:law_outlawed_dissent
			}
		}
		show_as_unavailable = { always = yes }
		highlighted_option = yes
	}

}
