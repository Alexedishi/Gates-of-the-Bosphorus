turkfm_expeditions.010 = { # Negotiations On Head-taking (Receiver)
	type = country_event
	placement = scope:montenegro_scope
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 1
	dlc = dlc_grefm
	
	title = turkfm_expeditions.010.t
	desc = turkfm_expeditions.010.d
	flavor = turkfm_expeditions.010.f
	
	event_image = {
		video = "asia_politician_parliament_motion"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		# triggered by turkfm_renounce_headtaking_button
	}
	
	immediate = {
		random_country = {
			limit = {
				has_variable = turkfm_negociated_headtaking
			}
			save_scope_as = turkfm_headtaking_negotiator
		}
		set_variable = {
			name = turkfm_headtaking_recipient
			months = 3
		}
	}
	
	option = { # Reject headtaking offer
		name = turkfm_expeditions.010.a
		default_option = yes
		add_modifier = {
			name = modifier_turkfm_retained_headtaking
			days = long_modifier_time
		}
		scope:turkfm_headtaking_negotiator = {
			trigger_event = {
				id = turkfm_expeditions.011 # Negotiations On Head-taking
			}
			set_variable = turkfm_headtaking_rejected
		}
	}
	
	option = { # Accept headtaking offer
		name = turkfm_expeditions.010.b
		add_modifier = {
			name = modifier_turkfm_abolished_headtaking
			days = normal_modifier_time
		}
		scope:turkfm_headtaking_negotiator = {
			trigger_event = {
				id = turkfm_expeditions.011 # Negotiations On Head-taking
			}
		}
	}
}

turkfm_expeditions.011 = { # Negotiations on Head-taking (MON)
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 2
	dlc = dlc_grefm
	
	title = turkfm_expeditions.011.t
	desc = {
		triggered_desc = {
			desc = turkfm_expeditions.011.d1
			trigger = {
				has_variable = turkfm_headtaking_rejected
			}
		}
		triggered_desc = {
			desc = turkfm_expeditions.011.d2
			trigger = {
				NOT = {
					has_variable = turkfm_headtaking_rejected
				}
			}
		}
	}
	flavor = turkfm_expeditions.011.f
	
	gui_window = event_window_2char_discussion
	left_icon = scope:turkfm_ruler_scope
	right_icon = scope:ottoman_ruler_scope
	
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"
	
	trigger = {
		# triggered by turkfm_expeditions.041
	}
	
	immediate = {
		random_country = {
			limit = {
				has_variable = turkfm_headtaking_recipient
			}
			save_scope_as = turkfm_headtaking_recipient_scope
		}
		ruler = {
			save_scope_as = turkfm_ruler_scope
		}
	}
	
	option = { # Head-taking offer rejected, retain headtaking
		name = turkfm_expeditions.011.a
		default_option = yes
		trigger = {
			has_variable = turkfm_headtaking_rejected
		}
		add_modifier = {
			name = modifier_turkfm_retained_headtaking
			days = normal_modifier_time
		}
		remove_variable = turkfm_headtaking_rejected
	}
	
	option = { # Head-taking offer rejected, abolish unilaterally
		name = turkfm_expeditions.011.b
		trigger = {
			has_variable = turkfm_headtaking_rejected
		}
		remove_modifier = modifier_turkfm_practicing_headtaking
		add_modifier = {
			name = modifier_turkfm_unilaterally_abolished_headtaking
			days = long_modifier_time
		}
		remove_variable = turkfm_headtaking_rejected
	}
	
	option = { # Head-taking offer accepted
		name = turkfm_expeditions.011.c
		trigger = {
			NOT = {
				has_variable = turkfm_headtaking_rejected
			}
		}
		remove_modifier = modifier_turkfm_practicing_headtaking
		add_modifier = {
			name = modifier_turkfm_abolished_headtaking
			days = normal_modifier_time
		}
	}
}

turkfm_expeditions.012 = { # Negotiations Over Cattaro (Recipient)
	type = country_event
	placement = scope:turkfm_cattaro_scope
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	duration = 2
	dlc = dlc_grefm
	
	title = turkfm_expeditions.012.t
	desc = turkfm_expeditions.012.d
	flavor = turkfm_expeditions.012.f
	
	event_image = {
		video = "unspecific_signed_contract"
	}
	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		# triggered by turkfm_negotiate_kotor_button
	}
	
	cancellation_trigger = {
		c:MON ?= {
			NOR = {
				is_subject_of = c:AUS
				is_in_same_power_bloc = c:AUS
				any_scope_treaty = {
					binds = root
					binds = c:AUS
					any_scope_article = {
						OR = {
							has_type = alliance
							has_type = defensive_pact
						}
					}
				}
			}
		}
	}
	
	immediate = {
		c:MON = {
			save_scope_as = montenegro_scope
		}
		p:x90A0E0 = { # Kotor
			save_scope_as = turkfm_cattaro_scope
		}
	}
	
	option = { # Accept the Deal
		name = turkfm_expeditions.012.a
		default_option = yes
		show_as_tooltip = {
			s:STATE_MONTENEGRO = {
				set_owner_of_provinces = {
					country = scope:montenegro_scope
					provinces = {
						x90A0E0 # Kotor
					}
				}
			}
			c:MON = {
				set_owes_obligation_to = {
					country = root
					setting = yes
				}
			}
		}
		create_diplomatic_catalyst = {
			type = catalyst_relations_level_increased
			target = scope:montenegro_scope
		}
		change_relations = {
			country = c:MON
			value = 25
		}
	}
	
	option = { # Reject the Deal
		name = turkfm_expeditions.012.b
		change_relations = {
			country = c:MON
			value = -25
		}
		create_diplomatic_catalyst = {
			type = catalyst_relations_level_decreased
			target = c:MON
		}
	}

	after = {
		trigger_event = {
			id = turkfm_expeditions.013
			days = { 5 15 }
		}
	}
}

turkfm_expeditions.020 = { # Marauding in the Hinterlands
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	duration = 3
	dlc = dlc_grefm
	
	title = turkfm_expeditions.020.t
	desc = turkfm_expeditions.020.d
	flavor = turkfm_expeditions.020.f
	
	event_image = {
		video = "ip3_officer_inspecting_peasant_conscripts"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		# triggered by turkfm_appoint_tribal_chief_button
	}
	
	cancellation_trigger = {
		has_global_variable = turkfm_montenegrin_raiding_settled
	}
	
	immediate = {
		if = {
			limit = {
				any_scope_character = {
					count < 1
					has_role = general
				}
			}
			create_character = {
				traits = {
					mountain_commander
				}
				on_created = {
					place_character_in_void = 120
					set_variable = turkfm_tribal_leader
				}
			}
			create_character = {
				traits = {
					mountain_commander
				}
				on_created = {
					place_character_in_void = 120
					set_variable = turkfm_tribal_leader
				}
			}
		}
		else_if = {
			limit = {
				any_scope_character = {
					count < 2
					has_role = general
				}
			}
			create_character = {
				traits = {
					mountain_commander
				}
				on_created = {
					place_character_in_void = 120
					set_variable = turkfm_tribal_leader
				}
			}
		}
		random_scope_character = {
			limit = {
				is_busy = no
				has_role = general
				is_ruler = no
			}
			save_scope_as = expedition_leader_1
		}
		if = {
			limit = {
				any_scope_character = {
					is_busy = no
					is_ruler = no
					has_role = general
					OR = {
						has_trait = surveyor
						has_trait = mountain_commander
						has_trait = basic_offensive_planner
						has_trait = experienced_offensive_planner
						has_trait = expert_offensive_planner
					}
					NOT = {
						scope:expedition_leader_1 ?= this
					}
				}
			}
			random_scope_character = {
				limit = {
					is_busy = no
					is_ruler = no
					has_role = general
					OR = {
						has_trait = surveyor
						has_trait = mountain_commander
						has_trait = basic_offensive_planner
						has_trait = experienced_offensive_planner
						has_trait = expert_offensive_planner
					}
					NOT = {
						scope:expedition_leader_1 ?= this
					}
				}
				save_scope_as = expedition_leader_2
			}
		}
		else = {
			random_scope_character = {
				limit = {
					is_busy = no
					is_ruler = no
					has_role = general
					NOT = {
						scope:expedition_leader_1 ?= this
					}
				}
				save_scope_as = expedition_leader_2
			}
		}
		random_character_in_void = {
			limit = {
				has_variable = turkfm_tribal_leader
				NOR = {
					scope:expedition_leader_1 ?= this
					scope:expedition_leader_2 ?= this
				}
			}
			save_scope_as = expedition_leader_3
		}
		random_character_in_void = {
			limit = {
				has_variable = turkfm_tribal_leader
				NOR = {
					scope:expedition_leader_1 ?= this
					scope:expedition_leader_2 ?= this
					scope:expedition_leader_3 ?= this
				}
			}
			save_scope_as = expedition_leader_4
		}
	}
	
	option = {
		name = turkfm_expeditions.020.a
		default_option = yes
		trigger = {
			exists = scope:expedition_leader_1
		}
		scope:expedition_leader_1 ?= {
			set_character_busy_and_immortal = yes
			add_modifier = modifier_turkfm_leading_black_mountain_expedition
		}
	}
	
	option = {
		name = turkfm_expeditions.020.b
		trigger = {
			exists = scope:expedition_leader_2
		}
		scope:expedition_leader_2 ?= {
			set_character_busy_and_immortal = yes
			add_modifier = modifier_turkfm_leading_black_mountain_expedition
		}
	}
	
	option = {
		name = turkfm_expeditions.020.c
		trigger = {
			exists = scope:expedition_leader_3
		}
		scope:expedition_leader_3 ?= {
			set_character_busy_and_immortal = yes
			add_modifier = modifier_turkfm_leading_black_mountain_expedition
		}
	}
	
	option = {
		name = turkfm_expeditions.020.e
		trigger = {

				has_religious_taboo = coffee
				has_religious_taboo = tea

			}
		}
		any_scope_state = {
			is_coastal = yes
			NOR = {
				has_modifier = modifier_turkfm_seizing_contraband
				has_modifier = modifier_turkfm_smuggling_hub

			}
		}

		has_global_variable = turkfm_jack_sparrow_whacked
	}

		years = 1
	}

		set_variable = {
			name = turkfm_piracy_cooldown_var
			months = 6
		}
		turkfm_piracy_scopes_effect = yes
		random_scope_state = {
			limit = {
				is_coastal = yes
				NOR = {
					has_modifier = modifier_turkfm_seizing_contraband
					has_modifier = modifier_turkfm_smuggling_hub
					has_modifier = modifier_turkfm_social_support
				}
			}
			save_scope_as = turkfm_piracy_state_scope
		}
		if = {
			limit = {
				religion = {
					has_religious_taboo = liquor
				}
			}
			g:liquor = {
				save_scope_as = turkfm_piracy_goods_scope
			}
		}
		else_if = {
			limit = {
				religion = {
					has_religious_taboo = wine
				}
			}
			g:wine = {
				save_scope_as = turkfm_piracy_goods_scope
			}
		}
		else_if = {
			limit = {
				religion = {
					has_religious_taboo = coffee
				}
			}
			g:coffee = {
				save_scope_as = turkfm_piracy_goods_scope
			}
		}
		else_if = {
			limit = {
				religion = {
					has_religious_taboo = tea
				}
			}
			g:tea = {
				save_scope_as = turkfm_piracy_goods_scope
			}
		}
		else = {
			g:meat = {
				save_scope_as = turkfm_piracy_goods_scope
			}
		}

	option = { # Arrest them
		name = turkfm_expeditions.117.a

			add_modifier = {
				name = modifier_turkfm_seizing_contraband
				days = normal_modifier_time
			}

				strata = lower
				value = small_radicals
			}

	}

	option = { # Not entirely sure if they play blackjack, but they for sure have hookers

		scope:turkfm_piracy_state_scope = {

			}
