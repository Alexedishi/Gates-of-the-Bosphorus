balkfm_legacies.001 = {
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	duration = 3
	dlc = dlc_grefm

	title = balkfm_legacies.001.t
	desc = balkfm_legacies.001.d
	flavor = balkfm_legacies.001.f

	event_image = {
		video = "unspecific_ruler_speaking_to_people"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		je:je_balkfm_legacies = {
			has_modifier = modifier_balkfm_tax_reform_agenda
		}
		NOR = {
			has_law_or_variant = law_type:law_consumption_based_taxation
			has_law_or_variant = law_type:law_land_based_taxation
		}
	}

	immediate = {
		je:je_balkfm_legacies = {
			remove_modifier = modifier_balkfm_tax_reform_agenda
		}
	}

	option = {
		name = balkfm_legacies.001.a
		default_option = yes
		show_as_tooltip = {
			remove_modifier = modifier_balkfm_tax_reform_agenda
		}
		add_modifier = {
			name = modifier_balkfm_tax_reform_completed
			months = gbbf_normal_modifier_time
		}
		add_loyalists = {
			value = small_radicals
			strata = lower
		}
		set_variable = balkfm_tax_reform_complete
	}
}

balkfm_legacies.010 = {
	type = country_event
	placement = scope:brigandage_state
	icon = "gfx/interface/icons/event_icons/event_fire.dds"
	duration = 3
	dlc = dlc_grefm

	title = balkfm_legacies.010.t
	desc = balkfm_legacies.010.d
	flavor = balkfm_legacies.010.f

	event_image = {
		video = "rp1_social_banditry"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		any_scope_state = {
			NOT = { has_modifier = modifier_balkfm_maurauding_bridgands }
		}
	}

	cancellation_trigger = {
		OR = {
			je:je_balkfm_legacies = {
				NOT = {
					has_modifier = modifier_balkfm_security_reform_agenda
				}
			}
			NOT = {
				any_scope_state = {
					is_in_geographic_region = geographic_region_rumelia
				}
			}
		}
	}

	cooldown = {
		years = 3
	}

	immediate = {
		random_scope_state = {
			limit = {
				NOT = { has_modifier = modifier_balkfm_maurauding_bridgands }
			}
			save_scope_as = brigandage_state
		}
	}

	option = {
		name = balkfm_legacies.010.a
		default_option = yes
		scope:brigandage_state = {
			add_modifier = {
				name = modifier_balkfm_maurauding_bridgands
				months = gbbf_normal_modifier_time
			}
			add_radicals_in_state = {
				value = small_radicals
			}
		}
		if = {
			limit = {
				has_journal_entry = je_grefm_bavarocracy_main
			}
			custom_tooltip = {
				text = je_grefm_bavarocracy_minus_50
				change_variable = {
					name = grefm_bavarocracy_var
					add = -50
				}
			}
		}
	}
}
