turkfm_eastern_question.009 = { # Adds the Eastern Question JE
	type = country_event
	placement = root
	hidden = yes

	trigger = {
		exists = c:TUR
		c:TUR ?= {
			has_diplomatic_relevance = root
			any_scope_state = {
				region = sr:region_balkans
			}
		}
		NOR = {
			has_global_variable = turkfm_separatism_complete
			has_journal_entry = je_turkfm_eastern_question
			has_journal_entry = je_turkfm_balkan_situation_main
		}
	}

	cooldown = {
		months = 3
	}

	immediate = {
		if = {
			limit = {
				has_variable = turkfm_eqfm_started
			}
			add_journal_entry = {
				type = je_turkfm_eastern_question
			}
		}
		else = {
			set_variable = turkfm_eqfm_started
			add_journal_entry = {
				type = je_turkfm_eastern_question
			}
			trigger_event = {
				id = turkfm_eastern_question.001 # The Eastern Question
			}
		}
	}

}

turkfm_eastern_question.010 = { # Brigands Raid Turkish Lands
	type = country_event
	placement = scope:turkfm_brigandage_event_scope

	title = turkfm_eastern_question.010.t
	desc = turkfm_eastern_question.010.d
	flavor = turkfm_eastern_question.010.f

	event_image = {
		video = "rp1_social_banditry"
	}

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		# triggered by turkfm_balkan_situation.119
	}

	cancellation_trigger = {
		NOT = {
			has_journal_entry = je_balkfm_brigandage
		}
	}

	immediate = {
		c:TUR = {
			save_scope_as = ottoman_scope
			random_scope_state = {
				limit = {
					has_variable = turkfm_brigandage_event_var
				}
				save_scope_as = turkfm_brigandage_event_scope
				remove_variable = turkfm_brigandage_event_var
			}
		}
		set_global_variable = {
			name = turkfm_global_reference_var
			value = money_amount_multiplier_small
		}
	}

	option = { # Offer Compensation
		name = turkfm_eastern_question.010.a
		default_option = yes
		add_modifier = {
			name = modifier_turkfm_brigandage_reparations_negative
			days = normal_modifier_time
			multiplier = global_var:turkfm_global_reference_var
		}
		c:TUR = {
			add_modifier = {
				name = modifier_turkfm_brigandage_reparations_positive
				days = normal_modifier_time
				multiplier = global_var:turkfm_global_reference_var
			}
			post_notification = notification_turkfm_brigandage_reparations
		}
		change_relations = {
			country = c:TUR
			value = 15
		}
	}

	option = { # Ignore the Complaint
		name = turkfm_eastern_question.010.b
		c:TUR = {
			post_notification = notification_turkfm_brigandage_permitted
		}
		change_relations = {
			country = c:TUR
			value = -15
		}
	}

	after = {
		remove_global_variable = turkfm_global_reference_var
	}

}

turkfm_eastern_question.104 = { # A Chance at Deconfliction (Third Party Event)
	type = country_event
	placement = scope:ottoman_scope

	flavor = turkfm_eastern_question.104.f

	event_image = {

	duration = 3

	dlc = dlc_grefm

	trigger = {
		has_global_variable = turkfm_crisis_var
		country_rank >= rank_value:great_power
		NOT = {
			has_global_variable = turkfm_deconfliction_var
		}
		global_var:turkfm_crisis_var >= 30
		global_var:turkfm_crisis_var < 90
		turkfm_crisis_deconfliction_trigger = yes
	}

	cancellation_trigger = {
		NOR = {
			global_var:turkfm_balkan_situation_phase_var != flag:turkfm_crisis
			has_global_variable = turkfm_balkan_situation_phase_var
			has_variable = turkfm_crisis_attacker
		}
	}

	cooldown = {
		years = 2
	}

	immediate = {
		c:TUR = {
			save_scope_as = ottoman_scope
		}
	}

	option = { # Attempt Deconfliction
		name = turkfm_eastern_question.104.a
		default_option = yes
		custom_tooltip = turkfm_eastern_question.104.a.tt
		if = {
			limit = {
				c:GBR ?= this
			}
			set_global_variable = {
				name = turkfm_deconfliction_var
				value = 1
			}
		}
		else_if = {
			limit = {
				c:FRA ?= this
			}
			set_global_variable = {
				name = turkfm_deconfliction_var
				value = 2
			}
		}
		else_if = {
			limit = {
				c:RUS ?= this
			}
			set_global_variable = {
				name = turkfm_deconfliction_var
				value = 3
			}
		}
		c:TUR = {
			trigger_event = {
				id = turkfm_eastern_question.105 # A Chance at Deconfliction (Ottoman Event)
			}
		}
	}

	option = { # Continue into the Fight
		name = turkfm_eastern_question.104.b
		custom_tooltip = turkfm_eastern_question.104.b.tt
	}

turkfm_eastern_question.106 = { # Balkan Crisis Superevent <- Creates DP
	type = country_event
	placement = root

	title = turkfm_eastern_question.106.t
	desc = turkfm_eastern_question.106.d
	flavor = turkfm_eastern_question.106.f

	event_image = {
		video = "turkfm_balkan_crisis"
	}

	icon = "gfx/interface/icons/event_icons/event_ottoman.dds"

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	dlc = dlc_grefm

	trigger = {
		has_global_variable = turkfm_crisis_var
		global_var:turkfm_crisis_var >= 100
	}

	immediate = {
		random_country = {
			limit = {
				has_variable = turkfm_crisis_leader
			}
			if = {
				limit = {
					has_truce_with = c:TUR
				}
				end_truce = c:TUR
			}
			save_scope_as = turkfm_crisis_leader_scope
		}
		scope:turkfm_crisis_leader_scope = {
			create_diplomatic_play = {
				type = dp_turkfm_crisis
				target_country = c:TUR
				war = yes
			}
		}
		remove_global_variable = turkfm_crisis_var
		remove_global_variable = turkfm_crisis_progress_var
		set_global_variable = turkfm_crisis_launched
		every_country = {
			limit = {
				OR = {
					has_variable = turkfm_crisis_attacker
					has_variable = turkfm_crisis_defender
				}
				has_truce_with = c:TUR
			}
			end_truce = c:TUR
			trigger_event = {
				id = turkfm_eastern_question.107 # Balkan Crisis Superevent <- Participant Event
			}
		}
	}

	option = {
		name = turkfm_eastern_question.106.a
		default_option = yes
	}

}

turkfm_eastern_question.200 = { # Congress System Monitor
	type = country_event
	hidden = yes

	trigger = {
		has_global_variable = turkfm_congress_var
		NOT = {
			has_global_variable = turkfm_congress_cooldown_var
		}
		any_country = {
			OR = {
				has_variable = turkfm_crisis_attacker
				has_variable = turkfm_crisis_deconflicted
			}
		}
	}

	cancellation_trigger = {
		NOT = {
			has_journal_entry = je_turkfm_international_congress
		}
	}

	immediate = {
		every_country = {
			limit = {
				has_variable = turkfm_congress_issue_var
			}
			remove_variable = turkfm_congress_issue_var
		}
		random_country = {
			limit = {
				OR = {
					has_variable = turkfm_crisis_attacker
					has_variable = turkfm_crisis_deconflicted
				}
			}
			save_scope_as = turkfm_congress_issue_scope
			set_variable = turkfm_congress_issue_var
		}
		if = {
			limit = {
				exists = scope:turkfm_congress_issue_scope
			}
			set_global_variable = {
				name = turkfm_congress_cooldown_var
				days = 45
			}
			set_global_variable = {
				name = turkfm_congress_vote_var
				value = 0
			}
			if = {
				limit = {
					scope:turkfm_congress_issue_scope = {
						has_variable = turkfm_crisis_attacker
					}
				}
				every_country = {
					limit = {
						has_journal_entry = je_turkfm_international_congress
					}
					trigger_event = {
						id = turkfm_eastern_question.202 # The Issue of X
					}
					trigger_event = {
						id = turkfm_eastern_question.208 # Resolution on the X Matter
						days = 45
					}
				}
			}
			else_if = {
				limit = {
					scope:turkfm_congress_issue_scope = {
						has_variable = turkfm_crisis_deconflicted
					}
				}
				every_country = {
					limit = {
						has_journal_entry = je_turkfm_international_congress
					}
					trigger_event = {
						id = turkfm_eastern_question.207 # The Convention of X
					}
				}
			}
		}
		else = {
			every_country = {
				limit = {
					has_journal_entry = je_turkfm_international_congress
				}
				trigger_event = {
					id = turkfm_eastern_question.209 # The Treaty of X
				}
			}
			turkfm_force_deconfliction_effect = yes
		}
	}

}
