		}
	}

REPLACE_OR_CREATE:STATE_temp_state_name_assign = {
	if = {
		limit = { # Greek
			owner = {
				gbbf_country_has_greek_culture_trigger = yes
			}
		}
		set_state_name = STATE_temp_greek
		set_hub_name = {
			type = city
			name = HUB_NAME_STATE_temp_city_greek
		}
		set_hub_name = {
			type = farm
			name = HUB_NAME_STATE_temp_farm_greek
		}
		set_hub_name = {
			type = wood
			name = HUB_NAME_STATE_temp_wood_greek
		}
		set_hub_name = {
			type = mine
			name = HUB_NAME_STATE_temp_mine_greek
		}
		set_hub_name = {
			type = port
			name = HUB_NAME_STATE_temp_port_greek
		}
	}
	else = {
		reset_state_name = yes
		reset_hub_names = yes
	}
}
