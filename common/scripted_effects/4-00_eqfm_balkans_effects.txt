## General Situation Effects
eqfm_initialize_balkan_variables_effect = {
	scope:journal_entry = {
		set_variable = {
			name = eqfm_balkans_phase_var
			value = flag:eqfm_stable # We start in the stable phase
		}
		set_variable = {
			name = eqfm_escalation_var
			value = 0
		}
		set_variable = {
			name = eqfm_deescalation_var
			value = 0
		}
		set_variable = {
			name = eqfm_riot_counter_var
			value = 0
		}
		set_variable = {
			name = eqfm_crisis_var
			value = 250
		}
		set_variable = {
			name = eqfm_crisis_progress_var
			value = 0
		}
		set_variable = {
			name = eqfm_crisis_duration_var
			value = 0
		}
		set_variable = {
			name = eqfm_crisis_counter_var
			value = 0
		}
	}
}

eqfm_validate_journal_involvement = {
	every_country = {
		limit = {
			OR = {
				is_involved_in_journal_entry = je_eqfm_balkan_situation
				eqfm_balkans_should_be_involved_trigger = yes
			}
		}
		if = {
			limit = {
				is_involved_in_journal_entry = je_eqfm_balkan_situation
				eqfm_balkans_should_be_involved_trigger = no
			}
			global_var:eqfm_balkan_situation = {
				remove_involved_country = prev
			}
		}
		if = {
			limit = {
				NOT = {
					is_involved_in_journal_entry = je_eqfm_balkan_situation
				}
				eqfm_balkans_should_be_involved_trigger = yes
			}
			global_var:eqfm_balkan_situation = {
				add_involved_country = prev
			}
		}
	}
}

eqfm_calculate_balkan_situation_variables_effect = {
	global_var:eqfm_balkan_situation = {
		set_variable = {
			name = eqfm_escalation_progress_var
			value = eqfm_escalation_progress_value
		}
		set_variable = {
			name = eqfm_deescalation_progress_var
			value = eqfm_deescalation_progress_value
		}
		set_variable = {
			name = eqfm_riot_counter_var
			value = eqfm_riot_counter_value
		}
	}
	if = {
		limit = {
			eqfm_crisis_phase_trigger = yes
		}
		global_var:eqfm_balkan_situation = {
			set_variable = {
				name = eqfm_crisis_progress_var
				value = eqfm_crisis_progress_value
			}
			change_variable = {
				name = eqfm_crisis_duration_var
				add = 1
			}
		}
	}
	if = {
		limit = {
			OR = {
				eqfm_hostility_phase_trigger = yes
				eqfm_crisis_phase_trigger = yes
			}
		}
		eqfm_update_riot_variables = yes
	}
	eqfm_clamp_balkan_situation_variables_effect = yes
	eqfm_update_balkan_situation_progressbar_effect = yes
}

eqfm_clamp_balkan_situation_variables_effect = {
	global_var:eqfm_balkan_situation = {
		clamp_variable = {
			name = eqfm_escalation_var
			max = 500
			min = 0
		}
		clamp_variable = {
			name = eqfm_deescalation_var
			max = 500
			min = 0
		}
		clamp_variable = {
			name = eqfm_crisis_var
			max = 250
			min = 0
		}
	}
}

eqfm_advance_balkan_situation_variables_effect = {
	global_var:eqfm_balkan_situation = {
		change_variable = {
			name = eqfm_escalation_var
			add = var:eqfm_escalation_progress_var
		}
		change_variable = {
			name = eqfm_deescalation_var
			add = var:eqfm_deescalation_progress_var
		}
		if = {
			limit = {
				eqfm_crisis_phase_trigger = yes
			}
			change_variable = {
				name = eqfm_crisis_var
				add = var:eqfm_crisis_progress_var
			}
			clamp_variable = {
				name = eqfm_crisis_var
				max = 250
				min = 0
			}
		}
	}
	
	eqfm_clamp_balkan_situation_variables_effect = yes
	eqfm_update_balkan_situation_progressbar_effect = yes
}

eqfm_update_balkan_situation_progressbar_effect = {
	global_var:eqfm_balkan_situation = {
		set_bar_progress = {
			name = eqfm_balkans_escalation_progress_bar
			value = var:eqfm_escalation_var
		}
		set_bar_progress = {
			name = eqfm_balkans_deescalation_progress_bar
			value = var:eqfm_deescalation_var
		}
		if = {
			limit = {
				eqfm_crisis_phase_trigger = yes
			}
			set_bar_progress = {
				name = eqfm_balkans_crisis_progress_bar
				value = var:eqfm_crisis_var
			}
		}
		else = {
			set_bar_progress = {
				name = eqfm_balkans_crisis_progress_bar
				value = 250
			}
		}
	}
}

eqfm_check_balkan_situation_phase_effect = {
	global_var:eqfm_balkan_situation = {
		# Run Crisis Failure logic check first
		if = {
			limit = {
				eqfm_crisis_phase_trigger = yes
				var:eqfm_crisis_var <= 0
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_balkan_situation
				}
				trigger_event = {
					id = eqfm_balkans.028 # Rumelia in Disarray
				}
			}
			eqfm_cleanup_crisis_effect = yes
			eqfm_force_deconfliction_effect = yes
		}
		else_if = {
			limit = {
				OR = {
					var:eqfm_escalation_var >= 500
					var:eqfm_deescalation_var >= 500
				}
			}
			## Attempt to Deescalate First
			if = {
				limit = {
					eqfm_can_deescalate_tension_trigger = yes
					var:eqfm_deescalation_var >= 500
				}
				if = { # From Unrest to Deconfliction
					limit = {
						var:eqfm_balkans_phase_var ?= flag:eqfm_unrest
					}
					set_variable = {
						name = eqfm_balkans_phase_var
						value = flag:eqfm_deconfliction
					}
					every_country = {
						limit = {
							is_involved_in_journal_entry = je_eqfm_balkan_situation
						}
						post_notification = notification_eqfm_transition_deconfliction
					}
					custom_tooltip = eqfm_transition_deconfliction_tt
				}
				else_if = { # From Hostility to Deconfliction
					limit = {
						var:eqfm_balkans_phase_var ?= flag:eqfm_hostility
					}
					set_variable = {
						name = eqfm_balkans_phase_var
						value = flag:eqfm_deconfliction
					}
					every_country = {
						limit = {
							is_involved_in_journal_entry = je_eqfm_balkan_situation
						}
						post_notification = notification_eqfm_transition_deconfliction
					}
					custom_tooltip = eqfm_transition_deconfliction_tt
				}
				else_if = { # From Crisis to Deconfliction
					limit = {
						var:eqfm_balkans_phase_var ?= flag:eqfm_crisis
					}
					set_variable = {
						name = eqfm_balkans_phase_var
						value = flag:eqfm_deconfliction
					}
					c:TUR = {
						trigger_event = {
							id = eqfm_balkans.029 # Order Returns to Rumelia
						}
					}
					every_country = {
						limit = {
							is_involved_in_journal_entry = je_eqfm_balkan_situation
							c:TUR != this
						}
						post_notification = notification_eqfm_transition_deconfliction
					}
					custom_tooltip = eqfm_transition_deconfliction_tt
				}
				c:TUR ?= {
					eqfm_cleanup_riots_effect = yes
				}
				set_variable = {
					name = eqfm_escalation_var
					value = 0
				}
				set_variable = {
					name = eqfm_deescalation_var
					value = 0
				}
			}
			
			## Ignore this if we deescalated
			else_if = { # Ignore this if we deescalated
				limit = {
					eqfm_can_escalate_tension_trigger = yes
					var:eqfm_escalation_var >= 500
				}
				if = { # From Stable to Unrest
					limit = {
						var:eqfm_balkans_phase_var ?= flag:eqfm_stable
					}
					set_variable = {
						name = eqfm_balkans_phase_var
						value = flag:eqfm_unrest
					}
					every_country = {
						limit = {
							is_involved_in_journal_entry = je_eqfm_balkan_situation
						}
						post_notification = notification_eqfm_transition_unrest
					}
					custom_tooltip = eqfm_transition_unrest_tt
				}
				else_if = { # From Unrest to Hostility
					limit = {
						var:eqfm_balkans_phase_var ?= flag:eqfm_unrest
					}
					set_variable = {
						name = eqfm_balkans_phase_var
						value = flag:eqfm_hostility
					}
					every_country = {
						limit = {
							is_involved_in_journal_entry = je_eqfm_balkan_situation
						}
						post_notification = notification_eqfm_transition_hostility
					}
					custom_tooltip = eqfm_transition_hostility_tt
				}
				else_if = { # From Hostility to Crisis
					limit = {
						eqfm_crisis_possible_trigger = yes
						var:eqfm_balkans_phase_var ?= flag:eqfm_hostility
					}
					set_variable = {
						name = eqfm_balkans_phase_var
						value = flag:eqfm_crisis
					}
					eqfm_initialize_crisis_effect = yes
					c:TUR ?= {
						trigger_event = {
							id = eqfm_balkans.020 # A Crisis in Rumelia
						}
					}
					every_country = {
						limit = {
							is_involved_in_journal_entry = je_eqfm_balkan_situation
							c:TUR != this
						}
						trigger_event = {
							id = eqfm_balkans.021 # A Crisis in Rumelia
						}
						post_notification = notification_eqfm_transition_crisis
					}
					custom_tooltip = eqfm_transition_crisis_tt
				}
				else_if = { # From Deconfliction to Stable
					limit = {
						var:eqfm_balkans_phase_var ?= flag:eqfm_deconfliction
					}
					set_variable = {
						name = eqfm_balkans_phase_var
						value = flag:eqfm_stable
					}
					every_country = {
						limit = {
							is_involved_in_journal_entry = je_eqfm_balkan_situation
						}
						post_notification = notification_eqfm_transition_stable
					}
					custom_tooltip = eqfm_transition_stable_tt
				}
				set_variable = {
					name = eqfm_escalation_var
					value = 0
				}
				set_variable = {
					name = eqfm_deescalation_var
					value = 0
				}
			}
		}
		eqfm_check_modifiers_effect = yes
	}
}

eqfm_check_modifiers_effect = {
	c:TUR ?= {
		hidden_effect = {
			remove_modifier = modifier_eqfm_stable_phase
			remove_modifier = modifier_eqfm_unrest_phase
			remove_modifier = modifier_eqfm_hostility_phase
			remove_modifier = modifier_eqfm_deconfliction_phase
			remove_modifier = modifier_eqfm_crisis_phase
			every_political_movement = {
				remove_modifier = modifier_eqfm_stable_phase_movement
				remove_modifier = modifier_eqfm_unrest_phase_movement
				remove_modifier = modifier_eqfm_hostility_phase_movement
				remove_modifier = modifier_eqfm_deconfliction_phase_movement
				remove_modifier = modifier_eqfm_crisis_phase_movement
			}
		}
	}
	global_var:eqfm_balkan_situation = {
		if = {
			limit = {
				var:eqfm_balkans_phase_var = flag:eqfm_stable
			}
			c:TUR ?= {
				hidden_effect = {
					add_modifier = modifier_eqfm_stable_phase
					every_political_movement = {
						limit = {
							eqfm_eligible_minority_movement_trigger = yes
						}
						set_local_variable = {
							name = eqfm_current_adjustment_var
							value = eqfm_movement_activism_multiplier_value
						}
						add_modifier = {
							name = modifier_eqfm_stable_phase_movement
							multiplier = local_var:eqfm_current_adjustment_var
						}
						set_variable = {
							name = eqfm_previous_adjustment_var
							value = local_var:eqfm_current_adjustment_var
						}
					}
				}
			}
		}
		else_if = {
			limit = {
				var:eqfm_balkans_phase_var = flag:eqfm_unrest
			}
			c:TUR ?= {
				add_modifier = modifier_eqfm_unrest_phase
				every_political_movement = {
					limit = {
						eqfm_eligible_minority_movement_trigger = yes
					}
					set_local_variable = {
						name = eqfm_current_adjustment_var
						value = eqfm_movement_activism_multiplier_value
					}
					add_modifier = {
						name = modifier_eqfm_unrest_phase_movement

					}
					set_variable = {
						name = eqfm_previous_adjustment_var
						value = local_var:eqfm_current_adjustment_var
					}
				}
			}
		}
		else_if = {
			limit = {
				var:eqfm_balkans_phase_var = flag:eqfm_hostility
			}
			c:TUR ?= {
				add_modifier = modifier_eqfm_hostility_phase
				every_political_movement = {
					limit = {
						eqfm_eligible_minority_movement_trigger = yes
					}
					set_local_variable = {
						name = eqfm_current_adjustment_var
						value = eqfm_movement_activism_multiplier_value
					}
					add_modifier = {
						name = modifier_eqfm_hostility_phase_movement
						multiplier = local_var:eqfm_current_adjustment_var
					}
					set_variable = {
						name = eqfm_previous_adjustment_var
						value = local_var:eqfm_current_adjustment_var
					}
				}
			}
		}
		else_if = {
			limit = {
				var:eqfm_balkans_phase_var = flag:eqfm_deconfliction
			}
			c:TUR ?= {
				add_modifier = modifier_eqfm_deconfliction_phase
				every_political_movement = {
					limit = {
						eqfm_eligible_minority_movement_trigger = yes
					}
					set_local_variable = {
						name = eqfm_current_adjustment_var
						value = eqfm_movement_activism_multiplier_value
					}
					add_modifier = {
						name = modifier_eqfm_deconfliction_phase_movement
						multiplier = local_var:eqfm_current_adjustment_var
					}
					set_variable = {
						name = eqfm_previous_adjustment_var
						value = local_var:eqfm_current_adjustment_var
					}
				}
			}
		}
		else_if = {

eqfm_update_riot_variables = {
	global_var:eqfm_balkan_situation = {
		remove_variable = eqfm_riot_state_1_var
		remove_variable = eqfm_riot_state_2_var
		remove_variable = eqfm_riot_state_3_var
		# Try to find state 1 and set variable
		c:TUR ?= {
			random_scope_state = {
				limit = {
					has_modifier = modifier_eqfm_race_riots
				}
				save_scope_as = eqfm_riot_state_1_scope
			}
		}
		if = {
			limit = {
				exists = scope:eqfm_riot_state_1_scope
			}
			set_variable = {
				name = eqfm_riot_state_1_var
				value = scope:eqfm_riot_state_1_scope
			}
			# Try to find state 2 and set variable
			c:TUR ?= {
				random_scope_state = {
					limit = {
						NOT = {
							scope:eqfm_riot_state_1_scope ?= this
						}
						has_modifier = modifier_eqfm_race_riots
					}
					save_scope_as = eqfm_riot_state_2_scope
				}
			}
			if = {
				limit = {
					exists = scope:eqfm_riot_state_2_scope
				}
				set_variable = {
					name = eqfm_riot_state_2_var
					value = scope:eqfm_riot_state_2_scope
				}
				# Try to find state 3 and set variable
				c:TUR ?= {
					random_scope_state = {
						limit = {
							NOR = {
								scope:eqfm_riot_state_1_scope ?= this
								scope:eqfm_riot_state_2_scope ?= this
							}
							has_modifier = modifier_eqfm_race_riots
						}
						save_scope_as = eqfm_riot_state_3_scope
					}
				}
				if = {
					limit = {
						exists = scope:eqfm_riot_state_3_scope
					}
					set_variable = {
						name = eqfm_riot_state_3_var
						value = scope:eqfm_riot_state_3_scope
					}
				}
			}
		}
	}
}

eqfm_tension_cooldown_effect = {
	if = {
		limit = {
			global_var:eqfm_balkan_situation = {
				OR = {
					var:eqfm_balkans_phase_var ?= flag:eqfm_stable
					var:eqfm_balkans_phase_var ?= flag:eqfm_unrest
					var:eqfm_balkans_phase_var ?= flag:eqfm_deconfliction
				}
			}
		}
		set_global_variable = {
			name = eqfm_balkans_cooldown_var
			years = 2
		}
	}
	if = {
		limit = {
			global_var:eqfm_balkan_situation = {
				OR = {
					var:eqfm_balkans_phase_var ?= flag:eqfm_stable
					var:eqfm_balkans_phase_var ?= flag:eqfm_unrest
					var:eqfm_balkans_phase_var ?= flag:eqfm_deconfliction
				}
			}
			
		}
		set_global_variable = {
			name = eqfm_balkans_cooldown_var
			years = 1
		}
	}
	else_if = {
		limit = {
			global_var:eqfm_balkan_situation = {
				OR = {
					var:eqfm_balkans_phase_var ?= flag:eqfm_hostility
					var:eqfm_balkans_phase_var ?= flag:eqfm_crisis
				}
			}
		}
		set_global_variable = {
			name = eqfm_balkans_cooldown_var
			months = 3
		}
	}
