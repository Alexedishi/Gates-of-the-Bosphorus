## General Situation Effects
turkfm_initialize_balkan_situation_effect = {
	set_global_variable = {
		name = eqfm_balkans_phase_var
		value = flag:eqfm_stable # We start in the stable phase
	}
	set_global_variable = {
		name = eqfm_escalation_var
		value = 0
	}
	set_global_variable = {
		name = eqfm_deescalation_var
		value = 0
	}
	set_global_variable = {
		name = eqfm_riot_counter_var
		value = 0
	}
	set_global_variable = {
		name = eqfm_crisis_counter_var
		value = 0
	}
	add_modifier = modifier_eqfm_stable_phase
}

eqfm_initialize_eastern_question_effect = {
	if = {
		limit = {
			has_global_variable = turkfm_balkans_situation_var
			exists = global_var:turkfm_balkans_situation_var
		}
		random_in_global_list = {
			variable = com_situation_cache
			limit = {
				var:com_id ?= flag:eqfm_balkan_situation
			}
			save_scope_as = com_situation
		}
		scope:journal_entry = {
			set_variable = {
				name = com_situation
				value = scope:com_situation
			}
		}
	}
	else = {
		create_com_situation = {
			situation_id = eqfm_balkan_situation
			icon = gbbf_eastern_question

			side_leader_country_left = c:TUR
			side_name_left = je_eqfm_eastern_question_porte
			side_desc_left = je_eqfm_eastern_question_porte_desc

			side_leader_country_right = c:RUS
			side_name_right = je_eqfm_eastern_question_balkans
			side_desc_right = je_eqfm_eastern_question_balkans_desc
		}
		set_com_situation_header = {
			text = je_eqfm_eastern_question_header
		}

		# Create Porte faction
		create_com_situation_faction = {
			side = left
			name = je_eqfm_eastern_question_faction_porte
			desc = je_eqfm_eastern_question_faction_porte_desc
		}
		add_com_situation_faction_country = {
			side = left
			name = je_eqfm_eastern_question_faction_porte
			country = c:GBR
		}
		add_com_situation_faction_country = {
			side = left
			name = je_eqfm_eastern_question_faction_porte
			country = c:FRA
		}

		# Create Balkans Faction
		create_com_situation_faction = {
			side = right
			name = je_eqfm_eastern_question_faction_balkans
			desc = je_eqfm_eastern_question_faction_balkans_desc
		}
		add_com_situation_faction_country = {
			side = right
			name = je_eqfm_eastern_question_faction_balkans
			country = c:MON
		}
		add_com_situation_faction_country = {
			side = right
			name = je_eqfm_eastern_question_faction_balkans
			country = c:SER
		}
		add_com_situation_faction_country = {
			side = right
			name = je_eqfm_eastern_question_faction_balkans
			country = c:MOL
		}
		add_com_situation_faction_country = {
			side = right
			name = je_eqfm_eastern_question_faction_balkans
			country = c:WAL
		}
		add_com_situation_faction_country = {
			side = right
			name = je_eqfm_eastern_question_faction_balkans
			country = c:BOS
		}
		add_com_situation_faction_country = {
			side = right
			name = je_eqfm_eastern_question_faction_balkans

		}
		add_com_situation_faction_country = {
			side = right
			name = je_eqfm_eastern_question_faction_balkans

eqfm_update_balkan_situation_lists_effect = {
	if = {
		limit = {
			NOT = {
				has_global_variable = turkfm_faction_lists_cooldown
			}
		}
	}
	else = {
		set_global_variable = {
			name = turkfm_faction_lists_cooldown
			days = 20
		}
		clear_global_variable_list = eqfm_balkan_situation_porte_side_list
		clear_global_variable_list = eqfm_balkan_situation_balkan_side_list
		every_country = {
			limit = {
				eqfm_balkan_situation_porte_side_trigger = yes
			}
			add_to_variable_list = {
				name = eqfm_balkan_situation_porte_side_list
				target = this
			}
		}
		every_country = {
			limit = {
				eqfm_balkan_situation_balkan_side_trigger = yes
			}
			add_to_variable_list = {
				name = eqfm_balkan_situation_balkan_side_list
				target = this
			}
		}
		sort_global_variable_list = {
			name = eqfm_balkan_situation_porte_side_list
			order = prestige
		}
		sort_global_variable_list = {
			name = eqfm_balkan_situation_balkan_side_list
			order = prestige
		}
	}
}

turkfm_update_balkan_situation_factions_effect = {
	# Starting from any country as root
	save_com_situation_scope = {
		situation_id = eqfm_balkan_situation
	}

	# Ensure lists are up to date and sorted
	eqfm_update_balkan_situation_lists_effect = yes

	clear_com_situation_factions_all = yes

	create_com_situation_faction = {
		side = left
		name = je_eqfm_eastern_question_faction_porte
		desc = je_eqfm_eastern_question_faction_porte_desc

			side = left

	ordered_in_global_list = {

			side = right
			leader = this
		}
	}

	# Update Balkan Side Faction Members
	create_com_situation_faction = {
		side = right
		name = je_eqfm_eastern_question_faction_balkans
		desc = je_eqfm_eastern_question_faction_balkans_desc
	}
	every_in_global_list = {
		variable = je_eqfm_eastern_question_faction_balkans
		add_com_situation_faction_country = {
			side = left
			name = je_eqfm_eastern_question_faction_porte

		}
	}
}

eqfm_check_balkan_situation_phase_effect = {
	if = {
		limit = {
			OR = {
				global_var:eqfm_escalation_var >= 500
				global_var:eqfm_deescalation_var >= 500
			}
		}

		## Attempt to Deescalate First
		if = {
			limit = {
				eqfm_can_deescalate_tension_trigger = yes
				global_var:eqfm_deescalation_var >= 500
			}
			if = { # From Unrest to Deconfliction
				limit = {
					global_var:eqfm_balkans_phase_var ?= flag:eqfm_unrest
				}
				set_global_variable = {
					name = eqfm_balkans_phase_var
					value = flag:eqfm_deconfliction
				}
				every_country = {
					limit = {
						eqfm_balkans_party_trigger = yes
					}
					post_notification = notification_eqfm_transition_deconfliction
				}
				custom_tooltip = eqfm_transition_deconfliction_tt
			}
			else_if = { # From Hostility to Deconfliction
				limit = {
					global_var:eqfm_balkans_phase_var ?= flag:eqfm_hostility
				}
				set_global_variable = {
					name = eqfm_balkans_phase_var
					value = flag:eqfm_deconfliction
				}
				every_country = {
					limit = {
						eqfm_balkans_party_trigger = yes
					}
					post_notification = notification_eqfm_transition_deconfliction
				}
				custom_tooltip = eqfm_transition_deconfliction_tt
			}
			else_if = { # From Crisis to Deconfliction
				limit = {
					global_var:eqfm_balkans_phase_var ?= flag:eqfm_crisis
				}
				set_global_variable = {
					name = eqfm_balkans_phase_var
					value = flag:eqfm_deconfliction
				}
				c:TUR = {
					trigger_event = {
						id = eqfm_balkans.029 # Order Returns to Rumelia
					}
				}
				every_country = {
					limit = {
						eqfm_balkans_party_trigger = yes
					}
					post_notification = notification_eqfm_transition_deconfliction
				}
				custom_tooltip = eqfm_transition_deconfliction_tt
			}
			set_global_variable = {
				name = eqfm_escalation_var
				value = 0
			}
			set_global_variable = {
				name = eqfm_deescalation_var
				value = 0
			}
		}

		## Ignore this if we deescalated
		else_if = { # Ignore this if we deescalated
			limit = {
				eqfm_can_escalate_tension_trigger = yes
				global_var:eqfm_escalation_var >= 500
			}
			if = { # From Stable to Unrest
				limit = {
					global_var:eqfm_balkans_phase_var ?= flag:eqfm_stable
				}
				set_global_variable = {
					name = eqfm_balkans_phase_var
					value = flag:eqfm_unrest
				}
				every_country = {
					limit = {
						eqfm_balkans_party_trigger = yes
					}
					post_notification = notification_eqfm_transition_unrest
				}
				custom_tooltip = eqfm_transition_unrest_tt
			}
			else_if = { # From Unrest to Hostility
				limit = {
					global_var:eqfm_balkans_phase_var ?= flag:eqfm_unrest
				}
				set_global_variable = {
					name = eqfm_balkans_phase_var
					value = flag:eqfm_hostility
				}
				every_country = {
					limit = {
						eqfm_balkans_party_trigger = yes
					}
					post_notification = notification_eqfm_transition_hostility
				}
				custom_tooltip = eqfm_transition_hostility_tt
			}
			else_if = { # From Hostility to Crisis
				limit = {
					eqfm_crisis_possible_trigger = yes
					global_var:eqfm_balkans_phase_var ?= flag:eqfm_hostility
				}
				set_global_variable = {
					name = eqfm_balkans_phase_var
					value = flag:eqfm_crisis
				}
				c:TUR ?= {
					trigger_event = {
						id = eqfm_balkans.020 # A Crisis in Rumelia
					}
					add_journal_entry = {
						type = je_eqfm_balkan_situation_crisis
					}
				}
				every_country = {
					limit = {
						eqfm_balkans_party_trigger = yes
					}
					post_notification = notification_eqfm_transition_crisis
				}
				every_country = {
					limit = {
						has_journal_entry = je_eqfm_eastern_question
					}
					trigger_event = {
						id = eqfm_balkans.021 # A Crisis in Rumelia
					}
				}
				custom_tooltip = eqfm_transition_crisis_tt
			}
			else_if = { # From Deconfliction to Stable
				limit = {
					global_var:eqfm_balkans_phase_var ?= flag:eqfm_deconfliction
				}
				set_global_variable = {
					name = eqfm_balkans_phase_var
					value = flag:eqfm_stable
				}
				every_country = {
					limit = {
						eqfm_balkans_party_trigger = yes
					}
					post_notification = notification_eqfm_transition_stable
				}
				custom_tooltip = eqfm_transition_stable_tt
			}
			set_global_variable = {
				name = eqfm_escalation_var
				value = 0
			}
			set_global_variable = {
				name = eqfm_deescalation_var
				value = 0
			}
		}
	}
	turkfm_check_modifiers_effect = yes
}

turkfm_check_modifiers_effect = {
	hidden_effect = {
		remove_modifier = modifier_eqfm_stable_phase
		remove_modifier = modifier_eqfm_unrest_phase
		remove_modifier = modifier_eqfm_hostility_phase
		remove_modifier = modifier_eqfm_deconfliction_phase
		remove_modifier = modifier_eqfm_crisis_phase
		every_political_movement = {
			remove_modifier = modifier_eqfm_stable_phase_movement
			remove_modifier = modifier_eqfm_unrest_phase_movement
			remove_modifier = modifier_eqfm_hostility_phase_movement
			remove_modifier = modifier_eqfm_deconfliction_phase_movement
			remove_modifier = modifier_eqfm_crisis_phase_movement
		}
		if = {
			limit = {
				global_var:eqfm_balkans_phase_var = flag:eqfm_stable
			}
			add_modifier = modifier_eqfm_stable_phase
			every_political_movement = {
				limit = {
					eqfm_eligible_minority_movement_trigger = yes
				}
				set_local_variable = {
					name = eqfm_current_adjustment_var
					value = eqfm_movement_activism_multiplier_value
				}

					name = modifier_eqfm_stable_phase_movement

					name = eqfm_previous_adjustment_var

					value = eqfm_movement_activism_multiplier_value

					name = eqfm_previous_adjustment_var

			}
			add_modifier = modifier_eqfm_deconfliction_phase

eqfm_tension_cooldown_effect = {
	if = {
		limit = {
			OR = {
				global_var:eqfm_balkans_phase_var ?= flag:eqfm_stable
				global_var:eqfm_balkans_phase_var ?= flag:eqfm_unrest
				global_var:eqfm_balkans_phase_var ?= flag:eqfm_deconfliction
			}
		}
		set_global_variable = {
			name = eqfm_balkans_cooldown_var
			years = 2
		}
	}
	if = {
		limit = {
			OR = {
				global_var:eqfm_balkans_phase_var ?= flag:eqfm_stable
				global_var:eqfm_balkans_phase_var ?= flag:eqfm_unrest
				global_var:eqfm_balkans_phase_var ?= flag:eqfm_deconfliction
			}
		}
		set_global_variable = {
			name = eqfm_balkans_cooldown_var
			years = 1
		}
	}
	else_if = {
		limit = {
			OR = {
				global_var:eqfm_balkans_phase_var ?= flag:eqfm_hostility

eqfm_update_riot_variables_effect = {
	# Step 1, clear all variables
	remove_variable = eqfm_riot_state_1_var
	remove_variable = eqfm_riot_state_1_progress_var
	remove_variable = eqfm_riot_state_2_var
	remove_variable = eqfm_riot_state_2_progress_var
	remove_variable = eqfm_riot_state_3_var
	remove_variable = eqfm_riot_state_3_progress_var

	# Step 2, Recapture states and set appropriate progress variables
	random_scope_state = {
		limit = {
			has_modifier = modifier_eqfm_race_riots
		}
		save_scope_as = eqfm_riot_state_1_scope
	}
	set_variable = {
		name = eqfm_riot_state_1_var
		value = scope:eqfm_riot_state_1_scope
	}
	set_variable = {
		name = eqfm_riot_state_1_progress_var
		value = eqfm_riot_1_progress_value
	}
	if = {
		limit = {
			exists = scope:eqfm_riot_state_1_scope
		}
		random_scope_state = {
			limit = {
				has_modifier = modifier_eqfm_race_riots
				NOT = {
					scope:eqfm_riot_state_1_scope ?= this
				}
			}
			save_scope_as = eqfm_riot_state_2_scope
		}
		if = {
			limit = {
				exists = scope:eqfm_riot_state_2_scope
			}
			set_variable = {
				name = eqfm_riot_state_2_var
				value = scope:eqfm_riot_state_2_scope
			}
			set_variable = {
				name = eqfm_riot_state_2_progress_var
				value = eqfm_riot_2_progress_value
			}
			random_scope_state = {
				limit = {
					has_modifier = modifier_eqfm_race_riots
					NOR = {
						scope:eqfm_riot_state_1_scope ?= this
						scope:eqfm_riot_state_2_scope ?= this
					}
				}
				save_scope_as = eqfm_riot_state_3_scope
			}
			if = {
				limit = {
					exists = scope:eqfm_riot_state_2_scope
				}
				set_variable = {
					name = eqfm_riot_state_3_var
					value = scope:eqfm_riot_state_3_scope
				}
				set_variable = {
					name = eqfm_riot_state_3_progress_var
					value = eqfm_riot_3_progress_value
				}
			}
		}
	}
	set_variable = {
		name = eqfm_riot_progress_var
		value = eqfm_riot_progress_value
	}
}
