## Bavarocracy Scripted Effects ##
grefm_initialize_bavarocracy_variables_effect = {
	set_variable = grefm_bavarocracy
	set_variable = {
		name = grefm_bavarocracy_var
		value = 2000
	}
	set_variable = {
		name = grefm_bavarocracy_support_var
		value = 0
	}
	set_variable = {
		name = grefm_bavarocracy_opposition_var
		value = 0
	}
	set_variable = {
		name = grefm_bavarocracy_progress_var
		value = 0
	}
	set_variable = {
		name = grefm_bavarocracy_instability_var
		value = 0
	}
	set_variable = {
		name = grefm_bavarocracy_instability_progress_var
		value = 0
	}
	if = {
		limit = {
			has_dlc_feature = ip3_content # National Awakening
		}
		set_variable = {
			name = grefm_bavarocracy_var# Functionality is reversed in the context of IP3
			value = 0
		}
		set_variable = no_heirs
	}
	set_variable = grefm_bavarocracy_started # This specifically tracks GRE
}

grefm_calculate_bavarocracy_variables_effect = {
	if = {
		limit = {
			has_journal_entry = je_grefm_bavarocracy_main
		}
		set_variable = {
			name = grefm_bavarocracy_support_var
			value = grefm_bavarocracy_support_value
		}
		set_variable = {
			name = grefm_bavarocracy_opposition_var
			value = grefm_bavarocracy_opposition_value
		}
		set_variable = {
			name = grefm_bavarocracy_progress_var
			value = grefm_bavarocracy_progress_value
		}
		if = {
			limit = {
				grefm_can_accrue_instability_trigger = yes
			}
			set_variable = {
				name = grefm_bavarocracy_instability_progress_var
				value = grefm_bavarocracy_instability_progress_value
			}
		}
		else = {
			set_variable = {
				name = grefm_bavarocracy_instability_progress_var
				value = 0
			}
			set_variable = {
				name = grefm_bavarocracy_instability_var
				value = 0
			}
		}
		clamp_variable = {
			name = grefm_bavarocracy_var
			max = 2000
			min = 0
		}
		clamp_variable = {
			name = grefm_bavarocracy_instability_var
			max = 200
			min = 0
		}
		je:je_grefm_bavarocracy_main = {
			set_bar_progress = {
				name = grefm_bavarocracy_main_progressbar
				value = owner.var:grefm_bavarocracy_var
			}
			set_bar_progress = {
				name = grefm_bavarocracy_instability_progressbar
				value = owner.var:grefm_bavarocracy_instability_var
			}
		}
	}
	else_if = {
		limit = {
			has_journal_entry = je_grefm_bavarocracy_ip3
		}
		set_variable = {
			name = grefm_bavarocracy_instability_progress_var
			value = grefm_bavarocractic_ire_progress_value
		}
		clamp_variable = {
			name = grefm_bavarocracy_instability_var
			max = 100
			min = 0
		}
		clamp_variable = {
			name = grefm_bavarocracy_var
			max = 50
			min = 0
		}
		je:je_grefm_bavarocracy_ip3 = {
			set_bar_progress = {
				name = grefm_bavarocracy_ip3_progressbar
				value = owner.var:grefm_bavarocracy_var
			}
		}
		je:je_grefm_bavarocracy_ip3 = {
			set_bar_progress = {
				name = grefm_bavarocracy_ip3_instability_progressbar
				value = owner.var:grefm_bavarocracy_instability_var
			}
		}
	}
}

grefm_update_bavarian_auxiliaries_effect = {
	if = {
		limit = {
			has_modifier = modifier_grefm_bavarian_auxiliaries
		}
		remove_modifier = modifier_grefm_bavarian_auxiliaries
		remove_modifier = modifier_grefm_funding_bavarian_auxiliaries
		if = {
			limit = {
				NOT = {
					has_variable = grefm_dismissed_bavarian_auxiliaries
				}
			}
			add_modifier = modifier_grefm_bavarian_auxiliaries
			add_modifier = {
				name = modifier_grefm_funding_bavarian_auxiliaries
				multiplier = money_amount_multiplier_medium
			}
		}
	}
}

grefm_expel_bavarocracy_effect = {
	custom_tooltip = {
		text = grefm_expel_bavarocracy

			limit = {
				OR = {
					has_role_of_type = ruler
					has_role_of_type = heir

## Goudi Coup Effects ##
grefm_initialize_military_league_effect = {
	set_variable = {
		name = coup_event_active
		months = 24
	}
	create_character = {
		template = template_grefm_zorbas_nikolas
		on_created = {
			add_character_role = general
			place_character_in_void = 100
		}
	}
	scope:journal_entry = {
		set_variable = {
			name = grefm_goudi_coup_var
			value = 50
		}
		random_character_in_void = {
			limit = {
				has_template = template_grefm_zorbas_nikolas
			}
			save_scope_as = grefm_goudi_leader_scope
			save_scope_as = golpista_general
			interest_group = {
				save_scope_as = grefm_goudi_ig_scope
				save_scope_as = golpista_ig
			}
		}
		add_to_variable_list = {
			name = com_journal_characters
			target = scope:grefm_goudi_leader_scope
		}
	}
}

## Venizelism / Metaxism Effects ##
grefm_add_venizelism_agenda = {
	if = {
		limit = {
			has_journal_entry = je_grefm_venizelism
		}
		if = {
			limit = {
				NOR = {
					has_variable = grefm_venizelism_$type$_agenda
					has_variable = grefm_venizelism_$type$_complete
				}
			}
			custom_tooltip = {
				text = grefm_add_venizelism_$type$_agenda
				set_variable = grefm_venizelism_$type$_agenda
			}
		}
		else = {
			error_log = "grefm_add_venizelism_agenda was called, but the specified agenda is either active or complete."
		}
	}
	else = {
		error_log = "grefm_add_venizelism_agenda was called, but the calling country does not have an active je_grefm_venizelism journal entry."
	}
	set_variable = {
		name = grefm_total_venizelism_agendas_var
		value = grefm_total_venizelism_agendas_value
	}
	set_variable = {
		name = grefm_completed_venizelism_agendas_var
		value = grefm_completed_venizelism_agendas_value
	}
}
