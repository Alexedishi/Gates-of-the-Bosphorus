gbbf_rehost_race_effect = {
	set_local_variable = {
		name = days_param
		value = $days$
	}
	custom_tooltip = {
		text = gbbf_rehost_race_effect
		trigger_event = {
			on_action = grefm_rehost_chariot_race
			days = $days$
		}
	}
}
