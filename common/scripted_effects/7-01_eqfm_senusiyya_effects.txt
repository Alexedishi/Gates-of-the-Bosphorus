mgrbfm_update_senussi_effect = {
	## Step 1, Make Sure Senussi Exists and Recapture/Regenerate as appropriate
	if = {
		limit = {
			any_character_in_void = {
				mgrbfm_character_is_senussi_trigger = yes
			}
		}
		random_character_in_void = {
			limit = {
				mgrbfm_character_is_senussi_trigger = yes
			}
			save_scope_as = mgrbfm_senussi_character_scope
		}
		set_variable = {
			name = mgrbfm_senussi_character_var
			value = scope:mgrbfm_senussi_character_scope
		}
	}
	else = {
		if = {
			limit = {
				global_var:mgrbfm_senussi_counter_var >= 5
			}
			create_character = {
				template = template_mgrbfm_senussi_dynamic
				on_created = {
					add_trait = gbbf_trait_marabout
					place_character_in_void = 240
				}
			}
		}
		else_if = {
			limit = {
				global_var:mgrbfm_senussi_counter_var = 4
			}
			create_character = {
				template = template_mgrbfm_senussi_hasan
				on_created = {
					add_trait = gbbf_trait_marabout
					place_character_in_void = 240
				}
			}
		}
		else_if = {
			limit = {
				global_var:mgrbfm_senussi_counter_var = 3
			}
			create_character = {
				template = template_mgrbfm_senussi_idris
				on_created = {
					add_trait = gbbf_trait_marabout
					place_character_in_void = 240
				}
			}
		}
		else_if = {
			limit = {
				global_var:mgrbfm_senussi_counter_var = 2
			}
			create_character = {
				template = template_mgrbfm_senussi_ahmed
				on_created = {
					add_trait = gbbf_trait_marabout
					place_character_in_void = 240
				}
			}
		}
		else_if = {
			limit = {
				global_var:mgrbfm_senussi_counter_var = 1
			}
			create_character = {
				template = template_mgrbfm_senussi_muhammad_ii
				on_created = {
					add_trait = gbbf_trait_marabout
					place_character_in_void = 240
				}
			}
		}
		else = {
			create_character = {
				template = template_mgrbfm_senussi_muhammad_i
				on_created = {
					add_trait = gbbf_trait_marabout
					place_character_in_void = 120
				}
			}
		}
		random_character_in_void = {
			limit = {
				mgrbfm_character_is_senussi_trigger = yes
			}
			save_scope_as = mgrbfm_senussi_character_scope
		}
		set_variable = {
			name = mgrbfm_senussi_character_var
			value = scope:mgrbfm_senussi_character_scope
		}
		change_global_variable = {
			name = mgrbfm_senussi_counter_var
			add = 1
		}
	}

	## Step 2, Update the journal list
	if = {
		limit = {
			has_journal_entry = je_mgrbfm_senusiyya_main
		}
		je:je_mgrbfm_senusiyya_main = {
			clear_variable_list = com_journal_characters
			add_to_variable_list = {
				name = com_journal_characters
				target = scope:mgrbfm_senussi_character_scope
			}
		}
	}
	else_if = {
		limit = {
			has_journal_entry = je_mgrbfm_senusiyya_sub
		}
		je:je_mgrbfm_senusiyya_sub = {
			clear_variable_list = com_journal_characters
			add_to_variable_list = {
				name = com_journal_characters
				target = scope:mgrbfm_senussi_character_scope
			}
		}
	}

	## Step 3, Figure out where Senussi should be located & update flags
	if = {
		limit = {
			var:mgrbfm_senussi_var >= 99
		}
		if = {
			limit = {
				var:mgrbfm_senussi_location_var != flag:mgrbfm_senussi_court
			}
			post_notification = notification_mgrbfm_senussi_entrenched
		}
		set_variable = {
			name = mgrbfm_senussi_location_var
			value = flag:mgrbfm_senussi_court
		}
	}
	else_if = {
		limit = {
			var:mgrbfm_senussi_var >= 75
		}
		if = {
			limit = {
				OR = {
					var:mgrbfm_senussi_location_var = flag:mgrbfm_senussi_ghosted
					var:mgrbfm_senussi_location_var = flag:mgrbfm_senussi_desert
					var:mgrbfm_senussi_location_var = flag:mgrbfm_senussi_villages
				}
			}
			post_notification = notification_mgrbfm_senussi_urbanized_up
		}
		else_if = {
			limit = {
				var:mgrbfm_senussi_location_var = flag:mgrbfm_senussi_court
			}
			post_notification = notification_mgrbfm_senussi_urbanized_down
		}
		set_variable = {
			name = mgrbfm_senussi_location_var
			value = flag:mgrbfm_senussi_urban
		}
	}
	else_if = {
		limit = {
			var:mgrbfm_senussi_var < 75
			var:mgrbfm_senussi_var >= 50
		}
		if = {
			limit = {
				OR = {
					var:mgrbfm_senussi_location_var = flag:mgrbfm_senussi_court
					var:mgrbfm_senussi_location_var = flag:mgrbfm_senussi_urban
				}
			}
			post_notification = notification_mgrbfm_senussi_villages_down
		}
		if = {
			limit = {
				OR = {
					var:mgrbfm_senussi_location_var = flag:mgrbfm_senussi_ghosted
					var:mgrbfm_senussi_location_var = flag:mgrbfm_senussi_desert
				}

			}
			post_notification = notification_mgrbfm_senussi_villages_up
		}
		set_variable = {
			name = mgrbfm_senussi_location_var
			value = flag:mgrbfm_senussi_villages
		}
	}
	else_if = {
		limit = {
			var:mgrbfm_senussi_var < 50
		}
		if = {
			limit = {
				var:mgrbfm_senussi_location_var = flag:mgrbfm_senussi_ghosted
			}
			post_notification = notification_mgrbfm_senussi_desert_up
		}
		else_if = {
			limit = {
				OR = {
					var:mgrbfm_senussi_location_var = flag:mgrbfm_senussi_court
					var:mgrbfm_senussi_location_var = flag:mgrbfm_senussi_urban
					var:mgrbfm_senussi_location_var = flag:mgrbfm_senussi_villages
				}
			}
			post_notification = notification_mgrbfm_senussi_desert_down
		}
		set_variable = {
			name = mgrbfm_senussi_location_var
			value = flag:mgrbfm_senussi_desert
		}
	}
	else = {
		if = {
			limit = {
				var:mgrbfm_senussi_location_var != flag:mgrbfm_senussi_ghosted
			}
			post_notification = notification_mgrbfm_senussi_ghosted
		}
		set_variable = {
			name = mgrbfm_senussi_location_var
			value = flag:mgrbfm_senussi_ghosted
		}
	}
	scope:mgrbfm_senussi_character_scope = {
		set_variable = {
			name = com_opinion
			value = root.var:mgrbfm_senussi_location_var
		}
	}

}

eqfm_validate_senusiyya_involvement_effect = {
	every_country = {
		limit = {
			OR = {
				AND = {
					NOT = {
						is_involved_in_journal_entry = je_eqfm_senusiyya
					}
					eqfm_senusiyya_involvement_trigger = yes
				}
				AND = {
					is_involved_in_journal_entry = je_eqfm_senusiyya
					eqfm_senusiyya_involvement_trigger = no
				}
			}
		}
		if = {
			limit = {
				NOT = {
					is_involved_in_journal_entry = je_eqfm_senusiyya
				}
				eqfm_senusiyya_involvement_trigger = yes
			}
			global_var:je_eqfm_senusiyya = {
				add_involved_country = prev
			}
		}
		else_if = {
			limit = {
				is_involved_in_journal_entry = je_eqfm_senusiyya
				eqfm_senusiyya_involvement_trigger = no
			}
			global_var:je_eqfm_senusiyya = {
				remove_involved_country = prev
			}
		}
	}
}
