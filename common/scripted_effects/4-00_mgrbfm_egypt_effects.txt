mgrbfm_nile_inundation_effect = {
	if = {
		limit = {
			global_var:mgrbfm_nile_flood_cooldown <= 0
		}
		random_list = {
			1 = {
				set_global_variable = {
					name = mgrbfm_nile_flood_var
					value = 1
				}
			}
			1 = {
				set_global_variable = {
					name = mgrbfm_nile_flood_var
					value = 2
				}
			}
			1 = {
				set_global_variable = {
					name = mgrbfm_nile_flood_var
					value = 3
				}
			}
			1 = {
				set_global_variable = {
					name = mgrbfm_nile_flood_var
					value = 4
				}
			}
			1 = {
				set_global_variable = {
					name = mgrbfm_nile_flood_var
					value = 5
				}
			}
			1 = {
				set_global_variable = {
					name = mgrbfm_nile_flood_var
					value = 6
				}
			}
			1 = {
				set_global_variable = {
					name = mgrbfm_nile_flood_var
					value = 7
				}
			}
			1 = {
				set_global_variable = {
					name = mgrbfm_nile_flood_var
					value = 8
				}
			}
			1 = {
				set_global_variable = {
					name = mgrbfm_nile_flood_var
					value = 9
				}
			}
			1 = {
				set_global_variable = {
					name = mgrbfm_nile_flood_var
					value = 10
				}
			}
		}
		set_global_variable = {
			name = mgrbfm_nile_flood_var
			value = {
				add = global_var:mgrbfm_nile_flood_var
				subtract = global_var:mgrbfm_nile_control_var
				max = 10
				min = 0
			}
		}
		set_global_variable = {
			name = mgrbfm_nile_flood_cooldown
			value = {
				add = global_var:mgrbfm_nile_flood_var
				divide = 2
				round = yes
				min = 0
			}
		}
		random_state_in_lower_nile = {
			limit = {
				state_region = {
					NOT = {
						has_harvest_condition = flood
					}
				}
			}
			save_scope_as = mgrbfm_flood_state
		}
		every_country = {
			limit = {
				NOT = {
					is_country_type = decentralized
				}
				any_scope_state = {
					is_in_geographic_region = geographic_region_lower_nile
				}
			}
			trigger_event = mgrbfm_egypt.110
		}
	}
	else = {
		set_global_variable = {
			name = mgrbfm_nile_flood_cooldown
			value = {
				add = global_var:mgrbfm_nile_flood_cooldown
				subtract = 1
				min = 0
			}
		}
	}
}
