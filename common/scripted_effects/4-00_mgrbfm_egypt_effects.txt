mgrbfm_nile_inundation_effect = {
	scope:journal_entry = {
		if = {
			limit = {
				var:mgrbfm_nile_flood_cooldown <= 0
			}
			random_list = {
				1 = {
					set_variable = {
						name = mgrbfm_nile_flood_var
						value = 1
					}
				}
				1 = {
					set_variable = {
						name = mgrbfm_nile_flood_var
						value = 2
					}
				}
				1 = {
					set_variable = {
						name = mgrbfm_nile_flood_var
						value = 3
					}
				}
				1 = {
					set_variable = {
						name = mgrbfm_nile_flood_var
						value = 4
					}
				}
				1 = {
					set_variable = {
						name = mgrbfm_nile_flood_var
						value = 5
					}
				}
				1 = {
					set_variable = {
						name = mgrbfm_nile_flood_var
						value = 6
					}
				}
				1 = {
					set_variable = {
						name = mgrbfm_nile_flood_var
						value = 7
					}
				}
				1 = {
					set_variable = {
						name = mgrbfm_nile_flood_var
						value = 8
					}
				}
				1 = {
					set_variable = {
						name = mgrbfm_nile_flood_var
						value = 9
					}
				}
				1 = {
					set_variable = {
						name = mgrbfm_nile_flood_var
						value = 10
					}
				}
			}
			set_variable = {
				name = mgrbfm_nile_flood_var
				value = {
					add = var:mgrbfm_nile_flood_var
					subtract = var:mgrbfm_nile_control_var
					max = 10
					min = 0
				}
			}
			if = {
				limit = {
					var:mgrbfm_nile_flood_var > 0
				}
				set_variable = {
					name = mgrbfm_nile_flood_cooldown
					value = {
						add = var:mgrbfm_nile_flood_var
						divide = 2
						round = yes
						min = 2
					}
				}
				random_state_region_in_lower_nile = {
					limit = {
						NOT = {
							has_harvest_condition = flood
						}
					}
					save_scope_as = mgrbfm_flood_state_region_scope
					start_harvest_condition = flood
					set_variable = mgrbfm_flood_state_region_var
				}
				set_variable = {
					name = mgrbfm_flood_epicenter_var
					value = scope:mgrbfm_flood_state_region_scope
				}
				every_country = {
					limit = {
						is_involved_in_journal_entry = je_mgrbfm_egypt_nile_river
					}
					trigger_event = mgrbfm_egypt.110
				}
				scope:mgrbfm_flood_state_region_scope = {
					remove_variable = mgrbfm_flood_state_region_var
				}
			}
			else = {
				every_country = {
					limit = {
						is_involved_in_journal_entry = je_mgrbfm_egypt_nile_river
					}
					post_notification = notification_mgrbfm_floods_mitigated
				}
				set_variable = {
					name = mgrbfm_nile_flood_cooldown
					value = 3
				}
			}
		}
		else = {
			set_variable = {
				name = mgrbfm_nile_flood_cooldown
				value = {
					add = var:mgrbfm_nile_flood_cooldown
					subtract = 1
					min = 0
				}
			}
		}
	}
}
