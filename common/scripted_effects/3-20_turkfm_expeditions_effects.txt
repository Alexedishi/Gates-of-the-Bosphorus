## The Black Mountain Effects ##
turkfm_initialize_black_mountain_effect = {
	# This is modified from je_montenegrin_raiding's immediate block
	# Vanilla variables are retained for compatibility
	# This effect is fired from global if IP3 is active
	set_variable = {
		name = raiding_intensity
		value = 50
	}
	set_variable = {
		name = last_week_raiding_income
		value = 1181
	}
	create_diplomatic_pact = {
		country = c:TUR
		type = raiding_pact
	}
	create_diplomatic_pact = {
		country = c:RUS
		type = disapproval_pact
	}
	add_modifier = {
		name = modifier_montenegrin_raiding
		multiplier = 1181
	}
	add_modifier = {
		name = modifier_montenegrin_disobedience
		multiplier = 50
	}
	random_neighbouring_state = {
		limit = {
			owner ?= c:TUR
		}
		save_scope_as = raid_target
		add_modifier = {
			name = raid_target
			multiplier = 1
		}
	}
	set_global_variable = {
		name = mon_raid_state
		value = scope:raid_target
	}
}

turkfm_weekly_black_mountain_effect = {
	# Ok look bruv, I don't like this either. You got a problem, email the complaint line. A pipebomb will be delivered to your mailbox within 7 to 12 Bosnian business days.
	if = {
		limit = {
			c:MON ?= this
		}
		## Reworked script from the IP3 journal
		if = {
			limit = {
				has_variable = mon_more_raiding
			}
			change_variable = {
				name = raiding_intensity
				add = 1
			}
		}
		else_if = {
			limit = {
				has_variable = mon_less_raiding
			}
			change_variable = {
				name = raiding_intensity
				subtract = 1
			}
		}
		clamp_variable = {
			name = raiding_intensity
			max = 100
			min = 0
		}
		if = {
			limit = {
				var:raiding_intensity >= 100
				has_variable = mon_more_raiding
			}
			remove_variable = mon_more_raiding
			remove_variable = turkfm_raiding_status_cooldown_var
		}
		if = {
			limit = {
				var:raiding_intensity <= 0
				has_variable = mon_less_raiding
			}
			remove_variable = mon_less_raiding
			remove_variable = turkfm_raiding_status_cooldown_var
		}
		
		## IP3 Raiding Income Effects ##
		turkfm_check_raiding_target_effect = yes
		turkfm_update_raiding_income_effect = yes
		
		## IP3 Raiding Event Script ##
		#Higher odds of ottoman Retribution Depending on Raiding Income. Note max Raiding income is 2500
		#Note that montenegrin_raiding.1 can only fire once, after which montenegrin_raiding.2 fires from then on.
		if = { #playing Russian Roulette each week
			limit = {
				raiding_income > 4200
			}
			random_list = {
				150 = { trigger_event = { id = montenegrin_raiding.1 popup = yes} } # Ottoman Attack
				100 = { trigger_event = { id = montenegrin_raiding.2 popup = yes } }
				600 = { } # nothing happens this week
			}
		}
		else_if = {
			limit = {
				raiding_income > 3500
			}
			random_list = {
				150 = { trigger_event = { id = montenegrin_raiding.1 popup = yes} } # Ottoman Attack
				100 = { trigger_event = { id = montenegrin_raiding.2 popup = yes } } # Ottoman Attack
				2000 = { } # nothing happens this week
			}
		}
		else_if = {
			limit = {
				raiding_income > 2800
			}
			random_list = {
				150 = { trigger_event = { id = montenegrin_raiding.1 popup = yes} } # Ottoman Attack
				100 = { trigger_event = { id = montenegrin_raiding.2 popup = yes } } # Ottoman Attack
				5000 = { } # nothing happens this week
			}
		}
		else_if = {
			limit = {
				raiding_income > 2000
			}
			random_list = {
				150 = { trigger_event = { id = montenegrin_raiding.1 popup = yes} } # Ottoman Attack
				100 = { trigger_event = { id = montenegrin_raiding.2 popup = yes } } # Ottoman Attack
				10000 = { } # nothing happens this week
			}
		}
		else_if = {
			limit = {
				raiding_income > 1200
			}
			random_list = {
				150 = { trigger_event = { id = montenegrin_raiding.1 popup = yes} } # Ottoman Attack
				100 = { trigger_event = { id = montenegrin_raiding.2 popup = yes } } # Ottoman Attack
				20000 = { } # nothing happens this week
			}
		}
		else = {
			random_list = {
				20000 = { } # nothing happens this week
			}
		}
	}
	if = {
		limit = {
			c:TUR ?= this
		}
	}
}

turkfm_check_raiding_target_effect = {
	scope:raid_target ?= {
		if = {
			limit = {
				NOT = {
					owner ?= c:TUR
				}
			}
			remove_modifier = raid_target
			random_neighbouring_state = {
				limit = {
					owner ?= c:TUR
				}
				save_scope_as = raid_target
				if = {
					limit = {
						raiding_level > 0
					}
					add_modifier = {
						name = raid_target
						multiplier = raiding_level
					}
				}
			}
		}
		#still state scope. add some radicals
		if = {
			limit = {
				raiding_level > 1
			}
			add_radicals_in_state = {
				value = 0.005
			}
		}
	}
}

turkfm_update_raiding_income_effect = {
	#If Raiding_income has changed, remove and reapply the modifiers with the new values. This is necessary because we do not have support for dynamic modifiers.
	if = {
		limit = {
			var:last_week_raiding_income != raiding_income
		}
		remove_modifier = modifier_montenegrin_raiding
		add_modifier = {
			name = modifier_montenegrin_raiding
			multiplier = raiding_income
		}
		remove_modifier = modifier_montenegrin_disobedience
		add_modifier = {
			name = modifier_montenegrin_disobedience
			multiplier = var:raiding_intensity
		}
		scope:journal_entry = {
			set_bar_progress = {
				value = root.var:raiding_intensity
				name = montenegrin_raiding_progress_bar
			}
		}
		set_variable = {
			name = last_week_raiding_income
			value = raiding_income
		}
		scope:raid_target ?= {
			remove_modifier = raid_target
			add_modifier = {
				name = raid_target
				multiplier = raiding_level
			}
		}
	}
}

turkfm_jack_sparrow_effect = {
	## Step 1, Check for pirate leader
	if = {
		limit = {
			any_character_in_void = {
				has_template = template_turkfm_jack_sparrow
			}

		}
		random_character_in_void = {
			limit = {
				has_template = template_turkfm_jack_sparrow
			}
			save_scope_as = turkfm_jack_sparrow_scope
		}
	}
	else = {
		create_character = {
			template = template_turkfm_jack_sparrow
			on_created = {
				place_character_in_void = 300
			}
		}
		random_character_in_void = {
			limit = {
				has_template = template_turkfm_jack_sparrow
			}
			save_scope_as = turkfm_jack_sparrow_scope
		}
		every_country = {
			limit = {
				has_journal_entry = je_turkfm_pirates_of_the_adriatic
			}
			post_notification = notification_turkfm_jack_sparrow_reincarnated
		}
	}
	set_variable = {
		name = turkfm_jack_sparrow_var
		value = scope:turkfm_jack_sparrow_scope
	}

	## Step 2, Check for ongoing expedition
	if = {
		limit = {
			any_scope_character = {
				has_modifier = modifier_turkfm_leading_adriatic_expedition
			}
		}
		random_scope_character = {
			limit = {
				has_modifier = modifier_turkfm_leading_adriatic_expedition
			}
			save_scope_as = turkfm_cutler_beckett_scope
		}
	}
	if = {
		limit = {
			exists = scope:turkfm_cutler_beckett_scope
		}
		set_variable = {
			name = turkfm_cutler_beckett_var
			value = scope:turkfm_cutler_beckett_scope
		}
	}
	else = {
		remove_variable = turkfm_cutler_beckett_var
	}

	## Step 3, Update Lists and progress bars
	je:je_turkfm_pirates_of_the_adriatic = {
		set_bar_progress = {
			name = turkfm_jack_sparrow_progress_bar
			value = global_var:turkfm_pirates_var
		}
		clear_variable_list = com_journal_characters
		add_to_variable_list = {
			name = com_journal_characters
			target = scope:turkfm_jack_sparrow_scope
		}
		if = {
			limit = {
				exists = scope:turkfm_cutler_beckett_scope
			}
			add_to_variable_list = {
				name = com_journal_characters
				target = scope:turkfm_cutler_beckett_scope
			}
		}
	}

	## Step 4, Set Opinion Flags
	scope:turkfm_jack_sparrow_scope ?= {
		set_variable = {
			name = com_opinion
			value = flag:turkfm_jack_sparrow_opinion
		}
	}
	scope:turkfm_cutler_beckett_scope ?= {
		set_variable = {
			name = com_opinion
			value = flag:turkfm_cutler_beckett_opinion
		}
	}
}
