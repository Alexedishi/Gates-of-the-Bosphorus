balkfm_initialize_dynastic_rivalry_variables_effect = {
	set_variable = {
		name = balkfm_obrenovic_support_var
		value = 0
	}
	set_variable = {
		name = balkfm_karadordevic_support_var
		value = 0
	}
	set_variable = {
		name = balkfm_obrenovic_progress_var
		value = 0
	}
	set_variable = {
		name = balkfm_karadordevic_progress_var
		value = 0
	}
}

balkfm_calculate_dynastic_rivalry_variables_effect = {
	set_variable = {
		name = balkfm_obrenovic_progress_var
		value = balkfm_obrenovic_progress_value
	}
	set_variable = {
		name = balkfm_karadordevic_progress_var
		value = balkfm_karadordevic_progress_value
	}
	scope:journal_entry = {
		set_bar_progress = {
			name = balkfm_dynastic_rivalry_progressbar
			value = owner.balkfm_leading_dynasty_value
		}
	}
}

balkfm_dynastic_rivalry_monthly_effect = {
	change_variable = {
		name = balkfm_obrenovic_support_var
		add = balkfm_obrenovic_net_progress_value
	}
	clamp_variable = {
		name = balkfm_obrenovic_support_var
		max = 500
		min = 0
	}
	change_variable = {
		name = balkfm_karadordevic_support_var
		add = balkfm_karadordevic_net_progress_value
	}
	clamp_variable = {
		name = balkfm_karadordevic_support_var
		max = 500
		min = 0
	}
	if = {
		limit = {
			NOT = {
				exists = heir
			}
			OR = {
				balkfm_has_karadordevic_ruler_trigger = yes
				balkfm_has_obrenovic_ruler_trigger = yes
			}
		}
		if = {
			limit = {
				balkfm_has_karadordevic_ruler_trigger = yes
			}
			create_character = {
				template = template_balkfm_karadordevic_dynamic_heir
			}
		}
		else_if = {
			limit = {
				balkfm_has_obrenovic_ruler_trigger = yes
			}
			create_character = {
				template = template_balkfm_obrenovic_dynamic_heir
			}
		}
	}
}

balkfm_minor_dynasty_support_loss_effect = {
	if = {
		limit = {
			balkfm_has_obrenovic_ruler_trigger = yes
		}
		custom_tooltip = {
			text = balkfm_obrenovic_minus_50
			hidden_effect = {
				change_variable = {
					name = balkfm_obrenovic_support_var
					subtract = 50
				}
			}
		}
	}
	else_if = {
		limit = {
			balkfm_has_karadordevic_ruler_trigger = yes
		}
		custom_tooltip = {
			text = balkfm_karadordevic_minus_50
			hidden_effect = {
				change_variable = {
					name = balkfm_karadordevic_support_var
					subtract = 50
				}
			}
		}
	}
	balkfm_calculate_dynastic_rivalry_variables_effect = yes
}

balkfm_cleanup_dynasty_variables_effect = {
	remove_variable = balkfm_obrenovic_support_var
	remove_variable = balkfm_karadordevic_support_var
	remove_variable = balkfm_obrenovic_progress_var
	remove_variable = balkfm_karadordevic_progress_var
}
