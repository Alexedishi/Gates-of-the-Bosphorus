balkfm_calculate_dynastic_rivalry_variables_effect = {
	set_variable = {
		name = balkfm_obrenovic_progress_var
		value = balkfm_obrenovic_progress_value
	}
	set_variable = {
		name = balkfm_karadordevic_progress_var
		value = balkfm_karadordevic_progress_value
	}
	je:je_balkfm_ser_monarchism = {
		set_bar_progress = {
			name = balkfm_dynastic_rivalry_progressbar
			value = owner.balkfm_leading_dynasty_value
		}
	}
}

balkfm_dynastic_rivalry_monthly_effect = {
	change_variable = {
		name = balkfm_obrenovic_support_var
		add = balkfm_obrenovic_net_progress_value
	}
	clamp_variable = {
		name = balkfm_obrenovic_support_var
		max = 500
		min = 0
	}
	change_variable = {
		name = balkfm_karadordevic_support_var
		add = balkfm_karadordevic_net_progress_value
	}
	clamp_variable = {
		name = balkfm_karadordevic_support_var
		max = 500
		min = 0
	}
	if = {
		limit = {
			NOT = {
				exists = heir
			}
			OR = {
				balkfm_has_karadordevic_ruler_trigger = yes
				balkfm_has_obrenovic_ruler_trigger = yes
			}
		}
		if = {
			limit = {
				balkfm_has_karadordevic_ruler_trigger = yes
			}
			create_character = {
				template = template_balkfm_karadordevic_dynamic_heir
			}
		}
		else_if = {
			limit = {
				balkfm_has_obrenovic_ruler_trigger = yes
			}
			create_character = {
				template = template_balkfm_obrenovic_dynamic_heir
			}
		}
	}
}

balkfm_update_dynasty_character_scopes_effect = {
	je:je_balkfm_ser_monarchism = {
		clear_variable_list = com_journal_characters
		remove_variable = balkfm_opposition_leader_var
	}
	if = {
		limit = {
			balkfm_has_obrenovic_ruler_trigger = yes
		}
		ruler = {
			save_scope_as = balkfm_obrenovic_scope
		}
	}
	else_if = {
		limit = {
			any_scope_character = {
				balkfm_is_obrenovic_character_trigger = yes
			}
		}
		random_scope_character = {
			limit = {
				balkfm_is_obrenovic_character_trigger = yes
			}
			save_scope_as = balkfm_obrenovic_scope
		}
	}
	else = {
		balkfm_find_obrenovic_effect = yes
	}
	if = {
		limit = {
			balkfm_has_karadordevic_ruler_trigger = yes
		}
		ruler = {
			save_scope_as = balkfm_karadordevic_scope
		}
	}
	else_if = {
		limit = {
			any_scope_character = {
				balkfm_is_karadordevic_character_trigger = yes
			}
		}
		random_scope_character = {
			limit = {
				balkfm_is_karadordevic_character_trigger = yes
			}
			save_scope_as = balkfm_karadordevic_scope
		}
	}
	else = {
		balkfm_find_karadordevic_effect = yes
	}
	balkfm_save_opposition_dynasty_scope_effect = yes
	je:je_balkfm_ser_monarchism = {
		add_to_variable_list = {
			name = com_journal_characters
			target = scope:balkfm_obrenovic_scope
		}
		add_to_variable_list = {
			name = com_journal_characters
			target = scope:balkfm_karadordevic_scope
		}
		set_variable = {
			name = balkfm_opposition_leader_var
			value = scope:balkfm_dynastic_opposition_scope
		}
	}
}

balkfm_find_obrenovic_effect = {
	if = {
		limit = {
			any_scope_character = {
				balkfm_is_obrenovic_character_trigger = yes
			}
		}
		random_character = {
			limit = {
				balkfm_is_obrenovic_character_trigger = yes
			}
			save_scope_as = balkfm_dynastic_opposition_scope
			save_scope_as = balkfm_obrenovic_scope
		}
	}
	else_if = {
		limit = {
			any_character_in_void = {
				balkfm_is_obrenovic_character_trigger = yes
			}
		}
		random_character_in_void = {
			limit = {
				balkfm_is_obrenovic_character_trigger = yes
			}
			save_scope_as = balkfm_dynastic_opposition_scope
			save_scope_as = balkfm_obrenovic_scope
		}
	}
	else = {
		if = {
			limit = {
				var:balkfm_obrenovic_counter_var ?= 1
			}
			create_character = { 
				template = template_balkfm_obrenovic_milan
				on_created = {
					place_character_in_void = 150 # characters will expire after 15 years
				}
			}
			change_variable = { 
				name = balkfm_obrenovic_counter_var
				add = 1
			}
		}
		if = {
			limit = {
				var:balkfm_obrenovic_counter_var ?= 2
			}
			create_character = {
				template = template_balkfm_obrenovic_mihailo
				on_created = {
					place_character_in_void = 150 # characters will expire after 15 years
				}
			}
			change_variable = {
				name = balkfm_obrenovic_counter_var
				add = 1
			}
		}
		if = {
			limit = {
				var:balkfm_obrenovic_counter_var ?= 3
			}
			create_character = {
				template = template_balkfm_obrenovic_milan_IV
				on_created = {
					place_character_in_void = 150 # characters will expire after 15 years
				}
			}
			change_variable = {
				name = balkfm_obrenovic_counter_var
				add = 1
			}
		}
		else = {
			create_character = {
				template = template_balkfm_obrenovic_dynamic
				on_created = {
					place_character_in_void = 150 # characters will expire after 15 years
				}
			}
			change_variable = {
				name = balkfm_obrenovic_counter_var
				add = 1
			}
		}
	}
	random_character_in_void = {
		limit = {
			balkfm_is_obrenovic_character_trigger = yes
		}
		save_scope_as = balkfm_dynastic_opposition_scope
		save_scope_as = balkfm_obrenovic_scope
	}
}
