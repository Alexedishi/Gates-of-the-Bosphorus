balkfm_validate_serbian_characters = {
	if = {
		limit = {
			has_variable = balkfm_ruling_dynasty_var
		}
		# Obrenović Validation Branch
		if = {
			limit = {
				var:balkfm_ruling_dynasty_var ?= flag:balkfm_house_obrenovic
			}
			# Ruler Validation
			if = {
				limit = {
					ruler = {
						balkfm_is_obrenovic_character_trigger = no
					}
				}
				create_character = {
					template = template_balkfm_obrenovic_dynamic
					ruler = yes
				}
				error_log = "Serbia is marked as being ruled by House Obrenović, but the ruler does not have an Obrenović template. Ruler for Serbia has been regenerated. House Obrenović rules Serbia."
			}
			# Heir Validation
			if = {
				limit = {
					exists = heir
					heir = {
						balkfm_is_obrenovic_character_trigger = no
					}
				}
				heir = {
					disinherit_character = yes
				}
				error_log = "Serbia has a ruler with an Obrenović template, but the heir does not. Heir has been liquidated."
			}
			# Heir Regeneration
			if = {
				limit = {
					NOR = {
						exists = heir
						has_variable = balkfm_constitution_pending # Blocks heir shenanigans until the initial event chain is complete
					}
				}
				if = {
					limit = {
						ruler = {
							has_template = template_balkfm_obrenovic_milosh
						}
					}
					create_character = {
						template = template_balkfm_obrenovic_milan
						heir = yes
					}
				}
				else_if = {
					limit = {
						ruler = {
							has_template = template_balkfm_obrenovic_milan
						}
					}
					create_character = {
						template = template_balkfm_obrenovic_mihailo
						heir = yes
					}
				}
				else_if = {
					limit = {
						ruler = {
							has_template = template_balkfm_obrenovic_mihailo
						}
					}
					if = {
						limit = {
							year >= 1855
						}
						create_character = {
							template = template_balkfm_obrenovic_dynamic_heir
						}
					}
					else = {
						create_character = {
							template = template_balkfm_obrenovic_milan_IV
							heir = yes
						}
					}
				}
				error_log = "Generated an Obrenović template heir for Serbia."
			}
			
		}
		# House Karadordevic Validation Branch
		else_if = {
			limit = {
				var:balkfm_ruling_dynasty_var ?= flag:balkfm_house_karadordevic
			}
			# Ruler Validation
			if = {
				limit = {
					ruler = {
						balkfm_is_karadordevic_character_trigger = no
					}
				}
				create_character = {
					template = template_balkfm_karadordevic_dynamic
					ruler = yes
				}
				error_log = "Serbia is marked as being ruled by House Karađorđević, but the ruler does not have a Karađorđević template. Ruler for Serbia has been regenerated. House Karađorđević rules Serbia."
			}
			# Heir Validation
			if = {
				limit = {
					exists = heir
					heir ?= {
						balkfm_is_karadordevic_character_trigger = no
					}
				}
				heir = {
					disinherit_character = yes
				}
				error_log = "Serbia has a ruler with a Karađorđević template, but the heir does not. Heir has been liquidated."
			}
			# Heir Regeneration
			if = {
				limit = {
					NOR = {
						exists = heir
						has_variable = balkfm_constitution_pending # Blocks heir shenanigans until the initial event chain is complete
					}
				}
				if = {
					limit = {
						ruler = {
							has_template = template_balkfm_karadordevic_aleksander
						}
					}
					if = {
						limit = {
							year >= 1844
						}
						create_character = {
							template = template_balkfm_karadordevic_petar
							heir = yes
						}
					}
					else = {
						create_character = {
							template = template_balkfm_karadordevic_dynamic_heir
						}
					}
				}
				else_if = {
					limit = {
						ruler = {
							has_template = template_balkfm_karadordevic_petar
						}
					}
					create_character = {
						template = template_balkfm_karadordevic_aleksander_I
						heir = yes
					}
				}
				else = {
					create_character = {
						template = template_balkfm_karadordevic_dynamic_heir
					}
				}
			}
			error_log = "Generated a Karađorđević template heir for Serbia."
		}
		else = {
			error_log = "Variable balkfm_ruling_dynasty_var is set, but its value does not correspond to a valid house flag. Unable to generate compatible ruler for Serbia."
		}
	}
	else = {
		error_log = "Variable balkfm_ruling_dynasty_var is not set. Unable to validate or generate compatible ruler for Serbia."
	}
}

balkfm_dynastic_rivalry_monthly_effect = {
	change_variable = {
		name = balkfm_obrenovic_support_var
		add = balkfm_obrenovic_net_progress_value
	}
	clamp_variable = {
		name = balkfm_obrenovic_support_var
		max = 500
		min = 0
	}
	change_variable = {
		name = balkfm_karadordevic_support_var
		add = balkfm_karadordevic_net_progress_value
	}
	clamp_variable = {
		name = balkfm_karadordevic_support_var
		max = 500
		min = 0
	}
	if = {
		limit = {
			NOT = {
				exists = heir
			}
			OR = {
				balkfm_has_karadordevic_ruler_trigger = yes
				balkfm_has_obrenovic_ruler_trigger = yes
			}
		}
		if = {
			limit = {
				balkfm_has_karadordevic_ruler_trigger = yes
			}
			create_character = {
				template = template_balkfm_karadordevic_dynamic_heir
			}
		}
		else_if = {
			limit = {
				balkfm_has_obrenovic_ruler_trigger = yes
			}
			create_character = {
				template = template_balkfm_obrenovic_dynamic_heir
			}
		}
	}
}

balkfm_update_dynasty_character_scopes_effect = {
	je:je_balkfm_ser_monarchism = {
		clear_variable_list = com_journal_characters
		remove_variable = balkfm_opposition_leader_var
	}
	if = {
		limit = {
			balkfm_has_obrenovic_ruler_trigger = yes
		}
		ruler = {
			save_scope_as = balkfm_obrenovic_scope
		}
	}
	else_if = {
		limit = {
			any_scope_character = {
				balkfm_is_obrenovic_character_trigger = yes
			}
		}
		random_scope_character = {
			limit = {
				balkfm_is_obrenovic_character_trigger = yes
			}
			save_scope_as = balkfm_obrenovic_scope
		}
	}
	else = {
		balkfm_find_obrenovic_effect = yes
	}
	if = {
		limit = {
			balkfm_has_karadordevic_ruler_trigger = yes
		}
		ruler = {
			save_scope_as = balkfm_karadordevic_scope
		}
	}
	else_if = {
		limit = {
			any_scope_character = {
				balkfm_is_karadordevic_character_trigger = yes
			}
		}
		random_scope_character = {
			limit = {
				balkfm_is_karadordevic_character_trigger = yes
			}
			save_scope_as = balkfm_karadordevic_scope
		}
	}
	else = {
		balkfm_find_karadordevic_effect = yes
	}
	balkfm_save_opposition_dynasty_scope_effect = yes
	je:je_balkfm_ser_monarchism = {
		add_to_variable_list = {
			name = com_journal_characters
			target = scope:balkfm_obrenovic_scope
		}
		add_to_variable_list = {
			name = com_journal_characters
			target = scope:balkfm_karadordevic_scope
		}
		set_variable = {
			name = balkfm_opposition_leader_var
			value = scope:balkfm_dynastic_opposition_scope
		}
	}
}

balkfm_find_obrenovic_effect = {
	if = {
		limit = {
			any_scope_character = {
				balkfm_is_obrenovic_character_trigger = yes
			}
		}
		random_character = {
			limit = {
				balkfm_is_obrenovic_character_trigger = yes
			}
			save_scope_as = balkfm_dynastic_opposition_scope
			save_scope_as = balkfm_obrenovic_scope
		}
	}
	else_if = {
		limit = {
			any_character_in_void = {
				balkfm_is_obrenovic_character_trigger = yes
			}
		}
		random_character_in_void = {
			limit = {
				balkfm_is_obrenovic_character_trigger = yes
			}
			save_scope_as = balkfm_dynastic_opposition_scope
			save_scope_as = balkfm_obrenovic_scope
		}
	}
	else = {
		if = {
			limit = {
				var:balkfm_obrenovic_counter_var ?= 1
			}
			create_character = { 
				template = template_balkfm_obrenovic_milan
				on_created = {
					place_character_in_void = 150 # characters will expire after 15 years
				}
			}
			change_variable = { 
				name = balkfm_obrenovic_counter_var
				add = 1
			}
		}
		if = {
			limit = {
				var:balkfm_obrenovic_counter_var ?= 2
			}
			create_character = {
				template = template_balkfm_obrenovic_mihailo
				on_created = {
					place_character_in_void = 150 # characters will expire after 15 years
				}
			}
			change_variable = {
				name = balkfm_obrenovic_counter_var
				add = 1
			}
		}
		if = {
			limit = {
				var:balkfm_obrenovic_counter_var ?= 3
			}
			create_character = {
				template = template_balkfm_obrenovic_milan_IV
				on_created = {
					place_character_in_void = 150 # characters will expire after 15 years
				}
			}
			change_variable = {
				name = balkfm_obrenovic_counter_var
				add = 1
			}
		}
		else = {
			create_character = {
				template = template_balkfm_obrenovic_dynamic
				on_created = {
					place_character_in_void = 150 # characters will expire after 15 years
				}
			}
			change_variable = {
				name = balkfm_obrenovic_counter_var
				add = 1
			}
		}
	}
	random_character_in_void = {
		limit = {
			balkfm_is_obrenovic_character_trigger = yes
		}
		save_scope_as = balkfm_dynastic_opposition_scope
		save_scope_as = balkfm_obrenovic_scope
	}
}
