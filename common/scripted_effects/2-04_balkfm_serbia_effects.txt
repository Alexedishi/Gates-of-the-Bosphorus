balkfm_validate_serbian_characters = {
	if = {
		limit = {
			is_key_in_variable_map = { 
				name = balkfm_dynastic_rivalry_varmap
				target = flag:balkfm_ruling_dynasty_var
			}
		}
		# Obrenović Validation Branch
		if = {
			limit = {
				"variable_map(balkfm_dynastic_rivalry_varmap|flag:balkfm_ruling_dynasty_var)" ?= flag:balkfm_house_obrenovic
			}
			# Ruler Validation
			if = {
				limit = {
					ruler = {
						balkfm_is_obrenovic_character_trigger = no
					}
				}
				create_character = {
					template = balkfm_obrenovic_dynamic
					ruler = yes
				}
				error_log = "Serbia is marked as being ruled by House Obrenović, but the ruler does not have an Obrenović template. Ruler for Serbia has been regenerated. House Obrenović rules Serbia."
			}
			# Heir Validation
			if = {
				limit = {
					exists = heir
					heir = {
						balkfm_is_obrenovic_character_trigger = no
					}
				}
				heir = {
					disinherit_character = yes
				}
				error_log = "Serbia has a ruler with an Obrenović template, but the heir does not. Heir has been liquidated."
			}
			# Heir Regeneration
			if = {
				limit = {
					NOR = {
						exists = heir
						has_variable = balkfm_constitution_pending # Blocks heir shenanigans until the initial event chain is complete
					}
				}
				if = {
					limit = {
						ruler = {
							has_template = balkfm_obrenovic_milosh
						}
					}
					create_character = {
						template = balkfm_obrenovic_milan
						heir = yes
					}
				}
				else_if = {
					limit = {
						ruler = {
							has_template = balkfm_obrenovic_milan
						}
					}
					create_character = {
						template = balkfm_obrenovic_mihailo
						heir = yes
					}
				}
				else_if = {
					limit = {
						ruler = {
							has_template = balkfm_obrenovic_mihailo
						}
					}
					if = {
						limit = {
							year >= 1855
						}
						create_character = {
							template = balkfm_obrenovic_dynamic_heir
						}
					}
					else = {
						create_character = {
							template = balkfm_obrenovic_milan_IV
							heir = yes
						}
					}
				}
				error_log = "Generated an Obrenović template heir for Serbia."
			}
			
		}
		# House Karadordevic Validation Branch
		else_if = {
			limit = {
				"variable_map(balkfm_dynastic_rivalry_varmap|flag:balkfm_ruling_dynasty_var)"  ?= flag:balkfm_house_karadordevic
			}
			# Ruler Validation
			if = {
				limit = {
					ruler = {
						balkfm_is_karadordevic_character_trigger = no
					}
				}
				create_character = {
					template = balkfm_karadordevic_dynamic
					ruler = yes
				}
				error_log = "Serbia is marked as being ruled by House Karađorđević, but the ruler does not have a Karađorđević template. Ruler for Serbia has been regenerated. House Karađorđević rules Serbia."
			}
			# Heir Validation
			if = {
				limit = {
					exists = heir
					heir ?= {
						balkfm_is_karadordevic_character_trigger = no
					}
				}
				heir = {
					disinherit_character = yes
				}
				error_log = "Serbia has a ruler with a Karađorđević template, but the heir does not. Heir has been liquidated."
			}
			# Heir Regeneration
			if = {
				limit = {
					NOR = {
						exists = heir
						has_variable = balkfm_constitution_pending # Blocks heir shenanigans until the initial event chain is complete
					}
				}
				if = {
					limit = {
						ruler = {
							has_template = balkfm_karadordevic_aleksander
						}
					}
					if = {
						limit = {
							year >= 1844
						}
						create_character = {
							template = balkfm_karadordevic_petar
							heir = yes
						}
					}
					else = {
						create_character = {
							template = balkfm_karadordevic_dynamic_heir
						}
					}
				}
				else_if = {
					limit = {
						ruler = {
							has_template = balkfm_karadordevic_petar
						}
					}
					create_character = {
						template = balkfm_karadordevic_aleksander_I
						heir = yes
					}
				}
				else = {
					create_character = {
						template = balkfm_karadordevic_dynamic_heir
					}
				}
				error_log = "Generated a Karađorđević template heir for Serbia."
			}
		}
		else = {
			error_log = "Key balkfm_ruling_dynasty_var is present in balkfm_dynastic_rivalry_varmap, but its value does not correspond to a valid house flag. Unable to generate compatible ruler for Serbia."
		}
	}
	else = {
		error_log = "Key balkfm_ruling_dynasty_var was not found in balkfm_dynastic_rivalry_varmap. Unable to validate or generate compatible ruler for Serbia."
	}
}

		name = balkfm_obrenovic_local_var
		value = {

			add = "variable_map(balkfm_dynastic_rivalry_varmap|flag:balkfm_obrenovic_progress_var)"
			max = 500
			min = 0
			round_to = 0.01
		}
	}
	set_local_variable = {
		name = balkfm_karadordevic_local_var
		value = {
			add = "variable_map(balkfm_dynastic_rivalry_varmap|flag:balkfm_karadordevic_support_var)"
			add = "variable_map(balkfm_dynastic_rivalry_varmap|flag:balkfm_karadordevic_progress_var)"
			max = 500
			min = 0
			round_to = 0.01
		}
	}
	com_update_value_in_variable_map = {
		name = balkfm_dynastic_rivalry_varmap
		key = flag:balkfm_obrenovic_support_var
		value = local_var:balkfm_obrenovic_local_var
	}
	com_update_value_in_variable_map = {
		name = balkfm_dynastic_rivalry_varmap
		key = flag:balkfm_karadordevic_support_var
		value = local_var:balkfm_karadordevic_local_var
	}
	if = {
		limit = {
			NOT = {
				exists = heir
			}
			OR = {
				balkfm_has_karadordevic_ruler_trigger = yes
				balkfm_has_obrenovic_ruler_trigger = yes
			}
		}
		if = {
			limit = {
				balkfm_has_karadordevic_ruler_trigger = yes
			}
			create_character = {
				template = balkfm_karadordevic_dynamic_heir
			}
		}
		else_if = {
			limit = {
				balkfm_has_obrenovic_ruler_trigger = yes
			}
			create_character = {
				template = balkfm_obrenovic_dynamic_heir
			}
		}
	}
}

		}

	}
	if = {
		limit = {
			NOT = {
				exists = scope:balkfm_obrenovic_scope
			}
		}
		ordered_character_in_void = { 
			limit = { 
				balkfm_is_obrenovic_character_trigger = yes
			}
			order_by = age
			save_scope_as = balkfm_obrenovic_scope
		}

			limit = {
				NOT = {
					exists = scope:balkfm_obrenovic_scope
				}
			}
			balkfm_generate_obrenovic_character_effect = yes
			random_scope_character = { 
				limit = {
					balkfm_is_obrenovic_character_trigger = yes
				}
				save_scope_as = balkfm_obrenovic_scope
			}
		}
	}
	com_update_value_in_variable_map = {
		name = balkfm_dynastic_rivalry_varmap
		key = flag:balkfm_obrenovic_head_var
		value = scope:balkfm_obrenovic_scope

	ordered_scope_character = {
		limit = {
			balkfm_is_karadordevic_character_trigger = yes
		}
		order_by = age
		save_scope_as = balkfm_karadordevic_scope
	}
	if = {
		limit = {
			NOT = {
				exists = scope:balkfm_karadordevic_scope
			}
		}
		ordered_character_in_void = {
			limit = {
				balkfm_is_karadordevic_character_trigger = yes
			}
			order_by = age
			save_scope_as = balkfm_karadordevic_scope
		}

				NOT = {
					exists = scope:balkfm_karadordevic_scope
				}

				limit = {

				save_scope_as = balkfm_karadordevic_scope

		value = scope:balkfm_karadordevic_scope
