## General Situation Effects
turkfm_initialize_balkan_situation_effect = {
	set_global_variable = {
		name = turkfm_separatism_phase_var
		value = flag:turkfm_stable # We start in the stable phase
	}
	set_global_variable = {
		name = turkfm_escalation_var
		value = 0
	}
	set_global_variable = {
		name = turkfm_deescalation_var
		value = 0
	}
	set_global_variable = {
		name = turkfm_riot_counter_var
		value = 0
	}
	set_global_variable = {
		name = turkfm_crisis_counter_var
		value = 0
	}
	set_global_variable = {
		name = turkfm_congress_counter_var
		value = 0
	}
	add_modifier = modifier_turkfm_stable_phase
}

turkfm_clamp_variables_effect = {
	clamp_global_variable = {
		name = turkfm_escalation_var
		max = 500
		min = 0
	}
	clamp_global_variable = {
		name = turkfm_deescalation_var
		max = 500
		min = 0
	}
}

turkfm_calculate_balkan_situation_variables_effect = {
	set_global_variable = {
		name = turkfm_escalation_progress_var
		value = turkfm_escalation_progress_value
	}
	set_global_variable = {
		name = turkfm_deescalation_progress_var
		value = turkfm_deescalation_progress_value
	}
}

turkfm_advance_balkan_situation_variables_effect = {
	change_global_variable = {
		name = turkfm_escalation_var
		add = global_var:turkfm_escalation_progress_var
	}
	change_global_variable = {
		name = turkfm_deescalation_var
		add = global_var:turkfm_deescalation_progress_var
	}
	turkfm_clamp_variables_effect = yes
}

turkfm_check_balkan_situation_phase_effect = {
	if = {
		limit = {
			OR = {
				global_var:turkfm_escalation_var >= 500
				global_var:turkfm_deescalation_var >= 500
			}
		}

		## Attempt to Deescalate First
		if = {
			limit = {
				turkfm_can_deescalate_tension_trigger = yes
				global_var:turkfm_deescalation_var >= 500
			}
			if = { # From Unrest to Deconfliction
				limit = {
					global_var:turkfm_separatism_phase_var ?= flag:turkfm_unrest
				}
				set_global_variable = {
					name = turkfm_separatism_phase_var
					value = flag:turkfm_deconfliction
				}
				every_country = {
					limit = {
						turkfm_balkans_party_trigger = yes
					}
					post_notification = notification_turkfm_transition_deconfliction
				}
				custom_tooltip = turkfm_transition_deconfliction_tt
			}
			else_if = { # From Hostility to Deconfliction
				limit = {
					global_var:turkfm_separatism_phase_var ?= flag:turkfm_hostility
				}
				set_global_variable = {
					name = turkfm_separatism_phase_var
					value = flag:turkfm_deconfliction
				}
				every_country = {
					limit = {
						turkfm_balkans_party_trigger = yes
					}
					post_notification = notification_turkfm_transition_deconfliction
				}
				custom_tooltip = turkfm_transition_deconfliction_tt
			}
			else_if = { # From Crisis to Deconfliction
				limit = {
					global_var:turkfm_separatism_phase_var ?= flag:turkfm_crisis
				}
				set_global_variable = {
					name = turkfm_separatism_phase_var
					value = flag:turkfm_deconfliction
				}
				every_country = {
					limit = {
						turkfm_balkans_party_trigger = yes
					}
					post_notification = notification_turkfm_transition_deconfliction
				}
				custom_tooltip = turkfm_transition_deconfliction_tt
			}
			set_global_variable = {
				name = turkfm_escalation_var
				value = 0
			}
			set_global_variable = {
				name = turkfm_deescalation_var
				value = 0
			}
		}

		## Ignore this if we deescalated
		else_if = { # Ignore this if we deescalated
			limit = {
				turkfm_can_escalate_tension_trigger = yes
				global_var:turkfm_escalation_var >= 500
			}
			if = { # From Stable to Unrest
				limit = {
					global_var:turkfm_separatism_phase_var ?= flag:turkfm_stable
				}
				set_global_variable = {
					name = turkfm_separatism_phase_var
					value = flag:turkfm_unrest
				}
				every_country = {
					limit = {
						turkfm_balkans_party_trigger = yes
					}
					post_notification = notification_turkfm_transition_unrest
				}
				custom_tooltip = turkfm_transition_unrest_tt
			}
			else_if = { # From Unrest to Hostility
				limit = {
					global_var:turkfm_separatism_phase_var ?= flag:turkfm_unrest
				}
				set_global_variable = {
					name = turkfm_separatism_phase_var
					value = flag:turkfm_hostility
				}
				every_country = {
					limit = {
						turkfm_balkans_party_trigger = yes
					}
					post_notification = notification_turkfm_transition_hostility
				}
				custom_tooltip = turkfm_transition_hostility_tt
			}
			else_if = { # From Hostility to Crisis
				limit = {
					turkfm_crisis_possible_trigger = yes
					global_var:turkfm_separatism_phase_var ?= flag:turkfm_hostility
				}
				set_global_variable = {
					name = turkfm_separatism_phase_var
					value = flag:turkfm_crisis
				}
				c:TUR ?= {
					trigger_event = {
						id = turkfm_eastern_question.100 # A Crisis in Rumelia
					}
				}
				every_country = {
					limit = {
						turkfm_balkans_party_trigger = yes
					}
					post_notification = notification_turkfm_transition_crisis
				}
				custom_tooltip = turkfm_transition_crisis_tt
			}
			else_if = { # From Deconfliction to Stable
				limit = {
					global_var:turkfm_separatism_phase_var ?= flag:turkfm_deconfliction
				}
				set_global_variable = {
					name = turkfm_separatism_phase_var
					value = flag:turkfm_stable
				}
				every_country = {
					limit = {
						turkfm_balkans_party_trigger = yes
					}
					post_notification = notification_turkfm_transition_stable
				}
				custom_tooltip = turkfm_transition_stable_tt
			}
			set_global_variable = {
				name = turkfm_escalation_var
				value = 0
			}
			set_global_variable = {
				name = turkfm_deescalation_var
				value = 0
			}
		}
	}
	turkfm_check_modifiers_effect = yes
}

turkfm_check_modifiers_effect = {
	hidden_effect = {
		remove_modifier = modifier_turkfm_stable_phase
		remove_modifier = modifier_turkfm_unrest_phase
		remove_modifier = modifier_turkfm_hosility_phase
		remove_modifier = modifier_turkfm_deconfliction_phase
		remove_modifier = modifier_turkfm_crisis_phase
		every_political_movement = {
			remove_modifier = modifier_turkfm_stable_phase_movement
			remove_modifier = modifier_turkfm_unrest_phase_movement
			remove_modifier = modifier_turkfm_hosility_phase_movement
			remove_modifier = modifier_turkfm_deconfliction_phase_movement
			remove_modifier = modifier_turkfm_crisis_phase_movement
		}
	}
	if = {
		limit = {
			global_var:turkfm_separatism_phase_var = flag:turkfm_stable
		}
		add_modifier = modifier_turkfm_stable_phase
		every_political_movement = {
			limit = {
				turkfm_eligible_minority_movement_trigger = yes
			}
			add_modifier = modifier_turkfm_stable_phase_movement
		}
	}
	else_if = {
		limit = {
			global_var:turkfm_separatism_phase_var = flag:turkfm_unrest
		}
		add_modifier = modifier_turkfm_unrest_phase
		every_political_movement = {
			limit = {
				turkfm_eligible_minority_movement_trigger = yes
			}
			add_modifier = modifier_turkfm_unrest_phase_movement
		}
	}
	else_if = {
		limit = {
			global_var:turkfm_separatism_phase_var = flag:turkfm_hostility
		}
		add_modifier = modifier_turkfm_hosility_phase
		every_political_movement = {
			limit = {
				turkfm_eligible_minority_movement_trigger = yes
			}
			add_modifier = modifier_turkfm_hosility_phase_movement
		}
	}
	else_if = {
		limit = {
			global_var:turkfm_separatism_phase_var = flag:turkfm_deconfliction
		}
		add_modifier = modifier_turkfm_deconfliction_phase
		every_political_movement = {
			limit = {
				turkfm_eligible_minority_movement_trigger = yes
			}
			add_modifier = modifier_turkfm_deconfliction_phase_movement
		}
	}
	else_if = {
		limit = {
			global_var:turkfm_separatism_phase_var = flag:turkfm_crisis
		}
		add_modifier = modifier_turkfm_crisis_phase
		every_political_movement = {
			limit = {
				turkfm_eligible_minority_movement_trigger = yes
			}
			add_modifier = modifier_turkfm_crisis_phase_movement
		}
	}
}

turkfm_tension_cooldown_effect = {
	if = {
		limit = {
			OR = {
				global_var:turkfm_separatism_phase_var ?= flag:turkfm_stable
				global_var:turkfm_separatism_phase_var ?= flag:turkfm_unrest
				global_var:turkfm_separatism_phase_var ?= flag:turkfm_deconfliction
			}
		}
		set_global_variable = {
			name = turkfm_tension_event_cooldown_var
			years = 2
		}
	}
	if = {
		limit = {
			OR = {
				global_var:turkfm_separatism_phase_var ?= flag:turkfm_stable
				global_var:turkfm_separatism_phase_var ?= flag:turkfm_unrest
				global_var:turkfm_separatism_phase_var ?= flag:turkfm_deconfliction
			}
		}
		set_global_variable = {
			name = turkfm_tension_event_cooldown_var
			years = 1
		}
	}
	else_if = {
		limit = {
			OR = {
				global_var:turkfm_separatism_phase_var ?= flag:turkfm_hostility
				global_var:turkfm_separatism_phase_var ?= flag:turkfm_crisis
			}
		}
		set_global_variable = {
			name = turkfm_tension_event_cooldown_var
			months = 3
		}
	}
}

turkfm_add_medium_movement_radicalism_effect = {
	every_political_movement = {
		limit = {
			OR = {
				is_political_movement_type = movement_cultural_minority
				is_political_movement_type = movement_religious_minority
			}
			gbbf_balkans_culture_trigger = yes
		}
		add_modifier = {
			name = modifier_turkfm_increased_radicalism
			days = normal_modifier_time
			is_decaying = yes
		}
	}
}

turkfm_update_riot_variables_effect = {
	# Step 1, clear all variables
	remove_variable = turkfm_riot_state_1_var
	remove_variable = turkfm_riot_state_1_progress_var
	remove_variable = turkfm_riot_state_2_var
	remove_variable = turkfm_riot_state_2_progress_var
	remove_variable = turkfm_riot_state_3_var
	remove_variable = turkfm_riot_state_3_progress_var

	# Step 2, Recapture states and set appropriate progress variables
	random_scope_state = {
		limit = {
			has_modifier = modifier_turkfm_race_riots
		}
		save_scope_as = turkfm_riot_state_1_scope
	}
	set_variable = {
		name = turkfm_riot_state_1_var
		value = scope:turkfm_riot_state_1_scope
	}
	set_variable = {
		name = turkfm_riot_state_1_progress_var
		value = turkfm_riot_1_progress_value
	}
	if = {
		limit = {
			exists = scope:turkfm_riot_state_1_scope
		}
		random_scope_state = {
			limit = {
				has_modifier = modifier_turkfm_race_riots
				NOT = {
					scope:turkfm_riot_state_1_scope ?= this
				}
			}
			save_scope_as = turkfm_riot_state_2_scope
		}
		if = {
			limit = {
				exists = scope:turkfm_riot_state_2_scope
			}
			set_variable = {
				name = turkfm_riot_state_2_var
				value = scope:turkfm_riot_state_2_scope
			}
			set_variable = {
				name = turkfm_riot_state_2_progress_var
				value = turkfm_riot_2_progress_value
			}
			random_scope_state = {
				limit = {
					has_modifier = modifier_turkfm_race_riots
					NOR = {
						scope:turkfm_riot_state_1_scope ?= this
						scope:turkfm_riot_state_2_scope ?= this
					}
				}
				save_scope_as = turkfm_riot_state_3_scope
			}
			if = {
				limit = {
					exists = scope:turkfm_riot_state_2_scope
				}
				set_variable = {
