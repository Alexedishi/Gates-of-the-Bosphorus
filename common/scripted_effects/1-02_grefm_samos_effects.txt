grefm_initialize_samos_variables_effect = {
	set_variable = {
		name = grefm_samos_support_var
		value = 0
	}
	set_variable = {
		name = grefm_samos_ruler_counter
		value = 0
	}
	set_variable = {
		name = grefm_samos_ruler_duration_var
		value = 0
	}
}
