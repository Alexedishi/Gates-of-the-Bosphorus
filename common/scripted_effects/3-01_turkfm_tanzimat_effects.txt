turkfm_add_tanzimat_agenda_effect = {
	if = {
		limit = {
			has_journal_entry = je_turkfm_tanzimat_reforms
		}
		custom_tooltip = {
			text = turkfm_add_$type$_agenda
			if = {
				limit = {
					NOR = {
						has_variable = turkfm_tanzimat_$type$_agenda
						has_variable = turkfm_tanzimat_$type$_complete
					}
				}
				set_variable = turkfm_tanzimat_$type$_agenda
			}
			else = {
				error_log = "turkfm_add_tanzimat_agenda_effect was called, but the specified agenda is either already active or complete."
			}
		}
	}
	else = {
		error_log = "turkfm_add_tanzimat_agenda_effect was called, but the calling country does not have an active je_turkfm_tanzimat_reforms journal entry."
	}
	set_variable = {
		name = turkfm_total_tanzimat_agendas_var
		value = turkfm_total_tanzimat_agendas_value
	}
	set_variable = {
		name = turkfm_completed_tanzimat_agendas_var
		value = turkfm_completed_tanzimat_agendas_value
	}
}

turkfm_complete_tanzimat_agenda_effect = {
	if = {
		limit = {
			has_journal_entry = je_turkfm_tanzimat_reforms
		}
		if = {
			limit = {
				has_variable = turkfm_tanzimat_$type$_agenda
			}
			remove_variable = turkfm_tanzimat_$type$_agenda
			set_variable = turkfm_tanzimat_$type$_complete
			turkfm_advance_ottomanism1_effect = yes
		}
		else = {
			error_log = "turkfm_complete_tanzimat_agenda_effect was called, but the specified agenda is not active in je_turkfm_tanzimat_reforms."
		}
	}
	else = {
		error_log = "turkfm_complete_tanzimat_agenda_effect was called, but the calling country does not have an active je_turkfm_tanzimat_reforms journal entry."
	}
	set_variable = {
		name = turkfm_total_tanzimat_agendas_var
		value = turkfm_total_tanzimat_agendas_value
	}
	set_variable = {
		name = turkfm_completed_tanzimat_agendas_var
		value = turkfm_completed_tanzimat_agendas_value
	}
}

turkfm_circassian_slave_trade_effect = {
	if = {
		limit = {
			active_law:lawgroup_slavery = {
				has_amendment = amendment_type:amendment_gbbf_esir_pazari
			}
			s:STATE_KUBAN = {
				has_variable = turkfm_circassian_slave_trade_var
			}
		}
		c:TUR ?= {
			capital = {
				save_scope_as = ottoman_capital_scope
			}
		}
		s:STATE_KUBAN = {
			set_variable = {
				name = turkfm_circassian_slave_trade_var
				value = {
					scope:ottoman_capital_scope = {
						every_scope_pop = {
							limit = {
								is_pop_type = aristocrats
							}
							add = standard_of_living
						}
						divide = {
							every_scope_pop = {
								limit = {
									is_pop_type = aristocrats
								}
								add = standard_of_living
							}
							min = 1
						}
					}
					round = yes # idk, unless you want a shipment of slaves plus like a hand and a few ears.
				}
			}
			random_scope_state = {
				limit = {
					any_scope_pop = {
						has_pop_culture = circassian
						total_size >= s:STATE_KUBAN.var:turkfm_circassian_slave_trade_var
					}
				}
				random_scope_pop = {
					limit = {
						has_pop_culture = circassian
						total_size >= s:STATE_KUBAN.var:turkfm_circassian_slave_trade_var
					}
					move_partial_pop = {
						state = scope:ottoman_capital_scope
						population = s:STATE_KUBAN.var:turkfm_circassian_slave_trade_var
					}
				}
			}
		}
	}
}
