turkfm_initialize_tanzimat_effect = {
	# Imperial Identity Variables
	set_variable = {
		name = turkfm_tanzimat_identity_var
		value = 300
	}
	set_variable = {
		name = turkfm_tanzimat_ottomanist_progress_var
		value = 0
	}
	set_variable = {
		name = turkfm_tanzimat_islamist_progress_var
		value = 0
	}
	set_variable = {
		name = turkfm_tanzimat_identity_progress_var
		value = 0
	}
	set_variable = {
		name = turkfm_tanzimat_identity_tilt_var
		value = 0
	}
	
	# Reform Progress Variables
	set_variable = {
		name = turkfm_tanzimat_reform_var
		value = 500
	}
	set_variable = {
		name = turkfm_tanzimat_reformist_progress_var
		value = 0
	}
	set_variable = {
		name = turkfm_tanzimat_conservatist_progress_var
		value = 0
	}
	set_variable = {
		name = turkfm_tanzimat_reform_progress_var
		value = 0
	}
	
	# Supporting Variables
	set_variable = {
		name = turkfm_tanzimat_completion_var
		value = 0
	}
	
	# Modifiers
	add_modifier = modifier_turkfm_feudal_argiculture
	add_modifier = modifier_turkfm_cizye_tax
	add_modifier = modifier_turkfm_outmoded_bureaucracy
	add_modifier = modifier_turkfm_stagnant_economy
	add_modifier = modifier_turkfm_illiterate_populace
	add_modifier = modifier_turkfm_insecure_countryside
}

turkfm_calculate_tanzimat_variables_effect = {
	# Imperial Identity Variables
	set_variable = {
		name = turkfm_tanzimat_ottomanist_progress_var
		value = turkfm_tanzimat_ottomanist_progress_value
	}
	set_variable = {
		name = turkfm_tanzimat_islamist_progress_var
		value = turkfm_tanzimat_islamist_progress_value
	}
	set_variable = {
		name = turkfm_tanzimat_identity_progress_var
		value = turkfm_tanzimat_identity_progress_value
	}
	set_variable = {
		name = turkfm_tanzimat_identity_tilt_var
		value = turkfm_tanzimat_identity_tilt_value
	}
	
	# Reform Progress Variables
	set_variable = {
		name = turkfm_tanzimat_reformist_progress_var
		value = turkfm_tanzimat_reformist_progress_value
	}
	set_variable = {
		name = turkfm_tanzimat_conservatist_progress_var
		value = turkfm_tanzimat_conservatist_progress_value
	}
	set_variable = {
		name = turkfm_tanzimat_reform_progress_var
		value = turkfm_tanzimat_reform_progress_value
	}
}

turkfm_cleanup_tanzimat_effect = {
	set_global_variable = turkfm_tanzimat_complete
	remove_variable = turkfm_tanzimat_identity_var
	remove_variable = turkfm_tanzimat_ottomanist_progress_var
	remove_variable = turkfm_tanzimat_islamist_progress_var
	remove_variable = turkfm_tanzimat_identity_progress_var
	remove_variable = turkfm_tanzimat_identity_tilt_var
	remove_variable = turkfm_tanzimat_reform_var
	remove_variable = turkfm_tanzimat_reformist_progress_var
	remove_variable = turkfm_tanzimat_conservatist_progress_var
	remove_variable = turkfm_tanzimat_reform_progress_var
	remove_variable = turkfm_tanzimat_completion_var
	hidden_effect = {
		if = {
			limit = {
				has_modifier = modifier_turkfm_feudal_argiculture
			}
			remove_modifier = modifier_turkfm_feudal_argiculture
		}
		if = {
			limit = {
				has_modifier = modifier_turkfm_cizye_tax
			}
			remove_modifier = modifier_turkfm_cizye_tax
		}
		if = {
			limit = {
				has_modifier = modifier_turkfm_outmoded_bureaucracy
			}
			remove_modifier = modifier_turkfm_outmoded_bureaucracy
		}
		if = {
			limit = {
				has_modifier = modifier_turkfm_stagnant_economy
			}
			remove_modifier = modifier_turkfm_stagnant_economy
		}
		if = {
			limit = {
				has_modifier = modifier_turkfm_illiterate_populace
			}
			remove_modifier = modifier_turkfm_illiterate_populace
		}
		if = {
			limit = {
				has_modifier = modifier_turkfm_insecure_countryside
			}
			remove_modifier = modifier_turkfm_insecure_countryside
		}
	}
}

turkfm_complete_reform_effect = {
	custom_tooltip = {
		text = turkfm_reform_completed
		change_variable = {
			name = turkfm_tanzimat_completion_var
			add = 1
		}
	}
}

turkfm_advance_ottomanism1_effect = {
	if = {
		limit = {
			has_variable = turkfm_tanzimat_reform_var
		}
		custom_tooltip = {
			text = turkfm_advance_tanzimat1
			hidden_effect = {
				change_variable = {
					name = turkfm_tanzimat_reform_var
					add = 50
				}
				clamp_global_variable = {
					name = turkfm_tanzimat_reform_var
					max = 1000
					min = 0
				}
				je:je_turkfm_tanzimat_main = {
					set_bar_progress = {
						name = turkfm_tanzimat_reforms_progress_bar
						value = owner.var:turkfm_tanzimat_reform_var
					}
				}
			}
		}
	}
}
