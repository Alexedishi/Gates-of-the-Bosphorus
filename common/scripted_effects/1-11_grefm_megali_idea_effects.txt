grefm_empire_name_tooltip_effect = {
	if = {
		limit = {
			country_has_primary_culture = cu:greek
			NOR = {
				country_has_primary_culture = cu:rhomaios
				country_has_primary_culture = cu:australian
			}
		}
		custom_tooltip = grefm_greek_empire_name_hellas
	}
	else_if = {
		limit = {
			country_has_primary_culture = cu:rhomaios
			NOT = {
				country_has_primary_culture = cu:greek
				country_has_primary_culture = cu:australian
			}
			has_game_rule = gbbf_rome_name_rule
		}
		custom_tooltip = grefm_greek_empire_name_rome
	}
	else_if = {
		limit = {
			country_has_primary_culture = cu:rhomaios
			NOT = {
				country_has_primary_culture = cu:greek
				country_has_primary_culture = cu:australian
			}
			has_game_rule = gbbf_byzantium_name_rule
		}
		custom_tooltip = grefm_greek_empire_name_byzantium
	}
	else_if = {
		limit = {
			country_has_primary_culture = cu:greek
			country_has_primary_culture = cu:rhomaios
		}
		custom_tooltip = grefm_greek_empire_name_graia
	}
	else_if = {
		limit = {
			country_has_primary_culture = cu:australian
		}
		custom_tooltip = grefm_greek_empire_name_emu
	}
}
