balkfm_set_up_ottoman_legacies_effect = {
	if = {
		limit = {
			NOT = {
				has_journal_entry = je_balkfm_legacies
				# There's a tiny corner case where Bulgaria can fire this effect twice
			}
		}
		add_journal_entry = {
			type = je_balkfm_legacies
		}
		## Tax Reform
		if = {
			limit = {
				always = no # TODO enable when ready
				NOT = {
					has_variable = balkfm_tax_reform_complete
				}
			}
			je:je_balkfm_legacies = {
				add_modifier = modifier_balkfm_timar
			}
		}
		
		## Land Reform
		if = {
			limit = {
				NOT = {
					has_variable = balkfm_land_reform_complete
				}
				OR = {
					has_law_or_variant = law_type:law_serfdom
					has_law_or_variant = law_type:law_tenant_farmers
					has_law_or_variant = law_type:law_traditionalism
					has_law_or_variant = law_type:law_agrarianism
				}
			}
			je:je_balkfm_legacies = {
				add_modifier = modifier_balkfm_outdated_argiculture
			}
		}
		
		## Civic Reform
		if = {
			limit = {
				NOT = {
					has_variable = balkfm_civic_reform_complete
				}
				OR = {
					has_law_or_variant = law_type:law_hereditary_bureaucrats
					ig:ig_landowners = {
						is_powerful = yes
					}
				}
			}
			je:je_balkfm_legacies = {
				add_modifier = modifier_balkfm_poor_administration
			}
		}
		
		## Brigandage
		if = {
			limit = {
				NOT = {
					has_variable = balkfm_brigandage_complete
				}
				any_scope_state = {
					is_in_geographic_region = geographic_region_rumelia
				}
				institution_investment_level = {
					institution = institution_police
					value < 3
				}
			}
			je:je_balkfm_legacies = {
				add_modifier = modifier_balkfm_insecure_countryside
			}
		}
		
		## State Reform
		if = {
			limit = {
				OR = {
					country_has_primary_culture = cu:romanian
				}
			}
			je:je_balkfm_legacies = {
				add_modifier = modifier_balkfm_incompetent_administration
				add_modifier = modifier_balkfm_informal_economy
				add_modifier = modifier_balkfm_rigid_society
			}
		}
		else_if = {
			limit = {
				country_has_primary_culture = cu:bosniak
			}
			je:je_balkfm_legacies = {
				add_modifier = modifier_balkfm_incompetent_administration
				add_modifier = modifier_balkfm_informal_economy
			}
		}
	}
	else = {
		error_log = "balkfm_set_up_ottoman_legacies_effect was called, but the country calling it already has an active je_balkfm_legacies journal entry."
	}
}

balkfm_add_legacy_agenda_effect = {
	# This effect does NOT work for Brigandage
	if = {
		limit = {
			has_journal_entry = je_balkfm_legacies
			NOR = {
				je:je_balkfm_legacies = {
					has_modifier = modifier_balkfm_$type$_reform_agenda
				}
				has_variable = balkfm_$type$_reform_complete
			}
		}
		custom_tooltip = {
			text = balkfm_add_$type$_reform_agenda
			je:je_balkfm_legacies = {
				add_modifier = modifier_balkfm_$type$_reform_agenda
			}
		}
	}
}

balkfm_add_brigandage_agenda_effect = {
	# Use this effect to add the Brigandage Agenda
	if = {
		limit = {
			has_journal_entry = je_balkfm_legacies
			NOR = {
				je:je_balkfm_legacies = {
					has_modifier = modifier_balkfm_brigandage_agenda
				}
				has_variable = balkfm_brigandage_complete
			}
		}
		custom_tooltip = {
			text = balkfm_add_brigandage_agenda
			je:je_balkfm_legacies = {
				add_modifier = modifier_balkfm_brigandage_agenda
			}
		}
	}
}
