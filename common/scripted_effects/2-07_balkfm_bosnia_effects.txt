balkfm_hersek_scopes_effect = {
	c:TUR ?= {
		save_scope_as = ottoman_scope
		ruler = {
			save_scope_as = sultan_scope
		}
	}
	c:BOS_HER ?= {
		save_scope_as = hersek_scope
		ruler = {
			save_scope_as = ali_pasha_scope
		}
		random_scope_character = {
			limit = {
				has_template = template_balkfm_cengic_smail
			}
			save_scope_as = smail_aga_scope
		}
	}
	if = {
		limit = {
			NOT = {
				exists = scope:smail_aga_scope
			}
		}
		random_character_in_void = {
			limit = {
				has_template = template_balkfm_cengic_smail
			}
			save_scope_as = smail_aga_scope
		}
		if = {
			limit = {
				NOT = {
					exists = scope:smail_aga_scope
				}
			}
			create_character = {
				template = template_balkfm_cengic_smail
				on_created = {
					place_character_in_void = 500
				}
			}
			random_character_in_void = {
				limit = {
					has_template = template_balkfm_cengic_smail
				}
				save_scope_as = smail_aga_scope
			}
		}
	}
	c:MON ?= {
		save_scope_as = montenegro_scope
		ruler = {
			save_scope_as = chad_vladika_petar_the_based_scope
		}
	}
}
