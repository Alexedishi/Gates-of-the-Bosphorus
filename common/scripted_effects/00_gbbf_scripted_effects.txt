	if = {
		limit = {
			is_player = yes
		}
		create_country = {
			tag = GRA_THE
			origin = c:RUS
			province = p:xB0DFDF
			province = p:xF56D13
			province = p:x706060
			on_created = {
				gbbf_setup_theodoro_effect = yes
				trigger_event = grefm_theodoro.001
				balkfm_set_up_ottoman_legacies_effect = yes
			}
		}
		s:STATE_CRIMEA = {
			region_state:RUS = {
				every_scope_pop = {
					limit = {
						has_pop_culture = urum
					}
					move_pop = s:STATE_CRIMEA.region_state:GRA_THE
				}
			}
		}
		play_as = c:GRA_THE
	}
	c:ALK = {
		annex = c:GBBF_THE
	}
}
gbbf_setup_theodoro_effect = {
	c:RUS = {
		create_diplomatic_pact = {
			country = c:GRA_THE
			type = puppet
		}
		set_secret_goal = {
			country = c:GRA_THE
			secret_goal = protect
		}
		change_relations = {
			country = c:GRA_THE
			value = 50
		}
	}
	s:STATE_CRIMEA = {
		region_state:RUS = {
			every_scope_pop = {
				limit = {
					OR = {
						has_pop_culture = greek
						has_pop_culture = rhomaios
						has_pop_culture = urum
					}
				}
				move_pop = s:STATE_CRIMEA.region_state:GRA_THE
			}
		}
	}
}
