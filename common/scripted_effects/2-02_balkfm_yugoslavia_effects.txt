balkfm_validate_yugoslav_nationalism_involvement_effect = {
	if = {
		limit = {
			exists = c:YUG # Error suppression in case YUG dies but the monthly pulse gets out before the journal closes
		}
		scope:journal_entry = {
			every_scope_je_involved = {
				limit = {
					NOR = {
						c:YUG = this
						is_subject_of = c:YUG
					}
				}
				global_var:je_balkfm_yugoslav_nationalism = {
					remove_involved_country = prev
				}
			}
		}
		c:YUG = {
			every_subject_or_below = {
				limit = {
					balkfm_country_has_yugoslav_culture_trigger = yes
					NOT = {
						is_involved_in_journal_entry = je_balkfm_yugoslav_nationalism
					}
				}
				global_var:je_balkfm_yugoslav_nationalism = {
					add_involved_country = prev
				}
			}
		}
	}
}

balkfm_calculate_yugoslav_populations_effect = {
	# This is a lot of math, so we only call it monthly; These are stacked in a way to avoid repeating math operations
	
	# Montenegrin
	set_variable = {
		name = balkfm_montenegrin_total_population_var
		value = {
			c:YUG ?= {
				every_scope_state = {
					every_scope_pop = {
						limit = {
							has_pop_culture = montenegrin
						}
						add = {
							value = total_size
						}
					}
				}
			}
		}
	}
	set_variable = {
		name = balkfm_montenegrin_loyalist_population_var
		value = {
			c:YUG ?= {
				every_scope_state = {
					every_scope_pop = {
						limit = {
							has_pop_culture = montenegrin
						}
						add = {
							value = total_size
							multiply = pop_loyalist_fraction
						}
					}
				}
				round_to = 1
			}
		}
	}
	set_variable = {
		name = balkfm_montenegrin_radical_population_var
		value = {
			c:YUG ?= {
				every_scope_state = {
					every_scope_pop = {
						limit = {
							has_pop_culture = montenegrin
						}
						add = {
							value = total_size
							multiply = pop_radical_fraction
						}
					}
				}
				round_to = 1
			}
		}
	}
	set_variable = { 
		name = balkfm_montenegrin_radical_fraction_var
		value = {
			add = var:balkfm_montenegrin_radical_population_var
			divide = {
				add = var:balkfm_montenegrin_total_population_var
				min = 1
			}
		}
	}
	set_variable = {
		name = balkfm_montenegrin_loyalist_fraction_var
		value = {
			add = var:balkfm_montenegrin_loyalist_population_var
			divide = {

			text = balkfm_create_federal_montenegro_effect
			s:STATE_MONTENEGRO = {
				random_scope_state = {
					limit = {
						owner = root
					}
					save_scope_as = montenegro_state_scope
				}
			}
			create_country = {
				tag = MON
				origin = root
				state = scope:montenegro_state_scope
			}
			create_diplomatic_pact = {
				country = c:MON
				type = puppet
			}
			create_bidirectional_truce = {
				country = c:MON
				months = 120
			}
		}
		custom_tooltip = balkfm_federal_transfer_montenegro
	}
}
