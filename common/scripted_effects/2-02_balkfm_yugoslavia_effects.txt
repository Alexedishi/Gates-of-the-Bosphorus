balkfm_initialize_yugoslav_unification_variables = {
	set_variable = {
		name = balkfm_mon_annex_var
		value = 0
	}
	set_variable = {
		name = balkfm_ser_annex_var
		value = 0
	}
	set_variable = {
		name = balkfm_bul_annex_var
		value = 0
	}
	set_variable = {
		name = balkfm_bos_annex_var
		value = 0
	}
	set_variable = {
		name = balkfm_cro_annex_var
		value = 0
	}
	set_variable = {
		name = balkfm_slo_annex_var
		value = 0
	}
	set_variable = {
		name = balkfm_ser_rsk_annex_var
		value = 0
	}
	set_variable = je_yugoslavia_is_active
}

balkfm_update_yugoslav_unification_variables = {
	# Montenegro Variable
	if = {
		limit = { 
			exists = c:MON
			c:MON != this
		}
		if = {
			limit = {
				NOT = {
					has_variable = balkfm_mon_annex_var
				}
			}
			set_variable = {
				name = balkfm_mon_annex_var
				value = 0
			}
		}
		else_if = {
			limit = {
				has_variable = balkfm_mon_annex_var
				c:MON.relations:root >= relations_threshold:cordial
			}
			change_variable = {
				name = balkfm_mon_annex_var
				add = 1
			}
		}
	}
	else = {
		remove_variable = balkfm_mon_annex_var

balkfm_cleanup_yugoslav_unification_variables_effect = {
	remove_variable = balkfm_mon_annex_var
	remove_variable = balkfm_ser_annex_var
	remove_variable = balkfm_bul_annex_var
	remove_variable = balkfm_bos_annex_var
	remove_variable = balkfm_cro_annex_var
	remove_variable = balkfm_slo_annex_var
}

			value = large_radicals
