balkfm_update_yugoslav_unification_variables = {
	# Montenegro Variable
	if = {
		limit = {
			exists = c:MON
			c:MON != this
		}
		if = {
			limit = {
				NOT = {
					has_variable = balkfm_mon_annex_var
				}
			}
			set_variable = {
				name = balkfm_mon_annex_var
				value = 0
			}
		}
		else_if = {
			limit = {
				has_variable = balkfm_mon_annex_var
				c:MON.relations:root >= relations_threshold:cordial
			}
			change_variable = {
				name = balkfm_mon_annex_var
				add = 1
			}

	# First Test for MON

	}

balkfm_create_federal_montenegro = {
	if = {
		limit = {
			has_state_in_state_region = STATE_MONTENEGRO
		}
		custom_tooltip = {
			text = balkfm_create_federal_montenegro
			s:STATE_MONTENEGRO = {
				random_scope_state = {
					limit = {
						owner = root
					}
				}
				save_scope_as = montenegro_state_scope
			}
			create_country = {
				tag = MON
				origin = root
				state = scope:montenegro_state_scope
			}
			create_diplomatic_pact = {
				country = c:MON
				type = puppet
			}
			create_bidirectional_truce = {
				country = c:MON
				years = 10
			}
		}
		custom_tooltip = balkfm_federal_transfer_montenegro
	}
}

				has_state_in_state_region = STATE_EASTERN_SERBIA
