gbbf_purge_turkish_homelands_effect = {
	every_scope_state = {
		limit = {
			AND = {
				NOT = { has_variable = grefm_turkish_reserve }
				state_region = {
					is_homeland = cu:turkish
				}
			}
		}
		add_radicals_in_state = {
			culture = cu:turkish
			value = small_radicals
		}
		state_region = {
			remove_homeland = cu:turkish
		}
	}
}
