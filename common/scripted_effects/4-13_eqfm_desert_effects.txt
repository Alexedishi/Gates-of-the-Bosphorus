eqfm_start_desert_expedition_effect = {
	remove_variable = eqfm_expedition_desert_transit_var
	set_variable = eqfm_desert_expedition_var
	set_variable = {
		name = eqfm_desert_expedition_progress_var
		value = 0
	}
	set_variable = {
		name = eqfm_desert_expedition_peril_var
		value = 0
	}
}

eqfm_update_desert_variables_effect = {
	set_variable = {
		name = eqfm_desert_provinces_var
		value = eqfm_desert_provinces_value
	}
	if = {
		limit = {
			has_variable = eqfm_desert_expedition_var
		}
		random_scope_character = {
			limit = {
				has_modifier = modifier_eqfm_leading_desert_expedition
			}
			save_scope_as = eqfm_desert_expedition_leader_scope
		}
		set_variable = {
			name = eqfm_desert_expedition_leader_var
			value = scope:eqfm_desert_expedition_leader_scope
		}
		je:je_eqfm_libyan_desert = {
			clear_variable_list = com_journal_characters
			add_to_variable_list = {
				name = com_journal_characters
				target = scope:eqfm_desert_expedition_leader_scope
			}
		}
	}
	else = {
		je:je_eqfm_libyan_desert = {
			clear_variable_list = com_journal_characters
		}
	}
}

eqfm_update_desert_modifier_effect = {
	s:STATE_LIBYAN_DESERT.region_state:TRI ?= {
		remove_modifier = modifier_eqfm_desert_scouting
	}
	s:STATE_LIBYAN_DESERT.region_state:TRI ?= {
		add_modifier = {
			name = modifier_eqfm_desert_scouting
			multiplier = eqfm_desert_multiplier_value
		}
	}
}

eqfm_desert_scopes_effect = {
	c:TRI_KUF ?= {
		save_scope_as = eqfm_desert_event_tribal_country
		random_scope_state = {
			save_scope_as = eqfm_desert_event_tribal_state
		}
	}
	s:STATE_LIBYAN_DESERT.region_state:TRI ?= {
		save_scope_as = eqfm_desert_event_state
	}
}

eqfm_complete_desert_expedition_effect = {
	every_scope_character = {
		limit = {
			has_modifier = modifier_eqfm_leading_desert_expedition
		}
		remove_modifier = modifier_eqfm_leading_desert_expedition
		set_character_busy_and_immortal = no
		add_modifier = {
			name = modifier_eqfm_completed_desert_expedition
			months = gbbf_duration_10_years
		}
		if = {
			limit = {
				has_modifier = modifier_eqfm_failed_desert_expedition_character
			}
			remove_modifier = modifier_eqfm_failed_desert_expedition_character
		}
	}
	eqfm_clear_desert_expedition_variables_effect = yes
	eqfm_update_expedition_costs_effect = yes
}

eqfm_fail_desert_expedition_effect = {
	every_scope_character = {
		limit = {
			has_modifier = modifier_eqfm_leading_desert_expedition
		}
		remove_modifier = modifier_eqfm_leading_desert_expedition
		set_character_busy_and_immortal = no
		add_modifier = {
			name = modifier_eqfm_failed_desert_expedition_character
			months = gbbf_duration_10_years
		}
	}
	add_modifier = {
		name = modifier_eqfm_failed_desert_expedition_country
		months = gbbf_duration_2_plus_half_years
	}
	eqfm_clear_desert_expedition_variables_effect = yes
	eqfm_update_expedition_costs_effect = yes
}

eqfm_cancel_desert_expedition_effect = {
	every_scope_character = {
		limit = {
			has_modifier = modifier_eqfm_leading_desert_expedition
		}
		remove_modifier = modifier_eqfm_leading_desert_expedition
		set_character_busy_and_immortal = no
	}
	add_modifier = {
		name = modifier_eqfm_cancelled_desert_expedition
		months = gbbf_duration_2_plus_half_years
	}
	eqfm_clear_desert_expedition_variables_effect = yes
	eqfm_update_expedition_costs_effect = yes
}

eqfm_clear_desert_expedition_variables_effect = {
	remove_variable = eqfm_desert_expedition_var
	remove_variable = eqfm_expedition_desert_transit_var
	remove_variable = eqfm_desert_expedition_progress_var
	remove_variable = eqfm_desert_expedition_peril_var
	remove_variable = eqfm_desert_expedition_leader_var
	remove_variable = eqfm_desert_event_cooldown_var
}

eqfm_cleanup_libyan_desert_effect = {
	remove_variable = eqfm_expedition_ticker_var
	eqfm_clear_desert_expedition_variables_effect = yes
	every_scope_character = {
		limit = {
			has_modifier = modifier_eqfm_leading_desert_expedition
		}
		remove_modifier = modifier_eqfm_leading_desert_expedition
		set_character_busy_and_immortal = no
	}
	eqfm_update_expedition_costs_effect = yes
}
