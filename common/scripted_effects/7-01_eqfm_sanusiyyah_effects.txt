	global_var:je_eqfm_sanusiyyah = {
		every_scope_je_involved = {
			limit = {
				NOR = {
					any_scope_state = {
						is_in_geographic_region = geographic_region_sanusiyyah
					}
					eqfm_sanusiyyah_is_relevant_due_to_subject_trigger = yes
				}
			}
		}
		global_var:je_eqfm_sanusiyyah = {
			remove_involved_country = prev
		}
	}
	every_state_in_sanusiyyah = {
		owner = {
			if = {
				limit = {
					NOT = {
						is_involved_in_journal_entry = je_eqfm_sanusiyyah
					}
				}
				global_var:je_eqfm_sanusiyyah = {
					add_involved_country = prev
				}
			}
			if = {
				limit = {
					eqfm_sanusiyyah_is_relevant_to_overlord_trigger = yes
				}
				overlord ?= {
					if = {
						limit = {
							NOT = {
								is_involved_in_journal_entry = je_eqfm_sanusiyyah
							}
						}
						global_var:je_eqfm_sanusiyyah = {
							add_involved_country = prev
						}
					}
				}
			}
		}
	}
}

eqfm_check_sanusiyyah_phase_effect = {
	global_var:je_eqfm_sanusiyyah ?= {
		if = {
			limit = {
				# Yes, this structure means that we can get pinned at either extreme; there is no natural de-escalation as in the balkans journal
				OR = {
					AND = {
						eqfm_sanusiyyah_can_escalate_trigger = yes
						var:eqfm_sanusiyyah_situation_var >= eqfm_sanusiyyah_situation_max_value
					}
					AND = {
						eqfm_sanusiyyah_can_deescalate_trigger = yes
						var:eqfm_sanusiyyah_situation_var <= eqfm_sanusiyyah_situation_min_value
					}
				}
			}
			# Attempt escalation first
			if = {
				limit = {
					eqfm_sanusiyyah_can_escalate_trigger = yes
					var:eqfm_sanusiyyah_situation_var >= eqfm_sanusiyyah_situation_max_value
				}
				if = {
					limit = {
						eqfm_sanusiyyah_tranquility_phase_trigger = yes
					}
					eqfm_set_sanusiyyah_tranquility_phase_effect = yes
				}
				else_if = {
					limit = {
						eqfm_sanusiyyah_unease_phase_trigger = yes
					}
					eqfm_set_sanusiyyah_unease_phase_effect = yes
				}
				else = {
					error_log = "eqfm_check_sanusiyyah_phase_effect detected valid conditions for phase transition and branched for escalation. However, neither escalation condition returned true."
				}
			}
			# Attempt deescalation second
			else_if = {
				limit = {
					eqfm_sanusiyyah_can_deescalate_trigger = yes
					var:eqfm_sanusiyyah_situation_var <= eqfm_sanusiyyah_situation_min_value
				}
				if = {
					limit = {
						eqfm_sanusiyyah_unease_phase_trigger = yes
					}
					eqfm_set_sanusiyyah_unease_phase_effect = yes
				}
				else_if = {
					limit = {
						eqfm_sanusiyyah_belligerency_phase_trigger = yes
					}
					eqfm_set_sanusiyyah_belligerency_phase_effect = yes
				}
				else = {
					error_log = "eqfm_check_sanusiyyah_phase_effect detected valid conditions for phase transition and branched for deescalation. However, neither deescalation condition returned true."
				}
			}
			else = {
				error_log = "eqfm_check_sanusiyyah_phase_effect detected valid conditions for phase transition. However, neither escalation nor deescalation branch conditions returned true. This should never happen."
			}
		}
	}
	eqfm_update_senussi_progress_bars_effect = yes
}

eqfm_update_senussi_character_effect = {
	global_var:je_eqfm_sanusiyyah ?= {
		## Step 1, Make Sure Senussi Exists and Recapture
		if = {
			limit = {
				any_character_in_void = {
					eqfm_character_is_senussi_trigger = yes
				}
			}
			random_character_in_void = {
				limit = {
					eqfm_character_is_senussi_trigger = yes
				}
				save_scope_as = eqfm_senussi_character_scope
			}
			set_variable = {
				name = eqfm_senussi_character_var
				value = scope:eqfm_senussi_character_scope
			}
		}
		
		## Step 2, Regenerate Senussi and increment counter if needed
		else = {
			ordered_country = {
				limit = {
					is_in_geographic_region = geographic_region_sanusiyyah
				}
				order_by = prestige
				position = 0
				save_scope_as = eqfm_senussi_generator_scope
			}
			scope:eqfm_senussi_generator_scope = {
				if = {
					limit = {
						global_var:je_eqfm_sanusiyyah.var:eqfm_senussi_chacter_counter_var >= 5
					}
					create_character = {
						template = eqfm_senussi_dynamic
						on_created = {
							add_trait = gbbf_trait_marabout
							place_character_in_void = 400
						}
					}
				}
				else_if = {
					limit = {
						global_var:je_eqfm_sanusiyyah.var:eqfm_senussi_chacter_counter_var = 4
					}
					create_character = {
						template = eqfm_senussi_hasan
						on_created = {
							add_trait = gbbf_trait_marabout
							place_character_in_void = 400
						}
					}
				}
				else_if = {
					limit = {
						global_var:je_eqfm_sanusiyyah.var:eqfm_senussi_chacter_counter_var = 3
					}
					create_character = {
						template = eqfm_senussi_idris
						on_created = {
							add_trait = gbbf_trait_marabout
							place_character_in_void = 400
						}
					}
				}
				else_if = {
					limit = {
						global_var:je_eqfm_sanusiyyah.var:eqfm_senussi_chacter_counter_var = 2
					}
					create_character = {
						template = eqfm_senussi_ahmed
						on_created = {
							add_trait = gbbf_trait_marabout
							place_character_in_void = 400
						}
					}
				}
				else_if = {
					limit = {
						global_var:je_eqfm_sanusiyyah.var:eqfm_senussi_chacter_counter_var = 1
					}
					create_character = {
						template = eqfm_senussi_muhammad_ii
						on_created = {
							add_trait = gbbf_trait_marabout
							place_character_in_void = 400
						}
					}
					every_country = {
						limit = {
							is_involved_in_journal_entry = je_eqfm_sanusiyyah
						}
						trigger_event = eqfm_sanusiyyah.003
					}
				}
				else = {
					create_character = {
						template = eqfm_senussi_muhammad_i
						on_created = {
							add_trait = gbbf_trait_marabout
							place_character_in_void = 400
						}
					}
				}
			}
			random_character_in_void = {
				limit = {
					eqfm_character_is_senussi_trigger = yes
				}
				save_scope_as = eqfm_senussi_character_scope
			}
			set_variable = {
				name = eqfm_senussi_character_var
				value = scope:eqfm_senussi_character_scope
			}
			com_change_variable = {
				name = eqfm_senussi_chacter_counter_var
				value = 1
			}
		}
		
		## Step 3, Update the journal list
		clear_variable_list = com_journal_characters
		add_to_variable_list = {
			name = com_journal_characters
			target = scope:eqfm_senussi_character_scope
		}
	}
}

eqfm_update_sanusiyya_georegion_variables_effect = {
	# This is a chonky effect; we only call it monthly since script math is stupendously slow
	every_state_region_in_sanusiyyah = {
		set_variable = {
			name = eqfm_centralized_provinces_var
			value = {
				every_scope_state = {
					limit = {
						owner = {
							NOT = {
								is_country_type = decentralized
							}
						}
					}
					add = num_provinces
				}
			}
		}
		set_variable = {
			name = eqfm_decentralized_provinces_var
			value = {
				every_scope_state = {
					limit = {
						owner = {
							is_country_type = decentralized
						}
					}
					add = num_provinces
				}
			}
		}
		set_variable = {
			name = eqfm_state_region_centralization_ratio_var
			value = {
				add = var:eqfm_centralized_provinces_var
				divide = num_provinces
				round_to = 0.01
			}
		}
		set_variable = {
			name = eqfm_state_region_total_bedouin_population_var
			value = {
				every_scope_state = {
					every_scope_pop = {
						limit = {
							has_pop_culture = bedouin
						}
						add = total_size
					}
				}
			}
		}
		set_variable = {
			name = eqfm_state_region_radical_bedouin_population_var
			value = {
				every_scope_state = {
					every_scope_pop = {
						limit = {
							has_pop_culture = bedouin
						}
						add = {
							value = total_size
							multiply = pop_radical_fraction
						}
					}
				}
				round_to = 1
			}
		}
		set_variable = {
			name = eqfm_state_region_bedouin_radical_fraction_var
			value = {
				add = var:eqfm_state_region_radical_bedouin_population_var
				divide = var:eqfm_state_region_total_bedouin_population_var
				round_to = 0.01
			}
		}
		set_variable = {
			name = eqfm_state_region_bedouin_loyalist_fraction_var
			value = {
				add = 1
				subtract = var:eqfm_state_region_bedouin_radical_fraction_var
				round_to = 0.01
			}
		}
		set_variable = {
			name = eqfm_state_region_bedouin_literacy_rate_var
			value = {
				every_scope_state = {
					every_scope_pop = {
						limit = {
							has_pop_culture = bedouin
						}
						add = {
							value = total_size
							multiply = literacy_rate
						}
					}
				}
				round_to = 0.01
			}
		}
	}
	scope:journal_entry = {
		set_variable = {
			name = eqfm_georegion_total_bedouin_population_var
			value = {
				every_state_region_in_sanusiyyah = {
					add = var:eqfm_state_region_total_bedouin_population_var
				}
			}
		}
		set_variable = {
			name = eqfm_georegion_bedouin_radical_fraction_var
			value = {
				every_state_region_in_sanusiyyah = {
					add = var:eqfm_state_region_radical_bedouin_population_var
				}
				divide = var:eqfm_georegion_total_bedouin_population_var
				round_to = 0.01
			}
		}
	}
}

eqfm_monthly_entrenchment_effect = {
	every_state_region_in_sanusiyyah = {

		if = {
			limit = {
				has_variable = eqfm_sanusiyyah_entrenchment_var
			}
			set_variable = {
				name = eqfm_sanusiyyah_entrenchment_progress_var
				value = eqfm_sanusiyyah_entrenchment_progress_value
			}
			com_change_and_clamp_variable = {
				name = eqfm_sanusiyyah_entrenchment_var
				value = var:eqfm_sanusiyyah_entrenchment_progress_var
				max = 100
				min = 0
			}
			every_scope_state = {
				add_modifier = {
					name = modifier_eqfm_sanusiyyah_stronghold
					multiplier = scope:eqfm_entrenchment_operation_scope.var:eqfm_sanusiyyah_entrenchment_var
				}
			}
		}
	}
}

eqfm_update_desert_scouting_costs_effect = {
	remove_modifier = modifier_eqfm_desert_scouts_cost
	if = {
		limit = {
			has_variable = eqfm_desert_scouts_leader_var
			NOT = {
				exists = var:eqfm_desert_scouts_leader_var
			}
		}
		remove_variable = eqfm_desert_scouts_leader_var
	}
	if = {
		limit = {
			has_variable = eqfm_desert_scouts_guide_var
			NOT = {
				exists = var:eqfm_desert_scouts_guide_var
			}
		}
		remove_variable = eqfm_desert_scouts_guide_var
	}
	if = {
		limit = {
			has_variable = eqfm_desert_scouts_logistician_var
			NOT = {
				exists = var:eqfm_desert_scouts_logistician_var
			}
		}
		remove_variable = eqfm_desert_scouts_logistician_var
	}
	if = {
		limit = {
			eqfm_country_is_scouting_desert_trigger = yes
		}
		add_modifier = {
			name = modifier_eqfm_desert_scouts_cost
			multiplier = {
				value = money_amount_multiplier_very_small
				multiply = {
					if = {
						limit = {
							has_variable = eqfm_desert_scouts_leader_var
						}
						add = 1
					}
					if = {
						limit = {
							has_variable = eqfm_desert_scouts_guide_var
						}
						add = 1
					}
					if = {
						limit = {
							has_variable = eqfm_desert_scouts_logistician_var
						}
						add = 1
					}
				}
			}
		}
	}
}

eqfm_monthly_reconnaissance_progress_effect = {
	# Check the state regions and clear non-existent expedition tags
	every_state_region_in_sanusiyyah = {
		limit = {
			has_variable = eqfm_sanusiyyah_expedition_var
			NOT = {
				exists = var:eqfm_sanusiyyah_expedition_var
			}
		}
		remove_variable = eqfm_sanusiyyah_expedition_var
	}
	# Math and cache what everyone's contributions will be (if any)
	every_scope_je_involved = {
		set_variable = {
			name = eqfm_desert_scouting_reconnaissance_generation_var
			value = eqfm_desert_scouting_reconnaissance_generation_value
		}
	}
	# Apply values to state regions
	every_state_region_in_sanusiyyah = {
		set_variable = {
			name = eqfm_desert_scouting_reconnaissance_progress_var
			value = eqfm_desert_scouting_reconnaissance_progress_value
		}
		com_change_and_clamp_variable = {
			name = eqfm_desert_scouting_reconnaissance_var
			value = var:eqfm_desert_scouting_reconnaissance_progress_var
			max = 100
			min = 0
		}
	}
}
