eqfm_update_global_sanusiyyah_variables_effect = {
	## First, do all the variable math
	set_variable = {
		name = eqfm_sanusiyyah_pressure_progress_var
		value = eqfm_sanusiyyah_pressure_progress_value
	}
	set_variable = {
		name = eqfm_georegion_bedouin_loyalist_fraction_var
		value = eqfm_georegion_bedouin_loyalist_fraction_value
	}
	set_variable = {
		name = eqfm_georegion_bedouin_radical_fraction_var
		value = eqfm_georegion_bedouin_radical_fraction_value
	}
	set_variable = {
		name = eqfm_state_libya_centralization_var
		value = eqfm_state_libya_centralization_value
	}
	set_variable = {
		name = eqfm_state_libyan_desert_centralization_var
		value = eqfm_state_libyan_desert_centralization_value
	}
	set_variable = {
		name = eqfm_state_matruh_centralization_var
		value = eqfm_state_matruh_centralization_value
	}
	set_variable = {
		name = eqfm_state_egyptian_desert_centralization_var
		value = eqfm_state_egyptian_desert_centralization_value
	}
	set_variable = {
		name = eqfm_state_chad_centralization_var
		value = eqfm_state_chad_centralization_value
	}
	
	## Figure out where Senussi should be located & update flags
	if = {
		limit = {
			var:eqfm_sanusiyyah_pressure_var >= 99
		}
		if = {
			limit = {
				var:eqfm_senussi_location_var != flag:eqfm_senussi_court
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_sanusiyyah
				}
				post_notification = eqfm_senussi_entrenched
			}
		}
		set_variable = {
			name = eqfm_senussi_location_var
			value = flag:eqfm_senussi_court
		}
	}
	else_if = {
		limit = {
			var:eqfm_sanusiyyah_pressure_var >= 75
		}
		if = {
			limit = {
				OR = {
					var:eqfm_senussi_location_var = flag:eqfm_senussi_ghosted
					var:eqfm_senussi_location_var = flag:eqfm_senussi_desert
					var:eqfm_senussi_location_var = flag:eqfm_senussi_villages
				}
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_sanusiyyah
				}
				post_notification = eqfm_senussi_urbanized_up
			}
		}
		else_if = {
			limit = {
				var:eqfm_senussi_location_var = flag:eqfm_senussi_court
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_sanusiyyah
				}
				post_notification = eqfm_senussi_urbanized_down
			}
		}
		set_variable = {
			name = eqfm_senussi_location_var
			value = flag:eqfm_senussi_urban
		}
	}
	else_if = {
		limit = {
			var:eqfm_sanusiyyah_pressure_var < 75
			var:eqfm_sanusiyyah_pressure_var >= 50
		}
		if = {
			limit = {
				OR = {
					var:eqfm_senussi_location_var = flag:eqfm_senussi_court
					var:eqfm_senussi_location_var = flag:eqfm_senussi_urban
				}
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_sanusiyyah
				}
				post_notification = eqfm_senussi_villages_down
			}
		}
		if = {
			limit = {
				OR = {
					var:eqfm_senussi_location_var = flag:eqfm_senussi_ghosted
					var:eqfm_senussi_location_var = flag:eqfm_senussi_desert
				}
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_sanusiyyah
				}
				post_notification = eqfm_senussi_villages_up
			}
		}
		set_variable = {
			name = eqfm_senussi_location_var
			value = flag:eqfm_senussi_villages
		}
	}
	else_if = {
		limit = {
			var:eqfm_sanusiyyah_pressure_var < 50
		}
		if = {
			limit = {
				var:eqfm_senussi_location_var = flag:eqfm_senussi_ghosted
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_sanusiyyah
				}
				post_notification = eqfm_senussi_desert_up
			}
		}
		else_if = {
			limit = {
				OR = {
					var:eqfm_senussi_location_var = flag:eqfm_senussi_court
					var:eqfm_senussi_location_var = flag:eqfm_senussi_urban
					var:eqfm_senussi_location_var = flag:eqfm_senussi_villages
				}
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_sanusiyyah
				}
				post_notification = eqfm_senussi_desert_down
			}
		}
		set_variable = {
			name = eqfm_senussi_location_var
			value = flag:eqfm_senussi_desert
		}
	}
	else = {
		if = {
			limit = {
				var:eqfm_senussi_location_var != flag:eqfm_senussi_ghosted
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_sanusiyyah
				}
				post_notification = eqfm_senussi_ghosted
			}
		}
		set_variable = {
			name = eqfm_senussi_location_var
			value = flag:eqfm_senussi_ghosted
		}
	}
	
	## Lastly, do the phase progress math and check phase
	set_variable = {
		name = eqfm_sanusiyyah_situation_progress_var
		value = eqfm_sanusiyyah_situation_progress_value
	}
	com_change_and_clamp_variable = {
		name = eqfm_sanusiyyah_situation_var
		value = var:eqfm_sanusiyyah_situation_progress_var
		max = 100
		min = 0
	}
	eqfm_check_sanusiyyah_phase_effect = yes
	eqfm_update_senussi_progress_bars_effect = yes
}

	global_var:je_eqfm_sanusiyyah ?= {
		set_bar_progress = {
			name = eqfm_sanusiyyah_pressure_progress_bar
			value = var:eqfm_sanusiyyah_pressure_var
		}

		}
	}
}

eqfm_check_sanusiyyah_phase_effect = {
	global_var:je_eqfm_sanusiyyah ?= {
		if = {
			limit = {
				# Yes, this structure means that we can get pinned at either extreme; there is no natural de-escalation as in the balkans journal
				OR = {
					AND = {
						eqfm_sanusiyyah_can_escalate_trigger = yes
						var:eqfm_sanusiyyah_situation_var >= eqfm_sanusiyyah_situation_max_value
					}
					AND = {
						eqfm_sanusiyyah_can_deescalate_trigger = yes
						var:eqfm_sanusiyyah_situation_var <= eqfm_sanusiyyah_situation_min_value
					}
				}
			}
			# Attempt escalation first
			if = {
				limit = {
					eqfm_sanusiyyah_can_escalate_trigger = yes
					var:eqfm_sanusiyyah_situation_var >= eqfm_sanusiyyah_situation_max_value
				}
				if = {
					limit = {
						eqfm_sanusiyyah_tranquility_phase_trigger = yes
					}
					eqfm_set_sanusiyyah_tranquility_phase_effect = yes
				}
				else_if = {
					limit = {
						eqfm_sanusiyyah_unease_phase_trigger = yes
					}
					eqfm_set_sanusiyyah_unease_phase_effect = yes
				}
				else = {
					error_log = "eqfm_check_sanusiyyah_phase_effect detected valid conditions for phase transition and branched for escalation. However, neither escalation condition returned true."
				}
			}
			# Attempt deescalation second
			else_if = {
				limit = {
					eqfm_sanusiyyah_can_deescalate_trigger = yes
					var:eqfm_sanusiyyah_situation_var <= eqfm_sanusiyyah_situation_min_value
				}
				if = {
					limit = {
						eqfm_sanusiyyah_unease_phase_trigger = yes
					}
					eqfm_set_sanusiyyah_unease_phase_effect = yes
				}
				else_if = {
					limit = {
						eqfm_sanusiyyah_belligerency_phase_trigger = yes
					}
					eqfm_set_sanusiyyah_belligerency_phase_effect = yes
				}
				else = {
					error_log = "eqfm_check_sanusiyyah_phase_effect detected valid conditions for phase transition and branched for deescalation. However, neither deescalation condition returned true."
				}
			}
			else = {
				error_log = "eqfm_check_sanusiyyah_phase_effect detected valid conditions for phase transition. However, neither escalation nor deescalation branch conditions returned true. This should never happen."
			}
		}
	}
	eqfm_update_senussi_progress_bars_effect = yes
}

eqfm_update_sanusiyyah_modifier_effect = {
	if = {
		limit = {
			has_modifier = modifier_eqfm_sanusiyyah_movement
		}
		remove_modifier = modifier_eqfm_sanusiyyah_movement
	}
	if = {
		limit = {
			has_variable = eqfm_sanusiyyah_influence_var
			has_global_variable = je_eqfm_sanusiyyah
		}
		add_modifier = {
			name = modifier_eqfm_sanusiyyah_movement
			multiplier = {
				value = 0
				add = global_var:je_eqfm_sanusiyyah.var:eqfm_sanusiyyah_pressure_var
				divide = 100
				round_to = 0.01
				min = 0
				max = 1
			}
		}
	}
}

eqfm_validate_sanusiyyah_involvement_effect = {
	every_country = {
		limit = {
			OR = {
				AND = {
					NOT = {
						is_involved_in_journal_entry = je_eqfm_sanusiyyah
					}
					eqfm_sanusiyyah_involvement_trigger = yes
				}
				AND = {
					is_involved_in_journal_entry = je_eqfm_sanusiyyah
					eqfm_sanusiyyah_involvement_trigger = no
				}
			}
		}
		if = {
			limit = {
				NOT = {
					is_involved_in_journal_entry = je_eqfm_sanusiyyah
				}
				eqfm_sanusiyyah_involvement_trigger = yes
			}

		if = {
			limit = {
				any_character_in_void = {
					eqfm_character_is_senussi_trigger = yes
				}
			}
			random_character_in_void = {
				limit = {
					eqfm_character_is_senussi_trigger = yes
				}
				save_scope_as = eqfm_senussi_character_scope
			}
			set_variable = {
				name = eqfm_senussi_character_var
				value = scope:eqfm_senussi_character_scope
			}
		}
		
		## Step 2, Regenerate Senussi and increment counter if needed
		else = {
			ordered_country = {
				limit = {
					is_in_geographic_region = geographic_region_sanusiyyah
				}
				order_by = prestige
				position = 0
				save_scope_as = eqfm_senussi_generator_scope
			}
			scope:eqfm_senussi_generator_scope = {
				if = {
					limit = {
						global_var:je_eqfm_sanusiyyah.var:eqfm_senussi_chacter_counter_var >= 5
					}
					create_character = {
						template = eqfm_senussi_dynamic
						on_created = {
							add_trait = gbbf_trait_marabout
							place_character_in_void = 400
						}
					}
				}
				else_if = {
					limit = {
						global_var:je_eqfm_sanusiyyah.var:eqfm_senussi_chacter_counter_var = 4
					}
					create_character = {
						template = eqfm_senussi_hasan
						on_created = {
							add_trait = gbbf_trait_marabout
							place_character_in_void = 400
						}
					}
				}
				else_if = {
					limit = {
						global_var:je_eqfm_sanusiyyah.var:eqfm_senussi_chacter_counter_var = 3
					}
					create_character = {
						template = eqfm_senussi_idris
						on_created = {
							add_trait = gbbf_trait_marabout
							place_character_in_void = 400
						}
					}
				}
				else_if = {
					limit = {
						global_var:je_eqfm_sanusiyyah.var:eqfm_senussi_chacter_counter_var = 2
					}
					create_character = {
						template = eqfm_senussi_ahmed
						on_created = {
							add_trait = gbbf_trait_marabout
							place_character_in_void = 400
						}
					}
				}
				else_if = {
					limit = {
						global_var:je_eqfm_sanusiyyah.var:eqfm_senussi_chacter_counter_var = 1
					}
					create_character = {
						template = eqfm_senussi_muhammad_ii
						on_created = {
							add_trait = gbbf_trait_marabout
							place_character_in_void = 400
						}
					}
					every_country = {
						limit = {
							is_involved_in_journal_entry = je_eqfm_sanusiyyah
						}
						trigger_event = eqfm_sanusiyyah.002
					}
				}
				else = {
					create_character = {
						template = eqfm_senussi_muhammad_i
						on_created = {
							add_trait = gbbf_trait_marabout
							place_character_in_void = 400
						}
					}
				}
			}
			random_character_in_void = {
				limit = {
					eqfm_character_is_senussi_trigger = yes
				}
				save_scope_as = eqfm_senussi_character_scope
			}
			set_variable = {
				name = eqfm_senussi_character_var
				value = scope:eqfm_senussi_character_scope
			}
			com_change_variable = {
				name = eqfm_senussi_chacter_counter_var
				value = 1
			}
		}

		clear_variable_list = com_journal_characters
		add_to_variable_list = {
			name = com_journal_characters
			target = scope:eqfm_senussi_character_scope
		}
	}

eqfm_update_sanusiyyah_expedition_variables_effect = {
	## Step 1, Check for Expedition Leader
	scope:journal_entry = {
		clear_variable_list = com_journal_characters
	}
	if = {
		limit = {
			any_scope_general = {
				has_modifier = modifier_eqfm_leading_sanusiyyah_expedition
			}
		}
		random_scope_general = {
			limit = {
				has_modifier = modifier_eqfm_leading_sanusiyyah_expedition
			}
			save_scope_as = eqfm_sanusiyyah_expedition_leader_scope
		}
		scope:journal_entry = {
			add_to_variable_list = {
				name = com_journal_characters
				target = scope:eqfm_sanusiyyah_expedition_leader_scope
			}
		}
	}
	
	## Step 2, Validate Expedition Variables
	if = {
		limit = {
			eqfm_has_sanusiyyah_expedition_trigger = yes
			NOT = {
				has_variable = eqfm_sanusiyyah_expedition_transit_var
			}
		}
		if = {
			limit = {
				NOT = {
					has_variable = eqfm_sanusiyyah_expedition_progress_var
				}
			}
			set_variable = {
				name = eqfm_sanusiyyah_expedition_progress_var
				value = 0
			}
		}
		if = {
			limit = {
				NOT = {
					has_variable = eqfm_sanusiyyah_expedition_peril_var
				}
			}
			set_variable = {
				name = eqfm_sanusiyyah_expedition_peril_var
				value = 0
			}
		}
	}
	
	## Step 3, Update Progress Bars
	if = {
		limit = {
			has_variable = eqfm_sanusiyyah_expedition_progress_var
		}
		clamp_variable = {
			name = eqfm_sanusiyyah_expedition_progress_var
			max = 8
			min = 0
		}
		scope:journal_entry = {
			set_bar_progress = {
				name = eqfm_sanusiyyah_expedition_progress_bar
				value = owner.var:eqfm_sanusiyyah_expedition_progress_var
			}
		}
	}
	if = {
		limit = {
			has_variable = eqfm_sanusiyyah_expedition_peril_var
		}
		clamp_variable = {
			name = eqfm_sanusiyyah_expedition_peril_var
			max = 5
			min = 0
		}
		scope:journal_entry = {
			set_bar_progress = {
				name = eqfm_sanusiyyah_expedition_peril_bar
				value = owner.var:eqfm_sanusiyyah_expedition_peril_var
			}
		}
	}
}

eqfm_annual_sanusiyyah_expedition_progress_effect = {
	if = {
		limit = {
			NOT = {
				has_variable = eqfm_sanusiyyah_expedition_transit_var
			}
		}
		if = {
			limit = {
				has_variable = eqfm_sanusiyyah_expedition_progress_timer
				
			}
			change_variable = {
				name = eqfm_sanusiyyah_expedition_progress_timer
				subtract = 1
			}
			if = {
				limit = {
					var:eqfm_sanusiyyah_expedition_progress_timer <= 0
				}
				eqfm_add_expedition_progress_effect = {
					type = sanusiyyah
					value = 1
				}
				set_variable = {
					name = eqfm_sanusiyyah_expedition_progress_timer
					value = 12
				}
				post_notification = eqfm_sanusiyyah_yearly_progress
			}
		}
		else = {
			set_variable = {
				name = eqfm_sanusiyyah_expedition_progress_timer
				value = 12
			}
			error_log = "Scripted effect 'eqfm_update_sanusiyyah_expedition_progress_effect' was called, but variable 'eqfm_sanusiyyah_expedition_progress_timer' was not set. Variable has been set with a value of 12."
		}
	}
}
