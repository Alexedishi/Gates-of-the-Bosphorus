turkfm_senussi_weekly_effect = {
	## Step 1, Check for Libyan Desert & Bedouin Involvement
	set_variable = {
		name = turkfm_senussi_support_var
		value = turkfm_senussi_support_value
	}
	set_variable = {
		name = turkfm_senussi_opposition_var
		value = turkfm_senussi_opposition_value
	}
	set_variable = {
		name = turkfm_senussi_desert_centralized_var
		value = turkfm_senussi_desert_centralized_value
	}
	set_variable = {
		name = turkfm_senussi_desert_decentralized_var
		value = turkfm_senussi_desert_decentralized_value
	}
	set_variable = {
		name = turkfm_bedouin_radical_fraction_var
		value = turkfm_bedouin_radical_fraction_value
	}

	## Step 2, Set Progress Variable
	set_variable = {
		name = turkfm_senussi_progress_var
		value = turkfm_senussi_progress_value
	}

	## Step 3, Update Senusiyya Modifier
	remove_modifier = modifier_turkfm_senusiyya_movement
	add_modifier = {
		name = modifier_turkfm_senusiyya_movement
		multiplier = turkfm_senussi_multiplier_value
	}

	##Step 4, Update the Progress Bar
	if = {
		limit = {
			has_journal_entry = je_turkfm_senusiyya_main
		}
		je:je_turkfm_senusiyya_main = {
			set_bar_progress = {
				name = turkfm_senussi_progressbar
				value = owner.var:turkfm_senussi_var
			}
		}

	}
	else_if = {
		limit = {
			has_journal_entry = je_turkfm_senusiyya_sub
		}
		je:je_turkfm_senusiyya_sub = {
			set_bar_progress = {
				name = turkfm_senussi_progressbar
				value = owner.var:turkfm_senussi_var
			}
		}

	}
}

turkfm_update_senussi_effect = {
	## Step 1, Make Sure Senussi Exists and Recapture/Regenerate as appropriate
	if = {
		limit = {
			any_character_in_void = {
				turkfm_is_senussi_trigger = yes
			}
		}
		random_character_in_void = {
			limit = {
				turkfm_is_senussi_trigger = yes
			}
			save_scope_as = turkfm_senussi_character_scope
		}
		set_variable = {
			name = turkfm_senussi_character_var
			value = scope:turkfm_senussi_character_scope
		}
	}
	else = {
		if = {
			limit = {
				global_var:turkfm_senussi_counter_var >= 5
			}
			create_character = {
				template = template_turkfm_senussi_dynamic
				on_created = {
					add_trait = gbbf_trait_marabout
					place_character_in_void = 240
				}
			}
		}
		else_if = {
			limit = {
				global_var:turkfm_senussi_counter_var = 4
			}
			create_character = {
				template = template_turkfm_senussi_hasan
				on_created = {
					add_trait = gbbf_trait_marabout
					place_character_in_void = 240
				}
			}
		}
		else_if = {
			limit = {
				global_var:turkfm_senussi_counter_var = 3
			}
			create_character = {
				template = template_turkfm_senussi_idris
				on_created = {
					add_trait = gbbf_trait_marabout
					place_character_in_void = 240
				}
			}
		}
		else_if = {
			limit = {
				global_var:turkfm_senussi_counter_var = 2
			}
			create_character = {
				template = template_turkfm_senussi_ahmed
				on_created = {
					add_trait = gbbf_trait_marabout
					place_character_in_void = 240
				}
			}
		}
		else_if = {
			limit = {
				global_var:turkfm_senussi_counter_var = 1
			}
			create_character = {
				template = template_turkfm_senussi_muhammad_ii
				on_created = {
					add_trait = gbbf_trait_marabout
					place_character_in_void = 240
				}
			}
		}
		else = {
			create_character = {
				template = template_turkfm_senussi_muhammad_i
				on_created = {
					add_trait = gbbf_trait_marabout
					place_character_in_void = 120
				}
			}
		}
		random_character_in_void = {
			limit = {
				turkfm_is_senussi_trigger = yes
			}
			save_scope_as = turkfm_senussi_character_scope
		}
		set_variable = {
			name = turkfm_senussi_character_var
			value = scope:turkfm_senussi_character_scope
		}
		change_global_variable = {
			name = turkfm_senussi_counter_var
			add = 1
		}
	}

	## Step 2, Update the journal list
	if = {
		limit = {
			has_journal_entry = je_turkfm_senusiyya_main
		}
		je:je_turkfm_senusiyya_main = {
			clear_variable_list = com_journal_characters
			add_to_variable_list = {
				name = com_journal_characters
				target = scope:turkfm_senussi_character_scope
			}
		}
	}
	else_if = {
		limit = {
			has_journal_entry = je_turkfm_senusiyya_sub
		}
		je:je_turkfm_senusiyya_sub = {
			clear_variable_list = com_journal_characters
			add_to_variable_list = {
				name = com_journal_characters
				target = scope:turkfm_senussi_character_scope
			}
		}
	}

	## Step 3, Figure out where Senussi should be located & update flags
	if = {
		limit = {
			var:turkfm_senussi_var >= 99
		}
		if = {
			limit = {
				var:turkfm_senussi_location_var != flag:turkfm_senussi_court
			}
			post_notification = notification_turkfm_senussi_entrenched
		}
		set_variable = {
			name = turkfm_senussi_location_var
			value = flag:turkfm_senussi_court
		}
	}
	else_if = {
		limit = {
			var:turkfm_senussi_var >= 75
		}
		if = {
			limit = {
				OR = {
					var:turkfm_senussi_location_var = flag:turkfm_senussi_ghosted
					var:turkfm_senussi_location_var = flag:turkfm_senussi_desert
					var:turkfm_senussi_location_var = flag:turkfm_senussi_villages
				}
			}
			post_notification = notification_turkfm_senussi_urbanized_up
		}
		else_if = {
			limit = {
				var:turkfm_senussi_location_var = flag:turkfm_senussi_court
			}
			post_notification = notification_turkfm_senussi_urbanized_down
		}
		set_variable = {
			name = turkfm_senussi_location_var
			value = flag:turkfm_senussi_urban
		}
	}
	else_if = {
		limit = {
			var:turkfm_senussi_var < 75
			var:turkfm_senussi_var >= 50
		}
		if = {
			limit = {
				OR = {
					var:turkfm_senussi_location_var = flag:turkfm_senussi_court
					var:turkfm_senussi_location_var = flag:turkfm_senussi_urban
				}
			}
			post_notification = notification_turkfm_senussi_villages_down
		}
		if = {
			limit = {
				OR = {
					var:turkfm_senussi_location_var = flag:turkfm_senussi_ghosted
					var:turkfm_senussi_location_var = flag:turkfm_senussi_desert
				}

			}
			post_notification = notification_turkfm_senussi_villages_up
		}
		set_variable = {
			name = turkfm_senussi_location_var
			value = flag:turkfm_senussi_villages
		}
	}
	else_if = {
		limit = {
			var:turkfm_senussi_var < 50
		}
		if = {
			limit = {
				var:turkfm_senussi_location_var = flag:turkfm_senussi_ghosted
			}
			post_notification = notification_turkfm_senussi_desert_up
		}
		else_if = {
			limit = {
				OR = {
					var:turkfm_senussi_location_var = flag:turkfm_senussi_court
					var:turkfm_senussi_location_var = flag:turkfm_senussi_urban
					var:turkfm_senussi_location_var = flag:turkfm_senussi_villages
				}
			}
			post_notification = notification_turkfm_senussi_desert_down
		}
		set_variable = {
			name = turkfm_senussi_location_var
			value = flag:turkfm_senussi_desert
		}
	}
	else = {
		if = {
			limit = {
				var:turkfm_senussi_location_var != flag:turkfm_senussi_ghosted
			}
			post_notification = notification_turkfm_senussi_ghosted
		}
		set_variable = {
			name = turkfm_senussi_location_var
			value = flag:turkfm_senussi_ghosted
		}
	}
	scope:turkfm_senussi_character_scope = {
		set_variable = {
			name = com_opinion
			value = root.var:turkfm_senussi_location_var
		}
	}

}

turkfm_cleanup_senussi_effect = {
	remove_modifier = modifier_turkfm_senusiyya_movement
	remove_variable = turkfm_senussi_var
	remove_variable = turkfm_senussi_progress_var
	remove_variable = turkfm_senussi_support_var
	remove_variable = turkfm_senussi_opposition_var
	remove_variable = turkfm_senussi_character_var
	remove_variable = turkfm_senussi_location_var
	remove_variable = turkfm_senussi_desert_centralized_var
	remove_variable = turkfm_senussi_desert_decentralized_var
	remove_variable = turkfm_bedouin_radical_fraction_var
	remove_variable = turkfm_senussi_outcome_var
	remove_global_variable = turkfm_intimidated_sensussi
}
