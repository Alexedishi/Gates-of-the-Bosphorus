eqfm_validate_balkan_powder_keg_involvement_effect = {
	every_country = {
		limit = {
			OR = {
				is_involved_in_journal_entry = je_eqfm_balkan_situation
				eqfm_balkans_should_be_involved_trigger = yes
			}
		}
		if = {
			limit = {
				is_involved_in_journal_entry = je_eqfm_balkan_situation
				eqfm_balkans_should_be_involved_trigger = no
			}
			global_var:eqfm_balkan_situation = {
				remove_involved_country = prev
			}
		}
		if = {
			limit = {
				NOT = {
					is_involved_in_journal_entry = je_eqfm_balkan_situation
				}
				eqfm_balkans_should_be_involved_trigger = yes
			}
			global_var:eqfm_balkan_situation = {
				add_involved_country = prev
			}
		}
	}
}

eqfm_check_balkan_powder_keg_phase_effect = {
	# First, run the Crisis Failure logic
	if = {
		limit = {
			eqfm_crisis_phase_trigger = yes
			"global_variable_map(eqfm_balkan_powder_keg_varmap|flag:eqfm_balkan_powder_keg_crisis_var)" <= eqfm_balkan_powder_keg_crisis_min_value
		}
		every_country = {
			limit = {
				is_involved_in_journal_entry = je_eqfm_balkan_situation
			}
			trigger_event = eqfm_balkans.028
		}
		eqfm_cleanup_crisis_effect = yes
		eqfm_force_deconfliction_effect = yes
	}
	# Next, check if either transition value is at maximum
	else_if = {
		limit = {
			OR = {
				AND = {
					eqfm_balkan_powder_keg_can_escalate_trigger = yes
					"global_variable_map(eqfm_balkan_powder_keg_varmap|flag:eqfm_balkan_powder_keg_escalation_var)" >= eqfm_balkan_powder_keg_escalation_max_value
				}
				AND = {
					eqfm_balkan_powder_keg_can_deescalate_trigger = yes
					"global_variable_map(eqfm_balkan_powder_keg_varmap|flag:eqfm_balkan_powder_keg_deescalation_var)" >= eqfm_balkan_powder_keg_deescalation_max_value
				}
			}
		}
		# Attempt to Escalate First
		if = {
			limit = {
				"global_variable_map(eqfm_balkan_powder_keg_varmap|flag:eqfm_balkan_powder_keg_escalation_var)" >= eqfm_balkan_powder_keg_escalation_max_value
			}
			# From Stable to Unrest
			if = {
				limit = {
					eqfm_stable_phase_trigger = yes
				}
				remove_from_global_variable_map = {
					name = eqfm_balkan_powder_keg_varmap
					key = flag:eqfm_balkan_powder_keg_phase_var
				}
				add_to_global_variable_map = {
					name = eqfm_balkan_powder_keg_varmap
					key = flag:eqfm_balkan_powder_keg_phase_var
					value = flag:eqfm_unrest
				}
				every_country = {
					limit = {
						is_involved_in_journal_entry = je_eqfm_balkan_situation
					}
					post_notification = eqfm_transition_unrest
				}
			}
			# From Unrest to Hostility
			else_if = {
				limit = {
					eqfm_unrest_phase_trigger = yes
				}
				remove_from_global_variable_map = {
					name = eqfm_balkan_powder_keg_varmap
					key = flag:eqfm_balkan_powder_keg_phase_var
				}
				add_to_global_variable_map = {
					name = eqfm_balkan_powder_keg_varmap
					key = flag:eqfm_balkan_powder_keg_phase_var
					value = flag:eqfm_hostility
				}
				every_country = {
					limit = {
						is_involved_in_journal_entry = je_eqfm_balkan_situation
					}
					post_notification = eqfm_transition_hostility
				}
			}
			# From Hostility to Crisis
			else_if = {
				limit = {
					eqfm_hostility_phase_trigger = yes
					eqfm_crisis_possible_trigger = yes
				}
				remove_from_global_variable_map = {
					name = eqfm_balkan_powder_keg_varmap
					key = flag:eqfm_balkan_powder_keg_phase_var
				}
				add_to_global_variable_map = {
					name = eqfm_balkan_powder_keg_varmap
					key = flag:eqfm_balkan_powder_keg_phase_var
					value = flag:eqfm_crisis
				}
				every_country = {
					limit = {
						is_involved_in_journal_entry = je_eqfm_balkan_situation
					}
					post_notification = eqfm_transition_crisis
				}
				eqfm_balkan_powder_keg_start_crisis_effect = yes
			}
			# From Deconfliction to Stable
			else_if = {
				limit = {
					eqfm_deconfliction_phase_trigger = yes
				}
				remove_from_global_variable_map = {
					name = eqfm_balkan_powder_keg_varmap
					key = flag:eqfm_balkan_powder_keg_phase_var
				}
				add_to_global_variable_map = {
					name = eqfm_balkan_powder_keg_varmap
					key = flag:eqfm_balkan_powder_keg_phase_var
					value = flag:eqfm_stable
				}
				every_country = {
					limit = {
						is_involved_in_journal_entry = je_eqfm_balkan_situation
					}
					post_notification = eqfm_transition_stable
				}
			}
		}
		
		# Attempt to De-escalate Second
		else_if = {
			limit = {
				eqfm_balkan_powder_keg_can_deescalate_trigger = yes
				"global_variable_map(eqfm_balkan_powder_keg_varmap|flag:eqfm_balkan_powder_keg_deescalation_var)" >= eqfm_balkan_powder_keg_deescalation_max_value
			}
			if = {
				limit = {
					eqfm_crisis_phase_trigger = yes
				}
				every_country = {
					limit = {
						is_involved_in_journal_entry = je_eqfm_balkan_situation
					}
					post_notification = eqfm_averted_crisis
				}
				c:TUR = {
					trigger_event = eqfm_balkans.029
				}
			}
			else = {
				every_country = {
					limit = {
						is_involved_in_journal_entry = je_eqfm_balkan_situation
					}
					post_notification = eqfm_transition_deconfliction
				}
			}
			remove_from_global_variable_map = {
				name = eqfm_balkan_powder_keg_varmap
				key = flag:eqfm_balkan_powder_keg_phase_var
			}
			add_to_global_variable_map = {
				name = eqfm_balkan_powder_keg_varmap
				key = flag:eqfm_balkan_powder_keg_phase_var
				value = flag:eqfm_deconfliction
			}
		}
		
		# Reset progress variables to zero
		eqfm_reset_powder_keg_phase_progress_effect = yes
	}
	eqfm_update_balkan_powder_keg_modifiers_effect = yes
}

			if = {
				limit = {

			}
			every_political_movement = {

					}
					remove_modifier = modifier_eqfm_stable_phase_movement

					limit = {

					}
					remove_modifier = modifier_eqfm_unrest_phase_movement
				}
				
				if = {

				}

turkfm_hellenic_separatism_effect = {
	if = {
		limit = {
			has_modifier = modifier_eqfm_supporting_gunrunners
		}
		remove_modifier = modifier_eqfm_supporting_gunrunners
		add_modifier = {
			name = modifier_eqfm_supporting_gunrunners
			multiplier = money_amount_multiplier_medium
		}
		if = {
			limit = {
				c:TUR ?= {
					OR = {
						owns_entire_state_region = STATE_THESSALIA
						owns_entire_state_region = STATE_MACEDONIA
						owns_entire_state_region = STATE_SKOPIA
						owns_entire_state_region = STATE_CRETE
						owns_entire_state_region = STATE_AYDIN
						owns_entire_state_region = STATE_CYPRUS
					}
				}
			}
			random_list = {
				1 = {
					trigger = {
						c:TUR = {
							owns_entire_state_region = STATE_THESSALIA
						}
					}
