turkfm_initialize_sick_man = {
	set_variable = {
		name = turkfm_sick_man_var
		value = 1
	}

	add_journal_entry = {
		type = je_turkfm_sick_man_syria
	}
	add_journal_entry = {
		type = je_turkfm_sick_man_crete
	}
	add_journal_entry = {
		type = je_turkfm_sick_man_kurdistan
	}
	add_journal_entry = {
		type = je_sick_man_main
	}

	if = {
		limit = {
			gbbf_controls_anatolia_trigger = yes
		}
		set_variable = gbbf_controls_anatolia
	}
	if = {
		limit = {
			gbbf_controls_balkan_highlands_trigger = yes
		}
		set_variable = gbbf_controls_balkan_highlands
	}
	if = {
		limit = {
			gbbf_controls_aegean_sea_trigger = yes
		}
		set_variable = gbbf_controls_aegean_sea
	}
	if = {
		limit = {
			gbbf_controls_levant_trigger = yes
		}
		set_variable = gbbf_controls_levant
	}
	if = {
		limit = {
			gbbf_controls_nile_river_trigger = yes
		}
		set_variable = gbbf_controls_nile_river
	}
	if = {
		limit = {
			gbbf_controls_middle_east_highlands_trigger = yes
		}
		set_variable = gbbf_controls_middle_east_highlands
	}
	if = {
		limit = {
			gbbf_controls_arabia_trigger = yes
		}
		set_variable = gbbf_controls_arabia
	}
	if = {
		limit = {
			gbbf_controls_danubia_trigger = yes
		}
		set_variable = gbbf_controls_danubia
	}
}

turkfm_calculate_sick_man_effect = {
	set_variable = {
		name = turkfm_sick_man_var
		value = turkfm_sick_man_value
	}
	je:je_turkfm_sick_man_main = {
		set_bar_progress = {
			name = turkfm_sick_main_progress_bar
			value = owner.var:turkfm_sick_man_var
		}
	}
}

turkfm_cleanup_sick_man = {
	remove_variable = turkfm_sick_man_var
	remove_variable = gbbf_controls_anatolia
	remove_variable = gbbf_controls_balkan_highlands
	remove_variable = gbbf_controls_aegean_sea
	remove_variable = gbbf_controls_levant
	remove_variable = gbbf_controls_nile_river
	remove_variable = gbbf_controls_arabia
	remove_variable = gbbf_controls_middle_east_highlands
	remove_variable = gbbf_controls_danubia
}
