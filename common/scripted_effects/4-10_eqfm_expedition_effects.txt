## Expedition Maintenance Effects ##
turkfm_update_expedition_costs_effect = {
	hidden_effect = {
		remove_modifier = modifier_turkfm_deployed_expedition
		add_modifier = {
			name = modifier_turkfm_deployed_expedition
			multiplier = turkfm_expedition_value
		}
	}
}

## Expedition Progress/Peril Effects ##
# GotB uses the following expedition keys:
# black_mountain
# adriatic
# desert

turkfm_add_small_peril_effect = {
	custom_tooltip = {
		text = turkfm_peril_plus_1
		change_variable = {
			name = eqfm_$type$_expedition_peril_var
			add = 1
		}
		clamp_variable = {
			name = eqfm_$type$_expedition_peril_var
			max = 5
			min = 0
		}
	}
}
