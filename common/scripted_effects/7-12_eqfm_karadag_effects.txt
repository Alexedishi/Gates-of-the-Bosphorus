eqfm_initialize_karadag_effect = {
	# This is modified from je_montenegrin_raiding's immediate block; effect is fired from global if IP3 is active
	c:MON ?= {
		set_variable = {
			name = eqfm_captured_nobles_var
			value = 0
		}
		set_variable = {
			name = raiding_intensity
			value = 50
		}
		set_variable = {
			name = last_week_raiding_income
			value = 1281
		}
		create_diplomatic_pact = {
			country = c:TUR
			type = raiding_pact
		}
		create_diplomatic_pact = {
			country = c:RUS
			type = disapproval_pact
		}
		add_modifier = {
			name = modifier_montenegrin_raiding
			multiplier = 1281
		}
		add_modifier = {
			name = modifier_montenegrin_disobedience
			multiplier = 50
		}
		random_neighbouring_state = {
			limit = {
				owner ?= c:TUR
			}
			save_scope_as = raid_target
			add_modifier = {
				name = raid_target
				multiplier = 1
			}
		}
		set_global_variable = {
			name = mon_raid_state
			value = scope:raid_target
		}
		
		
		add_journal_entry = {
			type = je_montenegrin_raiding
		}
		add_modifier = modifier_eqfm_practicing_headtaking_mon
	}
	c:BOS ?= {
	
	}
	c:TUR ?= {
		add_modifier = modifier_eqfm_practicing_headtaking_tur
	}
}

## Karadag Raiding Income Effects ##
eqfm_update_karadag_variables_effect = {
	if = {
		limit = {
			NOT = {
				has_global_variable = eqfm_montenegrin_raiding_settled
			}
		}
		# Montenegro Raiding Math
		c:MON ?= {
			## Reworked script from the IP3 journal
			if = {
				limit = {
					has_variable = mon_more_raiding
				}
				change_variable = {
					name = raiding_intensity
					add = 1
				}
			}
			else_if = {
				limit = {
					has_variable = mon_less_raiding
				}
				change_variable = {
					name = raiding_intensity
					subtract = 1
				}
			}
			clamp_variable = {
				name = raiding_intensity
				max = 100
				min = 0
			}
			if = {
				limit = {
					var:raiding_intensity >= 100
					has_variable = mon_more_raiding
				}
				remove_variable = mon_more_raiding
				remove_variable = eqfm_raiding_status_cooldown_var
			}
			if = {
				limit = {
					var:raiding_intensity <= 0
					has_variable = mon_less_raiding
				}
				remove_variable = mon_less_raiding
				remove_variable = eqfm_raiding_status_cooldown_var
			}
			set_variable = {
				name = eqfm_captured_nobles_var
				value = eqfm_captured_nobles_value
			}
			
			## IP3 Raiding Income Effects ##
			eqfm_check_raiding_target_effect = yes
			eqfm_update_raiding_income_effect = yes
		}
	}
}

eqfm_check_raiding_target_effect = {
	scope:raid_target ?= {
		if = {
			limit = {
				NOT = {
					owner ?= c:TUR
				}
			}
			remove_modifier = raid_target
			random_neighbouring_state = {
				limit = {
					owner ?= c:TUR
				}
				save_scope_as = raid_target
				if = {
					limit = {
						eqfm_raiding_level_value > 0
					}
					add_modifier = {
						name = raid_target
						multiplier = eqfm_raiding_level_value
					}
				}
			}
		}
		#still state scope. add some radicals
		if = {
			limit = {
				eqfm_raiding_level_value > 1
			}
			add_radicals_in_state = {
				value = 0.005
			}
		}
	}
}

eqfm_update_raiding_income_effect = {
	#If Raiding Income has changed, remove and reapply the modifiers with the new values. This is necessary because we do not have support for dynamic modifiers.
	if = {
		limit = {
			var:last_week_raiding_income != eqfm_raiding_income_value
		}
		remove_modifier = modifier_montenegrin_raiding
		add_modifier = {
			name = modifier_montenegrin_raiding
			multiplier = eqfm_raiding_income_value
		}
		remove_modifier = modifier_montenegrin_disobedience
		add_modifier = {
			name = modifier_montenegrin_disobedience
			multiplier = var:raiding_intensity
		}
		scope:journal_entry = {
			set_bar_progress = {
				value = root.var:raiding_intensity
				name = montenegrin_raiding_progress_bar
			}
		}
		set_variable = {
			name = last_week_raiding_income
			value = eqfm_raiding_income_value
		}
		scope:raid_target ?= {
			remove_modifier = raid_target
			add_modifier = {
				name = raid_target
				multiplier = eqfm_raiding_level_value
			}
		}
	}
}


## Karadag Expedition Effects ##
eqfm_check_karadag_expeditions_effect = {
	global_var:je_eqfm_karadag = {
		clear_variable_list = com_journal_characters
	}
	c:MON ?= {
		if = {
			limit = {
				eqfm_has_karadag_expedition_trigger = yes
			}
			random_scope_character = {
				limit = {
					has_modifier = modifier_eqfm_leading_karadag_expedition
				}
				save_scope_as = eqfm_karadag_mon_leader
			}
			if = {
				limit = {
					exists = scope:eqfm_karadag_mon_leader
				}
				set_variable = {
					name = eqfm_karadag_leader_var
					value = scope:eqfm_karadag_mon_leader
				}
			}
			else = {
				remove_variable = eqfm_karadag_leader_var
			}
			if = {
				limit = {
					NOT = {
						has_variable = eqfm_expedition_karadag_transit_var
					}
				}
				trigger_event = {
					on_action = eqfm_karadag_expedition_validation
				}
			}
		}
	}
	c:BOS ?= {
		if = {
			limit = {
				is_involved_in_journal_entry = je_eqfm_karadag
			}
			random_scope_character = {
				limit = {
					has_template = balkfm_cengic_smail
					has_modifier = modifier_eqfm_pillaging_frontier
				}
				save_scope_as = eqfm_karadag_bos_leader
			}
		}
	}
	c:TUR ?= {
		if = {
			limit = {
				eqfm_has_karadag_expedition_trigger = yes
			}
			random_scope_character = {
				limit = {
					has_modifier = modifier_eqfm_leading_karadag_expedition
				}
				save_scope_as = eqfm_karadag_tur_leader
			}
			if = {
				limit = {
					exists = scope:eqfm_karadag_tur_leader
				}
				set_variable = {
					name = eqfm_karadag_leader_var
					value = scope:eqfm_karadag_tur_leader
				}
			}
			else = {
				remove_variable = eqfm_karadag_leader_var
			}
			if = {
				limit = {
					NOT = {
						has_variable = eqfm_expedition_karadag_transit_var
					}
				}
				trigger_event = {
					on_action = eqfm_karadag_expedition_validation
				}
			}
		}
	}
	global_var:je_eqfm_karadag = {
		if = {
			limit = {
				exists = scope:eqfm_karadag_mon_leader
			}
			add_to_variable_list = {
				name = com_journal_characters
				target = scope:eqfm_karadag_mon_leader
			}
		}
		if = {
			limit = {
				exists = scope:eqfm_karadag_bos_leader
			}
			add_to_variable_list = {
				name = com_journal_characters
				target = scope:eqfm_karadag_bos_leader
			}
		}
		if = {
			limit = {
				exists = scope:eqfm_karadag_tur_leader
			}
			add_to_variable_list = {
				name = com_journal_characters
				target = scope:eqfm_karadag_tur_leader
			}
		}
	}
}
