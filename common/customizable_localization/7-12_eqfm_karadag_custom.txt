eqfm_karadag_raiding_income_type = {
	type = country
	
	text = {
		trigger = {
			has_modifier = modifier_montenegrin_raiding
			var:last_week_raiding_income < eqfm_max_raiding_income_value
		}
		localization_key = je_eqfm_karadag_raiding_amount
	}
	text = {
		#displays when max raiding value reached
		trigger = {
			has_modifier = modifier_montenegrin_raiding
			var:last_week_raiding_income >= eqfm_max_raiding_income_value
		}
		localization_key = je_eqfm_karadag_raiding_amount_max
	}
}
