mgrbfm_nile_flood_status_type = {
	type = country
	
	text = {
		localization_key = je_mgrbfm_egypt_nile_river_information
		trigger = {
			NOT = {
				any_state_region_in_lower_nile = {
					has_harvest_condition = flood
				}
			}
		}
	}
	text = {

				has_harvest_condition = flood

		trigger = {

			}
		}
	}
}
