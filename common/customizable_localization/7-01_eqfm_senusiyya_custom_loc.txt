eqfm_senusiyya_status_type = {
	type = country

	text = {
		localization_key = eqfm_senusiyya_status_influenced
		trigger = {
			has_variable = eqfm_senusiyya_influence_var
		}
	}
	text = {
		localization_key = eqfm_senusiyya_status_uninfluenced
		trigger = {
			NOT = {
				has_variable = eqfm_senusiyya_influence_var
			}
		}
	}
}

eqfm_senusiyya_libya_type = {
	type = country
	
	text = {
		localization_key = eqfm_senusiyya_libya_active
		trigger = {
			s:STATE_LIBYA = {
				has_variable = eqfm_senusiyya_entrenchment_var
			}
		}
	}
	text = {
		localization_key = eqfm_senusiyya_libya_inactive
		trigger = {
			s:STATE_LIBYA = {
				NOT = {
					has_variable = eqfm_senusiyya_entrenchment_var
				}
			}
		}
	}
}
