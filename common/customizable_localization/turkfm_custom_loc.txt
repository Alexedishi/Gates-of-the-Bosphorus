	text = {

		trigger = {
			var:turkfm_tanzimat_identity_progress_var = 0

		localization_key = turkfm_tanzimat_no_progress
	}
}

turkfm_tension_next_type = {
	type = country

	# Stable Phase
	text = {
		trigger = {
			global_var:turkfm_balkan_situation_phase_var ?= flag:turkfm_stable
		}
		localization_key = turkfm_custom_stable_next
	}

	# Unrest Phase
	text = {
		trigger = {
			global_var:turkfm_balkan_situation_phase_var ?= flag:turkfm_unrest
		}
		localization_key = turkfm_custom_unrest_next
	}

	# Hostility Phase
	text = {
		trigger = {
			global_var:turkfm_balkan_situation_phase_var ?= flag:turkfm_hostility
		}
		localization_key = turkfm_custom_hostility_next
	}

	# Hostility Phase with Crisis Available
	text = {
		trigger = {
			global_var:turkfm_balkan_situation_phase_var ?= flag:turkfm_hostility
			turkfm_crisis_possible_trigger = yes # Only appears if a crisis is possible
		}
		localization_key = turkfm_custom_hostility_crisis_next
	}

	# Deconfliction Phase
	text = {
		trigger = {
			global_var:turkfm_balkan_situation_phase_var ?= flag:turkfm_deconfliction
		}
		localization_key = turkfm_custom_crisis_next
	}

	# Crisis Phase
	text = {
		trigger = {
			global_var:turkfm_balkan_situation_phase_var ?= flag:turkfm_crisis
		}
		localization_key = turkfm_custom_deconfliction_next
	}
}

turkfm_deescalation_possible_type = {
	type = country

	text = {
		trigger = {
			turkfm_can_deescalate_tension_trigger = no
			c:TUR ?= {
				NOR = {
					has_any_secessionists_broken_out = yes
					has_any_secessionists_growing = yes
					is_at_war = yes
				}
			}
		}
		localization_key = turkfm_custom_cannot_deescalate
	}
	text = {
		trigger = {
			turkfm_crisis_phase_trigger = yes
			c:TUR ?= {
				OR = {
					has_any_secessionists_broken_out = yes
					has_any_secessionists_growing = yes
					is_at_war = yes
				}
			}
		}
		localization_key = turkfm_custom_cannot_deescalate_crisis
	}
}

eqfm_jack_sparrow_type = {
	type = country

	text = {
		trigger = {
			global_var:eqfm_pirates_var <= 3
			exists = var:eqfm_jack_sparrow_var
		}
		localization_key = eqfm_custom_jack_sparrow_cocky
	}
	text = {
		trigger = {
			global_var:eqfm_pirates_var >= 4
			global_var:eqfm_pirates_var <= 7
			exists = var:eqfm_jack_sparrow_var
		}
		localization_key = eqfm_custom_jack_sparrow_cautious
	}
	text = {
		trigger = {
			global_var:eqfm_pirates_var >= 8
			exists = var:eqfm_jack_sparrow_var
		}
		localization_key = eqfm_custom_jack_sparrow_scared
	}
}
