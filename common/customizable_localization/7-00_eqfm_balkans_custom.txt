eqfm_deescalation_possible_type = {
	type = country
	
	text = {
		trigger = {
			eqfm_balkan_powderkeg_can_deescalate_trigger = no
			c:TUR ?= {
				NOR = {
					has_any_secessionists_broken_out = yes
					has_any_secessionists_growing = yes
					is_at_war = yes
				}
			}
		}
		localization_key = eqfm_custom_cannot_deescalate
	}
	text = {
		trigger = {
			eqfm_crisis_phase_trigger = yes
			je:je_eqfm_balkan_situation.var:eqfm_crisis_var < 250
		}
		localization_key = eqfm_custom_cannot_deescalate_crisis
	}
}
