turkfm_tanzimat_identity_progress_type = {
	type = country

	text = {
		trigger = {
			var:turkfm_tanzimat_identity_progress_var < 0
		}
		localization_key = turkfm_tanzimat_islamism_progress
	}
	text = {
		trigger = {
			var:turkfm_tanzimat_identity_progress_var > 0
		}
		localization_key = turkfm_tanzimat_ottomanism_progress
	}
	text = {
		trigger = {
			var:turkfm_tanzimat_identity_progress_var = 0
		}
		localization_key = turkfm_tanzimat_no_progress
	}
}

## Agricultural Agenda
turkfm_tanzimat_agriculture_reform_status_type = {
	type = country
	text = {
		localization_key = je_turkfm_tanzimat_agriculture_reform_status
		trigger = {
			has_variable = turkfm_tanzimat_agriculture_agenda
		}
	}
	text = {
		localization_key = je_turkfm_tanzimat_agriculture_reform_success
		trigger = {
			has_variable = turkfm_tanzimat_agriculture_complete
		}
	}
}

turkfm_tanzimat_agriculture_reform_goal_1_type = {
	type = country
	text = {
		localization_key = gbbf_checkbox_empty
		trigger = {
			OR = {
				has_law_or_variant = law_type:law_traditionalism
				has_law_or_variant = law_type:law_agrarianism
			}
		}
	}
	text = {
		localization_key = gbbf_checkbox_green
		trigger = {
			NOR = {
				has_law_or_variant = law_type:law_traditionalism
				has_law_or_variant = law_type:law_agrarianism
			}
		}
	}
}

turkfm_tanzimat_agriculture_reform_goal_2_type = {
	type = country
	text = {
		localization_key = gbbf_checkbox_empty
		trigger = {
			active_law:lawgroup_land_reform ?= {
				has_amendment = amendment_type:amendment_gbbf_chiftlik
			}
		}
	}
	text = {
		localization_key = gbbf_checkbox_green
		trigger = {
			active_law:lawgroup_land_reform ?= {
				NOT = {
					has_amendment = amendment_type:amendment_gbbf_chiftlik
				}
			}
		}
	}
}

turkfm_tanzimat_agriculture_reform_goal_3_type = {
	type = country
	text = {
		localization_key = gbbf_checkbox_empty
		trigger = {
			NOT = {
				any_scope_building = {
					is_building_type = building_wheat_farm
					NOT = {
						has_active_production_method = pm_simple_farming
					}
					level >= 10
					weekly_profit > 5
					occupancy >= 0.9
				}
			}
		}
	}
	text = {
		localization_key = gbbf_checkbox_green
		trigger = {
			any_scope_building = {
				is_building_type = building_wheat_farm
				NOT = {
					has_active_production_method = pm_simple_farming
				}
				level >= 10
				weekly_profit > 5
				occupancy >= 0.9
			}
		}
	}
}
