####---------- Tax Reform Agenda ----------####
balkfm_tax_reform_status_type = {
	type = country
	text = { 
		localization_key = je_balkfm_tax_reform_status
		trigger = { 
			je:je_balkfm_legacies = {
				has_modifier = modifier_balkfm_timar
			}
		}
	}
	text = {
		localization_key = je_balkfm_tax_reform_success
		trigger = {
			has_variable = balkfm_tax_reform_complete
		}
	}
}

balkfm_tax_reform_goal_1_type = {
	type = country
	text = {
		localization_key = gbbf_checkbox_empty
		trigger = {
			OR = {
				has_law_or_variant = law_type:law_consumption_based_taxation
				has_law_or_variant = law_type:law_land_based_taxation
			}
		}
	}
	text = {
		localization_key = gbbf_checkbox_green
		trigger = {
			NOR = {
				has_law_or_variant = law_type:law_consumption_based_taxation
				has_law_or_variant = law_type:law_land_based_taxation
			}
		}
	}
