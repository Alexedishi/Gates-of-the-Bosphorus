####---------- Balkan Flavor Monthly Pulses ----------####
balkfm_monthly_country_pulse = {
	trigger = {
		balkfm_country_has_yugoslav_culture_trigger = yes
	}
	effect = {
		if = {
			limit = {
				OR = {
					has_global_variable = balkfm_yugoslavia_journal_var
					exists = c:YUG
				}
			}
			every_country = {
				limit = {
					balkfm_country_has_yugoslav_culture_trigger = yes
				}
				if = {
					limit = {
						has_modifier = modifier_balkfm_yugoslavist_separatism
					}
					remove_modifier = modifier_balkfm_yugoslavist_separatism
				}
				if = {
					limit = {
						is_subject = yes
						overlord ?= {
							NOR = {
								balkfm_country_has_yugoslav_culture_trigger = yes
								c:YUG ?= this
							}
						}
					}
					add_modifier = {
						name = modifier_balkfm_yugoslavist_separatism
						multiplier = {
							value = balkfm_yugoslav_fervor_value
							divide = 100
							round_to = 0.01
						}
					}
				}
			}
		}
		if = {
			limit = {
				is_involved_in_journal_entry = je_balkfm_yugoslav_project
			}
			balkfm_update_yugoslav_unification_variables = yes
		}
	}
}

####---------- Contextless Yearly Pulses ----------####
balkfm_state_region_law_pulse = {
	trigger = {
		NOT = {
			has_global_variable = has_formed_yug
		}
	}
	effect = {
		every_state_region_in_yugoslavia = {
			limit = {
				has_variable = balkfm_tax_law_var
			}
			ordered_scope_state = {
				order_by = num_provinces
				position = 0
				check_range_bounds = no
				owner = {
					active_law:lawgroup_taxation.type = {
						save_scope_as = balkfm_country_tax_law_scope
					}
				}
				set_variable = {
					name = balkfm_tax_law_var
					value = scope:balkfm_country_tax_law_scope
				}
			}
		}
	}
}
