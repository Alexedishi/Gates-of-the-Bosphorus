on_game_started_after_lobby = {
	on_actions = {
		gbbf_on_game_start
	}
}

##########################################################################

gbbf_on_game_start = {
	effect = {
		c:GBBF_ALB ?= {
			gbbf_play_yanya_effect = yes
		}
		c:GBBF_CAP ?= {
			gbbf_play_cappadocia_effect = yes
		}
		c:GBBF_SIL ?= {
			gbbf_play_silistra_effect = yes
		}
		c:GBBF_THE ?= {
			gbbf_play_theodoro_effect = yes
		}
		c:GBBF_TRB ?= {
			gbbf_play_trabzon_effect = yes
		}
		c:ALK = {
			if = {
				limit = {
					exists = c:GBBF_CAP
				}
				annex = c:GBBF_CAP
			}
			if = {
				limit = {
					exists = c:GBBF_SIL
				}
				annex = c:GBBF_SIL
			}
			if = {
				limit = {
					exists = c:GBBF_THE
				}
				annex = c:GBBF_THE
			}
			if = {
				limit = {
					exists = c:GBBF_TRB
				}
				annex = c:GBBF_TRB
			}
		}
		if = {
			limit = {
				has_game_rule = gbbf_balkan_eyalets
			}
			if = {
				limit = {
					NOT = {
						exists = c:BUL_SIL
					}
				}
				gbbf_spawn_silistra_effect = yes
			}
			if = {
				limit = {
					NOT = {
						exists = c:ALB_YAN
					}
				}
				gbbf_spawn_yanya_effect = yes
			}
		}
		c:GRE ?= {
			grefm_calculate_bavarocracy_variables_effect = yes
		}
		c:MON ?= {
            add_treasury = {
                value = -35000
            }
			create_character = {
				template = template_balkfm_petrovic_joko
			}
			random_scope_character = {
				limit = {
					has_template = template_balkfm_petrovic_joko
				}
				set_character_busy_and_immortal = yes
				add_modifier = modifier_turkfm_leading_black_mountain_expedition
			}
			turkfm_start_black_mountain_expedition_effect = yes
			turkfm_update_black_mountain_characters_effect = yes
		}
		c:SER ?= {
			change_variable = {
				name = balkfm_obrenovic_support_var
				add = 200
			}
			change_variable = {
				name = balkfm_karadordevic_support_var
				add = 50
			}
			balkfm_calculate_dynastic_rivalry_variables_effect = yes
			balkfm_update_dynasty_character_scopes_effect = yes
		}
		c:TUR ?= {
			turkfm_calculate_sick_man_effect = yes
			turkfm_calculate_tanzimat_variables_effect = yes
			turkfm_calculate_balkan_situation_variables_effect = yes
		}
		every_country = {
			limit = {
				gbbf_country_has_greek_culture_trigger = yes
			}
			grefm_check_annexation_variables_effect = yes
		}
	}
}
