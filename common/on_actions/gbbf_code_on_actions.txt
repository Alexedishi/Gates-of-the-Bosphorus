on_country_formed = {
	on_actions = {
		grefm_byz_culture_validation
	}
}

grefm_byz_culture_validation = {
	trigger = {
		c:BYZ ?= this
	}
	effect = {
		if = {
			limit = {
				has_variable = grefm_culture_var
			}
			if = {
				limit = {
					var:grefm_culture_var = 1 # Hellene
				}
				remove_primary_culture = cu:rhomaios
			}
			if = {
				limit = {
					var:grefm_culture_var = 2 # Hellene
				}
				remove_primary_culture = cu:greek
			}
		}
		else = {
			if = {
				limit = {
					OR = {
						was_formed_from = GRE
						was_formed_from = GRE_EPI
						was_formed_from = ION
						was_formed_from = CRE
						was_formed_from = CYP
						was_formed_from = GRE_SAM
					}
				}
				remove_primary_culture = cu:rhomaios
			}
			else_if = {
				limit = {
					OR = {
						was_formed_from = BYZ_TRB
						was_formed_from = BYZ_TRB
					}
				}
				remove_primary_culture = cu:greek
			}
			else_if = {
				limit = {
					was_formed_from = GRA_THE
				}
				add_primary_culture = cu:urum
			}
			else_if = {
				limit = {
					was_formed_from = BUL
				}
				add_primary_culture = cu:bulgarian
			}
			else_if = {
				limit = {
					was_formed_from = ROM
				}
				add_primary_culture = cu:romanian
			}
			else_if = {
				limit = {
					was_formed_from = AST
				}
				add_primary_culture = cu:australian
			}
		}
	}
}

on_country_formed = {
	on_actions = {
		gbbf_on_country_formed
	}
}

gbbf_on_country_formed = {
	effect = {
		if = {
			limit = {
				c:BYZ ?= root
				var:grefm_culture_var ?= 1 # Hellene Outcome
			}
			remove_primary_culture = cu:rhomaios
			add_primary_culture = cu:greek
		}
		if = {
			limit = {
				c:BYZ ?= root
				var:grefm_culture_var ?= 2 # Rhomaios Outcome
			}
			remove_primary_culture = cu:greek
			add_primary_culture = cu:rhomaios
		}
		if = {
			limit = {
				c:BYZ ?= root
				var:grefm_culture_var ?= 3 # Hybrid Outcome
			}
			add_primary_culture = cu:greek
			add_primary_culture = cu:rhomaios
		}
	}
}


####---------- on_country_released ----------##
on_country_released_as_independent = {
	on_actions = {
		balkfm_country_release_1
	}
}

on_country_released_as_own_subject = {
	on_actions = {
		balkfm_country_release_2
	}
}

balkfm_country_release_1 = {
	# BUL keeps BUL_RUM if fully independent
	effect = {
		if = {
			limit = {
				c:BUL ?= scope:target
				NOT = { has_global_variable = balkfm_lion_awakened }
			}
			scope:target = {
				balkfm_set_bulgarian_laws = yes
				balkfm_create_bulgarian_characters = yes
				trigger_event = {
					id = balkfm_bulgaria.001 # The Golden Lion Awakens
				}
			}
		}
	}
}

balkfm_country_release_2 = {
	# Splits BUL and BUL_RUM if BUL is released as a subject
	effect = {
		if = {
			limit = {
				c:BUL ?= scope:target
				NOT = { has_global_variable = balkfm_lion_awakened }
			}
			scope:target = {
				balkfm_set_bulgarian_laws = yes
				balkfm_create_bulgarian_characters = yes
				trigger_event = {
					id = balkfm_bulgaria.001 # The Golden Lion Awakens
				}
			}
			if = {
				limit = {
					c:TUR = {
						owns_entire_state_region = STATE_NORTHERN_THRACE
					}
				}
				create_country = {
					tag = BUL_RUM
					origin = this
					state = s:STATE_NORTHERN_THRACE.region_state:TUR
					on_created = {
					}
				}
				c:TUR ?= {
					create_diplomatic_pact = {
						country = c:BUL_RUM
						type = puppet
					}
				}
			}
			else_if = {
				limit = {
					c:BUL = {
						owns_entire_state_region = STATE_NORTHERN_THRACE
					}
				}
				create_country = {
					tag = BUL_RUM
					origin = this
					state = s:STATE_NORTHERN_THRACE.region_state:BUL
					on_created = {
					}
				}
				c:TUR ?= {
					create_diplomatic_pact = {
						country = c:BUL_RUM
						type = puppet
					}
				}
			}
		}
	}
}

on_law_enactment_pass = {
	on_actions = {
		gbbf_law_enactment_pass
	}
}

gbbf_law_enactment_pass = {
	trigger = {
		c:TUR ?= this
	}
	effect = {
		if = {
			limit = {
				OR = {
					ROOT.currently_enacting_law.type ?= law_type:law_no_schools
					ROOT.currently_enacting_law.type ?= law_type:law_religious_schools
					ROOT.currently_enacting_law.type ?= law_type:law_public_schools
					ROOT.currently_enacting_law.type ?= law_type:law_private_schools
				}
				active_law:lawgroup_education_system ?= {
					has_amendment = amendment_gbbf_elifba
				}
			}
			currently_enacting_law = {
				add_amendment = {
					type = amendment_gbbf_elifba
					sponsor = ig:ig_devout
				}
			}
		}
	}
}

	on_actions = {
		eqfm_secession_start
	}
}

eqfm_secession_start = {
	effect = {
		if = {
			limit = {
				c:TUR ?= this
				is_involved_in_journal_entry = je_eqfm_balkan_situation
				scope:target = {
					balkfm_balkans_native_trigger = yes
				}
			}
			scope:target = {
				set_variable = eqfm_rebellion_tag_tracker
				every_scope_state = {
					remove_modifier = modifier_eqfm_race_riots
				}
				add_modifier = {
					name = modifier_eqfm_nationalist_uprising
					months = gbbf_duration_5_years
				}
				balkfm_set_up_ottoman_legacies_effect = yes
			}
			eqfm_add_large_deescalation_effect = yes
			every_country = {
				limit = {
					eqfm_balkans_party_trigger = yes
				}
				post_notification = notification_eqfm_secession_start
			}
			trigger_event = {
				id = eqfm_balkans.006
			}
		}
	}
}
