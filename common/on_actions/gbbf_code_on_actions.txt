#### COUNTRY RELEASE ON ACTIONS

on_country_released_as_independent = {
	on_actions = {
		balkfm_country_release_1
	}
}

on_country_released_as_own_subject = {
	on_actions = {
		balkfm_country_release_2
	}
}

balkfm_country_release_1 = { # BUL keeps BUL_RUM if fully independent
	effect = {
		if = {
			limit = {
				c:BUL ?= scope:target
				NOT = { has_global_variable = balkfm_lion_awakened }
			}
			scope:target = {
				balkfm_set_bulgarian_laws = yes
				balkfm_create_bulgarian_characters = yes
				trigger_event = {
					id = balkfm.500 # The Golden Lion Awakens
				}
			}
		}
	}
}

balkfm_country_release_2 = { # Splits BUL and BUL_RUM if BUL is released as a subject
	effect = {
		if = {
			limit = {
				c:BUL ?= scope:target
				NOT = { has_global_variable = balkfm_lion_awakened }
			}
			scope:target = {
				balkfm_set_bulgarian_laws = yes
				balkfm_create_bulgarian_characters = yes
				trigger_event = {
					id = balkfm.500 # The Golden Lion Awakens
				}
			}
			if = {
				limit = {
					c:TUR = {
						owns_entire_state_region = STATE_NORTHERN_THRACE
					}
				}
				create_country = {
					tag = BUL_RUM
					origin = this
					state = s:STATE_NORTHERN_THRACE.region_state:TUR
					on_created = {
					}
				}
				c:TUR ?= {
					create_diplomatic_pact = {
						country = c:BUL_RUM
						type = puppet
					}
				}
			}
			else_if = {
				limit = {
					c:BUL = {
						owns_entire_state_region = STATE_NORTHERN_THRACE
					}
				}
				create_country = {
					tag = BUL_RUM
					origin = this
					state = s:STATE_NORTHERN_THRACE.region_state:BUL
					on_created = {
					}
				}
				c:TUR ?= {
					create_diplomatic_pact = {
						country = c:BUL_RUM
						type = puppet
					}
				}
			}
		}
	}
}

#### DIPLOMAT EXPLUSION ON ACTIONS
# Root = Diplomatic Action
# scope:target = the country whose diplomats were expelled
# scope:initiator = the country doing the expelling

on_diplomats_expelled = {
	on_actions = {
		turkfm_tension_diplomats_expelled
	}
}

turkfm_tension_diplomats_expelled = {
	trigger = {
		turkfm_can_escalate_tension_trigger = yes
		scope:target = {
			c:TUR ?= this
		}
	}
	effect = {
		if = {
			limit = {
				scope:initiator = {
					country_rank < rank_value:great_power
				}
			}
			change_global_variable = {
				name = turkfm_escalation_var
				add = 3
			}
			c:TUR ?= {
				turkfm_check_phase_effect = yes # This MUST fire from TUR scope
			}
			every_country = {
				limit = {
					turkfm_balkans_party_trigger = yes
				}
				post_notification = notification_turkfm_diplomats_expelled_1
			}
		}
		else_if = {
			limit = {
				scope:initiator = {
					country_rank >= rank_value:great_power
				}
			}
			change_global_variable = {
				name = turkfm_escalation_var
				add = 10
			}
			every_country = {
				limit = {
					turkfm_balkans_party_trigger = yes
				}
				post_notification = notification_turkfm_diplomats_expelled_2
			}
			c:TUR ?= {
				turkfm_check_phase_effect = yes # This MUST fire from TUR scope
			}
		}
	}
}

turkfm_crisis_wargoal_defender = { # From TUR, in the case of Ottoman victory
	trigger = {
		has_global_variable = turkfm_crisis_launched
		has_variable = turkfm_crisis_target
		scope:target = {
			has_variable = turkfm_crisis_leader
		}
	}
	effect = {
		remove_global_variable = turkfm_crisis_launched
		every_country = {
			limit = {
				turkfm_crisis_participant_trigger = yes
			}
			trigger_event = {
				id = turkfm.328 # The Porte Triumphant
			}
		}
		turkfm_force_stable_effect = yes # Skip the Congress, transition to Stable
	}
}

turkfm_crisis_wargoal_attacker = { # From Crisis initiator, in the case of Ottoman defeat
	trigger = {
		has_global_variable = turkfm_crisis_launched
		has_variable = turkfm_crisis_leader
		scope:target = {
			has_variable = turkfm_crisis_target
		}
	}
	effect = {
		remove_global_variable = turkfm_crisis_launched
		turkfm_crisis_armistice_effect = yes
	}
}
