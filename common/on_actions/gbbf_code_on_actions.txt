#### COUNTRY RELEASE ON ACTIONS

on_country_released_as_independent = {
	on_actions = {
		balkfm_country_release_1
	}
}

on_country_released_as_own_subject = {
	on_actions = {
		balkfm_country_release_2
	}
}

balkfm_country_release_1 = { # BUL keeps BUL_RUM if fully independent
	effect = {
		if = {
			limit = {
				c:BUL ?= scope:target
				NOT = { has_global_variable = balkfm_lion_awakened }
			}
			scope:target = {
				balkfm_set_bulgarian_laws = yes
				balkfm_create_bulgarian_characters = yes
				trigger_event = {
					id = balkfm_bulgaria.001 # The Golden Lion Awakens
				}
			}
		}
	}
}

balkfm_country_release_2 = { # Splits BUL and BUL_RUM if BUL is released as a subject
	effect = {
		if = {
			limit = {
				c:BUL ?= scope:target
				NOT = { has_global_variable = balkfm_lion_awakened }
			}
			scope:target = {
				balkfm_set_bulgarian_laws = yes
				balkfm_create_bulgarian_characters = yes
				trigger_event = {
					id = balkfm_bulgaria.001 # The Golden Lion Awakens
				}
			}
			if = {
				limit = {
					c:TUR = {
						owns_entire_state_region = STATE_NORTHERN_THRACE
					}
				}
				create_country = {
					tag = BUL_RUM
					origin = this
					state = s:STATE_NORTHERN_THRACE.region_state:TUR
					on_created = {
					}
				}
				c:TUR ?= {
					create_diplomatic_pact = {
						country = c:BUL_RUM
						type = puppet
					}
				}
			}
			else_if = {
				limit = {
					c:BUL = {
						owns_entire_state_region = STATE_NORTHERN_THRACE
					}
				}
				create_country = {
					tag = BUL_RUM
					origin = this
					state = s:STATE_NORTHERN_THRACE.region_state:BUL
					on_created = {
					}
				}
				c:TUR ?= {
					create_diplomatic_pact = {
						country = c:BUL_RUM
						type = puppet
					}
				}
			}
		}
	}
}

#### SECESSION ON ACTIONS
on_secession_start = {
	on_actions = {
		turkfm_secession_start
	}
}

turkfm_secession_start = {
	effect = {
		if = {
			limit = {
				has_journal_entry = je_eqfm_balkan_situation_main
				scope:target = {
					balkfm_balkans_native_trigger = yes
				}
			}
			scope:target = {
				set_variable = eqfm_rebellion_tag_tracker
				every_scope_state = {
					remove_modifier = modifier_eqfm_race_riots
				}
				add_modifier = {
					name = modifier_eqfm_nationalist_uprising
					months = 60
				}
				balkfm_set_up_ottoman_legacies_effect = yes
			}
			eqfm_add_large_deescalation_effect = yes
			every_country = {
				limit = {
					eqfm_balkans_party_trigger = yes
				}
				post_notification = notification_eqfm_secession_start
			}
			trigger_event = {
				id = eqfm_balkans.006
			}
		}
	}
}

#### DIPLOMAT EXPLUSION ON ACTIONS
# Root = Diplomatic Action
# scope:target = the country whose diplomats were expelled
# scope:initiator = the country doing the expelling

on_diplomats_expelled = {
	on_actions = {
		eqfm_tension_diplomats_expelled
	}
}

eqfm_tension_diplomats_expelled = {
	trigger = {
		eqfm_can_escalate_tension_trigger = yes
		scope:target = {
			c:TUR ?= this
		}
	}
	effect = {
		if = {
			limit = {
				scope:initiator = {
					country_rank < rank_value:great_power
				}
			}
			change_global_variable = {
				name = eqfm_escalation_var
				add = 5
			}
			c:TUR ?= {
				eqfm_check_balkan_situation_phase_effect = yes # This MUST fire from TUR scope
			}
			every_country = {
				limit = {
					eqfm_balkans_party_trigger = yes
				}
				post_notification = notification_eqfm_diplomats_expelled_1
			}
		}
		else_if = {
			limit = {
				scope:initiator = {
					country_rank >= rank_value:great_power
				}
			}
			change_global_variable = {
				name = eqfm_escalation_var
				add = 10
			}
			every_country = {
				limit = {
					eqfm_balkans_party_trigger = yes
				}
				post_notification = notification_eqfm_diplomats_expelled_2
			}
			c:TUR ?= {
				eqfm_check_balkan_situation_phase_effect = yes # This MUST fire from TUR scope
			}
		}
	}
}

	on_actions = {
		gbbf_on_country_formed
	}
}

gbbf_on_country_formed = {
	effect = {
		if = {
			limit = {
				c:BYZ ?= root
				var:grefm_culture_var ?= 1 # Hellene Outcome
			}
			remove_primary_culture = cu:rhomaios
			add_primary_culture = cu:greek
		}
		if = {
			limit = {
				c:BYZ ?= root
				var:grefm_culture_var ?= 2 # Rhomaios Outcome
			}
			remove_primary_culture = cu:greek
			add_primary_culture = cu:rhomaios
		}
		if = {
			limit = {
				c:BYZ ?= root
				var:grefm_culture_var ?= 3 # Hybrid Outcome
			}
			add_primary_culture = cu:greek
			add_primary_culture = cu:rhomaios
		}
	}
}
