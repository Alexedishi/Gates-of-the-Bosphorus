####---------- on_country_formed ----------####
on_country_formed = {
	on_actions = {
		grefm_byz_culture_validation
	}
}

grefm_byz_culture_validation = {
	trigger = {
		c:BYZ ?= this
	}
	effect = {
		if = {
			limit = {
				has_variable = grefm_culture_var
			}
			if = {
				limit = {
					var:grefm_culture_var = 1 # Hellene
				}
				remove_primary_culture = cu:rhomaios
			}
			if = {
				limit = {
					var:grefm_culture_var = 2 # Hellene
				}
				remove_primary_culture = cu:greek
			}
		}
		else = {
			if = {
				limit = {
					OR = {
						was_formed_from = GRE
						was_formed_from = GRE_EPI
						was_formed_from = ION
						was_formed_from = CRE
						was_formed_from = CYP
						was_formed_from = GRE_SAM
					}
				}
				remove_primary_culture = cu:rhomaios
			}
			else_if = {
				limit = {
					OR = {
						was_formed_from = BYZ_TRB
						was_formed_from = BYZ_TRB
					}
				}
				remove_primary_culture = cu:greek
			}
			else_if = {
				limit = {
					was_formed_from = GRA_THE
				}
				add_primary_culture = cu:urum
			}
			else_if = {
				limit = {
					was_formed_from = BUL
				}
				add_primary_culture = cu:bulgarian
			}
			else_if = {
				limit = {
					was_formed_from = ROM
				}
				add_primary_culture = cu:romanian
			}
			else_if = {
				limit = {
					was_formed_from = AST
				}
				add_primary_culture = cu:australian
			}
		}
	}
}


####---------- on_country_formed ----------####
on_country_formed = {
	on_actions = {
		gbbf_on_country_formed
	}
}

gbbf_on_country_formed = {
	effect = {
		if = {
			limit = {
				c:BYZ ?= root
			}
			if = {
				limit = {
					var:grefm_culture_var ?= 1 # Hellene Outcome
				}
				remove_primary_culture = cu:rhomaios
				add_primary_culture = cu:greek
			}
			else_if = {
				limit = {
					var:grefm_culture_var ?= 2 # Rhomaios Outcome
				}
				remove_primary_culture = cu:greek
				add_primary_culture = cu:rhomaios
			}
			else_if = {
				limit = {
					var:grefm_culture_var ?= 3 # Hybrid Outcome
				}
				add_primary_culture = cu:greek
				add_primary_culture = cu:rhomaios
			}
		}
		if = {
			limit = {
				c:YUG ?= root
			}
			if = {
				limit = {
					NOT = {
						has_law_or_variant = law_type:law_multicultural
					}
				}
				if = {
					limit = {
						was_formed_from = MON
					}
					balkfm_remove_yugoslav_cultures_effect = yes
					if = {
						limit = {
							has_global_variable = balkfm_montegrin_culture_var
						}
						add_primary_culture = cu:montenegrin
					}
					else = {
						add_primary_culture = cu:serb
					}
				}
				else_if = {
					limit = {
						OR = {
							was_formed_from = SER
							was_formed_from = SER_RSK
						}
					}
					balkfm_remove_yugoslav_cultures_effect = yes
					add_primary_culture = cu:serb
				}
				else_if = {
					limit = {
						was_formed_from = BUL
					}
					balkfm_remove_yugoslav_cultures_effect = yes
					add_primary_culture = cu:bulgarian
				}
				else_if = {
					limit = {
						was_formed_from = BOS
					}
					balkfm_remove_yugoslav_cultures_effect = yes
					add_primary_culture = cu:bosniak
				}
				else_if = {
					limit = {
						was_formed_from = CRO
					}
					balkfm_remove_yugoslav_cultures_effect = yes
					add_primary_culture = cu:croat
				}
				else_if = {
					limit = {
						was_formed_from = SLO
					}
					balkfm_remove_yugoslav_cultures_effect = yes
					add_primary_culture = cu:slovene
				}
				else = {
					error_log = "The gbbf_on_country_formed was evaluated for YUG and did not have law_multicultural, but none of the was_formed_from triggers were true."
				}
			}
			else = {
				set_variable = je_balkfm_yugoslav_nationalism_complete
			}
		}
	}
}

balkfm_country_release_1 = {
	# BUL keeps BUL_RUM if fully independent
	effect = {
		if = {
			limit = {
				c:BUL ?= scope:target
				NOT = { has_global_variable = balkfm_lion_awakened }
			}
			scope:target = {
				balkfm_set_bulgarian_laws = yes
				balkfm_create_bulgarian_characters = yes
				trigger_event = balkfm_bulgaria.001
			}
		}
	}
}

balkfm_country_release_2 = {
	# Splits BUL and BUL_RUM if BUL is released as a subject
	effect = {
		if = {
			limit = {
				c:BUL ?= scope:target
				NOT = { has_global_variable = balkfm_lion_awakened }
			}
			scope:target = {
				balkfm_set_bulgarian_laws = yes
				balkfm_create_bulgarian_characters = yes
				trigger_event = balkfm_bulgaria.001
			}
			if = {
				limit = {
					c:TUR = {
						owns_entire_state_region = STATE_NORTHERN_THRACE
					}
				}
				create_country = {
					tag = BUL_RUM
					origin = this
					state = s:STATE_NORTHERN_THRACE.region_state:TUR
					on_created = {
					}
				}
				c:TUR ?= {
					create_diplomatic_pact = {
						country = c:BUL_RUM

					}

					c:BUL = {
						owns_entire_state_region = STATE_NORTHERN_THRACE
					}
				}
				create_country = {
					tag = BUL_RUM
					origin = this

					}

					create_diplomatic_pact = {
						country = c:BUL_RUM
						type = puppet
					}
				}
			}
		}
	}
}

balkfm_trs_protectorate_autonomy = {
	trigger = {
		scope:first_country = {
			c:AUS ?= this
		}
		scope:second_country = {
			c:TRS ?= this
		}
	}
	effect = {
		if = {
			limit = {
				c:TRS = {
					NOT = {
						is_subject_type = subject_type_protectorate
					}
				}
			}
			c:TRS = {
				if = {
					limit = {
						is_subject = yes
					}
				}
				make_independent = yes
			}
			c:AUS = {
				create_diplomatic_pact = {
					type = protectorate
					country = c:TRS
				}
			}
			error_log = "GotB: Transylvania's autonomy was increased, but it became independent instead. Subjecthood has been fixed."
		}
	}
}

		}
	}
	effect = {
		if = {

				}

				value = 5
			}

					is_involved_in_journal_entry = je_eqfm_balkan_powder_keg
				}
				post_notification = eqfm_diplomats_expelled_1
			}
		}
		else_if = {
			limit = {
				scope:initiator = {
					country_rank >= rank_value:great_power

	trigger = {
		c:TUR ?= this
	}

		if = {
			limit = {
				OR = {
					ROOT.currently_enacting_law.type ?= law_type:law_no_schools

					ROOT.currently_enacting_law.type ?= law_type:law_private_schools
				}
				active_law:lawgroup_education_system ?= {
					has_amendment = amendment_type:amendment_gbbf_elifba
				}
			}
			currently_enacting_law = {
				add_amendment = {
					type = amendment_gbbf_elifba
					sponsor = prev.ig:ig_devout
				}
			}
		}
	}
}

		gbbf_evaluate_and_assign_state_hub_dynamic_names
		gbbf_validate_state_modifiers
	}
}

gbbf_evaluate_and_assign_state_hub_dynamic_names = {
	effect = {
		gbbf_evaluate_and_assign_state_hub_dynamic_names = yes
	}
}

gbbf_validate_state_modifiers = {
	trigger = {
		OR = {
			state_region = s:STATE_MATRUH
			state_region = s:STATE_EGYPTIAN_DESERT
		}
		has_modifier = modifier_mgrbfm_tribal_resistance
	}
	effect = {
		remove_modifier = modifier_mgrbfm_tribal_resistance
	}
}


####---------- on_state_created ----------####
on_state_created = {
	on_actions = {
		gbbf_evaluate_and_assign_state_hub_dynamic_names
	}
}


####---------- on_secession_start ----------####
on_secession_start = {
	on_actions = {
		eqfm_secession_start
	}
}

eqfm_secession_start = {
	effect = {
		if = {
			limit = {
				c:TUR ?= this
				is_involved_in_journal_entry = je_eqfm_balkan_powder_keg
				scope:target = {
					balkfm_balkans_native_trigger = yes
				}
			}
			scope:target = {
				set_variable = eqfm_rebellion_tag_tracker
				every_scope_state = {
					remove_modifier = modifier_eqfm_race_riots
				}
				add_modifier = {
					name = modifier_eqfm_nationalist_uprising
					months = gbbf_normal_modifier_time
				}
			}
			eqfm_add_large_deescalation_effect = yes
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_balkan_powder_keg
				}
				post_notification = eqfm_secession_start
			}
			trigger_event = eqfm_balkans.006
		}
	}
}


####---------- on_wargoal_enforced ----------####
# Root = Country
# scope:target = the country the wargoal was enforced against
# scope:diplomatic_play = the relevant diplomatic play
# scope:wargoal_impact = the impact of the wargoal enforced
on_wargoal_enforced = {
	on_actions = {
		eqfm_tension_wargoal_enforced_1 # Wargoal against TUR
		eqfm_tension_wargoal_enforced_2 # Wargoal from TUR
	}
}

eqfm_tension_wargoal_enforced_1 = {
	trigger = {
		eqfm_balkan_powder_keg_can_escalate_trigger = yes
		scope:target = {
			c:TUR ?= this
		}
	}
	effect = {
		if = {
			limit = {
				country_rank < rank_value:great_power
			}
			eqfm_add_escalation_effect = {
				value = 3
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_balkan_powder_keg
				}
				post_notification = eqfm_wargoal_enforced_by_non_gp
			}
		}
		else_if = {
			limit = {
				country_rank >= rank_value:great_power
			}
			eqfm_add_escalation_effect = {
				value = 5
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_balkan_powder_keg
				}
				post_notification = eqfm_wargoal_enforced_by_great_power
			}
		}
	}
}

eqfm_tension_wargoal_enforced_2 = {
	trigger = {
		eqfm_balkan_powder_keg_can_deescalate_trigger = yes
		c:TUR ?= this
		scope:target = {
			is_in_geographic_region = geographic_region_rumelia
		}
	}
	effect = {
		if = {
			limit = {
				scope:target = {
					country_rank >= rank_value:great_power
				}
			}
			eqfm_add_deescalation_effect = {
				value = 10
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_balkan_powder_keg
				}
				post_notification = eqfm_wargoal_enforced_against_great_power
			}
		}
		else_if = {
			limit = {
				scope:target = {
					is_subject_of = c:TUR
				}
			}
			eqfm_add_deescalation_effect = {
				value = 5
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_balkan_powder_keg
				}
				post_notification = eqfm_wargoal_enforced_against_subject
			}
		}
		else = { # Yes, this INCREASES escalation
			eqfm_add_escalation_effect = {
				value = 5
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_balkan_powder_keg
				}
				post_notification = eqfm_wargoal_enforced_against_non_gp
			}
		}
	}
}
