####---------- Balkan Flavor Monthly Pulses ----------####
balkfm_monthly_pulse = {
	trigger = {
		balkfm_country_has_yugoslav_culture_trigger = yes
	}
	effect = {
		if = {
			limit = {
				has_modifier = modifier_balkfm_yugoslavist_separatism
			}
			remove_modifier = modifier_balkfm_yugoslavist_separatism
		}
		if = {
			limit = {
				OR = {
					has_global_variable = balkfm_yugoslavia_journal_var
					exists = c:YUG
				}
			}
			every_country = {
				limit = {
					balkfm_country_has_yugoslav_culture_trigger = yes
					AND = {
						is_subject = yes
						overlord ?= {
							NOT = {
								c:YUG ?= this
							}
						}
					}
				}
				add_modifier = {
					name = modifier_balkfm_yugoslavist_separatism
					multiplier = {
						value = balkfm_yugoslav_fervor_value
						divide = 100
						round_to = 0.01
					}
				}
			}
		}
	}
}
