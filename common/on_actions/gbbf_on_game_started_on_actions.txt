on_game_started_after_lobby = {
	on_actions = {
		gbbf_on_game_start
	}
}

##########################################################################

gbbf_on_game_start = {
	effect = {
		c:GBBF_ALB ?= {
			gbbf_play_albania_effect = yes
		}
		c:GBBF_CAP ?= {
			gbbf_play_cappadocia_effect = yes
		}
		c:GBBF_SIL ?= {
			gbbf_play_silistra_effect = yes
		}
		c:GBBF_THE ?= {
			gbbf_play_theodoro_effect = yes
		}
		c:GBBF_TRB ?= {
			gbbf_play_trabzon_effect = yes
		}
		c:ALK = {
			if = {
				limit = {
					exists = c:GBBF_CAP
				}
				annex = c:GBBF_CAP
			}
			if = {
				limit = {
					exists = c:GBBF_SIL
				}
				annex = c:GBBF_SIL
			}
			if = {
				limit = {
					exists = c:GBBF_THE
				}
				annex = c:GBBF_THE
			}
			if = {
				limit = {
					exists = c:GBBF_TRB
				}
				annex = c:GBBF_TRB
			}
		}
		c:GRE ?= {
			grefm_calculate_bavarocracy_variables_effect = yes
		}
		c:SER ?= {
			balkfm_calculate_dynasty_variables_effect = yes
		}
		c:TUR ?= {
			turkfm_calculate_sick_man_effect = yes
			turkfm_calculate_tanzimat_variables_effect = yes
			turkfm_calculate_balkan_situation_variables_effect = yes
		}
	}
}
