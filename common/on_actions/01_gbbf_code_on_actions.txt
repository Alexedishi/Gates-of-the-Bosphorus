####---------- on_country_formed ----------####
on_country_formed = {
	on_actions = {
		grefm_byz_culture_validation
	}
}

grefm_byz_culture_validation = {
	trigger = {
		c:BYZ ?= this
	}
	effect = {
		if = {
			limit = {
				has_variable = grefm_culture_var
			}
			if = {
				limit = {
					var:grefm_culture_var = 1 # Hellene
				}
				remove_primary_culture = cu:rhomaios
			}
			if = {
				limit = {
					var:grefm_culture_var = 2 # Hellene
				}
				remove_primary_culture = cu:greek
			}
		}
		else = {
			if = {
				limit = {
					OR = {
						was_formed_from = GRE
						was_formed_from = GRE_EPI
						was_formed_from = ION
						was_formed_from = CRE
						was_formed_from = CYP
						was_formed_from = GRE_SAM
					}
				}
				remove_primary_culture = cu:rhomaios
			}
			else_if = {
				limit = {
					OR = {
						was_formed_from = BYZ_TRB
						was_formed_from = BYZ_TRB
					}
				}
				remove_primary_culture = cu:greek
			}
			else_if = {
				limit = {
					was_formed_from = GRA_THE
				}
				add_primary_culture = cu:urum
			}
			else_if = {
				limit = {
					was_formed_from = BUL
				}
				add_primary_culture = cu:bulgarian
			}
			else_if = {
				limit = {
					was_formed_from = ROM
				}
				add_primary_culture = cu:romanian
			}
			else_if = {
				limit = {
					was_formed_from = AST
				}
				add_primary_culture = cu:australian
			}
		}
	}
}


####---------- on_country_formed ----------####
on_country_formed = {
	on_actions = {
		gbbf_on_byz_formed
		gbbf_on_yug_formed
	}
}

gbbf_on_byz_formed = {
	trigger = {
		c:BYZ ?= root
	}
	effect = {
		if = {
			limit = {
				var:grefm_culture_var ?= 1 # Hellene Outcome
			}
			remove_primary_culture = cu:rhomaios
			add_primary_culture = cu:greek
		}
		else_if = {
			limit = {
				var:grefm_culture_var ?= 2 # Rhomaios Outcome
			}
			remove_primary_culture = cu:greek
			add_primary_culture = cu:rhomaios
		}
		else_if = {
			limit = {
				var:grefm_culture_var ?= 3 # Hybrid Outcome
			}
			add_primary_culture = cu:greek
			add_primary_culture = cu:rhomaios
		}
	}
}

gbbf_on_yug_formed = {
	trigger = {
		c:YUG ?= root
	}
	events = {
		balkfm_yugoslavia.014
	}
	effect = {
		if = {
			limit = {
				was_formed_from = MON
			}
			balkfm_remove_yugoslav_cultures_effect = yes
			if = {
				limit = {
					has_global_variable = balkfm_montegrin_culture_var
				}
				add_primary_culture = cu:montenegrin
			}
			else = {
				add_primary_culture = cu:serb
			}
			set_state_religion = rel:orthodox
		}
		else_if = {
			limit = {
				OR = {
					was_formed_from = SER
					was_formed_from = SER_RSK
				}
			}
			balkfm_remove_yugoslav_cultures_effect = yes
			add_primary_culture = cu:serb
			set_state_religion = rel:orthodox
		}
		else_if = {
			limit = {
				was_formed_from = BUL
			}
			balkfm_remove_yugoslav_cultures_effect = yes
			add_primary_culture = cu:bulgarian
		}
		else_if = {
			limit = {
				was_formed_from = BOS
			}
			balkfm_remove_yugoslav_cultures_effect = yes
			add_primary_culture = cu:bosniak
			set_state_religion = rel:sunni # HAVE FUN BRO
		}
		else_if = {
			limit = {
				was_formed_from = CRO
			}
			balkfm_remove_yugoslav_cultures_effect = yes
			add_primary_culture = cu:croat
			set_state_religion = rel:catholic
		}
		else_if = {
			limit = {
				was_formed_from = SLO
			}
			balkfm_remove_yugoslav_cultures_effect = yes
			add_primary_culture = cu:slovene
			set_state_religion = rel:catholic
		}
		else = {
			error_log = "The gbbf_on_country_formed was evaluated for YUG and did not have law_multicultural, but none of the was_formed_from triggers were true."
		}
		every_state_region_in_yugoslavia = {
			remove_variable = balkfm_tax_law_var
		}
	}
}

	on_actions = {
		balkfm_country_release_2
	}
}

balkfm_country_release_1 = {
	# BUL keeps BUL_RUM if fully independent
	effect = {
		if = {
			limit = {
				c:BUL ?= scope:target
				NOT = { has_global_variable = balkfm_lion_awakened }
			}
			scope:target = {
				balkfm_set_bulgarian_laws = yes
				balkfm_create_bulgarian_characters = yes
				trigger_event = balkfm_bulgaria.001
			}
		}
	}
}

balkfm_country_release_2 = {
	# Splits BUL and BUL_RUM if BUL is released as a subject
	effect = {
		if = {
			limit = {
				c:BUL ?= scope:target
				NOT = { has_global_variable = balkfm_lion_awakened }
			}
			scope:target = {
				balkfm_set_bulgarian_laws = yes
				balkfm_create_bulgarian_characters = yes
				trigger_event = balkfm_bulgaria.001
			}
			if = {
				limit = {
					c:TUR = {
						owns_entire_state_region = STATE_NORTHERN_THRACE
					}
				}
				create_country = {
					tag = BUL_RUM
					origin = this
					state = s:STATE_NORTHERN_THRACE.region_state:TUR
					on_created = {
					}
				}
				c:TUR ?= {
					create_diplomatic_pact = {
						country = c:BUL_RUM
						type = puppet
					}
				}
			}
			else_if = {
				limit = {
					c:BUL = {
						owns_entire_state_region = STATE_NORTHERN_THRACE
					}
				}
				create_country = {
					tag = BUL_RUM
					origin = this
					state = s:STATE_NORTHERN_THRACE.region_state:BUL
					on_created = {
					}
				}
				c:TUR ?= {
					create_diplomatic_pact = {
						country = c:BUL_RUM
						type = puppet
					}
				}
			}
		}
	}
}

balkfm_trs_protectorate_autonomy = {
	trigger = {
		scope:actor = {
			c:AUS ?= this
		}
		scope:recipient = {
			c:TRS ?= this
		}
	}
	effect = {
		if = {
			limit = {
				c:TRS = {
					NOT = {
						is_subject_type = subject_type_protectorate
					}
				}
			}
			c:TRS = {
				if = {
					limit = {
						is_subject = yes
					}
				}
				make_independent = yes
			}
			c:AUS = {
				create_diplomatic_pact = {
					type = protectorate
					country = c:TRS
				}
			}
			error_log = "GotB: Transylvania's autonomy was increased, but it became independent instead. Subjecthood has been fixed."
		}
	}
}

	}
}

	trigger = {
		eqfm_balkan_powder_keg_can_escalate_trigger = yes
		scope:target = {
			c:TUR ?= this
		}
	}
	effect = {
		if = {
			limit = {
				scope:initiator = {
					country_rank < rank_value:great_power
				}
			}
			eqfm_add_balkan_powder_keg_escalation_effect = {
				value = 5
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_balkan_powder_keg
				}
				post_notification = eqfm_diplomats_expelled_1
			}
		}
		else_if = {
			limit = {
				scope:initiator = {
					country_rank >= rank_value:great_power
				}
			}
			eqfm_add_balkan_powder_keg_escalation_effect = {
				value = 10
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_balkan_powder_keg
				}
				post_notification = eqfm_diplomats_expelled_2
			}
		}
	}
}

gbbf_law_enactment_pass = {
	trigger = {
		c:TUR ?= this
	}
	effect = {
		if = {
			limit = {
				OR = {
					ROOT.currently_enacting_law.type ?= law_type:law_no_schools
					ROOT.currently_enacting_law.type ?= law_type:law_religious_schools
					ROOT.currently_enacting_law.type ?= law_type:law_public_schools
					ROOT.currently_enacting_law.type ?= law_type:law_private_schools
				}
				active_law:lawgroup_education_system ?= {
					has_amendment = amendment_type:amendment_gbbf_elifba
				}
			}
			currently_enacting_law = {
				add_amendment = {
					type = amendment_gbbf_elifba
					sponsor = prev.ig:ig_landowners
				}
			}
		}
		if = {
			limit = {
				OR = {
					root.currently_enacting_law.type ?= law_type:law_legacy_slavery
					root.currently_enacting_law.type ?= law_type:law_colonial_slavery
					root.currently_enacting_law.type ?= law_type:law_debt_slavery
				}
				active_law:lawgroup_slavery ?= {
					has_amendment = amendment_type:amendment_gbbf_esir_pazari
				}
			}
			currently_enacting_law = {
				add_amendment = {
					type = amendment_gbbf_elifba
					sponsor = prev.ig:ig_landowners
				}
			}
		}
	}

# Root = Country
# scope:target = the country the wargoal was enforced against
# scope:diplomatic_play = the relevant diplomatic play
# scope:wargoal_impact = the impact of the wargoal enforced
on_wargoal_enforced = {
	on_actions = {
		eqfm_balkans_wargoal_enforced_1 # Wargoal against TUR
		eqfm_balkans_wargoal_enforced_2 # Wargoal from TUR
	}
}

eqfm_balkans_wargoal_enforced_1 = {
	trigger = {
		eqfm_balkan_powder_keg_can_escalate_trigger = yes
		scope:target = {
			c:TUR ?= this
		}
	}
	effect = {
		if = {
			limit = {
				country_rank < rank_value:great_power
			}
			eqfm_add_balkan_powder_keg_escalation_effect = {
				value = eqfm_balkan_powder_keg_tiny_escalation_value
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_balkan_powder_keg
				}
				post_notification = eqfm_wargoal_enforced_by_non_gp
			}
		}
		else_if = {
			limit = {
				country_rank >= rank_value:great_power
			}
			eqfm_add_balkan_powder_keg_escalation_effect = {
				value = eqfm_balkan_powder_keg_very_small_escalation_value
			}
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_balkan_powder_keg
				}
				post_notification = eqfm_wargoal_enforced_by_great_power
			}
		}
	}
}
