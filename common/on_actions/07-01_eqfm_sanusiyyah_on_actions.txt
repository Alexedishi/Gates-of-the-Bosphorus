eqfm_sanusiyyah_scouting_weekly_pulse = {
	trigger = {
		eqfm_country_has_desert_scouts_trigger = yes
	}
	events = {
		eqfm_libyan_desert.007 # Expedition Withdraws from the Desert
		eqfm_libyan_desert.008 # Expedition Leader Has Died!
	}
	effect = {
		eqfm_update_desert_variables_effect = yes
		eqfm_update_desert_modifier_effect = yes
		if = {
			limit = {
				eqfm_has_desert_expedition_trigger = yes
				NOR = {
					has_variable = eqfm_expedition_desert_transit_var
				}
			}
			if = {
				limit = {
					NOT = {
						has_variable = eqfm_desert_expedition_progress_var
					}
				}
				set_variable = {
					name = eqfm_desert_expedition_progress_var
					value = 0
				}
			}
			if = {
				limit = {
					NOT = {
						has_variable = eqfm_desert_expedition_peril_var
					}
				}
				set_variable = {
					name = eqfm_desert_expedition_peril_var
					value = 0
				}
			}
		}
		if = {
			limit = {
				has_modifier = modifier_eqfm_desert_raids
				eqfm_has_desert_expedition_trigger = no
				NOT = {
					owns_entire_state_region = STATE_LIBYAN_DESERT
				}
			}
			remove_modifier = modifier_eqfm_desert_raids
			post_notification = eqfm_desert_raids_halted
		}
		if = {
			limit = {
				eqfm_has_desert_expedition_trigger = yes
				var:eqfm_desert_expedition_progress_var < 8
			}
			if = {
				limit = {
					var:eqfm_expedition_ticker_var < 32
				}
				change_variable = {
					name = eqfm_expedition_ticker_var
					add = 1
				}
			}
			else = {
				set_variable = {
					name = eqfm_expedition_ticker_var
					value = 0
				}
				change_variable = {
					name = eqfm_desert_expedition_progress_var
					add = 1
				}
				post_notification = eqfm_yearly_progress
			}
		}
		else = {
			set_variable = {
				name = eqfm_expedition_ticker_var
				value = 0
			}
		}
	}
}
