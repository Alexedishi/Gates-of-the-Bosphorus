turkfm_kemalism_value = {
	value = 0
	add = {
		desc = turkfm_value_from_republicanism
		value = {
			if = {
				limit = {
					OR = {
						has_law_or_variant = law_type:law_parliamentary_republic
						has_law_or_variant = law_type:law_presidential_republic
					}
				}
				add = 1
			}
		}
	}
	add = {
		desc = gbbf_value_from_citizenship_restrictiveness
		value = {
			add = gbbf_citizenship_restrictiveness_value
			divide = 10
		}
	}
	add = {
		desc = turkfm_value_from_sol_ratio
		value = {
			add = {
				value = average_sol
				subtract = average_expected_sol
				divide = 100
				round_to = 0.01
			}
		}
	}
	add = {
		desc = turkfm_value_from_laicism
		value = {
			if = {
				limit = {
					has_law_or_variant = law_type:law_total_separation
				}
				add = 1
			}
		}
	}
	add = {
		desc = gbbf_value_from_institution_levels
		value = {
			add = gbbf_institution_investment_levels_value
			divide = 10
		}
	}
	add = {
		desc = gbbf_value_from_innovation
		value = {
			add = country_innovation
			divide = 100
			round_to = 0.01
		}
	}
	add = {
		desc = gbbf_value_from_majority_movement
		value = {
			every_political_movement = {
				limit = {
					is_political_movement_type = movement_cultural_majority
				}
				add = political_movement_support
				round_to = 0.01
			}
		}
	}
}

turkfm_turanism_value = {
	value = 0
	add = {
		desc = turkfm_value_from_turkic_states
		value = {
			every_scope_state = {
				limit = {
					is_split_state = no
					is_incorporated = yes
					state_region = {
						OR = {
							is_homeland = turkmen
							is_homeland = kazak
							is_homeland = uzbek
							is_homeland = tajik
							is_homeland = uighur
							is_homeland = tatar
							is_homeland = azerbaijani
						}
					}
				}
				add = 0.1
			}
		}
	}
	add = {
		desc = turkfm_value_from_army_projection
		value = {
			add = army_power_projection
			divide = 1000
			round_to = 0.01
		}
	}
	subtract = {
		desc = gbbf_value_from_citizenship_restrictiveness
		value = {
			add = gbbf_citizenship_restrictiveness_value
			divide = 10
		}
	}
}

turkfm_anatolianism_value = {
	value = 0
	add = {
		desc = turkfm_value_from_turkish_literacy
		value = {
			every_scope_state = {
				limit = {
					is_in_geographic_region = geographic_region_anatolia
				}
				every_scope_pop = {
					limit = {
						has_pop_culture = turkish
					}
					add = literacy_rate
				}
			}
			divide = {
				every_scope_state = {
					limit = {
						is_in_geographic_region = geographic_region_anatolia
					}
					every_scope_pop = {
						limit = {
							has_pop_culture = turkish
						}
						add = 1
					}
				}
				min = 1
			}
			round_to = 0.01
		}
		
	}
	add = {
		desc = turkfm_value_from_turkish_states
		value = {
			every_scope_state = {
				limit = {
					is_split_state = no
					is_incorporated = yes
					is_in_geographic_region = geographic_region_anatolia
				}
				add = 0.1
			}
		}
	}
	add = {
		desc = turkfm_value_from_university_levels
		value = {
			every_state_in_anatolia = {
				limit = {
					owner = root
				}
				every_scope_building = {
					limit = {
						is_building_type = building_university
					}
					add = level
				}
			}
			divide = 10
			round_to = 0.01
		}
	}
}
