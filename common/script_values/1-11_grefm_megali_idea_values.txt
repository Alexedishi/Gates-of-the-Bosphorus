grefm_megali_idea_possible_value = 3
grefm_megali_idea_complete_base_value = 7
grefm_megali_idea_complete_pan_greek_value = 15
grefm_megali_idea_complete_byzantium_value = 11
grefm_byzantine_restoration_balkans_value = 11
grefm_byzantine_restoration_anatolia_value = 9

grefm_megali_idea_complete_value = {
	if = {
		limit = {
			NOR = {
				has_variable = grefm_embraced_megali_idea
				has_variable = grefm_embraced_imperial_restoration
			}
		}
		add = grefm_megali_idea_complete_base_value
	}
	else_if = {
		limit = {
			has_variable = grefm_embraced_megali_idea
		}
		add = grefm_megali_idea_complete_pan_greek_value
	}
	else_if = {
		limit = {
			has_variable = grefm_embraced_imperial_restoration
		}
		add = grefm_megali_idea_complete_byzantium_value
	}
}

grefm_megali_idea_counter_value = {
	value = 0
	if = {
		limit = {
			has_variable = grefm_embraced_imperial_restoration
		}
		every_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_byzantine_empire
				is_split_state = no
			}
			add = 1
		}
	}
	else = {
		every_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_greek_heartlands
				is_split_state = no
			}
			add = 1
		}
	}
}
