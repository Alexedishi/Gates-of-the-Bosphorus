balkfm_obrenovic_progress_value = {
	value = 0
	if = {
		limit = {
			has_dlc_feature = ip3_content # National Awakening
		}

			limit = {
				balkfm_has_obrenovic_ruler_trigger = yes
			}
			## Only apply ruler popularity if House is in charge

				value = {
					add = ruler.popularity
					multiply = 0.01
				}

			}

			if = {
				limit = {
					OR = {
						has_law = law_type:law_autocracy
						has_law = law_type:law_oligarchy
					}
				}
				add = {
					value = {
						every_interest_group = {
							limit = {
								OR = {
									law_stance = {
										law = law_type:law_autocracy
										value > neutral
									}
									law_stance = {
										law = law_type:law_oligarchy
										value > neutral

								}

										law = law_type:law_oligarchy
										value > neutral

				}

				limit = {
					NOT = {
						has_law = law_type:law_no_police
					}
				}
				add = {
					value = {
						every_interest_group = {
							limit = {
								law_stance = {
									law = law_type:law_no_police
									value < neutral
								}
							}
							add = ig_clout
						}
					}
					desc = balkfm_value_from_police_support
				}
			}

						}
					}

				}
			}
			## Relevant IGs will support ruling Obrenovic land policy
			if = {
				limit = {
					OR = {
						has_law = law_type:law_serfdom

			## Relevant IGs will support non-ruling Obrenovic if SER has suffrage
			if = {

									law_stance = {

									}
									law_stance = {
										law = law_type:law_tenant_farmers
										value > neutral
									}
								}
							}
							add = ig_clout
						}
					}
					desc = balkfm_value_from_land_conversativism
				}
			}
		}
		## Consul Check
		if = {
			limit = {
				has_variable = balkfm_austria_consul_var
			}
			add = {
				value = 0.2
				desc = balkfm_value_from_aus_consul
			}
		}
		else_if = {
			limit = {
				has_variable = balkfm_russia_consul_var
			}
			add = {
				value = -0.2
				desc = balkfm_value_from_rus_consul
			}
		}
		## Austria Relations Check
		add = {
			value = {
				add = c:AUS.relations:root
				divide = 100
			}
			desc = balkfm_value_from_aus_relations
		}
		## Lobby Appeasement Check
		add = {
			value = {
				every_political_lobby = {
					limit = {
						is_political_lobby_type = lobby_pro_country
						target = {
							c:AUS ?= this
						}
					}
					add = appeasement
				}
				multiply = 0.1
			}
			desc = balkfm_value_from_aus_lobbies
		}
		## Southern Serbia Control Check
		# Alexander Karadordevic lost a huge chunk of his support for failing to secure Serbian gains after the Crimean War
		if = {
			limit = {
				has_variable = balkfm_obrenovic_vojvodina_var
				NOT = {
					owns_entire_state_region = STATE_EASTERN_SERBIA
				}
			}
			subtract = {
				value = 2
				desc = balkfm_value_from_weak_kaykay
			}
		}
		else_if = {
			limit = {
				has_variable = balkfm_karadordevic_vojvodina_var
				owns_entire_state_region = STATE_EASTERN_SERBIA
			}
			add = {
				value = 2
				desc = balkfm_value_from_strong_kaykay
			}
		}
		round_to = 0.01
	}
}

balkfm_obrenovic_net_progress_value = {
	value = 0
	add = var:balkfm_obrenovic_progress_var
	subtract = var:balkfm_karadordevic_progress_var
}

balkfm_karadordevic_net_progress_value = {
	value = 0
	add = var:balkfm_karadordevic_progress_var
	subtract = var:balkfm_obrenovic_progress_var
}

balkfm_dynastic_rivalry_progress_value = {
	value = 0
	if = {
		limit = {
			var:balkfm_obrenovic_progress_var > var:balkfm_karadordevic_progress_var
		}
		add = var:balkfm_obrenovic_progress_var
		subtract = var:balkfm_karadordevic_progress_var
	}
	if = {
		limit = {
			var:balkfm_obrenovic_progress_var < var:balkfm_karadordevic_progress_var
		}
		add = var:balkfm_karadordevic_progress_var
		subtract = var:balkfm_obrenovic_progress_var
	}
	round_to = 0.01
}

balkfm_leading_dynasty_value = {
	if = {
		limit = {
			var:balkfm_obrenovic_support_var > var:balkfm_karadordevic_support_var
		}
		add = var:balkfm_obrenovic_support_var
	}
	else = {
		add = var:balkfm_karadordevic_support_var
	}
}
