					has_modifier = serbia_aided_magyarized_serbs_mod
				}
				add = {
					desc = "balkfm_value_from_aided_magyarized_serbs"
					value = 0.25
				}
			}
			# Aided Maygarized Serbs (serbian_throne.6.b)
			if = {
				limit = {
					has_modifier = serbia_greatly_aided_magyarized_serbs_mod
				}
				add = {
					desc = "balkfm_value_from_greatly_aided_magyarized_serbs"
					value = 0.75
				}
			}

			# + If SOL is higher than 7.5
			add = {
				desc = "balkfm_value_from_sol"
				value = {
					add = average_sol
					subtract = 10
					divide = 1.5
					round_to = 0.01
				}
			}

			# Negative bureaucracy
			if = {
				limit = {
					bureaucracy < 0
				}
				add = {
					desc = "balkfm_value_from_negative_bureaucracy"
					value = -1
				}
			}

			# Negative authority
			if = {
				limit = {
					authority < 0
				}
				add = {
					desc = "balkfm_value_from_negative_authority"
					value = -1
				}
			}

			# Legitimacy below 90
			add = {
				desc = "balkfm_value_from_legitimacy"
				value = legitimacy
				subtract = 80
				divide = 30
			}
		}

	}

	# Apply these values if House Obrenovic is ruling
	if = {
		limit = {
			balkfm_has_obrenovic_ruler_trigger = yes
		}
		## Relevant IGs will support non-ruling Karadordevic if no voting is present
		if = {
			limit = {
				NOR = {
					has_law_or_variant = law_type:law_landed_voting
					has_law_or_variant = law_type:law_wealth_voting
					has_law_or_variant = law_type:law_census_voting
					has_law_or_variant = law_type:law_universal_suffrage
				}
			}
			add = {
				value = {
					every_interest_group = {
						limit = {
							OR = {
								law_stance = {
									law = law_type:law_landed_voting
									value > neutral
								}
								law_stance = {
									law = law_type:law_wealth_voting
									value > neutral
								}
								law_stance = {
									law = law_type:law_census_voting
									value > neutral
								}
								law_stance = {
									law = law_type:law_universal_suffrage
									value > neutral
								}
							}
						}
						add = ig_clout
					}
				}
				desc = balkfm_value_from_disenfranchisment
			}
		}
		## Relevant IGs will support non-ruling Karadordevic if speech is suppressed
		if = {
			limit = {
				NOR = {
					has_law_or_variant = law_type:law_right_of_assembly
					has_law_or_variant = law_type:law_protected_speech
				}
			}
			add = {
				value = {
					every_interest_group = {
						limit = {
							OR = {
								law_stance = {
									law = law_type:law_right_of_assembly
									value > neutral
								}
								law_stance = {
									law = law_type:law_protected_speech
									value > neutral
								}
							}
						}
						add = ig_clout
					}
				}
				desc = balkfm_value_from_censorship
			}
		}
		## Relevant IGs will support non-ruling Karadorevic if land reform is suppressed
		if = {
			limit = {
				OR = {
					has_law_or_variant = law_type:law_serfdom
					has_law_or_variant = law_type:law_tenant_farmers
				}
			}
			add = {
				value = {
					every_interest_group = {
						limit = {
							OR = {
								law_stance = {
									law = law_type:law_serfdom
									value < neutral
								}
								law_stance = {
									law = law_type:law_tenant_farmers
									value < neutral
								}
							}
						}
						add = ig_clout
					}
				}
				desc = balkfm_value_from_land_conversativism
			}
		}
	}

	# Always apply these values
	## Consul Check
	if = {
		limit = {
			has_variable = balkfm_russia_consul_var
		}
		add = {
			value = 0.2
			desc = balkfm_value_from_rus_consul
		}
	}
	else_if = {
		limit = {
			has_variable = balkfm_austria_consul_var
		}
		add = {
			value = -0.2
			desc = balkfm_value_from_aus_consul
		}
	}
	## Russia Relations Check
	add = {
		value = {
			add = c:RUS.relations:root
			divide = 100
		}
		desc = balkfm_value_from_rus_relations
	}
	## Lobby Appeasement Check
	add = {
		value = {
			every_political_lobby = {
				limit = {
					is_political_lobby_type = lobby_pro_country
					target = {
						c:RUS ?= this
					}
				}
				add = appeasement
			}
			divide = 10
		}
		desc = balkfm_value_from_rus_lobbies
	}
	## Southern Serbia Control Check
	# Alexander Karadordevic lost a huge chunk of his support for failing to secure Serbian gains after the Crimean War
	if = {
		limit = {
			has_variable = balkfm_karadordevic_vojvodina_var
			NOT = {
				owns_entire_state_region = STATE_EASTERN_SERBIA
			}
		}
		subtract = {
			value = 2
			desc = balkfm_value_from_weak_kaykay
		}
	}
	else_if = {
		limit = {
			has_variable = balkfm_karadordevic_vojvodina_var
			owns_entire_state_region = STATE_EASTERN_SERBIA
		}
		add = {
			value = 2
			desc = balkfm_value_from_strong_kaykay
		}
	}
	round_to = 0.01
}
