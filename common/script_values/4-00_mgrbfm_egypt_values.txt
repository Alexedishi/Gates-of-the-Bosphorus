mgrbfm_egypt_modernization_progress_value = {
	value = 0
	add = {
		desc = mgrbfm_value_from_prestige
		value = {
			add = prestige
			multiply = 0.001
			round_to = 0.01
		}
	}
	add = {
		desc = mgrbfm_value_from_literacy_rate
		value = literacy_rate
		subtract = 0.30
		round_to = 0.01
	}
	add = {
		desc = mgrbfm_value_from_misri_urbanization
		value = {
			every_scope_state = {
				limit = {
					region = sr:region_nile_basin
					state_region = {
						is_homeland = misri
					}
				}
				add = state_urbanization_rate
			}
			multiply = 0.001
			round_to = 0.01
		}
	}
	add = {
		desc = mgrbfm_value_from_average_vs_expected_sol
		value = {
			add = average_sol
			subtract = average_expected_sol
			multiply = 0.01
			round_to = 0.01
		}
	}
	add = {
		desc = gbbf_value_from_institution_levels
		value = {
			add = gbbf_institution_investment_levels_value
			multiply = 0.1
		}
	}
	if = {
		limit = {
			is_country_type = unrecognized
		}
		subtract = {
			desc = mgrbfm_value_from_unrecognized
			value = 0.5
		}
	}
	subtract = {
		desc = mgrbfm_value_from_misri_radicalism
		value = {
			every_scope_state = {
				limit = {
					state_region = {
						is_homeland = misri
					}
				}
				add = radical_fraction
			}
			divide = {
				every_scope_state = {
					limit = {
						state_region = {
							is_homeland = misri
						}
					}
					add = 1
				}
				min = 1 # Div/0 guard
			}
			round_to = 0.01
		}
	}
	round_to = 0.01
}
