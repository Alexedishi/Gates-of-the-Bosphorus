mgrbfm_egypt_modernization_progress_value = {
	value = 0
	add = {
		desc = mgrbfm_value_from_prestige
		value = {
			add = prestige
			multiply = 0.001
			round_to = 0.01
		}
	}
	add = {
		desc = mgrbfm_value_from_literacy_rate
		value = literacy_rate
		subtract = 0.30
		round_to = 0.01
	}
	add = {
		desc = mgrbfm_value_from_misri_urbanization
		value = {
			every_scope_state = {
				limit = {
					state_region = {
						is_homeland = misri
					}
				}
				add = state_urbanization_rate
			}
			multiply = 0.001
			round_to = 0.01
		}
	}
	add = {
		desc = mgrbfm_value_from_average_vs_expected_sol
		value = {
			add = average_sol
			subtract = average_expected_sol
			multiply = 0.01
			round_to = 0.01
		}
	}
	add = {
		desc = gbbf_value_from_institution_levels
		value = {
			add = gbbf_institution_investment_levels_value
			multiply = 0.1
		}
	}
	add = {
		desc = gbbf_value_from_country_monopolies
		value = {
			subtract = var:je_mgrbfm_egypt_modernization.var:gbbf_country_monopolies_var
			multiply = 0.1
		}
	}
	if = {
		limit = {
			is_country_type = unrecognized
		}
		subtract = {
			desc = mgrbfm_value_from_unrecognized
			value = 0.5
		}
	}
	subtract = {
		desc = mgrbfm_value_from_misri_radicalism
		value = {
			every_scope_state = {
				limit = {
					state_region = {
						is_homeland = misri
					}
				}
				add = radical_fraction
			}
			divide = {
				every_scope_state = {
					limit = {
						state_region = {
							is_homeland = misri
						}
					}
					add = 1
				}
				min = 1 # Div/0 guard
			}
			round_to = 0.01
		}
	}
	round_to = 0.01
}

mgrbfm_misri_identity_value = {
	value = 0
	add = {
		desc = mgrbfm_value_from_misri_homelands
		value = {
			every_scope_state = {
				limit = {
					is_incorporated = yes
					is_split_state = no
					state_region = {
						is_homeland = misri
					}
				}
				add = 0.1
			}
		}
	}
	add = {
		desc = mgrbfm_value_from_misri_urbanization
		value = {
			every_scope_state = {
				limit = {
					state_region = {
						is_homeland = misri
					}
				}
				add = state_urbanization_rate
			}
			multiply = 0.01
			round_to = 0.01
		}
	}
	add = {
		desc = mgrbfm_value_from_secular_academia_levels
		value = {
			every_scope_building = {
				limit = {
					is_building_type = building_university
					has_active_production_method = pm_secular_academia
				}
				add = level
				multiply = 0.1
			}
		}
	}
	add = {
		desc = mgrbfm_value_from_misri_formations
		value = {
			every_scope_pop = {
				limit = {
					has_pop_culture = misri
					OR = {
						is_pop_type = officers
						is_pop_type = soldiers
					}
				}
				add = total_size
				divide = 100000
				round_to = 0.01
			}
		}
	}
	if = {
		limit = {
			morgenrote_is_active = yes
		}
		add = {
			desc = mgrbfm_value_from_egypt_artifact_major
			value = var:lepsius_egyptian_artifact_major
		}
		add = {
			desc = mgrbfm_value_from_egypt_artifact_minor
			value = {
				add = var:lepsius_egyptian_artifact_minor
				multiply = 0.1
			}
		}
	}
	add = {
		desc = mgrbfm_value_from_literacy_rate
		value = literacy_rate
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_freedom_of_conscience
		}
		add = {
			desc = mgrbfm_value_from_religious_law
			value = 1
		}
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_multicultural
		}
		divide = {
			desc = mgrbfm_value_from_multicultural
			value = 2
		}
	}
}
