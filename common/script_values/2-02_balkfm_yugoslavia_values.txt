####---------- Yugoslav Nationalism Values ----------####
balkfm_yugoslavism_progress_value = {
	value = 0
	## Integralist Progress Sources
	add = {
		desc = balkfm_value_from_yugoslav_loyalists
		value = balkfm_yugoslav_loyalists_value
	}
	add = {
		value = literacy_rate
		desc = balkfm_value_from_literacy
	}
	add = {
		desc = balkfm_value_from_legitimacy
		value = {
			add = government_legitimacy
			divide = 100
			round_to = 0.01
		}
	}
	if = {
		limit = {
			average_sol > average_expected_sol
		}
		add = {
			desc = balkfm_value_from_sol_surplus
			value = {
				add = average_sol
				subtract = average_expected_sol
				divide = 100
				round_to = 0.01
			}
			
		}
	}
	if = {
		limit = {
			average_sol < average_expected_sol
		}
		add = {
			desc = balkfm_value_from_sol_deficit
			value = {
				add = average_sol
				subtract = average_expected_sol
				divide = 100
				round_to = 0.01
			}
		}
	}
	add = {
		desc = balkfm_value_from_yugoslav_national_movement
		value = {
			every_political_movement = {
				limit = {
					is_political_movement_type = movement_yugoslav_national
				}
				add = political_movement_support
			}
		}
	}
	add = {
		desc = balkfm_value_from_yugoslav_states
		value = {
			every_scope_state = {
				limit = {
					is_in_geographic_region = geographic_region_yugoslavia
					is_incorporated = yes
					is_split_state = no
				}
				add = 0.1
			}
		}
	}
	if = {
		limit = {
			active_law:lawgroup_bureaucracy = {
				has_amendment = amendment_type:amendment_gbbf_disbanded_assemblies
			}
		}
		add = {
			desc = balkfm_value_from_disbanded_assemblies
			value = 0.25
		}
	}
	
	## Swing Sources
	subtract = {
		desc = balkfm_value_from_insurrectionary_yugoslav_movements
		value = {
			every_political_movement = {
				limit = {
					is_political_movement_type = movement_cultural_minority
					culture ?= {
						balkfm_is_yugoslav_culture_trigger = yes
					}
					is_insurrectionary = yes
				}
				add = political_movement_support
			}
			round_to = 0.01
		}
	}
	
	## Federalist Sources
	subtract = {
		desc = balkfm_value_from_yugoslav_cultures_beyond_first
		value = {
			add = balkfm_yugoslav_primary_cultures_value
			subtract = 1
			divide = 10
		}
	}
	subtract = {
		desc = balkfm_value_from_yugoslav_non_primary_radicals
		value = balkfm_yugoslav_non_primary_radicals_value
	}
	if = {
		limit = {
			has_global_variable = balkfm_montegrin_culture_var
			NOT = {
				country_has_primary_culture = cu:montenegrin
			}
		}
		subtract = {
			desc = balkfm_value_from_montenegrin_fervor_impact
			value = balkfm_montenegrin_fervor_impact_value
		}
	}
	if = {
		limit = {
			NOT = {
				country_has_primary_culture = cu:serb
			}
		}
		subtract = {
			desc = balkfm_value_from_serb_fervor_impact
			value = balkfm_serb_fervor_impact_value
		}
	}
	if = {
		limit = {
			has_variable = bul_pursued_greater_yugoslavism # Inherited from IP3
			NOT = {
				country_has_primary_culture = cu:bulgarian
			}
		}
		subtract = {
			desc = balkfm_value_from_bulgarian_fervor_impact
			value = balkfm_bulgarian_fervor_impact_value
		}
	}
	if = {
		limit = {
			NOT = {
				country_has_primary_culture = cu:bosniak
			}
		}
		subtract = {
			desc = balkfm_value_from_bosniak_fervor_impact
			value = balkfm_bosniak_fervor_impact_value
		}
	}
	if = {
		limit = {
			NOT = {
				country_has_primary_culture = cu:croat
			}
		}
		subtract = {
			desc = balkfm_value_from_croat_fervor_impact
			value = balkfm_croat_fervor_impact_value
		}
	}
	if = {
		limit = {
			NOT = {
				country_has_primary_culture = cu:slovene
			}
		}
		subtract = {
			desc = balkfm_value_from_slovene_fervor_impact
			value = balkfm_slovene_fervor_impact_value
		}
	}
	if = {
		limit = {
			active_law:lawgroup_bureaucracy = {
				has_amendment = amendment_type:amendment_gbbf_advisory_assemblies
			}
		}
		subtract = {
			desc = balkfm_value_from_advisory_assemblies
			value = 0.10
		}
	}
	if = {
		limit = {
			active_law:lawgroup_bureaucracy = {
				has_amendment = amendment_type:amendment_gbbf_autonomous_assemblies
			}
		}
		subtract = {
			desc = balkfm_value_from_autonomous_assemblies
			value = 0.50
		}
	}
}

balkfm_yugoslav_loyalists_value = {
	value = 0
	every_scope_state = {
		limit = {
			is_in_geographic_region = geographic_region_rumelia
			any_scope_pop = {
				balkfm_pop_has_yugoslav_culture_trigger = yes
			}
		}
		every_scope_pop = {
			limit = {
				balkfm_pop_has_yugoslav_culture_trigger = yes
			}
			add = pop_loyalist_fraction
		}

		every_scope_state = {

				is_in_geographic_region = geographic_region_rumelia

				}
			}
			every_scope_pop = {
				limit = {
					balkfm_pop_has_yugoslav_culture_trigger = yes
				}
				add = 1
			}
		}
		min = 1
	}
	round_to = 0.01
}
