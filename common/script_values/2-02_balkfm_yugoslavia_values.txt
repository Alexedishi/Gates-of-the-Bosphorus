balkfm_yugoslavism_integralist_value = {
	value = 0
	add = {
		desc = balkfm_value_from_yugoslav_loyalists
		value = balkfm_yugoslav_loyalists_value
	}
	add = {
		value = literacy_rate
		desc = balkfm_value_from_literacy
	}
	add = {
		desc = balkfm_value_from_legitimacy
		value = {
			add = government_legitimacy
			divide = 100
			round_to = 0.01
		}
	}
	add = {
		desc = balkfm_value_from_yugoslav_cultures_beyond_first
		value = {
			add = balkfm_yugoslav_primary_cultures_value
			subtract = 1
			divide = 10
		}
	}
	if = {
		limit = {
			average_sol > average_expected_sol
		}
		add = {
			desc = balkfm_value_from_sol_surplus
			value = {
				add = average_sol
				subtract = average_expected_sol
				divide = 100
				round_to = 0.01
			}
			
		}
	}
	if = {
		limit = {
			average_sol < average_expected_sol
		}
		add = {
			desc = balkfm_value_from_sol_deficit
			value = {
				add = average_sol
				subtract = average_expected_sol
				divide = 100
				round_to = 0.01
			}
		}
	}
	add = {
		desc = balkfm_value_from_yugoslav_national_movement
		value = {
			every_political_movement = {
				limit = {
					is_political_movement_type = movement_yugoslav_national
				}
				add = political_movement_support
			}
		}
	}
	add = {
		desc = balkfm_value_from_yugoslav_states
		value = {
			every_scope_state = {
				limit = {
					is_in_geographic_region = geographic_region_yugoslavia
					is_split_state = no
				}
				add = 0.1
			}
		}
	}
}
