eqfm_escalation_progress_value = {
	value = 0
	c:TUR ?= {
		add = {
			value = 1
			desc = gbbf_value_from_base_monthly
		}
		add = {
			value = {
				every_political_movement = {
					limit = {
						OR = {
							is_political_movement_type = movement_cultural_minority
							is_political_movement_type = movement_religious_minority
						}
						
						gbbf_balkans_culture_trigger = yes
					}
					add = eqfm_movement_impact_value
				}
				multiply = 100 # This might be overkill, we'll find out soon
				round_to = 0.01
			}
			desc = eqfm_value_from_movement_impact
		}
		add = {
			value = {
				every_country = {
					limit = {
						has_diplomatic_pact = {
							who = c:TUR
							type = gbbf_accelerate_tensions
						}
					}
					save_temporary_scope_as = eqfm_accelerating_country
					add = {
						value = 2
						desc = eqfm_value_from_accelerating_tensions
					}
				}
			}
			desc = eqfm_value_from_overall_accelerating_tensions
		}
		add = {
			value = {
				every_scope_state = {
					limit = {
						has_modifier = modifier_eqfm_race_riots
					}
					save_temporary_scope_as = eqfm_riot_state
					add = {
						value = radical_fraction
						desc = eqfm_value_from_race_riots
					}
				}
				round_to = 0.01
			}
			desc = eqfm_value_from_ongoing_riots
		}
		subtract = {
			value = {
				every_country = {
					limit = {
						has_diplomatic_pact = {
							who = c:TUR
							type = gbbf_decelerate_tensions
						}
					}
					save_temporary_scope_as = eqfm_decelerating_country
					add = {
						value = 1
						desc = eqfm_value_from_decelerating_tensions
					}
				}
			}
			desc = eqfm_value_from_overall_decelerating_tensions
		}
	}
	if = {
		limit = {
			eqfm_balkan_powder_keg_can_escalate_trigger = no
		}
		multiply = 0
	}
	min = 0
}

## Crisis Values
eqfm_crisis_progress_value = {
	value = 0
	if = {
		limit = {
			# No relevant movement can be >= 75 for the progress bar to advance
			c:TUR ?= {
				NOT = {
					any_political_movement = {
						AND  = {
							is_secessionist = yes
							NOT = {
								political_movement_radicalism > define:NPolitics|MIN_RADICALISM_FOR_CIVIL_WAR_PROGRESSION
							}
						}
					}
				}
			}
		}
		if = {
			limit = {
				c:TUR ?= {
					legitimacy >= control_eastern_crisis_legitimacy_min
				}
			}
			add = {
				desc = "progress_from_ottoman_legitimacy_tt"
				value = 1
			}
		}
		if = {
			limit = {
				c:TUR ?= {
					is_at_war = no
				}
			}
			add = {
				desc = "progress_from_being_at_peace_tt"
				value = 1
			}
		}
		else = {
			add = {
				desc = "progress_from_being_at_war_tt"
				value = -3
			}
		}
		if = {
			limit = {
				c:TUR ?= {
					has_any_secessionists_growing = no
				}
			}
			add = {
				desc = "progress_from_no_secessionists_growing"
				value = 1
			}
		}
		if = {
			limit = {
				c:TUR ?= {
					has_variable = ottoman_crushed_secession
				}
			}
			add = {
				desc = "progress_from_crushed_secession"
				value = 2
			}
		}
	}
}

## Minority Movement Values (Activism Adjustments)
eqfm_movement_activism_target = {
	value = 0
	if = {
		limit = {
			eqfm_deconfliction_phase_trigger = yes
		}
		add = 0
	}
	if = {
		limit = {
			eqfm_stable_phase_trigger = yes
		}
		add = 10
	}
	else_if = {
		limit = {
			eqfm_unrest_phase_trigger = yes
		}
		add = 25
	}
	else_if = {
		limit = {
			eqfm_hostility_phase_trigger = yes
		}
		add = 50
	}
	else_if = {
		limit = {
			eqfm_crisis_phase_trigger = yes
		}
		add = 75
	}
}

eqfm_movement_activism_adjustment_value = {
	value = 0
	if = {
		limit = {
			political_movement_radicalism < eqfm_movement_activism_target
		}
		add = eqfm_movement_activism_target
		subtract = political_movement_radicalism
		if = {
			limit = {
				has_variable = eqfm_previous_adjustment_var
			}
			subtract = var:eqfm_previous_adjustment_var
		}
	}
	if = {
		limit = {
			political_movement_radicalism > eqfm_movement_activism_target
		}
		subtract = 1
	}
}

eqfm_movement_activism_multiplier_value = {
	value = 0
	if = {
		limit = {
			has_variable = eqfm_previous_adjustment_var
		}
		add = var:eqfm_previous_adjustment_var
	}
	add = eqfm_movement_activism_adjustment_value
	min = 0
}
