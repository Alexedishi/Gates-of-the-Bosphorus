## Primary Mechanic Values
turkfm_escalation_progress_value = {
	value = 0
	add = {
		value = 1
		desc = turkfm_value_from_base_progress
	}
	add = {
		value = {
			every_political_movement = {
				limit = {
					OR = {
						is_political_movement_type = movement_cultural_minority
						is_political_movement_type = movement_religious_minority
					}

					gbbf_balkans_culture_trigger = yes
				}
				add = turkfm_movement_impact_value
			}
			multiply = 100 # This might be overkill, we'll find out soon
			round_to = 0.01
		}
		desc = turkfm_value_from_movement_impact
	}
	add = {
		value = {
			every_country = {
				limit = {
					has_diplomatic_pact = {
						who = c:TUR
						type = turkfm_accelerate_tensions
					}
				}
				save_temporary_scope_as = turkfm_accelerating_country
				add = {
					value = 2
					desc = turkfm_value_from_accelerating_tensions
				}
			}
		}
		desc = turkfm_value_from_overall_accelerating_tensions
	}
	add = {
		value = {
			every_scope_state = {
				limit = {
					has_modifier = modifier_turkfm_race_riots
				}
				save_temporary_scope_as = turkfm_riot_state
				add = {
					value = radical_fraction
					desc = turkfm_value_from_race_riots
				}
			}
			round_to = 0.01
		}
		desc = turkfm_value_from_ongoing_riots
	}
	subtract = {
		value = {
			every_country = {
				limit = {
					has_diplomatic_pact = {
						who = c:TUR
						type = turkfm_decelerate_tensions
					}
				}
				save_temporary_scope_as = turkfm_decelerating_country
				add = {
					value = 1
					desc = turkfm_value_from_decelerating_tensions
				}
			}
		}
		desc = turkfm_value_from_overall_decelerating_tensions
	}
	if = {
		limit = {
			turkfm_can_escalate_tension_trigger = no
		}
		multiply = 0
	}
	min = 0
}

turkfm_deescalation_progress_value = {
	value = 0
	if = {
		limit = {
			global_var:turkfm_balkan_situation_phase_var != flag:turkfm_crisis
		}
		add = {
			value = 1
			desc = turkfm_value_from_base
		}

		add = {
			value = {
				every_country = {
					limit = {
						has_diplomatic_pact = {
							who = c:TUR
							type = turkfm_decelerate_tensions
						}
					}
					save_temporary_scope_as = turkfm_decelerating_country
					add = {
						value = 2
						desc = turkfm_value_from_decelerating_tensions
					}
				}
			}
			desc = turkfm_value_from_overall_decelerating_tensions
		}
		subtract = {
			value = {
				every_country = {
					limit = {
						has_diplomatic_pact = {
							who = c:TUR
							type = turkfm_accelerate_tensions
						}
					}
					save_temporary_scope_as = turkfm_accelerating_country
					add = {
						value = 1
						desc = turkfm_value_from_accelerating_tensions
					}
				}
			}
			desc = turkfm_value_from_overall_accelerating_tensions
		}
	}
	if = {
		limit = {
			global_var:turkfm_balkan_situation_phase_var = flag:turkfm_crisis
		}
		if = {
			limit = {
				has_any_secessionists_broken_out = no
				has_any_secessionists_growing = no
				is_at_war = no
			}
			add = {
				value = 50
				desc = turkfm_value_from_base_progress
			}
			subtract = {
				value = {
					every_political_movement = {
						limit = {
							turkfm_eligible_minority_movement_trigger = yes
							is_secessionist = yes
						}
					}
				}
				desc = turkfm_value_from_secessionist_movements
			}
		}
	}
	if = {
		limit = {
			turkfm_can_deescalate_tension_trigger = no
		}
		multiply = 0
	}
	min = 0
}

turkfm_tension_progress_value = {
	value = 0
	if = {
		limit = {
			global_var:turkfm_deescalation_var > global_var:turkfm_escalation_var
		}
		add = global_var:turkfm_deescalation_var
	}
	else = {
		add = global_var:turkfm_escalation_var
	}
}
