turkfm_senussi_support_value = {
	value = 0
	add = {
		value = {
			every_character_in_void = {
				limit = {
					turkfm_is_senussi_trigger = yes
				}
				add = popularity
			}
			divide = 100
			round_to = 0.01
		}
		desc = turkfm_value_from_senussi_popularity
	}
	if = {
		limit = {
			has_variable = turkfm_senussi_desert_decentralized_var
		}
		add = {
			value = var:turkfm_senussi_desert_decentralized_var
			desc = turkfm_value_from_decentralized_provinces
		}
	}
	add = {
		value = {
			every_scope_state = {
				limit = {
					has_modifier = modifier_turkfm_senussi_stronghold
				}
				add = 1
			}
		}
		desc = turkfm_value_from_strongholds
	}
	add = {
		value = var:turkfm_bedouin_radical_fraction_var
		desc = turkfm_value_from_bedouin_radicals
	}
}

turkfm_senussi_opposition_value = {
	value = 0
	add = {
		value = {
			every_political_movement = {
				limit = {
					is_political_movement_type = movement_cultural_majority
				}
			}
			add = political_movement_support
			if = {
				limit = {
					has_technology_researched = nationalism
				}
				multiply = 2
			}
		}
		desc = turkfm_value_from_cultural_majority_movement
	}
	if = {
		limit = {
			has_variable = turkfm_senussi_desert_centralized_var
		}
		add = {
			value = var:turkfm_senussi_desert_centralized_var
			desc = turkfm_value_from_centralized_provinces
		}
	}
	add = {
		value = {
			ig:ig_devout = {
				add = ig_clout
			}
			round_to = 0.01
		}
		desc = turkfm_value_from_devout_clout
	}
}

turkfm_senussi_progress_value = {
	add = {
		value = var:turkfm_senussi_support_var
		desc = turkfm_value_from_senussi_support
	}
	subtract = {
		value = var:turkfm_senussi_opposition_var
		desc = turkfm_value_from_senussi_opposition
	}
}

turkfm_bedouin_radical_fraction_value = {
	value = 0
	every_scope_state = {
		limit = {
			gbbf_senussi_geography_trigger = yes
			any_scope_pop = {
				has_pop_culture = bedouin
			}
		}
		every_scope_pop = {
			limit = {
				has_pop_culture = bedouin
			}
			add = pop_radical_fraction
		}
	}
	divide = {
		every_scope_state = {
			limit = {
				gbbf_senussi_geography_trigger = yes
				any_scope_pop = {
					has_pop_culture = bedouin
				}
			}
			every_scope_pop = {
				limit = {
					has_pop_culture = bedouin
				}
				add = 1
			}
		}
		min = 1
	}
	round_to = 0.01
}

turkfm_bedouin_radicals_as_fraction_of_country_value = {
	value = 0
	every_scope_state = {
		limit = {
			gbbf_senussi_geography_trigger = yes
			any_scope_pop = {
				has_pop_culture = bedouin
			}
		}
		every_scope_pop = {
			limit = {
				has_pop_culture = bedouin
			}
			add = pop_radical_fraction
		}
	}
	divide = {
		every_scope_state = {
			limit = {
				gbbf_senussi_geography_trigger = yes
				any_scope_pop = {
					has_pop_culture = bedouin
				}
			}
			every_scope_pop = {
				limit = {
					has_pop_culture = bedouin
				}
				add = 1
			}
		}
		min = 1
	}
	multiply = {
		every_scope_state = {
			limit = {
				gbbf_senussi_geography_trigger = yes
				any_scope_pop = {
					has_pop_culture = bedouin
				}
			}
			every_scope_pop = {
				limit = {
					has_pop_culture = bedouin
				}
				add = total_size
			}
		}
		min = 0
	}
	divide = total_population
	round_to = 0.01
	min = 0.01
}
