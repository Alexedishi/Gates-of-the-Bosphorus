	}
	if = {
		limit = {

				relations:root > relations_threshold:neutral

		}

		}
	}
	if = { 
		limit = {
			ig:ig_landowners = {
				is_powerful = yes

			}
		}
		add = {
			value = 1
			desc = balkfm_bul_from_powerful_landowners_in_gov
		}
	}
	if = {
		limit = {
			has_modifier = modifier_balkfm_russian_attaches
		}
		add = {
			value = 1
			desc = balkfm_bul_from_russian_attaches
		}
	}
	if = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_protected_speech
				has_law_or_variant = law_type:law_right_of_assembly
			}
		}
		add = {
			value = 1
			desc = balkfm_bul_from_good_speech_laws
		}
	}
	if = {
		limit = {
			has_variable = balkfm_railways_built
		}
		add = {
			value = 1
			desc = balkfm_bul_from_building_railways
		}
	}
	if = {
		limit = {
			has_variable = balkfm_security_reform_complete
		}
		add = {
			value = 1
			desc = balkfm_bul_from_brigandage_completed
		}
	}
	
	## PENALTIES ##
	
	if = {
		limit = {
			is_in_power_bloc = yes
			exists = c:RUS
			power_bloc = {
				power_bloc_leader = {
					c:RUS != this
				}
			}
		}
		subtract = {
			value = 1
			desc = balkfm_bul_from_non_russian_bloc
		}
	}
	if = {
		limit = {
			c:RUS = {
				relations:root < relations_threshold:neutral
			}
		}
		subtract = {
			value = 1
			desc = balkfm_bul_from_bad_russian_relations
		}
	}
	if = {
		limit = {
			ig:ig_armed_forces = {
				is_powerful = yes
				is_in_government = no
			}
		}
		subtract = {
			value = 1
			desc = balkfm_bul_from_powerful_military_not_in_gov
		}
	}
	if = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_censorship
				has_law_or_variant = law_type:law_outlawed_dissent
			}
		}
		subtract = {
			value = 1
			desc = balkfm_bul_from_bad_speech_laws
		}
	}
	if = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_autocracy
				has_law_or_variant = law_type:law_oligarchy
			}
		}
		subtract = {
			value = 1
			desc = balkfm_bul_from_no_voting
		}
	}
	if = {
		limit = {
			AND = {
				owns_entire_state_region = STATE_BULGARIA
				country_or_subject_owns_entire_state_region = STATE_NORTHERN_THRACE
			}
		}
		subtract = {
			value = 1
			desc = balkfm_bul_from_controlling_rumelia
		}
	}
	if = {
		limit = {
			has_modifier = modifier_balkfm_railway_buyout
		}
		subtract = {
			value = 1
			desc = balkfm_bul_from_railways_buyout
		}
	}
	
	round_to = 0.01
	min = 0
	max = 10
	
}
