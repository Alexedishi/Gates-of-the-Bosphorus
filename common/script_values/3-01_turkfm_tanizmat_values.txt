turkfm_tanzimat_tilt_value = {
	value = 0
	if = {
		limit = {
			var:turkfm_tanzimat_var > 500
		}
		add = var:turkfm_tanzimat_var
		subtract = 500
	}
	if = {
		limit = {
			var:turkfm_tanzimat_var < 500
		}
		add = 500
		subtract = var:turkfm_tanzimat_var
	}
	round_to = 0.01
}
