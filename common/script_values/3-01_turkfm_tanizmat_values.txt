turkfm_tanzimat_reformist_progress_value = {
	value = 0
	add = {
		value = {
			every_scope_building = {
				limit = {
					is_building_type = building_government_administration
					has_active_production_method = pm_secular_bureaucrats
				}
				add = 0.1
			}
		}
		desc = turkfm_value_from_secular_bureaucrats
	}
	add = {
		value = {
			every_scope_building = {
				limit = {
					is_building_type = building_university
					has_active_production_method = pm_secular_academia
				}
				add = 0.1
			}
		}
		desc = turkfm_value_from_secular_academia
	}
	add = {
		value = {
			if = {
				limit = {
					has_institution = institution_colonial_affairs
				}
				institution:institution_colonial_affairs = {
					add = investment
				}
			}
			if = {
				limit = {
					has_institution = institution_social_security
				}
				institution:institution_social_security = {
					add = investment
				}
			}
			if = {
				limit = {
					has_institution = institution_workplace_safety
				}
				institution:institution_workplace_safety = {
					add = investment
				}
			}
			if = {
				limit = {
					has_institution = institution_schools
					OR = {
						has_law_or_variant = law_type:law_public_schools
						has_law_or_variant = law_type:law_private_schools
					}
				}
				institution:institution_schools = {
					add = investment
				}
			}
			if = {
				limit = {
					has_institution = institution_police
				}
				institution:institution_police = {
					add = investment
				}
			}
			if = {
				limit = {
					has_institution = institution_health_system
					OR = {
						has_law_or_variant = law_type:law_public_health_insurance
						has_law_or_variant = law_type:law_private_health_insurance
					}
				}
				institution:institution_health_system = {
					add = investment
				}
			}
			if = {
				limit = {
					has_institution = institution_home_affairs
				}
				institution:institution_home_affairs = {
					add = investment
				}
			}
			multiply = 0.1
		}
		desc = turkfm_value_from_institution_levels
	}
	round_to = 0.01
}

turkfm_tanzimat_conservatist_progress_value = {
	value = 0
	add = {
		value = {
			every_interest_group = {
				limit = {
					law_stance = {
						law = law_type:law_state_religion
						value > neutral
					}
					NOR = {
						law_stance = {
							law = law_type:law_landed_voting
							value > neutral
						}
						law_stance = {
							law = law_type:law_wealth_voting
							value > neutral
						}
						law_stance = {
							law = law_type:law_census_voting
							value > neutral
						}
						law_stance = {
							law = law_type:law_universal_suffrage
							value > neutral
						}
					}
				}
				add = ig_clout
			}
			round_to = 0.01
		}
		desc = turkfm_value_from_islamism
	}
	add = {
		value = {
			if = {
				limit = {
					has_institution = institution_schools
					has_law_or_variant = law_type:law_religious_schools
				}
				institution:institution_schools = {
					add = investment
				}
			}
			if = {
				limit = {
					has_institution = institution_health_system
					has_law_or_variant = law_type:law_charitable_health_system
				}
				institution:institution_health_system = {
					add = investment
				}
			}
		}
		desc = turkfm_value_from_institution_levels
	}
	add = {
		value = {
			every_scope_building = {
				limit = {
					is_building_type = building_government_administration
					has_active_production_method = pm_religious_bureaucrats
				}
				add = 0.1
			}
		}
		desc = turkfm_value_from_religious_bureaucrats
	}
	add = {
		value = {
			every_scope_building = {
				limit = {
					is_building_type = building_university
					has_active_production_method = pm_religious_academia
				}
				add = 0.1
			}
		}
		desc = turkfm_value_from_religious_academia
	}
	add = {
		value = {
			if = {
				limit = {
					has_law = law_type:law_millet_system
				}
				add = 2
			}
			else_if = {
				limit = {
					has_law_or_variant = law_type:law_state_religion
				}
				add = 1
			}
			else_if = {
				limit = {
					has_law_or_variant = law_type:law_freedom_of_conscience
				}
				add = -1
			}
			else_if = {
				limit = {
					has_law_or_variant = law_type:law_total_separation
				}
				subtract = -2
			}
			else = {
				subtract = 5
			}
		}
		desc = turkfm_value_from_religious_law
	}
	round_to = 0.01
}

turkfm_young_ottomans_progress_value = {
	value = 0
	add = {
		value = {
			every_interest_group = {
				limit = {
					OR = {
						law_stance = {
							law = law_type:law_wealth_voting
							value > neutral
						}
						law_stance = {
							law = law_type:law_census_voting
							value > neutral
						}
						law_stance = {
							law = law_type:law_universal_suffrage
							value > neutral
						}
					}
					law_stance = {
						law = law_type:law_state_religion
						value > neutral
					}
					OR = {
						law_stance = {
							law = law_type:law_parliamentary_republic
							value > neutral
						}
						law_stance = {
							law = law_type:law_presidential_republic
							value > neutral
						}
					}
					NOR = {
						is_interest_group_type = ig_armed_forces
						ig_approval > 0
					}
				}
				add = ig_clout
			}
		}
		desc = turkfm_value_from_pro_young_ottomans
	}
	if = {
		limit = {
			ig:ig_armed_forces = {
				ig_approval > 0
			}
		}
		add = {
			value = ig:ig_armed_forces.ig_clout
			desc = turkfm_value_from_armed_forces
		}
	}
	subtract = {
		value = 0.20
		desc = turkfm_value_from_powerful_ig_baseline
	}
	round_to = 0.01
}
