eqfm_senusiyya_support_value = {
	value = 0
	# General Values
	if = {
		limit = {
			global_var:je_eqfm_senusiyya ?= {
				has_variable = eqfm_senussi_character_var
				exists = var:eqfm_senussi_character_var
				var:eqfm_senussi_character_var = {
					popularity > 0
				}
			}
		}
		add = {
			value = {
				add = global_var:je_eqfm_senusiyya.var:eqfm_senussi_character_var.popularity
				multiply = 0.01
				round_to = 0.01
			}
			desc = eqfm_value_from_senussi_popularity_surplus
		}
	}
	add = {
		value = {
			every_state_region = {
				limit = {
					OR = {
						region = sr:region_north_africa
						region = sr:region_nile_basin
					}
					any_scope_state = {
						has_modifier = modifier_eqfm_senussi_stronghold
					}
				}
				add = 0.5
			}
		}
		desc = eqfm_value_from_strongholds
	}
	add = {
		value = var:mgrbfm_bedouin_radical_fraction_var
		desc = eqfm_value_from_bedouin_radicals
	}
	# Province Centralization Values
	if = {
		limit = { 
			global_var:je_eqfm_senusiyya = {
				var:eqfm_state_libya_centralization_var > 0
			}
		}
		add = {
			desc = eqfm_value_from_libya_centralized
			value = global_var:je_eqfm_senusiyya.var:eqfm_state_libya_centralization_var
		}
	}
	if = {
		limit = {
			global_var:je_eqfm_senusiyya = {
				var:eqfm_state_libyan_desert_centralization_var > 0
			}
		}
		add = {
			desc = eqfm_value_from_libyan_desert_centralized
			value = global_var:je_eqfm_senusiyya.var:eqfm_state_libyan_desert_centralization_var
		}
	}
	if = {
		limit = {
			global_var:je_eqfm_senusiyya = {
				var:eqfm_state_tripoli_centralization_var > 0
			}
		}
		add = {
			desc = eqfm_value_from_tripoli_centralized
			value = global_var:je_eqfm_senusiyya.var:eqfm_state_tripoli_centralization_var
		}
	}
	if = {
		limit = {
			global_var:je_eqfm_senusiyya = {
				var:eqfm_state_matruh_centralization_var > 0
			}
		}
		add = {
			desc = eqfm_value_from_matruh_centralized
			value = global_var:je_eqfm_senusiyya.var:eqfm_state_matruh_centralization_var
		}
	}
	if = {
		limit = {
			global_var:je_eqfm_senusiyya = {
				var:eqfm_state_egyptian_desert_centralization_var > 0
			}
		}
		add = {
			desc = eqfm_value_from_egyptian_desert_centralized
			value = global_var:je_eqfm_senusiyya.var:eqfm_state_egyptian_desert_centralization_var
		}
	}
	if = {
		limit = {
			global_var:je_eqfm_senusiyya = {
				var:eqfm_state_tunisia_centralization_var > 0
			}
		}
		add = {
			desc = eqfm_value_from_tunisia_centralized
			value = global_var:je_eqfm_senusiyya.var:eqfm_state_tunisia_centralization_var
		}
	}
	round_to = 0.01
}

eqfm_senussi_opposition_value = {
	value = 0
	add = {
		value = {
			every_political_movement = {
				limit = {
					is_political_movement_type = movement_cultural_majority
				}
				add = political_movement_support
			}
			if = {
				limit = {
					has_technology_researched = nationalism
				}
				multiply = 2
			}
		}
		desc = turkfm_value_from_cultural_majority_movement
	}
	if = {
		limit = {
			has_variable = eqfm_senussi_desert_centralized_var
		}
		add = {
			value = var:eqfm_senussi_desert_centralized_var
			desc = eqfm_value_from_centralized_provinces
		}
	}
	add = {
		value = {
			ig:ig_devout = {
				add = ig_clout
			}
			round_to = 0.01
		}
		desc = eqfm_value_from_devout_clout
	}
	round_to = 0.01
}

eqfm_senusiyya_progress_value = {
	add = {
		value = global_var:je_eqfm_senusiyya.var:eqfm_senusiyya_support_var
		desc = eqfm_value_from_senussi_support
	}
	subtract = {
		value = global_var:je_eqfm_senusiyya.var:eqfm_senusiyya_opposition_var
		desc = eqfm_value_from_senussi_opposition
	}
	round_to = 0.01
}

eqfm_state_libya_centralization_value = {
	value = 0
	add = {
		desc = eqfm_value_from_libya_centralized
		value = {
			s:STATE_LIBYA= {
				every_scope_state = {
					limit = {
						owner = {
							NOT = {
								is_country_type = decentralized
							}
						}
					}
					add = num_provinces
				}
			}
		}
	}
	divide = {
		desc = eqfm_value_from_libya_decentralized
		value = {
			s:STATE_LIBYA= {
				every_scope_state = {
					limit = {
						owner = {
							is_country_type = decentralized
							country_has_primary_culture = cu:bedouin
						}
					}
					add = num_provinces
				}
			}
			min = 1 # Div0 safety
		}
	}
	round_to = 0.01
}

		}
		every_scope_pop = {
			limit = {
				has_pop_culture = bedouin
			}
			add = pop_radical_fraction
		}
	}
	divide = {
		value = {
			every_state_in_senusiyya = {
				limit = {
					owner = {
						NOT = {
							is_country_type = decentralized
						}

					limit = {
						has_pop_culture = bedouin
					}
					add = 1
				}
			}
			min = 1 # Div0 safety
		}
	}
	round_to = 0.01
}
