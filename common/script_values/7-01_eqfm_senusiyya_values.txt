eqfm_senusiyya_progress_value = {
	value = 0
	# General Values
	if = {
		limit = {
			global_var:je_eqfm_senusiyya ?= {
				has_variable = eqfm_senussi_character_var
				exists = var:eqfm_senussi_character_var
			}
		}
		add = {
			desc = eqfm_value_from_senussi_popularity
			value = {
				add = global_var:je_eqfm_senusiyya.var:eqfm_senussi_character_var.popularity
				multiply = 0.01
				round_to = 0.01
			}
		}
	}
	# State Region Values
	if = {
		limit = {
			s:STATE_LIBYA = {
				has_variable = eqfm_senusiyya_entrenchment_var
			}
		}
		add = {
			desc = eqfm_value_from_senusiyya_entrenchment_libya
			value = s:STATE_LIBYA.var:eqfm_senusiyya_entrenchment_var
		}
	}
	if = {
		limit = {
			s:STATE_LIBYAN_DESERT = {
				has_variable = eqfm_senusiyya_entrenchment_var
			}
		}
		add = {
			desc = eqfm_value_from_senusiyya_entrenchment_libyan_desert
			value = s:STATE_LIBYAN_DESERT.var:eqfm_senusiyya_entrenchment_var
		}
	}
	if = {
		limit = {
			s:STATE_TRIPOLI = {
				has_variable = eqfm_senusiyya_entrenchment_var
			}
		}
		add = {
			desc = eqfm_value_from_senusiyya_entrenchment_tripoli
			value = s:STATE_TRIPOLI.var:eqfm_senusiyya_entrenchment_var
		}
	}
	if = {
		limit = {
			s:STATE_MATRUH = {
				has_variable = eqfm_senusiyya_entrenchment_var
			}
		}
		add = {
			desc = eqfm_value_from_senusiyya_entrenchment_matruh
			value = s:STATE_MATRUH.var:eqfm_senusiyya_entrenchment_var
		}
	}
	if = {
		limit = {
			s:STATE_EGYPTIAN_DESERT = {
				has_variable = eqfm_senusiyya_entrenchment_var
			}
		}
		add = {
			desc = eqfm_value_from_senusiyya_entrenchment_egyptian_desert
			value = s:STATE_EGYPTIAN_DESERT.var:eqfm_senusiyya_entrenchment_var
		}
	}
	if = {
		limit = {
			s:STATE_TUNISIA = {
				has_variable = eqfm_senusiyya_entrenchment_var
			}
		}
		add = {
			desc = eqfm_value_from_senusiyya_entrenchment_tunisia
			value = s:STATE_TUNISIA.var:eqfm_senusiyya_entrenchment_var
		}
	}
	round_to = 0.01
}

eqfm_senusiyya_entrenchment_progress_value = {
	# This is evaluated in the !!STATE REGION!! scope
	value = 0
	add = {
		desc = eqfm_value_from_bedouin_literacy
		value = {
			every_scope_state = {
				limit = {
					owner = {
						NOT = {
							country_has_primary_culture = cu:bedouin
						}
					}
				}
				every_scope_pop = {
					limit = {
						has_pop_culture = bedouin
					}
					add = literacy_rate
				}
			}
			divide = {
				every_scope_state = {
					limit = {
						owner = {
							NOT = {
								country_has_primary_culture = cu:bedouin
							}
						}
					}
					every_scope_pop = {
						limit = {
							has_pop_culture = bedouin
						}
						add = 1
					}
				}
				min = 1  # Div0 safety
			}
			round_to = 0.01
		}
	}
	add = {
		desc = eqfm_from_senusiyya_strongholds
		value = {
			every_scope_state = {
				limit = {
					has_modifier = modifier_eqfm_senusiyya_stronghold
				}
				add = modifier:state_senusiyya_pressure_add
			}
			divide = {
				every_scope_state = {
					limit = {
						has_modifier = modifier_eqfm_senusiyya_stronghold
					}
					add = 1
				}
				min = 1 # Div0 safety
			}
			round_to = 0.01
		}
	}
	add = {
		desc = eqfm_value_from_bedouin_radicals
		value = {
			every_scope_state = {
				limit = {
					owner = {
						NOR = {
							is_country_type = decentralized
							country_has_primary_culture = cu:bedouin
						}
					}
				}
				every_scope_pop = {
					limit = {
						has_pop_culture = bedouin
					}
					add = pop_radical_fraction
				}
			}
			divide = {
				every_scope_state = {
					limit = {
						owner = {
							NOR = {
								is_country_type = decentralized
								country_has_primary_culture = cu:bedouin
							}
						}
					}
					every_scope_pop = {
						limit = {
							has_pop_culture = bedouin
						}
						add = 1
					}
				}
				min = 1 # Div0 safety
			}
			round_to = 0.01
		}
	}
	if = {
		limit = {
			s:STATE_LIBYA = this
		}
		add = {
			desc = eqfm_value_from_libya_centralization_ratio
			value = global_var:je_eqfm_senusiyya.var:eqfm_state_libya_centralization_var
		}
	}
	if = {
		limit = {
			s:STATE_LIBYAN_DESERT = this
		}
		add = {
			desc = eqfm_value_from_libyan_desert_centralization_ratio
			value = global_var:je_eqfm_senusiyya.var:eqfm_state_libyan_desert_centralization_var
		}
	}
	if = {
		limit = {
			s:STATE_TRIPOLI = this
		}
		add = {
			desc = eqfm_value_from_tripoli_centralization_ratio
			value = global_var:je_eqfm_senusiyya.var:eqfm_state_tripoli_centralization_var
		}
	}
	if = {
		limit = {
			s:STATE_MATRUH = this
		}
		add = {
			desc = eqfm_value_from_matruh_centralization_ratio
			value = global_var:je_eqfm_senusiyya.var:eqfm_state_matruh_centralization_var
		}
	}
	if = {
		limit = {
			s:STATE_EGYPTIAN_DESERT = this
		}
		add = {
			desc = eqfm_value_from_egyptian_desert_centralization_ratio
			value = global_var:je_eqfm_senusiyya.var:eqfm_state_egyptian_desert_centralization_var
		}
	}
	if = {
		limit = {
			s:STATE_TUNISIA = this
		}
		add = {
			desc = eqfm_value_from_tunisia_centralization_ratio
			value = global_var:je_eqfm_senusiyya.var:eqfm_state_tunisia_centralization_var
		}
	}
	subtract = {
		desc = eqfm_value_from_average_urbanization
		value = {
			every_scope_state = {
				limit = {
					owner = {
						NOR = {
							is_country_type = decentralized
							country_has_primary_culture = cu:bedouin
						}
					}
				}
				add = state_urbanization_rate
			}
			divide = {
				every_scope_state = {
					limit = {
						owner = {
							NOR = {
								is_country_type = decentralized
								country_has_primary_culture = cu:bedouin
							}
						}
					}
					add = 1
				}
				min = 1 # Div0 safety
			}
			divide = 10
			round_to = 0.01
		}
	}
	subtract = {
		desc = eqfm_value_from_bedouin_loyalists
		value = {
			every_scope_state = {
				limit = {
					owner = {
						NOR = {
							is_country_type = decentralized
							country_has_primary_culture = cu:bedouin
						}
					}
				}
				every_scope_pop = {
					limit = {
						has_pop_culture = bedouin
					}
					add = pop_radical_fraction
				}
			}
			divide = {
				every_scope_state = {
					limit = {
						owner = {
							NOR = {
								is_country_type = decentralized
								country_has_primary_culture = cu:bedouin
							}
						}
					}
					every_scope_pop = {
						limit = {
							has_pop_culture = bedouin
						}
						add = 1
					}
				}
				min = 1 # Div0 safety
			}
			round_to = 0.01
		}
	}
}

eqfm_senusiyya_penetration_progress_value = {
	value = 0
	add = {
		desc = eqfm_value_from_senusiyya_pressure
		add = global_var:je_eqfm_senusiyya.var:eqfm_senusiyya_var
	}
	add = {
		desc = eqfm_value_from_legitimacy
		value = {
			add = government_legitimacy
			subtract = 70
			divide = 100
			round_to = 0.01
		}
	}
	add = {
		desc = eqfm_value_from_senusiyya_entrenchment
		value = {
			every_scope_state = {
				limit = {
					state_region = {
						has_variable = eqfm_senusiyya_entrenchment_var
					}
				}
				add = var:eqfm_senusiyya_entrenchment_var
				subtract = 0.5
			}
		}
	}
}
