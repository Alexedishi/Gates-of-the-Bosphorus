		desc = eqfm_value_from_sanusiyyah_pressure

eqfm_sanusiyyah_entrenchment_progress_value = {
	# This is evaluated in the !!STATE REGION!! scope
	value = 0
	add = {
		desc = eqfm_value_from_bedouin_literacy
		value = {
			every_scope_state = {
				limit = {
					owner = {
						NOT = {
							country_has_primary_culture = cu:bedouin
						}
					}
				}
				every_scope_pop = {
					limit = {
						has_pop_culture = bedouin
					}
					add = literacy_rate
				}
			}
			divide = {
				every_scope_state = {
					limit = {
						owner = {
							NOT = {
								country_has_primary_culture = cu:bedouin
							}
						}
					}
					every_scope_pop = {
						limit = {
							has_pop_culture = bedouin
						}
						add = 1
					}
				}
				min = 1  # Div0 safety
			}
			round_to = 0.01
		}
	}
	add = {
		desc = eqfm_from_sanusiyyah_strongholds
		value = {
			every_scope_state = {
				limit = {
					has_modifier = modifier_eqfm_sanusiyyah_stronghold
				}
				add = modifier:state_sanusiyyah_pressure_growth_add
			}
			divide = {
				every_scope_state = {
					limit = {
						has_modifier = modifier_eqfm_sanusiyyah_stronghold
					}

				}
				min = 1 # Div0 safety

			round_to = 0.01

							}
						}
					}

						limit = {
							has_pop_culture = bedouin
						}
						add = 1
					}
				}
				min = 1 # Div0 safety
			}
			round_to = 0.01
		}
	}
	if = {
		limit = {
			s:STATE_LIBYA = this
		}
		add = {
			desc = eqfm_value_from_libya_centralization_ratio
			value = global_var:je_eqfm_sanusiyyah.var:eqfm_state_libya_centralization_var
		}
	}

		}

			desc = eqfm_value_from_libyan_desert_centralization_ratio
