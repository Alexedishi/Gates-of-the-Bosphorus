turkfm_karadag_tur_value = {
	value = 0

	#Sources
	if = {
		limit = {
			NOT = {
				has_journal_entry = je_turkfm_sick_man_main
			}
		}
		add = {
			desc = turkfm_value_from_sick_man
			value = 4
		}
	}
	else_if = {
		limit = {
			has_variable = turkfm_sick_man_var
		}
		add = {
			desc = turkfm_value_from_sick_man
			value = var:turkfm_sick_man_var
		}
	}
	if = {
		limit = {
			NOT = {
				has_journal_entry = je_turkfm_tanzimat_main
			}
		}
		add = {
			desc = turkfm_value_from_tanzimat
			value = 5
		}
	}
	if = {
		limit = {
			has_variable = turkfm_tanzimat_completion_var
		}
		add = {
			desc = turkfm_value_from_tanzimat
			value = {
				add = var:turkfm_tanzimat_completion_var
				max = 5
			}
		}
	}

	#Penalities
	if = {
		limit = {
			has_journal_entry = je_sick_man_main
		}
		subtract = {
			desc = turkfm_value_from_sick_man_open
			value = 4
		}
	}
	if = {
		limit = {
			has_modifier = modifier_turkfm_accepted_ransom
		}
		subtract = {
			desc = turkfm_value_from_ransoming_nobles
			value = 1
		}
	}
}

turkfm_karadag_mon_value = {
	value = 0

	# Sources
	add = {
		desc = turkfm_value_from_bishopric_journals
		value = {
			if = {
				limit = {
					has_variable = balkfm_mon_soc_completed
				}
				add = 1
			}
			if = {
				limit = {
					has_variable = balkfm_mon_gov_completed
				}
				add = 1
			}
			if = {
				limit = {
					has_variable = balkfm_mon_mil_completed
				}
				add = 1
			}
		}
	}
	add = {
		desc = turkfm_value_from_capital_infrastructure
		value = {
			add = root.capital.relative_infrastructure
			divide = 10
			max = 10 # to prevent abuse
		}
	}
	add = {
		desc = turkfm_value_from_standing_army
		value = {
			add = root.army_power_projection
			divide = 10
			max = 100 # lol
		}
	}
	if = {
		limit = {
			has_modifier = modifier_turkfm_ransomed_noble
		}
		add = {
			desc = turkfm_value_from_ransoming_nobles
			value = 1
		}
	}
	if = {
		limit = {
			has_modifier = modifier_turkfm_executed_noble
		}
		add = {
			desc = turkfm_value_from_beheading_noble
			value = 2
		}
	}
	if = {
		limit = {
			has_modifier = modifier_balkfm_skulduggery
		}
		add = {
			desc = turkfm_value_from_skulduggery
			value = 1
		}
	}
	if = {
		limit = {
			any_country = {
				c:TUR != this
				OR = {
					any_scope_treaty = {
						binds = root
						binds = c:TUR
						any_scope_article = {
							has_type = alliance
						}
					}
					any_scope_treaty = {
						binds = root
						binds = c:TUR
						any_scope_article = {
							has_type = defensive_pact
						}
					}
				}
				has_diplomatic_pact = {
					who = c:TUR
					type = rivalry
				}
			}
		}
		add = {
			desc = turkfm_value_from_diplomatic_pacts
			value = 5
		}
	}
	if = {
		limit = {
			is_in_power_bloc = yes
			power_bloc = {
				power_bloc_leader = {
					has_diplomatic_pact = {
						who = c:TUR
						type = rivalry
					}
				}
			}
		}
		add = {
			desc = turkfm_value_from_power_bloc
			value = 5
		}
	}

	#Penalities
	if = {
		limit = {
			has_modifier = modifier_balkfm_tolerated_abuses
		}
		subtract = {
			desc = turkfm_value_from_tolerating_abuses
			value = 1
		}
	}
	if = {
		limit = {
			has_modifier = modifier_balkfm_display_of_poverty
		}
		subtract = {
			desc = turkfm_value_from_poverty_display
			value = 1
		}
	}
}
