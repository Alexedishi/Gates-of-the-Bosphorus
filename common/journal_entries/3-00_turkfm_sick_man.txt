je_turkfm_sick_man_main = {
	group = je_group_gbbf_sick_man
	icon = "gfx/interface/icons/event_icons/event_ottoman.dds"
	weight = 100
	
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	transferable = no
	
	is_shown_in_lobby = {
		c:TUR ?= this
	}
	
	scripted_progress_bar = turkfm_sick_main_progress_bar
	
	immediate = {
		gbbf_assign_dlc_icon_effect = yes
	}
	
	modifiers_while_active = {
		modifier_turkfm_sick_man
	}
	
	complete = {
		country_rank >= rank_value:great_power
		gbbf_controls_anatolia_trigger = yes
		calc_true_if = {
			amount >= 4
			gbbf_controls_aegean_sea_trigger = yes
			gbbf_controls_balkan_highlands_trigger = yes
			gbbf_controls_levant_trigger = yes
			gbbf_controls_nile_river_trigger = yes
			gbbf_controls_armenian_highlands_trigger = yes
			gbbf_controls_arabia_trigger = yes
		}
		custom_tooltip = {
			text = turkfm_tanzimat_completed_or_failed
			has_global_variable = turkfm_tanzimat_complete
		}
		custom_tooltip = {
			text = eqfm_separatism_completed_or_failed
			has_global_variable = eqfm_balkan_situation_resolved
		}
	}
	
	on_complete = {
		trigger_event = turkfm_sick_man.009
		set_global_variable = {
			name = turkfm_sick_man_complete
			value = flag:turkfm_sick_man_completed
		}
	}
	
	fail = {
		OR = {
			country_rank < rank_value:major_power
			custom_tooltip = {
				text = turkfm_grand_collapse_failed
				has_variable = turkfm_grand_collapse_failed
			}
			NOR = {
				gbbf_controls_anatolia_trigger = yes
				gbbf_controls_aegean_sea_trigger = yes
				gbbf_controls_balkan_highlands_trigger = yes
				gbbf_controls_levant_trigger = yes
				gbbf_controls_nile_river_trigger = yes
				gbbf_controls_armenian_highlands_trigger = yes
				gbbf_controls_arabia_trigger = yes
			}
		}
	}
	
	on_fail = {
		trigger_event = turkfm_sick_man.008
		if = {
			limit = {
				is_in_power_bloc = yes
				is_power_bloc_leader = yes
			}
			custom_tooltip = sick_man_of_europe_lose_power_bloc
		}
		set_global_variable = {
			name = turkfm_sick_man_complete
			value = flag:turkfm_sick_man_failed
		}
	}
	
	invalid = {
		hidden_trigger = {
			has_global_variable = turkfm_sick_man_complete
		}
	}
	
	on_weekly_pulse = {
		events = {
			turkfm_sick_man.005 # Region Lost
			turkfm_sick_man.006 # Region Secured
		}
		effect = {
			turkfm_calculate_sick_man_effect = yes
		}
	}
	
	on_monthly_pulse = {
		events = {
			turkfm_sick_man.018 # A Consuming Sickness
		}
	}
}

je_turkfm_sick_man_grand_collapse = {
	# This is a dupe of the vanilla journal... for now
	group = je_group_gbbf_sick_man
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	weight = 10000
	
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = no
	transferable = no
	
	is_shown_when_inactive = {
		NOT = {
			has_global_variable = turkfm_tanzimat_complete
		}
		c:TUR ?= this
	}
	
	scripted_button = je_the_grand_collapse_bribe_politicians_button
	scripted_button = je_the_grand_collapse_issue_khutbah_button
	scripted_button = je_the_grand_collapse_law_speed_start_button
	scripted_button = je_the_grand_collapse_law_speed_stop_button
	scripted_button = je_the_grand_collapse_relation_speed_start_button
	scripted_button = je_the_grand_collapse_relation_Speed_stop_button
	
	possible = {
		custom_tooltip = {
			text = turkfm_tanzimat_failed
			has_global_variable = turkfm_tanzimat_failed
		}
		OR = {
			NOT = {
				has_law_or_variant = law_type:law_monarchy
			}
			any_civil_war = {
				count >= 1
			}
		}
	}
	
	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		trigger_event = the_grand_collapse.1
		hidden_effect = {
			add_journal_entry = {
				type = je_the_grand_collapse
			}
		}
	}
	
	complete = {
		any_civil_war = {
			count = 0
		}
		trigger_if = {
			limit = {
				is_player = yes
			}
			has_law_or_variant = law_type:law_monarchy
			radical_fraction = {
				value < 0.1
			}
			ig:ig_intelligentsia = {
				is_in_government = no
			}
		}
		government_legitimacy > 80
		trigger_if = {
			limit = {
				any_subject_or_below = {
					count > 0
				}
			}
			any_subject_or_below = {
				relations:root >= relations_threshold:cordial
			}
		}
		any_interest_group = {
			ig_approval >= happy
			count >= 3
		}
	}
	
	on_complete = {
		trigger_event = the_grand_collapse.4
		hidden_effect = {
			the_grand_collapse_mod_cleanser = yes
		}
		set_variable = turkfm_grand_collapse_completed
	}
	
	timeout = 3650
	
	on_timeout = {
		trigger_event = the_grand_collapse.5
		hidden_effect = {
			the_grand_collapse_mod_cleanser = yes
		}
		set_variable = turkfm_grand_collapse_failed
	}
	
	invalid = {
		NOT = {
			any_scope_state = {
				is_in_geographic_region = geographic_region_rumelia
			}
		}
	}
	
	on_invalid = {
		hidden_effect = {
			the_grand_collapse_mod_cleanser = yes
		}
	}
	
	on_monthly_pulse = {
		random_events = {
			200 = 0
			20 = the_grand_collapse.2 #Any Port in a Storm
		}
	}
}
