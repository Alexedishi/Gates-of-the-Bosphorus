je_balkfm_ser_monarchism = {
	icon = "gfx/interface/icons/event_icons/event_serbia.dds"
	group = je_group_historical_content
	weight = 1000

	should_be_pinned_by_default = yes
	can_revolution_inherit = no
	transferable = no
	
	scripted_button = balkfm_ser_slander_button
	scripted_button = balkfm_ser_campaign_button
	scripted_button = balkfm_ser_flip_consul_button
	scripted_button = balkfm_cement_monarchy_button

	scripted_progress_bar = balkfm_dynastic_rivalry_progressbar

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
	}

	modifiers_while_active = {
		modifier_balkfm_dynastic_feud
	}

	complete = {
		has_law_or_variant = law_type:law_monarchy
		custom_tooltip = {
			text = je_balkfm_ser_monarchism_complete
			OR = {
				has_modifier = modifier_balkfm_obrenovic_succession
				has_modifier = modifier_balkfm_karadordevic_succession
			}
		}
	}

	on_complete = {
		if = {
			limit = {
				has_dlc_feature = ip3_content # National Awakening
			}
			custom_tooltip = {
				text = je_balkfm_ser_monarchism_obrenovic_outcome
				if = {
					limit = {
						balkfm_has_obrenovic_ruler_trigger = yes
					}
				}
				trigger_event = {
					id = balkfm_serbia.029 # The Reckoning
				}
			}
			custom_tooltip = {
				text = je_balkfm_ser_monarchism_karadordevic_outcome
				if = {
					limit = {
						balkfm_has_karadordevic_ruler_trigger = yes
					}
				}
				trigger_event = {
					id = balkfm_serbia.028 # Here to Stay
				}
			}
		}
		else = {
			trigger_event = {
				id = balkfm_serbia.019 # Mandate Secured
			}
		}
		hidden_effect = {
			balkfm_cleanup_dynastic_rivalry_effect = yes
		}
	}

	fail = {
		NOT = {
			has_law_or_variant = law_type:law_monarchy
		}
	}

	on_fail = {
		hidden_effect = {
			balkfm_cleanup_dynastic_rivalry_effect = yes
		}
	}

	on_weekly_pulse = {
		effect = {
			balkfm_calculate_dynastic_rivalry_variables_effect = yes
			balkfm_update_dynasty_character_scopes_effect = yes
		}
	}

	on_monthly_pulse = {
		events = {
			balkfm_serbia.002 # Constitution Imposed
			balkfm_serbia.007 # Defenders of the Constitution
			balkfm_serbia.010 # House Karađorđević Takes the Throne
			balkfm_serbia.011 # House Obrenović Takes the Throne
			balkfm_serbia.012 # Austria/Russia Requests Primacy
		}
		random_events = {
			400 = 0
			5 =	balkfm_serbia.013 # Plague Outbreak!
			5 = balkfm_serbia.014 # Abusive Garrison
			5 = balkfm_serbia.020 # Clerical Publicity
			5 = balkfm_serbia.022 # Exposing Hypocrites
			5 = balkfm_serbia.023 # Renegade Hajduks
			5 = balkfm_serbia.024 # Urban Rebels
			5 = balkfm_serbia.025 # Plea for Aid
		}
		effect = {
			balkfm_dynastic_rivalry_monthly_effect = yes
		}
	}

	status_desc = {
		desc = je_balkfm_ser_monarchism_status
	}
}

je_balkfm_ser_defenders = {
	group = je_group_internal_affairs
	icon = "gfx/interface/icons/event_icons/event_serbia.dds"
	weight = 500

	should_be_pinned_by_default = yes
	can_revolution_inherit = no
	transferable = no

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
	}

	modifiers_while_active = {
		modifier_balkfm_defenders
	}

	complete = {
		NOT = { has_modifier = modifier_balkfm_royal_turmoil }
		custom_tooltip = {
			text = je_balkfm_ser_defenders_tt1
			hidden_trigger = {
				has_variable = balkfm_state_reform_complete
			}
		}
		custom_tooltip = {
			text = je_balkfm_ser_defenders_tt2
			hidden_trigger = {
				has_variable = balkfm_civic_reform_complete
			}
		}
		custom_tooltip = {
			text = je_balkfm_ser_defenders_tt3
			hidden_trigger = {
				has_variable = balkfm_land_reform_complete
			}
		}
	}

	on_complete = {
		trigger_event = {
			id = balkfm_serbia.009 # Serbia Reforged
		}
		set_variable = balkfm_defenders_completed
	}

	timeout = 3650 # mod's been out for over a year and this my first timeout journal since the Bavarocracy journals... anyway, this number is based on the "Defenders" being dominant in Serbian politics from roughly 1838-1858

	on_timeout = {
		trigger_event = {
			id = balkfm_serbia.008 # Absent Reform
		}
		set_variable = balkfm_defenders_completed
	}

	invalid = {
		OR = {
			NOT = {
				has_law_or_variant = law_type:law_monarchy
			}
			has_variable = balkfm_defenders_completed
		}
	}

	on_invalid = {
		set_variable = balkfm_defenders_completed
	}
}
