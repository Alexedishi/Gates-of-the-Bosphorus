je_turkfm_tun_modernization = {
	icon = "gfx/interface/icons/event_icons/event_tunis.dds"
	group = je_group_turkfm_tunis
	should_be_pinned_by_default = yes

	is_shown_in_lobby = {
		c:TUN ?= this
	}

	modifiers_while_active = {
		modifier_turkfm_tribal_administration
	}

	scripted_button = turkfm_visit_france_button

	complete = {
		OR = {
			is_country_type = recognized
			AND = {
				NOR = {
					has_law_or_variant = law_type:law_hereditary_bureaucrats
					has_law_or_variant = law_type:law_consumption_based_taxation
					has_law_or_variant = law_type:law_land_based_taxation
				}
				any_scope_building = {
					is_building_type = building_university
					NOR = {
						has_active_production_method = pm_simple_organization
						has_active_production_method = pm_religious_bureaucrats
					}
					occupancy > 0.9
					level >= 5
				}
				any_scope_building = {
					OR = {
						is_building_type = gbbf_building_mohamedia_palace
						is_building_type = building_government_administration
					}
					NOR = {
						has_active_production_method = pm_simple_organization
						has_active_production_method = pm_religious_bureaucrats
					}
					occupancy >= 0.9
				}
				bureaucracy >= 0
				approaching_bureaucracy_shortage = no
			}
		}
	}

	on_complete = {
		trigger_event = {
			id = turkfm_tunis.009 # The Tunisian Fundamental Pact
		}
		if = {
			limit = {
				has_journal_entry = je_earn_recognition
			}
			custom_tooltip = turkfm_recognition_gain_50
		}
		set_variable = turkfm_tun_modernization_completed
	}

	timeout = 10950 # 30 years

	on_timeout = {
		trigger_event = {
			id = turkfm_tunis.008 # A Stagnant State
		}
		set_variable = turkfm_tun_modernization_completed
	}

	invalid = {
		has_variable = turkfm_tun_modernization_completed
	}

	on_monthly_pulse = {
		random_events = {
			chance_of_no_event = 90
			5 = turkfm_tunis.002 # An Amibitous Bey
		}
	}

	weight = 100

}
