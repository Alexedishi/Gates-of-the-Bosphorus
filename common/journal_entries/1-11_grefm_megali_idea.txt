je_grefm_megali_idea = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_hellenic.dds"
	weight = 10000

	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no

	scripted_button = grefm_annex_heptanese_button
	scripted_button = grefm_annex_cyprus_button
	scripted_button = grefm_annex_crete_button
	scripted_button = grefm_annex_samos_button
	scripted_button = grefm_annex_trabzon_button
	scripted_button = grefm_annex_cappadocia_button
	scripted_button = grefm_annex_theodoro_button

	scripted_button = grefm_ionian_enosis_button
	scripted_button = grefm_ionian_enosis_alternate_button
	scripted_button = grefm_samos_enosis_button

	is_shown_when_inactive = {
		NOT = {
			has_global_variable = grefm_megali_idea_complete
		}
		OR = {
			country_has_primary_culture = cu:greek
			country_has_primary_culture = cu:rhomaios
			AND = {
				anzfp_is_active = yes
				c:AST ?= this
			}
			has_variable = grefm_megali_idea_aspirant
		}
	}

	possible = {
		has_technology_researched = nationalism
		trigger_if = {
			limit = {
				gbbf_country_has_greek_culture_trigger = yes
			}
			custom_tooltip = {
				text = je_grefm_megali_idea_possible
				any_scope_state = {
					is_greek_homeland = yes
					count >= greek_homeland_states_owned_to_activate_min
					owner = root
				}
			}
		}
		trigger_if = {
			limit = {
				gbbf_country_has_greek_culture_trigger = no
			}
			any_scope_state = {
				region = sr:region_balkans
				is_incorporated = yes
			}
		}
		is_subject = no
		is_diplomatic_play_committed_participant = no
	}

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		set_variable = grefm_mainland_claims_received
		set_variable = grefm_rhomaios_claims_received # blocks grefm.280
		if = {
			limit = {
				country_has_primary_culture = cu:rhomaios
			}
			trigger_event = {
				id = grefm_megali_idea.011 # The Legacy of Rhōmanía
			}
		}
		else_if = {
			limit = {
				c:AST ?= this
			}
			trigger_event = {
				id = grefm_emu_empire.001 # 'Straya has Landed
			}
			set_variable = grefm_megali_idea_aspirant
		}
		else_if = {
			limit = {
				c:BUL ?= this
			}
			trigger_event = {
				id = grefm_megali_idea.002 # The Megali Idea
			}
		}
		else_if = {
			limit = {
				c:ROM ?= this
			}
			trigger_event = {
				id = balkfm_romania.720 # The Imperial Standard
			}
		}
		else = {
			trigger_event = {
				id = grefm_megali_idea.002 # The Megali Idea
			}
		}
		set_variable = {
			name = grefm_megali_idea_counter_var
			value = grefm_megali_idea_counter_value
		}
		add_journal_entry = {
			type = je_greek_nationalism
		}
	}

	complete = {
		is_at_war = no
		is_subject = no
		gold_reserves > 0
		owns_entire_state_region = STATE_EASTERN_THRACE
		trigger_if = {
			# This is the standard non-BYZ form of the Megali Idea
			limit = {
				NOR = {
					has_variable = grefm_weaksauce
					has_variable = grefm_embraced_imperial_restoration
				}
			}
			custom_tooltip = {
				text = greek_state_complete_tt
				any_scope_state = {
					is_greek_homeland = yes
					count >= greek_homeland_states_owned_to_complete_min
					owner = root
				}
			}
			country_rank >= rank_value:minor_power
		}
		trigger_if = {
			# This is the limited form of the Megali Idea
			limit = {
				has_variable = grefm_weaksauce
			}
			custom_tooltip = {
				text = greek_state_complete_tt_2
				any_scope_state = {
					is_greek_homeland = yes
					count >= greek_homeland_states_owned_to_complete_max
					owner = root
				}
			}
			country_rank >= rank_value:minor_power
		}
		trigger_if = {
			# This is the full BYZ form of the Megali Idea
			limit = {
				has_variable = grefm_embraced_imperial_restoration
			}
			custom_tooltip = {
				text = grefm_greek_empire_formed
				c:BYZ ?= this
			}
			custom_tooltip = {
				text = greek_state_complete_tt_3
				any_scope_state = {
					is_byzantium_state_region = yes
					count >= byzantium_state_region_owned_to_complete
					owner = root
				}
			}
			country_rank >= rank_value:major_power
		}
		custom_tooltip = {
			text = no_insurrectionary_movement_tt
			NOT = {
				any_political_movement = {
					is_insurrectionary = yes
				}
			}
		}
		hidden_trigger = {
			OR = {
				has_variable = grefm_weaksauce
				has_variable = grefm_embraced_imperial_restoration
				has_variable = grefm_imperial_restoration_declined
			}
			has_variable = grefm_megali_idea_expansion_considered
		}
	}

	on_complete = {
		set_global_variable = grefm_megali_idea_complete
		if = {
			limit = {
				country_has_primary_culture = cu:rhomaios
			}
			trigger_event = {
				id = grefm_megali_idea.012 # The Throne of the East
			}
		}
		else_if = {
			limit = {
				c:AST ?= this
			}
			trigger_event = {
				id = grefm_emu_empire.001 # The Empire from Down Under
			}
			show_as_tooltip = {
				change_tag = BYZ
			}
		}
		else_if = {
			limit = {
				c:BUL ?= this
			}
			trigger_event = {
				id = balkfm_bulgaria.218 # The Queen of Cities
			}
			show_as_tooltip = {
				change_tag = BYZ
			}
		}
		else_if = {
			limit = {
				c:ROM ?= this
			}
			trigger_event = {
				id = balkfm_romania.721 # The Queen of Cities
			}
			show_as_tooltip = {
				change_tag = BYZ
			}
		}
		else = {
			trigger_event = {
				id = grefm_megali_idea.003 # The Future of Greece
			}
		}
		grefm_empire_name_tooltip_effect = yes
	}

	fail = {
		OR = {
			trigger_if = {
				limit = {
					NOT = {
						c:BYZ ?= this
					}
				}
				grefm_greek_empire_exists_trigger = yes
			}
			is_subject = yes
			AND = {
				country_rank < rank_value:minor_power
				gold_reserves < 0
			}
		}
	}

	on_fail = {
		trigger_event = {
			id = grefm_megali_idea.005 # The Greek Lands Lost
		}
	}

	invalid = {
		custom_tooltip = {
			text = grefm_megali_idea_rejected_trigger
			has_variable = grefm_megali_idea_rejected
		}
	}
	
	on_weekly_pulse = {
		effect = {
			# Dude, WHY THE FUCK is this cluttering the monthly on_action in base game when you can just wedge it here???
			# COMPLETE WITH THROWING THE VARIABLE CHECKS INTO THE SCRIPT VALUE ITSELF?!?!?!
			set_variable = {
				name = grefm_megali_idea_counter_var
				value = grefm_megali_idea_counter_value
			}
		}
	}

	on_monthly_pulse = {
		events = {
			grefm_megali_idea.004 # The Empire of the Greeks
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_variable = grefm_embraced_imperial_restoration
				}
				desc = je_grefm_megali_idea_status_embraced_imperial_restoration
			}
			triggered_desc = {
				trigger = {
					has_variable = grefm_embraced_megali_idea
				}
				desc = je_grefm_megali_idea_status_embraced_megali_idea
			}
			triggered_desc = {
				desc = je_grefm_megali_idea_status_default
			}
		}
	}

}

je_grefm_byzantine_restoration = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_aquila.dds"
	weight = 10000

	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no

	scripted_button = grefm_balkans_triumph_button
	scripted_button = grefm_anatolia_triumph_button
	scripted_button = grefm_levant_triumph_button
	scripted_button = grefm_egypt_triumph_button
	scripted_button = grefm_italy_triumph_button

	is_shown_when_inactive = {
		OR = {
			gbbf_country_has_greek_culture_trigger = yes
			has_variable = grefm_megali_idea_aspirant
		}
		NOT = {
			has_variable = grefm_weaksauce
		}
	}

	possible = {
		custom_tooltip = {
			text = grefm_greek_empire_formed
			c:BYZ ?= this
		}
		trigger_if = {
			limit = {
				has_journal_entry = je_grefm_greek_nationalism
			}
			custom_tooltip = {
				text = grefm_greek_identity_resolved
				has_variable = grefm_nationalism_complete
			}
		}
		is_subject = no
		is_diplomatic_play_committed_participant = no
	}

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		if = {
			limit = {
				country_has_primary_culture = cu:australian
			}
			scope:journal_entry = {
				add_modifier = modifier_grefm_wanakterian_aspiriations
			}
			trigger_event = {
				id = grefm_emu_empire.002 # The Empire from Down Under
			}
		}
		else_if = {
			limit = {
				country_has_primary_culture = cu:greek
				NOT = {
					country_has_primary_culture = cu:rhomaios
				}
			}
			scope:journal_entry = {
				add_modifier = modifier_grefm_hellenic_aspiriations
			}
			trigger_event = {
				id = grefm_hellenic_empire.001 # The Revenge of the Greeks
			}
		}
		else_if = {
			limit = {
				country_has_primary_culture = cu:rhomaios
				NOT = {
					country_has_primary_culture = cu:greek
				}
			}
			scope:journal_entry = {
				add_modifier = modifier_grefm_imperial_aspiriations
			}
			trigger_event = {
				id = grefm_roman_empire.001 # The Purple Pheonix
			}
		}
		else_if = {
			limit = {
				country_has_primary_culture = cu:rhomaios
				country_has_primary_culture = cu:greek
			}
			scope:journal_entry = {
				add_modifier = modifier_grefm_imperial_aspiriations
			}
			trigger_event = {
				id = grefm_hybrid_empire.001 # The New Greece Rises
			}
		}
	}

	modifiers_while_active = {
		modifier_grefm_imperial_aspiriations
	}

	complete = {
		owns_entire_state_region = STATE_EASTERN_THRACE
		trigger_if = {
			limit = {
				has_variable = grefm_embraced_roman_reconquest
			}
			owns_entire_state_region = STATE_LAZIO
			country_rank >= rank_value:great_power
			has_modifier = modifier_grefm_reclaimed_balkans
			has_modifier = modifier_grefm_reclaimed_anatolia
			has_modifier = modifier_grefm_reclaimed_levant
			has_modifier = modifier_grefm_reclaimed_egypt
			has_modifier = modifier_grefm_reclaimed_italy
		}
		trigger_else = {
			country_rank >= rank_value:major_power
			has_modifier = modifier_grefm_reclaimed_balkans
			has_modifier = modifier_grefm_reclaimed_anatolia
			has_modifier = modifier_grefm_reclaimed_levant
			has_modifier = modifier_grefm_reclaimed_egypt
		}
		is_at_war = no
		is_subject = no
		gold_reserves > 0
	}

	on_complete = {
		if = {
			limit = {
				country_has_primary_culture = cu:australian
			}
			trigger_event = {
				id = grefm_emu_empire.009
			}
		}
		else_if = {
			limit = {
				country_has_primary_culture = cu:greek
				NOT = {
					country_has_primary_culture = cu:rhomaios
				}
			}
			trigger_event = {
				id = grefm_hellenic_empire.009
			}
		}
		else_if = {
			limit = {
				country_has_primary_culture = cu:rhomaios
				NOT = {
					country_has_primary_culture = cu:greek
				}
			}
			trigger_event = {
				id = grefm_roman_empire.009
			}
		}
		else_if = {
			limit = {
				country_has_primary_culture = cu:rhomaios
				country_has_primary_culture = cu:greek
			}
			trigger_event = {
				id = grefm_hybrid_empire.009 # The Greek Realm
			}
		}
		set_global_variable = grefm_byzantium_restored
	}

	fail = {
		OR = {
			is_subject = yes
			AND = {
				country_rank < rank_value:minor_power
				gold_reserves < 0
			}
		}
	}

	on_fail = {
	}

	invalid = {
		custom_tooltip = {
			text = je_grefm_byzantine_restoration_rejected_trigger
			has_variable = grefm_byzantine_restoration_rejected
		}
	}

	on_monthly_pulse = {
		events = {
			grefm_hellenic_empire.002
			grefm_roman_empire.002
			grefm_hybrid_empire.002
			grefm_emu_empire.003
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_variable = grefm_embraced_roman_reconquest
				}
				desc = je_grefm_byzantine_restoration_status_expanded
			}
			triggered_desc = {
				desc = je_grefm_byzantine_restoration_status_base
			}
		}
	}

}
