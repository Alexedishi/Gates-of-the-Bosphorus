je_grefm_greek_nationalism = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/invention_icons/nationalism.dds"
	weight = 3000

	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no

	scripted_progress_bar = grefm_hellenism_progress_bar
	scripted_progress_bar = grefm_romanism_progress_bar

	scripted_button = grefm_embrace_hellenic_identity_button
	scripted_button = grefm_embrace_roman_identity_button
	scripted_button = grefm_embrace_hybrid_identity_button

	is_shown_in_lobby = {
		OR = {
			country_has_primary_culture = cu:greek
			country_has_primary_culture = cu:rhomaios
		}
	}
	
	is_shown_when_inactive = {
		OR = {
			country_has_primary_culture = cu:greek
			country_has_primary_culture = cu:rhomaios
		}
	}

	possible = {
		has_technology_researched = mass_communication
	}

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		set_variable = {
			name = grefm_hellene_culture_var
			value = grefm_hellenism_value
		}
		set_variable = {
			name = grefm_rhomaios_culture_var
			value = grefm_romanism_value
		}
		trigger_event = {
			id = grefm_greek_nationalism.001 # What is a Greek
		}
	}

	complete = {
		OR = {
			custom_tooltip = {
				text = je_grefm_greek_nationalism_complete
				hidden_trigger = {
					has_variable = grefm_nationalism_complete
				}
			}
			has_law_or_variant = law_type:law_ethnostate
			has_law_or_variant = law_type:law_multicultural
		}
	}

	on_complete = {
		custom_tooltip = {
			text = je_grefm_greek_nationalism_ethostate
			if = {
				limit = {
					has_law_or_variant = law_type:law_ethnostate
				}
				if = {
					limit = {
						country_has_primary_culture = cu:greek
						NOT = {
							country_has_primary_culture = cu:rhomaios
						}
					}
					set_variable = {
						name = grefm_culture_var
						value = 1
					}
					trigger_event = {
						id = grefm_greek_nationalism.009 # The Greek Ethnogenesis
					}
				}
				else_if = {
					limit = {
						country_has_primary_culture = cu:rhomaios
						NOT = {
							country_has_primary_culture = cu:greek
						}
					}
					set_variable = {
						name = grefm_culture_var
						value = 2
					}
					trigger_event = {
						id = grefm_greek_nationalism.009 # The Greek Ethnogenesis
					}
				}
				else_if = {
					limit = {
						country_has_primary_culture = cu:greek
						country_has_primary_culture = cu:rhomaios
					}
					set_variable = {
						name = grefm_culture_var
						value = 3
					}
					trigger_event = {
						id = grefm_greek_nationalism.009 # The Greek Ethnogenesis
					}
				}
			}
		}
		custom_tooltip = {
			text = je_grefm_greek_nationalism_multicultural
			if = {
				limit = {
					has_law_or_variant = law_type:law_multicultural
				}
				set_variable = {
					name = grefm_culture_var
					value = 3
				}
				trigger_event = {
					id = grefm_greek_nationalism.009 # The Greek Ethnogenesis
				}
			}
		}
	}

	on_weekly_pulse = {
		effect = {
			set_variable = {
				name = grefm_hellene_culture_var
				value = grefm_hellenism_value
			}
			set_variable = {
				name = grefm_rhomaios_culture_var
				value = grefm_romanism_value
			}
			je:je_grefm_greek_nationalism = {
				set_bar_progress = {
					name = grefm_hellenism_progress_bar
					value = owner.var:grefm_hellene_culture_var
				}
			}
			je:je_grefm_greek_nationalism = {
				set_bar_progress = {
					name = grefm_romanism_progress_bar
					value = owner.var:grefm_rhomaios_culture_var
				}
			}
		}
	}
}

		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					has_modifier = modifier_grefm_active_excavation
				}

					name = grefm_excavation_var

				}

				}

			10 = grefm_greek_nationalism.050 # Excavation Proceeding Smoothly
			5 = grefm_greek_nationalism.051 # Industrial Equipment Requested
			1 = grefm_greek_nationalism.052 # Ancient Tomb Uncovered

			1 = grefm_greek_nationalism.056 # Palace Uncovered
			5 = grefm_greek_nationalism.058 # IG Offers Funding
			5 = grefm_greek_nationalism.059 # Reconstruction Garners Criticism
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_modifier = modifier_grefm_active_excavation
				}
				desc = je_grefm_archaeology_status
			}
		}
	}
