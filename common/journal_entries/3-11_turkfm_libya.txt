je_turkfm_tri_centralization = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_fez.dds"
	weight = 2000

	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no

	is_shown_in_lobby = {
		c:TRI ?= this
	}

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		scope:journal_entry = {
			set_variable = {
				name = turkfm_institution_investment_var
				value = owner.turkfm_total_institution_investment_value
			}
		}
	}

	modifiers_while_active = {
		modifier_turkfm_informal_administration
	}

	complete = {
		NOR = {
			has_law_or_variant = law_type:law_hereditary_bureaucrats
			has_law_or_variant = law_type:law_peasant_levies
			has_law_or_variant = law_type:law_no_police
		}
		government_legitimacy >= 75
		custom_tooltip = {
			text = je_turkfm_tri_centralization_complete
			scope:journal_entry.var:turkfm_institution_investment_var >= 10
		}
		bureaucracy >= 0
		approaching_bureaucracy_shortage = no
		ig:ig_intelligentsia = {
			is_marginal = no
		}
		NOT = {
			any_interest_group = {
				OR = {
					is_insurrectionary = yes
					is_revolutionary = yes
				}
			}
		}
	}

	on_complete = {
		trigger_event = {
			id = turkfm_libya.009 # A Centralized State
		}
	}
	
	on_weekly_pulse = {
		effect = {
			scope:journal_entry = {
				set_variable = {
					name = turkfm_institution_investment_var
					value = owner.turkfm_total_institution_investment_value
				}
			}
		}
	}
}

je_turkfm_senusiyya_main = {
	group = je_group_internal_affairs
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	weight = 1000
	
	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no

	is_shown_in_lobby = {
		c:TRI ?= this
	}
	
	scripted_progress_bar = turkfm_senussi_progressbar
	
	scripted_button = turkfm_embrace_senussi_button
	scripted_button = turkfm_sweep_desert_button
	scripted_button = turkfm_supress_senusiyya_button
	scripted_button = turkfm_armed_delegation_button

	is_shown_when_inactive = {
		c:TRI ?= this
		NOR = {
			has_variable = turkfm_embraced_senussi
			has_variable = turkfm_swept_senussi
			owns_entire_state_region = STATE_LIBYAN_DESERT
		}
	}

	possible = {
		has_technology_researched = mass_communication
	}

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		trigger_event = {
			id = turkfm_libya.100 # The Senussi Order
		}
		turkfm_update_senussi_effect = yes
		set_variable = turkfm_senussi_has_landed
	}

	complete = {
		custom_tooltip = {
			text = je_turkfm_senusiyya_complete
			OR = {
				has_variable = turkfm_embraced_senussi
				has_variable = turkfm_swept_senussi
			}
		}
	}

	on_complete = {
		custom_tooltip = {
			text = je_turkfm_senusiyya_on_complete_1
			if = {
				limit = {
					has_variable = turkfm_embraced_senussi
				}
				trigger_event = {
					id = turkfm_libya.109 # House Senussi of Country
				}
			}
		}
		custom_tooltip = {
			text = je_turkfm_senusiyya_on_complete_2
			if = {
				limit = {
					has_variable = turkfm_swept_senussi
				}
				trigger_event = {
					id = turkfm_libya.108 # The Senusiyya Extinguished
				}
			}
		}
		turkfm_cleanup_senussi_effect = yes
	}

	fail = {
		OR = {
			has_law_or_variant = law_type:law_total_separation
			has_law_or_variant = law_type:law_state_atheism
		}
	}

	on_fail = {
		turkfm_cleanup_senussi_effect = yes
	}

	on_weekly_pulse = {
		effect = {
			turkfm_senussi_weekly_effect = yes
			turkfm_update_senussi_effect = yes
		}
	}

	on_monthly_pulse = {
		events = {
			turkfm_libya.101 # The Grand Senussi
			turkfm_libya.104 # The State-within-a-State
			turkfm_libya.110 # Senusiyya Spreads to Country
			turkfm_libya.112 # Senusiyya Takes Root in State
			turkfm_libya.113 # Senusiyya Expunged from State
		}
		random_events = {
			chance_of_no_event = 90
			5 = turkfm_libya.102 # Senussi Withdraws to the Desert
		}
		effect = {
			change_variable = {
				name = turkfm_senussi_var
				add = var:turkfm_senussi_progress_var
			}
			clamp_variable = {
				name = turkfm_senussi_var
				max = 100
				min = 0
			}
		}
	}

	status_desc = {
		triggered_desc = {
			desc = turkfm_senussi_location_status
			trigger = {
				has_journal_entry = je_turkfm_senusiyya_main
			}
		}
	}
}
