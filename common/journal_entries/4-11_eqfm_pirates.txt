je_eqfm_pirates_of_the_adriatic = {
	group = je_group_global_international_situations
	icon = "gfx/interface/icons/event_icons/event_piracy.dds"
	weight = 1000

	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no

	is_shown_in_lobby = {
		any_scope_state = {
			is_in_geographic_region = geographic_region_adriatic
		}
	}

	scripted_progress_bar = eqfm_jack_sparrow_progress_bar

	scripted_button = eqfm_deploy_adriatic_expedition_button
	scripted_button = eqfm_recall_adriatic_expedition_button
	scripted_button = eqfm_hire_retired_pirates_button
	scripted_button = eqfm_resupply_adriatic_expedition_button
	
	is_shown_when_inactive = {
		custom_tooltip = {
			text = je_eqfm_pirates_of_the_adriatic_possible
			NOT = {
				has_global_variable = eqfm_jack_sparrow_whacked
			}
		}
	}
	
	possible = {
		year >= 1837
	}

	should_be_involved = {
		any_scope_state = {
			is_in_geographic_region = geographic_region_adriatic
		}
	}
	
	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		if = {
			limit = {
				NOT = {
					has_global_variable = eqfm_pirates_var
				}
			}
			set_global_variable = {
				name = eqfm_pirates_var
				value = 0
			}
		}
	}

	immediate_all_involved = {
		if = {
			limit = {
				NOT = {
					has_variable = eqfm_piracy_started
				}
			}
			trigger_event = {
				id = eqfm_pirates.001 # The Pirates of the Adriatic
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = je_eqfm_pirates_of_the_adriatic_complete
			has_global_variable = eqfm_jack_sparrow_whacked
		}
	}

	on_complete = {
		eqfm_cleanup_piracy_effect = yes
		eqfm_update_expedition_costs_effect = yes
	}

	fail = {
		NOT = {
			any_scope_state = {
				gbbf_adriatic_geography_trigger = yes
				is_coastal = yes
			}
		}
	}

	on_fail = {
		eqfm_cleanup_piracy_effect = yes
		eqfm_update_expedition_costs_effect = yes
	}

	on_monthly_pulse = {
		effect = {
			every_country = {
				limit = {
					OR = {
						is_involved_in_journal_entry = je_eqfm_pirates_of_the_adriatic
						any_scope_state = {
							is_in_geographic_region = geographic_region_adriatic
						}
					}
				}
				if = {
					limit = {
						is_involved_in_journal_entry = je_eqfm_pirates_of_the_adriatic
						NOT = {
							any_scope_state = {
								is_in_geographic_region = geographic_region_adriatic
							}
						}
					}
				}
				if = {
					limit = {
						NOT = {
							is_involved_in_journal_entry = je_eqfm_pirates_of_the_adriatic
						}
						any_scope_state = {
							is_in_geographic_region = geographic_region_adriatic
						}
					}
				}
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = eqfm_cutler_beckett_expedition_chase
				trigger = {
					has_variable = eqfm_piracy_chase_var
				}
			}
			triggered_desc = {
				desc = eqfm_cutler_beckett_expedition_default
				trigger = {
					has_variable = eqfm_cutler_beckett_var
					exists = var:eqfm_cutler_beckett_var
				}
			}
			triggered_desc = {
				desc = eqfm_jack_sparrow_scared
				trigger = {
					has_global_variable = eqfm_pirates_var
					global_var:eqfm_pirates_var >= 3
					exists = var:eqfm_jack_sparrow_var
				}
			}
			triggered_desc = {
				desc = eqfm_jack_sparrow_cautious
				trigger = {
					has_global_variable = eqfm_pirates_var
					global_var:eqfm_pirates_var >= 2
					exists = var:eqfm_jack_sparrow_var
				}
			}
			triggered_desc = {
				desc = eqfm_jack_sparrow_cocky
				trigger = {
					has_global_variable = eqfm_pirates_var
					global_var:eqfm_pirates_var >= 0
					exists = var:eqfm_jack_sparrow_var
				}
			}
		}
	}
}
