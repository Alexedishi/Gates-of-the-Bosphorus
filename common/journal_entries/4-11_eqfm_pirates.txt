je_eqfm_pirates_of_the_adriatic = {
	group = je_group_expeditions
	icon = "gfx/interface/icons/event_icons/event_piracy.dds"
	weight = 1000

	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no

	scripted_progress_bar = eqfm_jack_sparrow_progress_bar

	scripted_button = eqfm_deploy_adriatic_expedition_button
	scripted_button = eqfm_recall_adriatic_expedition_button
	scripted_button = eqfm_hire_retired_pirates_button

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		if = {
			limit = {
				NOT = {
					has_global_variable = eqfm_pirates_var
				}
			}
			set_global_variable = {
				name = eqfm_pirates_var
				value = 0
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = je_eqfm_pirates_of_the_adriatic_complete_tt
			has_global_variable = eqfm_jack_sparrow_whacked
		}
	}

	on_complete = {
		eqfm_cleanup_piracy_effect = yes
		eqfm_update_expedition_costs_effect = yes
	}

	fail = {
		NOT = {
			any_scope_state = {
				gbbf_adriatic_geography_trigger = yes
				is_coastal = yes
			}
		}
	}

	on_fail = {
		eqfm_cleanup_piracy_effect = yes
		eqfm_update_expedition_costs_effect = yes
	}

	on_weekly_pulse = {
		effect = {
			eqfm_jack_sparrow_effect = yes
		}
		events = {
			eqfm_pirates.004 # Cutler Beckett Withdraws from the Adriatic
			eqfm_pirates.005 # Cutler Beckett Has Died!
			eqfm_pirates.006 # Pirate Hideout Located
			eqfm_pirates.007 # Pirate Fleet Destroyed
		}
		random_events = { # Chase Events
			15 = eqfm_pirates.031 # Man Overboard!
			35 = eqfm_pirates.032 # The Pillagers' Wake
			# 5 = eqfm_pirates.033 #
			5 = eqfm_pirates.034 # Without a Trace
			45 = eqfm_pirates.035 # Bearing Down
			10 = eqfm_pirates.036 # Cut and Run
			30 = eqfm_pirates.037 # Action Along the Coast
			10 = eqfm_pirates.038 # Naval Duel on the High Seas!
		}
	}

	on_monthly_pulse = {
		events = {}
		random_events = { # Expedition Events
			# Weight for Roughly 95% against General Events and 50% against Expedition Events
			900 = 0

			# General Piracy Events - Balance to a total weight of 100 Max
			20 = eqfm_pirates.010 # Coastal Village Raided!
			20 = eqfm_pirates.011 # Merchant Vessels Raided!
			5 = eqfm_pirates.012 # Nobleman Held Hostage
			1 = eqfm_pirates.013 # A Pirate's Booty
			5 = eqfm_pirates.014 # Diplomatic Communiques Intercepted
			10 = eqfm_pirates.015 # Slavers Intercepted
			5 = eqfm_pirates.016 # Greener Pastures
			10 = eqfm_pirates.017 # The Goods Smugglers
			20 = eqfm_pirates.018 # The Peasant's Pirate

			# Adriatic Expedition Events - Balance to a total weight of 900 Min
			100 = eqfm_pirates.020 # Local Informants
			100 = eqfm_pirates.021 # A Suspicious Inlet
			10 = eqfm_pirates.022 # The Runaway
			50 = eqfm_pirates.023 # Spoiled Provisions
			100 = eqfm_pirates.024 # Circular Bearing
			50 = eqfm_pirates.025 # Windstorms in the Adriatic
			50 = eqfm_pirates.026 # Foul Waters
			100 = eqfm_pirates.027 # Scouts Pick Up Leads
			50 = eqfm_pirates.028 # Emergency Repairs
			100 = eqfm_pirates.029 # Port of Call
			200 = eqfm_pirates.030 # Pirates Spotted!
		}
	}

	on_yearly_pulse = {
		effect = {
			if = {
				limit = {
					has_variable = eqfm_adriatic_expedition_var
				}
				change_variable = {
					name = eqfm_adriatic_expedition_progress_var
					add = 1
				}
				post_notification = notification_eqfm_piracy_yearly_progress
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = eqfm_cutler_beckett_expedition_1
				trigger = {
					has_variable = eqfm_cutler_beckett_var
					exists = var:eqfm_cutler_beckett_var
				}
			}
			triggered_desc = {
				desc = eqfm_jack_sparrow_scared
				trigger = {
					has_global_variable = eqfm_pirates_var
					global_var:eqfm_pirates_var >= 3
					exists = var:eqfm_jack_sparrow_var
				}
			}
			triggered_desc = {
				desc = eqfm_jack_sparrow_cautious
				trigger = {
					has_global_variable = eqfm_pirates_var
					global_var:eqfm_pirates_var >= 2
					exists = var:eqfm_jack_sparrow_var
				}
			}
			triggered_desc = {
				desc = eqfm_jack_sparrow_cocky
				trigger = {
					has_global_variable = eqfm_pirates_var
					global_var:eqfm_pirates_var >= 0
					exists = var:eqfm_jack_sparrow_var
				}
			}
		}
	}
}
