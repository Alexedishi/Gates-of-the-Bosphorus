je_eqfm_pirates_of_the_adriatic = {
	group = je_group_expeditions
	icon = "gfx/interface/icons/event_icons/event_piracy.dds"
	weight = 1000

	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no

	scripted_progress_bar = eqfm_jack_sparrow_progress_bar

	scripted_button = eqfm_adriatic_expedition_button
	scripted_button = eqfm_recall_adriatic_expedition_button
	scripted_button = eqfm_hire_retired_pirates_button

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		if = {
			limit = {
				NOT = {
					has_global_variable = eqfm_pirates_var
				}
			}
			set_global_variable = {
				name = eqfm_pirates_var
				value = 0
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = je_eqfm_pirates_of_the_adriatic_complete_tt
			has_global_variable = eqfm_jack_sparrow_whacked
		}
	}

	on_complete = {
		eqfm_cleanup_piracy_effect = yes
		eqfm_update_expedition_costs_effect = yes
	}

	fail = {
		NOT = {
			any_scope_state = {
				gbbf_adriatic_geography_trigger = yes
				is_coastal = yes
			}
		}
	}

	on_fail = {
		eqfm_cleanup_piracy_effect = yes
		eqfm_update_expedition_costs_effect = yes
	}

	on_weekly_pulse = {
		effect = {
			eqfm_jack_sparrow_effect = yes
		}
		events = {
			eqfm_pirates.103 # Cutler Beckett Withdraws from the Adriatic
			eqfm_pirates.104 # Cutler Beckett Has Died!
			eqfm_pirates.106 # Pirate Hideout Located
			eqfm_pirates.107 # Pirate Fleet Destroyed
		}
		random_events = { # Chase Events
			15 = eqfm_pirates.131 # Man Overboard!
			35 = eqfm_pirates.132 # The Pillagers' Wake
			# 5 = eqfm_pirates.133 #
			5 = eqfm_pirates.134 # Without a Trace
			45 = eqfm_pirates.135 # Bearing Down
			10 = eqfm_pirates.136 # Cut and Run
			30 = eqfm_pirates.137 # Action Along the Coast
			10 = eqfm_pirates.138 # Naval Duel on the High Seas!
		}
	}

	on_monthly_pulse = {
		events = {}
		random_events = { # Expedition Events
			# Weight for Roughly 95% against General Events and 50% against Expedition Events
			900 = 0

			# General Piracy Events - Balance to a total weight of 100 Max
			20 = eqfm_pirates.110 # Coastal Village Raided!
			20 = eqfm_pirates.111 # Merchant Vessels Raided!
			5 = eqfm_pirates.112 # Nobleman Held Hostage
			1 = eqfm_pirates.113 # A Pirate's Booty
			5 = eqfm_pirates.114 # Diplomatic Communiques Intercepted
			10 = eqfm_pirates.115 # Slavers Intercepted
			5 = eqfm_pirates.116 # Greener Pastures
			10 = eqfm_pirates.117 # The Goods Smugglers
			20 = eqfm_pirates.118 # The Peasant's Pirate

			# Adriatic Expedition Events - Balance to a total weight of 900 Min
			100 = eqfm_pirates.120 # Local Informants
			100 = eqfm_pirates.121 # A Suspicious Inlet
			10 = eqfm_pirates.122 # The Runaway
			50 = eqfm_pirates.123 # Spoiled Provisions
			100 = eqfm_pirates.124 # Circular Bearing
			50 = eqfm_pirates.125 # Windstorms in the Adriatic
			50 = eqfm_pirates.126 # Foul Waters
			100 = eqfm_pirates.127 # Scouts Pick Up Leads
			50 = eqfm_pirates.128 # Emergency Repairs
			100 = eqfm_pirates.129 # Port of Call
			200 = eqfm_pirates.130 # Pirates Spotted!
		}
	}

	on_yearly_pulse = {
		effect = {
			if = {
				limit = {
					has_variable = eqfm_adriatic_expedition_var
				}
				change_variable = {
					name = eqfm_adriatic_expedition_progress_var
					add = 1
				}
				post_notification = notification_eqfm_piracy_yearly_progress
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = eqfm_cutler_beckett_expedition_1
				trigger = {
					has_variable = eqfm_cutler_beckett_var
					exists = var:eqfm_cutler_beckett_var
				}
			}
			triggered_desc = {
				desc = eqfm_jack_sparrow_scared
				trigger = {
					has_global_variable = eqfm_pirates_var
					global_var:eqfm_pirates_var >= 3
					exists = var:eqfm_jack_sparrow_var
				}
			}
			triggered_desc = {
				desc = eqfm_jack_sparrow_cautious
				trigger = {
					has_global_variable = eqfm_pirates_var
					global_var:eqfm_pirates_var >= 2
					exists = var:eqfm_jack_sparrow_var
				}
			}
			triggered_desc = {
				desc = eqfm_jack_sparrow_cocky
				trigger = {
					has_global_variable = eqfm_pirates_var
					global_var:eqfm_pirates_var >= 0
					exists = var:eqfm_jack_sparrow_var
				}
			}
		}
	}
}
