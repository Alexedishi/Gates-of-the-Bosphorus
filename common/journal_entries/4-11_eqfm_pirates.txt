je_turkfm_pirates_of_the_adriatic = {
	group = je_group_expeditions
	icon = "gfx/interface/icons/event_icons/event_piracy.dds"
	weight = 1000

	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no

	scripted_progress_bar = turkfm_jack_sparrow_progress_bar

	scripted_button = turkfm_adriatic_expedition_button
	scripted_button = turkfm_recall_adriatic_expedition_button
	scripted_button = turkfm_hire_retired_pirates_button

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		if = {
			limit = {
				NOT = {
					has_global_variable = turkfm_pirates_var
				}
			}
			set_global_variable = {
				name = turkfm_pirates_var
				value = 0
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = je_turkfm_pirates_of_the_adriatic_complete_tt
			has_global_variable = turkfm_jack_sparrow_whacked
		}
	}

	on_complete = {
		turkfm_cleanup_piracy_effect = yes
		turkfm_update_expedition_costs_effect = yes
	}

	fail = {
		NOT = {
			any_scope_state = {
				gbbf_adriatic_geography_trigger = yes
				is_coastal = yes
			}
		}
	}

	on_fail = {
		turkfm_cleanup_piracy_effect = yes
		turkfm_update_expedition_costs_effect = yes
	}

	on_weekly_pulse = {
		effect = {
			turkfm_jack_sparrow_effect = yes
		}
		events = {
			turkfm_expeditions.103 # Cutler Beckett Withdraws from the Adriatic
			turkfm_expeditions.104 # Cutler Beckett Has Died!
			turkfm_expeditions.106 # Pirate Hideout Located
			turkfm_expeditions.107 # Pirate Fleet Destroyed
		}
		random_events = { # Chase Events
			15 = turkfm_expeditions.131 # Man Overboard!
			35 = turkfm_expeditions.132 # The Pillagers' Wake
			# 5 = turkfm_expeditions.133 #
			5 = turkfm_expeditions.134 # Without a Trace
			45 = turkfm_expeditions.135 # Bearing Down
			10 = turkfm_expeditions.136 # Cut and Run
			30 = turkfm_expeditions.137 # Action Along the Coast
			10 = turkfm_expeditions.138 # Naval Duel on the High Seas!
		}
	}

	on_monthly_pulse = {
		events = {}
		random_events = { # Expedition Events
			# Weight for Roughly 95% against General Events and 50% against Expedition Events
			900 = 0

			# General Piracy Events - Balance to a total weight of 100 Max
			20 = turkfm_expeditions.110 # Coastal Village Raided!
			20 = turkfm_expeditions.111 # Merchant Vessels Raided!
			5 = turkfm_expeditions.112 # Nobleman Held Hostage
			1 = turkfm_expeditions.113 # A Pirate's Booty
			5 = turkfm_expeditions.114 # Diplomatic Communiques Intercepted
			10 = turkfm_expeditions.115 # Slavers Intercepted
			5 = turkfm_expeditions.116 # Greener Pastures
			10 = turkfm_expeditions.117 # The Goods Smugglers
			20 = turkfm_expeditions.118 # The Peasant's Pirate

			# Adriatic Expedition Events - Balance to a total weight of 900 Min
			100 = turkfm_expeditions.120 # Local Informants
			100 = turkfm_expeditions.121 # A Suspicious Inlet
			10 = turkfm_expeditions.122 # The Runaway
			50 = turkfm_expeditions.123 # Spoiled Provisions
			100 = turkfm_expeditions.124 # Circular Bearing
			50 = turkfm_expeditions.125 # Windstorms in the Adriatic
			50 = turkfm_expeditions.126 # Foul Waters
			100 = turkfm_expeditions.127 # Scouts Pick Up Leads
			50 = turkfm_expeditions.128 # Emergency Repairs
			100 = turkfm_expeditions.129 # Port of Call
			200 = turkfm_expeditions.130 # Pirates Spotted!
		}
	}

	on_yearly_pulse = {
		effect = {
			if = {
				limit = {
					has_variable = eqfm_adriatic_expedition_var
				}
				change_variable = {
					name = eqfm_adriatic_expedition_progress_var
					add = 1
				}
				post_notification = notification_turkfm_piracy_yearly_progress
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = turkfm_cutler_beckett_expedition
				trigger = {
					has_variable = turkfm_cutler_beckett_var
					exists = var:turkfm_cutler_beckett_var
				}
			}
			triggered_desc = {
				desc = turkfm_jack_sparrow_scared
				trigger = {
					has_global_variable = turkfm_pirates_var
					global_var:turkfm_pirates_var >= 3
					exists = var:turkfm_jack_sparrow_var
				}
			}
			triggered_desc = {
				desc = turkfm_jack_sparrow_cautious
				trigger = {
					has_global_variable = turkfm_pirates_var
					global_var:turkfm_pirates_var >= 2
					exists = var:turkfm_jack_sparrow_var
				}
			}
			triggered_desc = {
				desc = turkfm_jack_sparrow_cocky
				trigger = {
					has_global_variable = turkfm_pirates_var
					global_var:turkfm_pirates_var >= 0
					exists = var:turkfm_jack_sparrow_var
				}
			}
		}
	}
}
