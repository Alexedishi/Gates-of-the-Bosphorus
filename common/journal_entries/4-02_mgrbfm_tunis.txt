je_mgrbfm_tunis_modernization = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_tunis.dds"
	weight = 2000
	
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	transferable = no

	is_shown_in_lobby = {
		c:TUN ?= this
	}
	
	scripted_button = mgrbfm_visit_france_button

	immediate = {
	}
	
	modifiers_while_active = {
		modifier_mgrbfm_tribal_administration
	}

	complete = {
		OR = {
			is_country_type = recognized
			AND = {
				NOR = {
					has_law_or_variant = law_type:law_hereditary_bureaucrats
					has_law_or_variant = law_type:law_consumption_based_taxation
					has_law_or_variant = law_type:law_land_based_taxation
				}
				any_scope_building = {
					is_building_type = building_university
					NOR = {
						has_active_production_method = pm_simple_organization
						has_active_production_method = pm_religious_bureaucrats
					}
					occupancy > 0.9
					level >= 5
				}
				any_scope_building = {
					OR = {
						is_building_type = gbbf_building_mohamedia_palace
						is_building_type = building_government_administration
					}
					NOR = {
						has_active_production_method = pm_simple_organization
						has_active_production_method = pm_religious_bureaucrats
					}
					occupancy >= 0.9
				}
				bureaucracy >= 0
				approaching_bureaucracy_shortage = no
			}
		}
	}

	on_complete = {
		trigger_event = mgrbfm_tunis.009
		if = {
			limit = {
				has_journal_entry = je_earn_recognition
			}
			custom_tooltip = mgrbfm_recognition_gain_50
		}
		set_variable = mgrbfm_modernization_completed
	}

	timeout = 10950 # 30 years

	on_timeout = {
		trigger_event = mgrbfm_tunis.008
		set_variable = mgrbfm_modernization_completed
	}

	invalid = {
		has_variable = mgrbfm_modernization_completed
	}

	on_monthly_pulse = {
		random_events = {
			chance_of_no_event = 90
			5 = mgrbfm_tunis.002 # An Amibitous Bey
		}
	}
}
