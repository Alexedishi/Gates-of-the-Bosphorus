je_eqfm_pirates_of_the_adriatic = {
	group = je_group_global_international_situations
	icon = "gfx/interface/icons/event_icons/event_piracy.dds"
	weight = 1000

	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no

	is_shown_in_lobby = {
		any_scope_state = {
			is_in_geographic_region = geographic_region_adriatic_sea
		}
	}

	scripted_progress_bar = eqfm_jack_sparrow_progress_bar
	
	scripted_button = eqfm_hire_retired_pirates_button
	scripted_button = eqfm_deploy_adriatic_expedition_button

	should_be_involved = {
		any_scope_state = {
			is_in_geographic_region = geographic_region_adriatic_sea
		}
	}
	
	should_show_when_not_involved = {
		any_scope_state = {
			is_in_geographic_region = geographic_region_europe
		}
	}
	
	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		eqfm_initialize_pirates_effect = yes
	}
	
	immediate_all_involved = {
		trigger_event = {
			id = eqfm_pirates.001 # The Pirates of the Adriatic
			days = 15
		}
	}
	
	on_become_involved_after_activation = {
		if = {
			limit = {
				NOT = {
					has_variable = eqfm_piracy_started
				}
			}
			trigger_event = {
				id = eqfm_pirates.001 # The Pirates of the Adriatic
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = je_eqfm_pirates_of_the_adriatic_complete
			has_global_variable = eqfm_jack_sparrow_whacked
		}
	}
	
	on_complete = {
		remove_global_variable = eqfm_pirates_situation
	}
	
	on_complete_all_involved = {
		eqfm_clear_piracy_expedition_effect = yes
	}
	
	on_no_longer_involved = {
		eqfm_clear_piracy_expedition_effect = yes
	}
	
	on_weekly_pulse = {
		effect = {
			eqfm_update_piracy_variables_effect = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			eqfm_validate_piracy_involvement = yes
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = eqfm_jack_sparrow_scared
				trigger = {
					has_global_variable = eqfm_pirates_situation
					global_var:eqfm_pirates_situation.var:eqfm_pirates_var >= 3
					exists = global_var:eqfm_pirates_situation.var:eqfm_jack_sparrow_var
				}
			}
			triggered_desc = {
				desc = eqfm_jack_sparrow_cautious
				trigger = {
					has_global_variable = eqfm_pirates_situation
					global_var:eqfm_pirates_situation.var:eqfm_pirates_var >= 2
					exists = global_var:eqfm_pirates_situation.var:eqfm_jack_sparrow_var
				}
			}
			triggered_desc = {
				desc = eqfm_jack_sparrow_cocky
				trigger = {
					has_global_variable = eqfm_pirates_situation
					global_var:eqfm_pirates_situation.var:eqfm_pirates_var >= 0
					exists = global_var:eqfm_pirates_situation.var:eqfm_jack_sparrow_var
				}
			}
		}
	}
}

je_eqfm_expedition_to_the_adriatic = {
	group = je_group_expeditions
	icon = "gfx/interface/icons/event_icons/event_piracy.dds"
	weight = 1000
	
	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no
	can_deactivate = no
	
	scripted_progress_bar = eqfm_cutler_beckett_progress_bar
	scripted_progress_bar = eqfm_cutler_beckett_peril_bar
	
	scripted_button = eqfm_recall_adriatic_expedition_button
	scripted_button = eqfm_resupply_adriatic_expedition_button
	
	immediate = {
		gbbf_assign_dlc_icon_effect = yes
	}
	
	complete = {
		eqfm_expedition_progress_greater_than_or_equals_trigger = {
			type = adriatic
			value = 8
		}
	}
	
	on_complete = {
		if = {
			limit = {
				has_variable = eqfm_piracy_chase_var
			}
			trigger_event = {
				id = eqfm_pirates.007 # Pirate Fleet Destroyed
			}
		}
		else = {
			trigger_event = {
				id = eqfm_pirates.006 # Pirate Hideout Located
			}
		}
	}
	
	fail = {
		OR = {
			NOT = {
				eqfm_has_expedition_trigger = {
					type = adriatic
				}
			}
			eqfm_expedition_peril_greater_than_or_equals_trigger = {
				type = adriatic
				value = 5
			}
			NOT = {
				eqfm_expedition_has_leader_trigger = {
					type = adriatic
				}
			}
		}
	}
	
	on_fail = {
		if = {
			limit = {
				eqfm_expedition_has_leader_trigger = {
					type = adriatic
				}
			}
			trigger_event = {
				id = eqfm_pirates.004 # Cutler Beckett Withdraws from the Adriatic
			}
		}
		else = {
			trigger_event = {
				id = eqfm_pirates.005 # Cutler Beckett Has Died!
			}
		}
	}
	
	invalid = {
		OR = {
			NOT = {
				is_involved_in_journal_entry = je_eqfm_pirates_of_the_adriatic
			}
			custom_tooltip = {
				text = je_eqfm_pirates_of_the_adriatic_complete
				has_global_variable = eqfm_jack_sparrow_whacked
			}
		}
	}
	
	on_invalid = {
		eqfm_clear_piracy_expedition_effect = yes
	}
	
	on_weekly_pulse = {
		random_events = { # Chase Events
			15 = eqfm_pirates.031 # Man Overboard!
			35 = eqfm_pirates.032 # The Pillagers' Wake
			# 5 = eqfm_pirates.033 #
			5 = eqfm_pirates.034 # Without a Trace
			45 = eqfm_pirates.035 # Bearing Down
			10 = eqfm_pirates.036 # Cut and Run
			30 = eqfm_pirates.037 # Action Along the Coast
			10 = eqfm_pirates.038 # Naval Duel on the High Seas!
		}
		effect = {
			eqfm_update_adriatic_expedition_variables_effect = yes
		}
	}
	
	on_monthly_pulse = {
		random_events = { # Expedition Events
			chance_of_no_event = 50
			150 = eqfm_pirates.020 # Local Informants
			150 = eqfm_pirates.021 # A Suspicious Inlet
			50 = eqfm_pirates.022 # The Runaway
			75 = eqfm_pirates.023 # Spoiled Provisions
			50 = eqfm_pirates.024 # Circular Bearing
			50 = eqfm_pirates.025 # Windstorms in the Adriatic
			50 = eqfm_pirates.026 # Foul Waters
			150 = eqfm_pirates.027 # Scouts Pick Up Leads
			75 = eqfm_pirates.028 # Emergency Repairs
			125 = eqfm_pirates.029 # Port of Call
			100 = eqfm_pirates.030 # Pirates Spotted!
		}
		effect = {
			eqfm_annual_piracy_expedition_progress_effect = yes
		}
	}
	
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = eqfm_cutler_beckett_expedition_chase
				trigger = {
					has_variable = eqfm_piracy_chase_var
				}
			}
			triggered_desc = {
				desc = eqfm_cutler_beckett_expedition_default
				trigger = {
					has_variable = eqfm_cutler_beckett_var
					exists = var:eqfm_cutler_beckett_var
				}
			}
		}
	}
}
