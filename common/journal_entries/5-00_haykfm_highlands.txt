je_haykfm_highlands = {
	group = je_group_global_international_situations
	icon = "gfx/interface/icons/invention_icons/wargaming.dds"
	dlc = dlc_grefm
	weight = 1000
	
	activation_description = {
		always = yes
	}

	should_be_pinned_by_default_involved = yes
	can_revolution_inherit = yes
	transferable = no

	is_shown_in_lobby = {
		any_scope_state = {
			is_in_geographic_region = geographic_region_armenian_highlands
		}
	}

	scripted_progress_bar = haykfm_higlands_progress_bar

	should_be_involved = {
		any_scope_state = {
			is_in_geographic_region = geographic_region_armenian_highlands
		}
	}

	should_show_when_not_involved = {
		has_interest_marker_in_region = region_near_east
		has_interest_marker_in_region = region_arabia
		has_interest_marker_in_region = region_greater_persia
	}

	immediate = {
		com_save_journal_to_global_variable = {
			name = je_haykfm_highlands
		}
		c:RUS = {
			save_scope_as = russia_scope
		}
		c:TUR = {
			save_scope_as = ottoman_scope
		}
		scope:journal_entry = {
			set_variable = {
				name = haykfm_highlands_var
				value = haykfm_highlands_value
			}
			set_bar_progress = {
				name = haykfm_higlands_progress_bar
				value = var:haykfm_highlands_var
			}
		}
		set_global_variable = haykfm_highlands_started
		create_com_progress_bar = {
			progress_bar = haykfm_higlands_progress_bar
			index = 0
			color = blue
		}
		set_com_progress_bar_color = {
			progress_bar = haykfm_higlands_progress_bar
			color = gold
			side = right
		}
	}
	
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = gold_positive
		
		trigger = {
			OR = {
				custom_tooltip = {
					text = je_haykfm_highlands_complete_tur
					scope:journal_entry = {
						var:haykfm_highlands_var >= 10
					}
				}
				custom_tooltip = {
					text = je_haykfm_highlands_complete_rus
					scope:journal_entry = {
						var:haykfm_highlands_var <= 0
					}
				}
				custom_tooltip = {
					text = je_haykfm_highlands_complete_tie
					scope:journal_entry = {
						any_state = {
							filter = {
								is_in_geographic_region = geographic_region_armenian_highlands
							}
							count = all
							is_incorporated = yes
						}
					}
				}
			}
		}
		effect = { }
		effect_all_involved = {
			custom_tooltip = {
				text = je_haykfm_highlands_on_complete_tur
				if = {
					limit = {
						scope:journal_entry = {
							var:haykfm_highlands_var >= 10
						}
						
					}
				}
			}
			custom_tooltip = {
				text = je_haykfm_highlands_on_complete_rus
				if = {
					limit = {
						scope:journal_entry = {
							var:haykfm_highlands_var <= 0
						}
						
					}
				}
			}
			custom_tooltip = {
				text = je_haykfm_highlands_on_complete_tie
				if = {
					limit = {
						scope:journal_entry = {
							NOR = {
								var:haykfm_highlands_var >= 10
								var:haykfm_highlands_var <= 0
							}
						}
						
					}
				}
			}
		}
	}
	
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = blue_negative
		
		flags = { }
		
		trigger = {
			OR = {
				custom_tooltip = {
					text = je_haykfm_highlands_fail_tur_dead
					NOT = {
						exists = c:TUR
					}
				}
				custom_tooltip = {
					text = je_haykfm_highlands_fail_tur_loss
					c:TUR ?= {
						NOR = {
							any_scope_state = {
								is_in_geographic_region = geographic_region_armenian_highlands
							}
							any_subject_or_below = {
								is_in_geographic_region = geographic_region_armenian_highlands
							}
						}
					}
				}
				custom_tooltip = {
					text = je_haykfm_highlands_fail_rus_dead
					NOT = {
						exists = c:RUS
					}
				}
				custom_tooltip = {
					text = je_haykfm_highlands_fail_rus_loss
					c:RUS ?= {
						NOR = {
							any_scope_state = {
								is_in_geographic_region = geographic_region_armenian_highlands
							}
							any_subject_or_below = {
								is_in_geographic_region = geographic_region_armenian_highlands
							}
						}
					}
				}
			}
		}
		effect = { }
		effect_all_involved = {}
	}

	on_monthly_pulse = {
		effect = {
			every_country = {
				if = {
					limit = {
						is_involved_in_journal_entry = je_haykfm_highlands
						NOT = {
							any_scope_state = {
								is_in_geographic_region = geographic_region_armenian_highlands
							}
						}
					}
					scope:journal_entry = {
						remove_involved_country = prev
					}
				}
				if = {
					limit = {
						any_scope_state = {
							is_in_geographic_region = geographic_region_armenian_highlands
						}
						NOT = {
							is_involved_in_journal_entry = je_haykfm_highlands
						}
					}
					scope:journal_entry = {
						add_involved_country = prev
					}
				}
			}
		}
	}
}
