	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_vergina_sun.dds"
	weight = 100
	
	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no
	
	is_shown_when_inactive = {
		c:BYZ ?= this
		country_has_primary_culture = cu:greek
		NOR = {
			country_has_primary_culture = cu:rhomaios
			country_has_primary_culture = cu:australian
			grefm_reconquest_rejected_trigger = yes
		}
		OR = {
			has_journal_entry = je_grefm_byzantine_restoration
			has_global_variable = grefm_byzantium_restored
		}
	}
	
	possible = {
		grefm_byzantine_restoration_completed_trigger = yes
		calc_true_if = {
			amount >= 5
			owns_entire_state_region = STATE_EASTERN_THRACE
			owns_entire_state_region = STATE_NORTHERN_THRACE
			owns_entire_state_region = STATE_HUDAVENDIGAR
			owns_entire_state_region = STATE_KASTAMONU
			owns_entire_state_region = STATE_TRABZON
			owns_entire_state_region = STATE_BULGARIA
			owns_entire_state_region = STATE_DOBRUDJA
		}
	}
	
	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		trigger_event = {
			id = grefm_hellenic_empire.013 # The Shores of the Hospitable Sea
		}
	}
	
	complete = {
		owns_entire_state_region = STATE_KARS
		owns_entire_state_region = STATE_GREATER_CAUCASUS
		owns_entire_state_region = STATE_CRIMEA
	}
	
	on_complete = {
		s:STATE_KARS = {
			every_scope_state = {
				add_modifier = {
					name = modifier_grefm_bosporan_legacy
					months = gbbf_long_modifier_time
				}
			}
		}
		s:STATE_GREATER_CAUCASUS = {
			every_scope_state = {
				add_modifier = {
					name = modifier_grefm_bosporan_legacy
					months = gbbf_long_modifier_time
				}
			}
		}
		s:STATE_CRIMEA = {
			every_scope_state = {
				add_modifier = {
					name = modifier_grefm_bosporan_legacy
					months = gbbf_long_modifier_time
				}
			}
		}
	}
	
	invalid = {
		grefm_reconquest_rejected_trigger = yes
	}
}
