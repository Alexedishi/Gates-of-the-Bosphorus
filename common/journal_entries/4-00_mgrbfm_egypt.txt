je_mgrbfm_egypt_muhammad_ali = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_hegemon.dds"
	weight = 100
	
	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no
	can_deactivate = no
	
	is_shown_in_lobby = {
		c:EGY ?= this
	}
	
	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		scope:journal_entry = {
			set_variable = {
				name = mgrbfm_egy_army_projection
				value = owner.army_power_projection
			}
			set_variable = {
				name = mgrbfm_egy_navy_projection
				value = owner.naval_power_projection
			}
			set_variable = {
				name = mgrbfm_tur_army_projection
				value = c:TUR.army_power_projection
			}
			set_variable = {
				name = mgrbfm_tur_navy_projection
				value = c:TUR.naval_power_projection
			}
		}
		c:TUR = {
			save_scope_as = ottoman_scope
		}
	}
	
	modifiers_while_active = {
		modifier_mgrbfm_imperial_ambitions
	}
	
	complete = {
		OR = {
			is_subject = no
			is_country_type = recognized
		}
		trigger_if = { # Covering for this bizarre edge case
			limit = {
				exists = c:TUR
			}
			custom_tooltip = {
				text = je_mgrbfm_egypt_muhammad_ali_complete_1
				army_power_projection > c:TUR.army_power_projection
			}
			custom_tooltip = {
				text = je_mgrbfm_egypt_muhammad_ali_complete_2
				naval_power_projection > c:TUR.naval_power_projection
			}
		}
		custom_tooltip = {
			text = je_mgrbfm_egypt_muhammad_ali_complete_3
			any_scope_state = {
				count >= 4
				region = sr:region_nile_basin
				is_split_state = no
			}
		}
		custom_tooltip = {
			text = je_mgrbfm_egypt_muhammad_ali_complete_4
			any_scope_state = {
				count >= 6
				is_in_geographic_region = geographic_region_levant
				is_split_state = no
			}
		}
	}
	
	on_complete = {
		trigger_event = mgrbfm_egypt.009
		set_variable = mgrbfm_egy_ambition_completed
	}
	
	fail = {
		custom_tooltip = {
			text = je_mgrbfm_egypt_muhammad_ali_fail
			ruler ?= {
				NOR = {
					has_template = egy_muhammad_ali_template
					has_template = egy_ibrahim_alawi_template
				}
			}
		}
	}
	
	on_fail = {
		trigger_event = mgrbfm_egypt.008
	}

	on_weekly_pulse = {
		effect = {
			scope:journal_entry = {
				set_variable = {
					name = mgrbfm_egy_army_projection
					value = owner.army_power_projection
				}
				set_variable = {
					name = mgrbfm_egy_navy_projection
					value = owner.naval_power_projection
				}
				set_variable = {
					name = mgrbfm_tur_army_projection
					value = c:TUR.army_power_projection
				}
				set_variable = {
					name = mgrbfm_tur_navy_projection
					value = c:TUR.naval_power_projection
				}
			}
		}
	}

	status_desc = {
		triggered_desc = {
			desc = je_mgrbfm_egypt_muhammad_ali_status
			trigger = {
				ruler ?= {
					OR = {
						has_template = egy_muhammad_ali_template

					}
				}
			}
		}
	}

je_mgrbfm_egypt_sudan = {

	weight = 1000
	
	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no
	
	trigger_event = mgrbfm_egypt.210
	
	complete = {
		trigger_if = {
			limit = {
				has_state_in_state_region = STATE_KORDOFAN
			}
			custom_tooltip = {
				text = je_mgrbfm_egypt_sudan_kordofan
				any_scope_state = {
					filter = {
						state_region = s:STATE_KORDOFAN
					}
					count = all
					is_incorporated = yes
					radical_fraction = {
						value < 0.25
					}
				}
			}
		}
		trigger_if = {
			limit = {
				has_state_in_state_region = STATE_BLUE_NILE
			}
			custom_tooltip = {
				text = je_mgrbfm_egypt_sudan_blue_nile
				any_scope_state = {
					filter = {
						state_region = s:STATE_BLUE_NILE
					}
					count = all
					is_incorporated = yes
					radical_fraction = {
						value < 0.25
					}
				}
			}
		}
		trigger_if = {
			limit = {
				has_state_in_state_region = STATE_DONGOLA
			}
			custom_tooltip = {
				text = je_mgrbfm_egypt_sudan_dongola
				any_scope_state = {
					filter = {
						state_region = s:STATE_DONGOLA
					}
					count = all
					is_incorporated = yes
					radical_fraction = {
						value < 0.25
					}
				}
			}
		}
	}

	on_complete = {
		trigger_event = mgrbfm_egypt.211
	}
	
	fail = {
		NOR = {
			has_state_in_state_region = STATE_KORDOFAN
			has_state_in_state_region = STATE_BLUE_NILE
			has_state_in_state_region = STATE_DONGOLA
		}
	}

	on_fail = {
		if = {
			limit = {
				exists = c:SUD
			}
			if = {
				limit = {
					c:SUD ?= {
						is_subject_of = root
						is_diplomatic_play_enemy_of = root
					}
				}
				trigger_event = mgrbfm_egypt.212
			}
			else_if = {
				limit = {
					c:SUD ?= {
						is_subject_of = root
					}
				}
				trigger_event = mgrbfm_egypt.213
			}
			else_if = {
				limit = {
					c:SUD ?= {
						NOT = {
							is_subject_of = root
						}
					}
				}
				trigger_event = mgrbfm_egypt.214
			}
		}
		else = {
			trigger_event = mgrbfm_egypt.215
		}
	}
}
