je_mgrbfm_egypt_muhammad_ali = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_hegemon.dds"
	weight = 100
	
	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no
	can_deactivate = no
	
	is_shown_in_lobby = {
		c:EGY ?= this
	}
	
	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		scope:journal_entry = {
			set_variable = {
				name = mgrbfm_egy_army_projection
				value = owner.army_power_projection
			}
			set_variable = {
				name = mgrbfm_egy_navy_projection
				value = owner.naval_power_projection
			}
			set_variable = {
				name = mgrbfm_tur_army_projection
				value = c:TUR.army_power_projection
			}
			set_variable = {
				name = mgrbfm_tur_navy_projection
				value = c:TUR.naval_power_projection
			}
		}
		c:TUR = {
			save_scope_as = ottoman_scope
		}
	}
	
	modifiers_while_active = {
		modifier_mgrbfm_imperial_ambitions
	}
	
	complete = {
		OR = {
			is_subject = no
			is_country_type = recognized
		}
		trigger_if = { # Covering for this bizarre edge case
			limit = {
				exists = c:TUR
			}
			custom_tooltip = {
				text = je_mgrbfm_egypt_muhammad_ali_complete_1
				army_power_projection > c:TUR.army_power_projection
			}
			custom_tooltip = {
				text = je_mgrbfm_egypt_muhammad_ali_complete_2
				naval_power_projection > c:TUR.naval_power_projection
			}
		}
		custom_tooltip = {
			text = je_mgrbfm_egypt_muhammad_ali_complete_3
			any_scope_state = {
				count >= 4
				region = sr:region_nile_basin
				is_split_state = no
			}
		}
		custom_tooltip = {
			text = je_mgrbfm_egypt_muhammad_ali_complete_4
			any_scope_state = {
				count >= 6
				is_in_geographic_region = geographic_region_levant
				is_split_state = no
			}
		}
	}
	
	on_complete = {
		trigger_event = mgrbfm_egypt.009
	}
	
	fail = {
		custom_tooltip = {
			text = je_mgrbfm_egypt_muhammad_ali_fail
			ruler ?= {
				NOR = {
					has_template = egy_muhammad_ali_template
					has_template = egy_ibrahim_alawi_template
				}
			}
		}
	}
	
	on_fail = {
		trigger_event = mgrbfm_egypt.008
	}
	
	invalid = {
		NOT = {
			exists = c:TUR
		}
	}

	on_weekly_pulse = {
		effect = {
			scope:journal_entry = {
				set_variable = {
					name = mgrbfm_egy_army_projection
					value = owner.army_power_projection
				}
				set_variable = {
					name = mgrbfm_egy_navy_projection
					value = owner.naval_power_projection
				}
				set_variable = {
					name = mgrbfm_tur_army_projection
					value = c:TUR.army_power_projection
				}
				set_variable = {
					name = mgrbfm_tur_navy_projection
					value = c:TUR.naval_power_projection
				}
			}
		}
	}

	status_desc = {
		triggered_desc = {
			desc = je_mgrbfm_egypt_muhammad_ali_status
			trigger = {
				ruler ?= {
					OR = {
						has_template = egy_muhammad_ali_template
						has_template = egy_ibrahim_alawi_template
					}
				}
			}
		}
	}
}

je_mgrbfm_egypt_modernization = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/invention_icons/nationalism.dds"
	weight = 100
	
	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no
	can_deactivate = no
	
	is_shown_in_lobby = {
		c:EGY ?= this
	}
	
	scripted_progress_bar = mgrbfm_egypt_modernization_progress_bar

#	scripted_button =
	
	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		scope:journal_entry = {
			set_variable = {
				name = mgrbfm_egypt_modernization_var
				value = 0
			}
			set_variable = {
				name = mgrbfm_egypt_modernization_progress_var
				value = 0
			}
			save_scope_as = je_mgrbfm_egypt_modernization
		}
		set_variable = {
			name = je_mgrbfm_egypt_modernization
			value = scope:je_mgrbfm_egypt_modernization
		}
	}

	modifiers_while_active = {
		modifier_mgrbfm_modernizing_drive
	}
	
	complete = {
		custom_tooltip = {
			text = je_mgrbfm_egypt_modernization_complete
			scope:journal_entry = {
				var:mgrbfm_egypt_modernization_var >= 100
			}
		}
	}
	
	on_complete = {
		trigger_event = mgrbfm_egypt.010
		remove_variable = je_mgrbfm_egypt_modernization
	}
	
	fail = {
	}
	
	on_fail = {
		trigger_event = mgrbfm_egypt.011
		remove_variable = je_mgrbfm_egypt_modernization
	}

	on_weekly_pulse = {
		effect = {
			scope:journal_entry = {
				set_variable = {
					name = mgrbfm_egypt_modernization_progress_var
					value = owner.mgrbfm_egypt_modernization_progress_value
				}
				set_bar_progress = {
					name = mgrbfm_egypt_modernization_progress_bar
					value = var:mgrbfm_egypt_modernization_var
				}
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			scope:journal_entry = {
				set_variable = {
					name = mgrbfm_egypt_modernization_progress_var
					value = owner.mgrbfm_egypt_modernization_progress_value
				}
				change_variable = {
					name = mgrbfm_egypt_modernization_var
					add = var:mgrbfm_egypt_modernization_progress_var
				}
				clamp_variable = {
					name = mgrbfm_egypt_modernization_var
					max = 100
					min = 0
				}
				set_bar_progress = {
					name = mgrbfm_egypt_modernization_progress_bar
					value = var:mgrbfm_egypt_modernization_var
				}
			}
		}
	}
}

				limit = {
					var:mgrbfm_national_identity_var ?= cu:arab
				}
#				trigger_event =
			}
		}
		custom_tooltip = {
			text = je_mgrbfm_egypt_nationalism_on_complete_3

				}
#				trigger_event =
			}
		}
	}

	fail = {

	}

	on_fail = {
	}

	invalid = {
	}

	on_invalid = {

	}
