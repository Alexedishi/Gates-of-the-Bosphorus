je_turkfm_eastern_question = {
	icon = "gfx/interface/icons/event_icons/event_ottoman.dds"

	group = je_group_turkfm_eastern_question

	should_be_pinned_by_default = yes

	immediate = {
		set_variable = turkfm_eqfm_started
	}

	scripted_progress_bar = turkfm_tension_progress_bar

	scripted_button = turkfm_greek_gunrunners_button
	scripted_button = turkfm_epirus_provoke_button
	scripted_button = turkfm_macedonia_provoke_button
	scripted_button = turkfm_skopia_provoke_button
	scripted_button = turkfm_smyrna_provoke_button
	scripted_button = turkfm_cyprus_provoke_button
	scripted_button = turkfm_crete_provoke_button

	scripted_button = turkfm_crisis_defender_button
	scripted_button = turkfm_crisis_attacker_button

	complete = {
		custom_tooltip = {
			text = turkfm_separatism_completed_third_tt
			hidden_trigger = {
				has_global_variable = turkfm_separatism_complete
			}
		}
	}

	on_complete = {
		turkfm_cleanup_eqfm_effect = yes
	}

	fail = {
		OR = {
			NOT = {
				exists = c:TUR
			}
			NOT = {
				has_diplomatic_relevance = c:TUR
			}
			custom_tooltip = {
				text = turkfm_sick_man_completed_third_tt
				hidden_trigger = {
					has_global_variable = turkfm_sick_man_complete
				}
			}
		}
	}

	on_fail = {
		turkfm_cleanup_eqfm_effect = yes
	}

	invalid = {
		hidden_trigger = {
			c:TUR ?= this
			any_scope_state = {
				region = sr:region_balkans
			}
		}
	}

	on_invalid = {
		hidden_effect = {
			trigger_event = {
				id = turkfm.204 # Adds je_turkfm_separatism_main
			}
		}
		turkfm_cleanup_eqfm_effect = yes
	}

	on_weekly_pulse = {
		events = {
			turkfm.280 # Ottoman Humiliation Monitor Event
			turkfm.281 # Ottoman Rivalry Monitor Event
			turkfm.282 # Ottoman Rivalry Monitor Event
		}
		effect = {
			turkfm_update_tension_progressbar_effect = yes
		}
	}

	on_monthly_pulse = {
		random_events = {
			200 = 0
			5 = turkfm.312 # Diplomat Offers Mediation
			50 = turkfm.324 # A Chance at Deconfliction
		}
	}

	status_desc = {
		triggered_desc = {
			desc = turkfm_tension_status
			trigger = {
				always = yes
			}
		}
		triggered_desc = {
			desc = turkfm_tension_crisis_status
			trigger = {
				has_global_variable = turkfm_crisis_var
			}
		}
	}

	weight = 1000

}
