je_balkfm_land_reform = {
	group = je_group_technology
	icon = "gfx/interface/icons/invention_icons/enclosure.dds"
	weight = 1000

	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
	}

	modifiers_while_active = {
		modifier_balkfm_outdated_argiculture
	}

	complete = {
		NOR = {
			has_law_or_variant = law_type:law_traditionalism
			has_law_or_variant = law_type:law_agrarianism
		}
		NOR = {
			has_law_or_variant = law_type:law_serfdom
			has_law_or_variant = law_type:law_tenant_farmers
		}
		any_scope_building = {
			is_building_type = building_wheat_farm
			NOT = {
				has_active_production_method = pm_simple_farming
			}
			level >= 5
			weekly_profit > 5
			occupancy >= 0.9
		}
	}

	on_complete = {
		trigger_event = balkfm_legacies.002
		set_variable = balkfm_land_reform_complete
	}

	invalid = {
		hidden_trigger = {
			has_variable = balkfm_land_reform_complete
		}
	}
}

je_balkfm_brigandage = {
	group = je_group_technology
	icon = "gfx/interface/icons/event_icons/event_raiding.dds"
	weight = 1000

	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
	}

	modifiers_while_active = {
		modifier_balkfm_insecure_countryside
	}

	complete = {
		NOR = {
			has_law_or_variant = law_type:law_no_police
			has_law_or_variant = law_type:law_local_police
		}
		institution_investment_level = {
			institution = institution_police
			value >= 3
		}
		any_scope_state = {
			filter = {
				is_in_geographic_region = geographic_region_rumelia
			}
			count = all
			turmoil < 0.20
		}
	}

	on_complete = {
		trigger_event = balkfm_legacies.004
		set_variable = balkfm_brigandage_complete
	}

	invalid = {
		OR = {
			hidden_trigger = {
				has_variable = balkfm_brigandage_complete
			}
			NOT = {
				any_scope_state = {
					is_in_geographic_region = geographic_region_rumelia
				}
			}
		}
	}

	on_invalid = {
		hidden_effect = {
			every_scope_state = {
				limit = {
					has_modifier = modifier_balkfm_brigandage
				}
				remove_modifier = modifier_balkfm_brigandage
			}
		}
	}

	on_monthly_pulse = {
	
	}
}
