je_balkfm_nacertanije = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_flag_yug.dds"
	weight = 100

	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no

	scripted_button = balkfm_austrian_gunrunners_button
	scripted_button = balkfm_ottoman_gunrunners_button
	scripted_button = balkfm_form_yugoslavia_button # this is a fallback for Bulgaria

	is_shown_in_lobby = {
		has_dlc_feature = ip3_content
		OR = {
			country_has_primary_culture = cu:serb
			country_has_primary_culture = cu:croat
			country_has_primary_culture = cu:bosniak
			country_has_primary_culture = cu:slovene
			country_has_primary_culture = cu:bulgarian
		}
	}

	is_shown_when_inactive = {
		NOR = { # not Danubian State
			has_global_variable = balkfm_yugoslavism_failed
			exists = c:YUG
			has_global_variable = has_formed_yug
			c:KUK ?= this
			c:HRE ?= this # https://media.tenor.com/Z16slmCnZFcAAAAM/stare-christian-bale.gif
			has_journal_entry = je_danubian_federation
		}
		OR = {
			country_has_primary_culture = cu:serb
			country_has_primary_culture = cu:croat
			country_has_primary_culture = cu:bosniak
			country_has_primary_culture = cu:slovene
			AND = {
				has_variable = bul_pursued_greater_yugoslavism # Inherited from IP3
				country_has_primary_culture = cu:bulgarian
			}
		}
		aggressive_diplomatic_plays_permitted = yes
	}

	possible = {
		has_technology_researched = nationalism
		is_subject = no
		hidden_trigger = { # blocks journal before a tag has independence
			is_secessionist = no
		}
	}

	immediate = {
		root = {
			save_scope_as = root_country_scope
		}
		cu:serb = {
			save_scope_as = serbian_scope
		}
		cu:bulgarian = {
			save_scope_as = bulgarian_scope
		}
		cu:bosniak = {
			save_scope_as = bosniak_scope
		}
		cu:croat = {
			save_scope_as = croatian_scope
		}
		cu:slovene = {
			save_scope_as = slovene_scope
		}
		trigger_event = {
			id = balkfm_yugoslavia.001 # Yugoslavism Starter Event
		}
		set_global_variable = balkfm_yugoslavism_started
		set_variable = je_yugoslavia_is_active
	}

	complete = {
		custom_tooltip = {
			text = je_balkfm_nacertanije_complete_tt
			c:YUG ?= this
		}
		government_legitimacy >= 50
		is_at_war = no
		custom_tooltip = {
			text = no_insurrectionary_movement_tt
			NOT = {
				any_political_movement = {
					is_insurrectionary = yes
				}
			}
		}
		trigger_if = {
			limit = {
				has_variable = bul_pursued_greater_yugoslavism # Inherited from IP3
			}
			owns_entire_state_region = STATE_BULGARIA
			owns_entire_state_region = STATE_NORTHERN_THRACE
			owns_entire_state_region = STATE_DOBRUDJA
		}
	}

	on_complete = {
		if = {
			limit = {
				has_variable = bul_pursued_greater_yugoslavism # Inherited from IP3
			}
			trigger_event = {
				id = balkfm_bulgaria.217 # The Slavic Balkans
			}
		}
		else = {
			trigger_event = {
				id = balkfm_yugoslavia.009 # The Birth of Yugoslavia
			}
		}
		balkfm_cleanup_yugoslavism_effect = yes
	}

	fail = {
		OR = {
			is_subject = yes
			balkfm_is_national_empire_trigger = yes
		}
	}

	on_fail = {
		balkfm_cleanup_yugoslavism_effect = yes
		post_notification = yugoslavia_failed_message
	}

	invalid = {
		OR = {
			custom_tooltip = {
				text = balkfm_yugoslavism_completed_tt
				hidden_trigger = {
					has_global_variable = has_formed_yug
					NOT = {
						c:YUG ?= this
					}
				}
			}
			custom_tooltip = {
				text = balkfm_yugoslavism_failed_tt
				hidden_trigger = {
					has_global_variable = balkfm_yugoslavism_failed
				}
			}
			custom_tooltip = {
				text = balkfm_weaksauce_trigger_tt
				hidden_trigger = {
					has_variable = balkfm_weaksauce
				}
			}
		}
	}

	on_invalid = {
		balkfm_cleanup_yugoslavism_effect = yes
	}

	on_weekly_pulse = {
		effect = {
			balkfm_determine_yugoslav_leader_effect = yes
		}
	}

	on_monthly_pulse = {
		random_events = {
			chance_of_no_event = 50
			5 = balkfm_yugoslavia.015 # Yugoslav Country Seeks Union (Same Culture)
			5 = balkfm_yugoslavia.016 # Yugoslav Country Seeks Union (Different Culture)
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc ={
				desc = je_balkfm_nacertanije_no_leader_tt
				trigger = {
					NOT = {
						has_global_variable = balkfm_yugoslav_leader_var
					}
				}
			}
			triggered_desc ={
				desc = je_balkfm_nacertanije_no_leader_tt
				trigger = {
					NOT = {
						exists = global_var:balkfm_yugoslav_leader_var
					}
				}
			}
			triggered_desc = {
				desc = je_balkfm_nacertanije_leader_tt
				trigger = {
					exists = global_var:balkfm_yugoslav_leader_var
				}
			}
		}
	}

}
