je_balkfm_serbia_and_montenegro = {
	group = je_group_global_international_situations
	icon = "gfx/interface/icons/event_icons/event_montenegro.dds"
	weight = 100
	
	should_be_pinned_by_default = yes
	
	is_shown_in_lobby = {
		has_dlc_feature = ip3_content # National Awakening
		country_has_primary_culture = cu:serb
	}
	
	should_be_involved = {
		country_has_primary_culture = cu:serb
	}
	
	should_show_when_not_involved = {
		has_interest_marker_in_region = region_balkans
	}
	
	immediate = {
		c:RUS ?= {
			save_scope_as = russia
		}
		c:TUR ?= {
			save_scope_as = ottoman
		}
		c:MON ?= {
			save_scope_as = montenegro
		}
		c:SER ?= {
			save_scope_as = serbia
		}
	}
	
	complete = {
		c:MON ?= {
			is_subject = no
		}
		c:SER ?= {
			is_subject = no
		}
		custom_tooltip = {
			text = je_balkfm_serbia_and_montenegro_complete
			c:MON ?= {
				any_neighbouring_state = { owner = c:SER }
			}
		}
	}
	
	on_complete = {
		custom_tooltip = {
			text = je_balkfm_serbia_and_montenegro_on_complete
			if = {
				limit = {
					c:MON = {
						is_player = yes
					}
				}
				c:MON = {
					trigger_event = {
						id = balkfm_yugoslavia.007
						popup = yes
					}
				}
			}
			else_if = {
				limit = {
					c:SER = {
						is_player = yes
					}
				}
				c:SER = {
					trigger_event = {
						id = balkfm_yugoslavia.007
						popup = yes
					}
				}
			}
			else_if = {
				limit = {
					c:MON.prestige >= c:SER.prestige
				}
				c:MON = {
					trigger_event = {
						id = balkfm_yugoslavia.007
					}
				}
			}
			else = {
				c:SER = {
					trigger_event = {
						id = balkfm_yugoslavia.007
					}
				}
			}
		}
	}
	
	fail = {
		OR = {
			AND = {
				c:MON ?= {
					is_subject = no
				}
				c:SER ?= {
					is_subject = no
					has_war_with = c:MON
				}
			}
			custom_tooltip = {
				text = je_balkfm_serbia_and_montenegro_fail
				c:MON ?= {
					NOT = {
						country_has_primary_culture = cu:serb
					}
				}
			}
		}
	}
	
	on_fail = {
		custom_tooltip = {
			text = je_balkfm_serbia_and_montenegro_on_fail
			c:MON ?= {
				create_diplomatic_catalyst = {
					type = catalyst_event_negative
					target = c:SER
				}
			}
			c:SER ?= {
				create_diplomatic_catalyst = {
					type = catalyst_event_negative
					target = c:MON
				}
			}
		}
	}
	
	invalid = {
		OR = {
			c:MON ?= {
				is_country_alive = no
			}
			c:SER ?= {
				is_country_alive = no
			}
			AND = { # Retained from IP3 just in case this somehow gets added after history execution
				c:MON ?= {
					is_player = yes
				}
				c:SER ?= {
					is_player = yes
				}
			}
		}
	}
}

je_balkfm_unification_of_yugoslavia = {
	group = je_group_global_international_situations
	icon = "gfx/interface/icons/event_icons/event_flag_yug.dds"
	weight = 1000

	should_be_pinned_by_default = yes

	is_shown_in_lobby = {
		has_dlc_feature = ip3_content # National Awakening
		OR = {
			country_has_primary_culture = cu:serb
			country_has_primary_culture = cu:croat
			country_has_primary_culture = cu:bosniak
			country_has_primary_culture = cu:slovene
			country_has_primary_culture = cu:bulgarian
		}
	}

	scripted_button = balkfm_form_yugoslavia_button # this is a fallback for Bulgaria
	scripted_button = balkfm_annex_mon_button
	scripted_button = balkfm_annex_ser_button
	scripted_button = balkfm_annex_bul_button
	scripted_button = balkfm_annex_bos_button
	scripted_button = balkfm_annex_cro_button
	scripted_button = balkfm_annex_slo_button

	is_shown_when_inactive = {
		NOR = {
			has_global_variable = has_formed_yug
			has_global_variable = balkfm_yugoslavism_failed
		}
	}

	should_be_involved = {
		OR = {
			AND = {
				balkfm_country_is_eligible_for_yugoslavism_trigger = yes
				has_technology_researched = nationalism
			}
			any_scope_state = {
				is_in_geographic_region = geographic_region_yugoslavia
			}
			any_subject_or_below = {
				balkfm_country_has_yugoslav_culture_trigger = yes
				has_technology_researched = nationalism
			}
		}
	}

	should_show_when_not_involved = {
		has_interest_marker_in_region = region_balkans
	}

	possible = {
		any_country_in_balkans = {
			balkfm_country_is_eligible_for_yugoslavism_trigger = yes
		}
	}

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		set_global_variable = balkfm_yugoslavism_started
		scope:journal_entry = {
			save_scope_as = balkfm_yugoslavia_journal_scope
		}
		set_global_variable = {
			name = balkfm_yugoslavia_journal_var
			value = scope:balkfm_yugoslavia_journal_scope
		}
	}

	immediate_all_involved = {
		trigger_event = {
			id = balkfm_yugoslavia.001 # Yugoslavism Starter Event
		}
	}

	on_become_involved_after_activation = {
		if = {
			limit = {
				balkfm_country_has_yugoslav_culture_trigger = yes
				NOT = {
					has_variable = je_yugoslavia_is_active
				}
			}
			trigger_event = {
				id = balkfm_yugoslavia.001 # Yugoslavism Starter Event
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = je_balkfm_unification_of_yugoslavia_complete
			exists = c:YUG
		}
	}

	on_complete = {
		set_global_variable = {
			name = je_yugoslavism_cooldown_var # Blocks event spam
			months = 3
		}
		if = {
			limit = {
				exists = c:YUG # Error suppression
			}
			every_country = {
				limit = {
					has_diplomatic_relevance = c:YUG
					balkfm_country_has_yugoslav_culture_trigger = no
				}
				c:YUG ?= {
					save_scope_as = yug_scope
					ruler = {
						save_scope_as = ruler_scope
					}
				}
				post_notification = yug_unification
			}
			every_country = {
				limit = {
					balkfm_country_has_yugoslav_culture_trigger = yes
				}
				balkfm_cleanup_yugoslav_unification_variables_effect = yes
			}
		}
	}

	on_complete_all_involved = {
		custom_tooltip = {
			text = je_balkfm_unification_of_yugoslavia_on_complete_all_involved
			if = {
				limit = {
					exists = c:YUG # Error suppression
				}
				if = {
					limit = {
						balkfm_country_has_yugoslav_culture_trigger = yes
						c:YUG != this
					}
					trigger_event = {
						id = balkfm_yugoslavia.009
					}
				}
			}
		}
	}

	fail = {
		NOT = {
			any_country_in_balkans = {
				balkfm_country_has_yugoslav_culture_trigger = yes
			}
		}
	}

	on_fail = {
		custom_tooltip = {
			text = je_balkfm_unification_of_yugoslavia_on_fail
			every_country = {
				limit = {
					has_interest_marker_in_region = region_balkans
				}
				trigger_event = {
					id = balkfm_yugoslavia.008
				}
			}
			every_country = {
				limit = {
					balkfm_country_has_yugoslav_culture_trigger = yes
				}
				balkfm_cleanup_yugoslav_unification_variables_effect = yes
			}
		}
	}

	on_weekly_pulse = {
		effect = {
			ordered_country = {
				limit = {
					is_involved_in_journal_entry = je_balkfm_unification_of_yugoslavia
					balkfm_country_has_yugoslav_culture_trigger = yes
				}
				order_by = prestige
				position = 0
				check_range_bounds = no
				save_scope_as = balkfm_yugoslav_leader_scope
			}
			if = {
				limit = {
					exists = scope:balkfm_yugoslav_leader_scope
				}
				scope:journal_entry = {
					set_variable = {
						name = balkfm_yugoslav_leader_var
						value = scope:balkfm_yugoslav_leader_scope
					}
					set_variable = {
						name = balkfm_yugoslav_leader_prestige_var
						value = var:balkfm_yugoslav_leader_var.prestige
					}
				}
			}
			else = {
				scope:journal_entry = {
					remove_variable = balkfm_yugoslav_leader_var
					remove_variable = balkfm_yugoslav_leader_prestige_var
				}
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			every_country = {
				if = {
					limit = {
						is_involved_in_journal_entry = je_balkfm_unification_of_yugoslavia
						NOR = {
							balkfm_country_is_eligible_for_yugoslavism_trigger = yes
							any_scope_state = {
								is_in_geographic_region = geographic_region_yugoslavia
							}
							any_subject_or_below = {
								balkfm_country_has_yugoslav_culture_trigger = yes
							}
						}
					}
					global_var:balkfm_yugoslavia_journal_var = {
						remove_involved_country = prev
					}
				}
				if = {
					limit = {
						NOT = {
							is_involved_in_journal_entry = je_balkfm_unification_of_yugoslavia
						}
						OR = {
							balkfm_country_has_yugoslav_culture_trigger = yes
							has_variable = bul_pursued_greater_yugoslavism
							any_scope_state = {
								is_in_geographic_region = geographic_region_yugoslavia
							}
							any_subject_or_below = {
								balkfm_country_has_yugoslav_culture_trigger = yes
							}
						}
					}
					global_var:balkfm_yugoslavia_journal_var = {
						add_involved_country = prev
					}
				}
				if = {
					limit = {
						balkfm_country_has_yugoslav_culture_trigger = yes
					}
					balkfm_update_yugoslav_unification_variables = yes
				}
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc ={
				desc = je_balkfm_unification_of_yugoslavia_status
				trigger = {
					has_global_variable = balkfm_yugoslavism_started
					scope:journal_entry = {
						has_variable = balkfm_yugoslav_leader_var
						exists = var:balkfm_yugoslav_leader_var
					}
				}
			}
			triggered_desc = {
				desc = je_balkfm_unification_of_yugoslavia_status_idle
				trigger = {
					always = yes
				}
			}
		}
	}
}

je_balkfm_yugoslav_nationalism = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_flag_yug.dds"
	weight = 1000
	
	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no
	can_deactivate = no
	
	scripted_progress_bar = balkfm_yugoslavism_progressbar
	
	is_shown_when_inactive = {
		c:YUG ?= this
		NOT = {
			has_global_variable = je_yugoslavism_cooldown_var
		}
	}
	
	possible = {
		NOT = {
			has_journal_entry = je_balkfm_unification_of_yugoslavia
		}
		has_technology_researched = pan-nationalism
		has_technology_researched = central_archives
	}
	
	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		scope:journal_entry = {
			set_variable = {
				name = balkfm_yugoslav_identity_var
				value = 100
			}
			set_variable = {
				name = balkfm_yugoslav_identity_progress_var
				value = 0
			}
			set_variable = {
				name = balkfm_yugoslav_identity_tilt_var
				value = 0
			}
			set_variable = {
				name = balkfm_yugoslav_identity_integralist_var
				value = 0
			}
			set_variable = {
				name = balkfm_yugoslav_identity_federalist_var
				value = 0
			}
			save_scope_as = balkfm_yugoslav_nationalism_scope
		}
		set_variable = {
			name = balkfm_yugoslav_nationalism
			value = scope:balkfm_yugoslav_nationalism_scope
		}
	}
	
	modifiers_while_active = {
		modifier_balkfm_unstable_national_identity
	}
	
	complete = {
		OR = {
			custom_tooltip = {
				text = je_balkfm_yugoslav_nationalism_complete_integralist
				scope:journal_entry = {
					has_variable = balkfm_yugoslav_identity_var
					var:balkfm_yugoslav_identity_var >= 200
				}
			}
			custom_tooltip = {
				text = je_balkfm_yugoslav_nationalism_complete_federalist
				scope:journal_entry = {
					has_variable = balkfm_yugoslav_identity_var
					var:balkfm_yugoslav_identity_var <= 0
				}
			}
		}
	}
	
	on_complete = {
		custom_tooltip = {
			text = je_balkfm_yugoslav_nationalism_on_complete_integralist
			if = {
				limit = {
					var:balkfm_yugoslav_identity_var >= 200
				}
			}
		}
		custom_tooltip = {
			text = je_balkfm_yugoslav_nationalism_on_complete_federalist
			if = {
				limit = {
					var:balkfm_yugoslav_identity_var <= 0
				}
			}
		}
		remove_variable = balkfm_yugoslav_nationalism
	}
	
	fail = {
		custom_tooltip = {
			text = je_balkfm_yugoslav_nationalism_fail
			NOT = {
				c:YUG ?= this
			}
		}
	}
	
	on_fail = {
		remove_variable = balkfm_yugoslav_nationalism
	}
	
	on_weekly_pulse = {
		events = {
		}
		random_events = {
		}
		effect = {
			scope:journal_entry = {
				set_variable = {
					name = balkfm_yugoslav_identity_integralist_var
					value = owner.balkfm_yugoslav_identity_integralist_value
				}
				set_variable = {
					name = balkfm_yugoslav_identity_federalist_var
					value = owner.balkfm_yugoslav_identity_federalist_value
				}
				set_variable = {
					name = balkfm_yugoslav_identity_progress_var
					value = owner.balkfm_yugoslav_identity_progress_value
				}
			}
		}
	}
	
	on_monthly_pulse = {
		events = {
		}
		random_events = {
		}
		effect = {
			scope:journal_entry = {
				change_variable = {
					name = balkfm_yugoslav_identity_var
					add = var:balkfm_yugoslav_identity_progress_var
				}
				clamp_variable = {
					name = balkfm_yugoslav_identity_var
					max = 200
					min = 0
				}
				set_bar_progress = {
					name = balkfm_yugoslavism_progressbar
					value = var:balkfm_yugoslav_identity_var
				}
				set_variable = {
					name = balkfm_yugoslav_identity_tilt_var
					value = owner.balkfm_yugoslav_identity_tilt_value
				}
			}
		}
	}
	
	status_desc = {
		desc = je_balkfm_yugoslav_nationalism_status
	}
}
