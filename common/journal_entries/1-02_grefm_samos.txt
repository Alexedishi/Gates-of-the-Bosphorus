je_grefm_samos_main = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	dlc = dlc_grefm
	weight = 1000
	
	activation_description = {
		always = yes
	}

	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	transferable = no

	is_shown_in_lobby = {
		c:GRE_SAM ?= this
	}

	modifiers_while_active = {
		modifier_grefm_submission
	}
	
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = gold_positive
		
		flags = { historical }
		
		trigger = {
			OR = {
				is_subject = no
				trigger_if = {
					limit = {
						exists = c:GRE
					}
					is_subject_of = c:GRE
				}
			}
		}
		effect = {
			trigger_event = grefm_samos.008
		}
	}
	
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = blue_negative
		
		trigger = {
			OR = {
				is_subject_type = subject_type_puppet
				is_subject_type = subject_type_vassal
			}
		}
	}

	on_monthly_pulse = {
		events = {
			grefm_samos.006 # New Ruler Appointed
			grefm_samos.007 # Prince of Samos Deposed
		}
		effect = {
			if = {
				limit = {
					NOT = {
						has_variable = grefm_stefanos_var
					}
					has_law_or_variant = law_type:law_monarchy
				}
				change_variable = {
					name = grefm_samos_support_var
					add = grefm_samos_support_value
				}
				change_variable = {
					name = grefm_samos_ruler_duration_var
					subtract = 1
				}
			}
			if = {
				limit = {
					has_variable = grefm_stefanos_var
					NOT = {
						has_law_or_variant = law_type:law_monarchy
					}
				}
				remove_variable = grefm_stefanos_var
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_grefm_samos_main_status_1
				trigger = {
					has_variable = grefm_stefanos_var
					has_law_or_variant = law_type:law_monarchy
				}
			}
			triggered_desc = {
				desc = je_grefm_samos_main_status_2
				trigger = {
					NOT = {
						has_variable = grefm_stefanos_var
						has_law_or_variant = law_type:law_monarchy
					}
				}
			}
		}
	}
}
