	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	dlc = dlc_grefm
	weight = 1000
	
	activation_description = {
		always = yes
	}
	
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	transferable = no
	
	is_shown_in_lobby = {
		country_has_primary_culture = cu:romanian
	}
	
	scripted_button = balkfm_unite_principalities_button
	scripted_button = balkfm_all_for_one_button
	scripted_button = balkfm_bessarabian_retrocession_button
	scripted_button = balkfm_marea_unire_button
	
	is_shown_when_inactive = {
		country_has_primary_culture = cu:romanian
	}
	
	possible = {
		has_technology_researched = nationalism
		NOT = {
			any_overlord_or_above ?= {
				# Can't unite the principalities if you're already the lesser in a pact with another Romanian state
				country_has_primary_culture = cu:romanian
			}
		}
	}
	
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = gold_positive
		
		flags = { historical }
		
		trigger = {
			custom_tooltip = {
				text = balkfm_is_romania
				c:ROM ?= this
			}
		}
		effect = { }
	}
	
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = blue_negative
		
		trigger = {
			OR = {
				custom_tooltip = {
					text = balkfm_no_other_romanians_tt
					hidden_trigger = {
						NOT = {
							any_country = {
								NOT = { this = ROOT }
								country_has_primary_culture = cu:romanian
							}
						}
					}
				}
				custom_tooltip = {
					text = balkfm_rom_formed_tt
					hidden_trigger = {
						AND = {
							exists = c:ROM
							NOT = { c:ROM ?= this }
						}
					}
				}
				custom_tooltip = {
					text = balkfm_rom_lost_union_tt
					hidden_trigger = {
						any_country = {
							AND = {
								has_variable = romania_union_partner
								NOT = { is_subject_of = ROOT }
							}
						}
					}
				}
			}
		}
		effect = {
			remove_variable = ROM_united_principality
		}
	}
	
	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					has_variable = romanian_union_var
				}
				change_variable = {
					name = romanian_union_var
					add = 1
				}
				clamp_variable = {
					name = romanian_union_var
					min = 0
					max = 120
				}
			}
		}
	}
}

	scripted_button = balkfm_bessarabian_retrocession_button
	scripted_button = balkfm_marea_unire_button
	
	possible = {
		has_technology_researched = nationalism
		custom_tooltip = {
			text = je_balkfm_rom_greater_romania_possible_tt

					is_homeland = cu:romanian

		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = gold_positive
		
		flags = { historical }
		
		trigger = {
			is_subject = no
			calc_true_if = {
				amount >= 6
				owns_entire_state_region = STATE_BANAT
				owns_entire_state_region = STATE_BESSARABIA
				owns_entire_state_region = STATE_BEKES
				owns_entire_state_region = STATE_MOLDAVIA
				owns_entire_state_region = STATE_NORTHERN_TRANSYLVANIA
				owns_entire_state_region = STATE_SOUTHERN_TRANSYLVANIA
				owns_entire_state_region = STATE_WALLACHIA
			}
		}
		effect = {
			trigger_event = balkfm_romania.709
		}
	}
}
