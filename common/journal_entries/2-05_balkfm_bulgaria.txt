je_balkfm_bul_battenburg_monarchy = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_bulgaria.dds"
	weight = 1000

	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = no
	transferable = no

	scripted_progress_bar = balkfm_battenburg_legitimacy_progressbar
	scripted_progress_bar = balkfm_battenburg_support_progressbar

	is_shown_in_lobby = {
		c:BUL ?= this
	}

	is_shown_when_inactive = {
		c:BUL ?= this
		has_law_or_variant = law_type:law_monarchy
	}

	possible = {
		is_country_type = recognized
	}

	immediate = {
		set_variable = {
			name = balkfm_battenburg_support_var
			value = balkfm_battenburg_support_value
		}
		set_variable = {
			name = balkfm_coup_var
			value = 120
		}
		trigger_event = {
			id = balkfm_bulgaria.209
			days = 30
		}
	}

	complete = {
		has_law_or_variant = law_type:law_monarchy
		custom_tooltip = {
			text = battenburg_monarchy_success_tt1
			hidden_trigger = {
				var:balkfm_battenburg_support_var > 5
			}
		}
		custom_tooltip = {
			text = battenburg_monarchy_success_tt2
			hidden_trigger = {
				has_variable = balkfm_alex_yeeted
			}
		}
	}

	on_complete = {
		trigger_event = balkfm_bulgaria.008
	}

	fail = { # Battenburgs get the boot
		custom_tooltip = {
			text = battenburg_monarchy_fail_tt1
			hidden_trigger = {
				balkfm_battenburg_support_value <= 5
			}
		}
		custom_tooltip = {
			text = battenburg_monarchy_fail_tt2
			hidden_trigger = {
				has_variable = balkfm_alex_yeeted
			}
		}
	}

	on_fail = {
		trigger_event = balkfm_bulgaria.020
	}

	invalid = { # 404 monarch not found
		NOT = { has_law_or_variant = law_type:law_monarchy }
	}

	on_invalid = {
		set_variable = balkfm_alex_yeeted
	}

	on_weekly_pulse = {
		events = {
			balkfm_bulgaria.009 # The August Coup
		}
		effect = {
			set_variable = {
				name = balkfm_battenburg_support_var
				value = balkfm_battenburg_support_value
			}
			je:je_balkfm_bul_battenburg_monarchy = {
				set_bar_progress = {
					name = balkfm_battenburg_legitimacy_progressbar
					value = root.var:balkfm_coup_var
				}
				set_bar_progress = {
					name = balkfm_battenburg_support_progressbar
					value = root.var:balkfm_battenburg_support_var
				}
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			change_variable = {
				name = balkfm_coup_var
				subtract = 1
			}
			clamp_variable = {
				name = balkfm_coup_var
				max = 120
				min = 0
			}
		}
		events = {
			balkfm_bulgaria.002 # Democratic Obstructions
			balkfm_bulgaria.003 # Dissident Publishing
			balkfm_bulgaria.004 # Tsarist Influence
			balkfm_bulgaria.005 # Russia Demands Subservience
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = balkfm_coup_warning_1
				trigger = {
					var:balkfm_coup_var > 60
				}
			}
			triggered_desc = {
				desc = balkfm_coup_warning_2
				trigger = {
					var:balkfm_coup_var > 30
				}
			}
			triggered_desc = {
				desc = balkfm_coup_warning_3
				trigger = {
					var:balkfm_coup_var > 15
				}
			}
			triggered_desc = {
				desc = balkfm_coup_warning_4
				trigger = {
					var:balkfm_coup_var <= 15
				}
			}
		}
	}
}
