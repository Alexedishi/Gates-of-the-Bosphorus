	# Hi everyone, Alexedishi here with some documentation:
	# There are three distinct values to keep track of:
	# "Pressure" which is the global strength of the Senusiyya movement
	# "Entrenchment" which is the strength of the Senusiyya movement in a specific state_region
	# "Influence" which is the strength of the Senusiyya movement in a specific country

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	weight = 100

	}

	}

				is_in_geographic_region = geographic_region_senusiyya
			}
			any_subject_or_below = {
				any_scope_state = {
					is_in_geographic_region = geographic_region_senusiyya
				}
			}
		}
	}
	
	should_show_when_not_involved = {
		OR = {
			any_scope_state = {
				is_in_geographic_region = geographic_region_senusiyya
			}
			any_subject_or_below = {
				any_scope_state = {
					is_in_geographic_region = geographic_region_senusiyya
				}
				gbbf_is_muslim_trigger = yes
			}
			has_strategic_region_interest_tier = {
				strategic_region = sr:region_north_africa
				value >= 2
			}
			has_strategic_region_interest_tier = {
				strategic_region = sr:region_nile_basin
				value >= 2
			}
		}
	}

	immediate = {
		com_save_journal_to_global_variable = {
			name = je_eqfm_senusiyya
		}
		eqfm_initialize_senusiyya_variables_effect = yes
		s:STATE_TRIPOLI = {
			set_variable = {
				name = eqfm_senusiyya_entrenchment_var
				value = 25
			}
		}
		hidden_effect = {
			scope:journal_entry = {
				add_modifier = modifier_eqfm_senusiyya_tranquility_phase
			}
			every_state_region_in_senusiyya = {
				add_state_trait = gbbf_state_trait_tranquility_phase
				save_scope_as = eqfm_senusiyya_state_region_scope
				global_var:je_eqfm_senusiyya = {
					add_to_variable_list = {
						name = eqfm_senusiyya_state_regions_list
						target = scope:eqfm_senusiyya_state_region_scope
					}
				}
			}
		}
		eqfm_update_senussi_character_effect = yes
	}
	
	immediate_all_involved = {
		eqfm_initialize_local_senusiyya_variables_effect = yes
	}
	
	on_become_involved_after_activation = {
		eqfm_initialize_local_senusiyya_variables_effect = yes
	}
	
	on_no_longer_involved = {
		eqfm_cleanup_senusiyya_effect = yes
	}
	
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = gold_positive
		
		flags = { historical }
		
		trigger = {
			any_country_in_senusiyya = {
				count = all
				custom_tooltip = {
					text = eqfm_embraced_senusiyya_situation
					has_variable = eqfm_embraced_senusiyya
				}
			}
		}
		effect = {
			custom_tooltip = {
				text = je_eqfm_senusiyya_on_complete_senusiyya_survives
				if = {
					limit = {
						any_country_in_senusiyya = {
							has_variable = eqfm_embraced_senusiyya
						}
					}
					every_country = {
						limit = {
							is_involved_in_journal_entry = je_eqfm_senusiyya

					}

			}
			eqfm_cleanup_global_senusiyya_variables_effect = yes
		}

			}
		}
	}

		conclusion_desc = je_standard_failure_conclusion_desc

					any_state_in_senusiyya = {
						count = all
						state_region = {
							has_variable = eqfm_expelled_senusiyya
						}
					}
					
				}
			}
		}
		effect = {
			custom_tooltip = {
				text = je_eqfm_senusiyya_on_complete_senusiyya_extinguished
				if = {
					limit = {
						NOT = {
							any_country_in_senusiyya = {
								has_variable = eqfm_embraced_senusiyya
							}
						}
					}
					every_country = {
						limit = {
							is_involved_in_journal_entry = je_eqfm_senusiyya
						}
						trigger_event = eqfm_senusiyya.009
					}
				}
			}
			eqfm_cleanup_global_senusiyya_variables_effect = yes
		}
		effect_all_involved = {
			hidden_effect = {
				eqfm_cleanup_senusiyya_effect = yes
			}
		}
	}
	
	on_weekly_pulse = {
		effect = {
			eqfm_update_senussi_character_effect = yes
			eqfm_update_global_senusiyya_variables_effect = yes
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_senusiyya
				}
				eqfm_update_senusiyya_modifier_effect = yes
			}
		}
	}
	
	on_monthly_pulse = {
		effect = {
			eqfm_monthly_entrenchment_effect = yes
			eqfm_validate_senusiyya_involvement_effect = yes
			scope:journal_entry = {
				change_variable = {
					name = eqfm_senusiyya_pressure_var
					add = var:eqfm_senusiyya_pressure_progress_var
				}
				clamp_variable = {
					name = eqfm_senusiyya_pressure_var
					max = 100
					min = 0
				}
				set_bar_progress = {
					name = eqfm_senusiyya_pressure_progress_bar
					value = var:eqfm_senusiyya_pressure_var
				}
			}
			if = {
				limit = {
					NOT = {
						has_global_variable = eqfm_intimidated_sensussi
					}
					global_var:je_eqfm_senusiyya.var:eqfm_senusiyya_pressure_var >= 50
				}
				set_global_variable = eqfm_intimidated_sensussi
				s:STATE_TRIPOLI = {
					remove_variable = eqfm_senusiyya_entrenchment_var
				}
				s:STATE_LIBYA = {
					set_variable = {
						name = eqfm_senusiyya_entrenchment_var
						value = 25
					}
				}
				every_country = {
					limit = {
						is_involved_in_journal_entry = je_eqfm_senusiyya
					}
					trigger_event = eqfm_senusiyya.003
				}
			}
			if = {
				limit = {
					NOT = {
						has_global_variable = eqfm_senussi_brokered_peace
					}
					has_global_variable = eqfm_intimidated_sensussi
					exists = c:ZWY
					exists = c:TBI
					s:STATE_LIBYA = {
						has_variable = eqfm_senusiyya_entrenchment_var
						var:eqfm_senusiyya_entrenchment_var >= 50
					}
					s:STATE_CHAD = {
						NOR = {
							has_variable = eqfm_senusiyya_entrenchment_var
							has_variable = eqfm_senusiyya_explusion_var
						}
					}
				}
				set_global_variable = eqfm_senussi_brokered_peace
				s:STATE_CHAD = {
					set_variable = {
						name = eqfm_senusiyya_entrenchment_var
						value = 25
					}
				}
				every_country = {
					limit = {
						is_involved_in_journal_entry = je_eqfm_senusiyya
					}
					trigger_event = eqfm_senusiyya.006
				}
			}
		}
	}
	
	status_desc = {
		triggered_desc = {
			desc = je_eqfm_senusiyya_status
			trigger = {
				has_global_variable = je_eqfm_senusiyya
			}
		}
	}

je_eqfm_senusiyya_expedition = {
	group = je_group_expeditions
	icon = "gfx/interface/icons/invention_icons/colonization.dds"
	dlc = dlc_grefm
	weight = 100
	
	activation_description = {
		always = yes
	}
	
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	transferable = no
	can_deactivate = no
	
	scripted_progress_bar = eqfm_senusiyya_expedition_progress_bar
	scripted_progress_bar = eqfm_senusiyya_expedition_peril_bar

	scripted_button = eqfm_resupply_senusiyya_expedition_button
	scripted_button = eqfm_recall_senusiyya_expedition_button
	
	immediate = {
		trigger_event = {
			id = eqfm_senusiyya.401
			days = { 20 30 }
		}
	}
	
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = gold_positive
		
		trigger = {
			custom_tooltip = {
				text = je_eqfm_senusiyya_expedition_complete
				has_variable = eqfm_senusiyya_expedition_progress_var
				var:eqfm_senusiyya_expedition_progress_var >= 8
			}
		}
		effect = {
			trigger_event = eqfm_senusiyya.404
			set_variable = {
				name = eqfm_senusiyya_expedition_transit_var
				months = 3
			}
		}
	}
	
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = blue_negative
		
		trigger = {
			OR = {
				eqfm_expedition_peril_greater_than_or_equals_trigger = {
					type = senusiyya
					value = 5
				}
				NOT = {
					eqfm_expedition_has_leader_trigger = {
						type = senusiyya
					}
				}
			}
		}
		effect = {
			hidden_effect = {
				if = {
					limit = {
						has_variable = eqfm_senusiyya_expedition_peril_var
						var:eqfm_senusiyya_expedition_peril_var >= 5
					}
					trigger_event = eqfm_senusiyya.403
				}
				else = {
					trigger_event = eqfm_senusiyya.402
				}
			}
		}
	}
	
	outcome = {
		conclusion_desc = je_standard_failure_conclusion_desc
		flags = { invalid_outcome hidden }
		
		trigger = {
			OR = {
				eqfm_has_senusiyya_expedition_trigger = no
				NOT = {
					is_involved_in_journal_entry = je_eqfm_senusiyya
				}
			}
		}
		effect = {
			eqfm_clear_senusiyya_expedition_effect = yes
		}
	}
	
	on_weekly_pulse = {
		random_events = {
			chance_of_no_event = 50
			5 = eqfm_senusiyya.405
			5 = eqfm_senusiyya.406
			5 = eqfm_senusiyya.407
			5 = eqfm_senusiyya.408
			5 = eqfm_senusiyya.409
			5 = eqfm_senusiyya.410
			5 = eqfm_senusiyya.411
			5 = eqfm_senusiyya.412
			5 = eqfm_senusiyya.413
			5 = eqfm_senusiyya.414
			5 = eqfm_senusiyya.415
			5 = eqfm_senusiyya.416
		}
		effect = {
			eqfm_update_senusiyya_expedition_variables_effect = yes
		}
	}
	
	on_monthly_pulse = {
		effect = {
			eqfm_annual_senusiyya_expedition_progress_effect = yes
		}
	}
	
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_eqfm_senusiyya_expedition_status_transit
				trigger = {
					has_variable = eqfm_senusiyya_expedition_transit_var
				}
			}
			triggered_desc = {
				desc = je_eqfm_senusiyya_expedition_status_default
				trigger = {
					has_variable = eqfm_senusiyya_expedition_var
				}
			}
		}
	}
