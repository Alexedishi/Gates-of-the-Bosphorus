je_eqfm_sanusiyyah = {
	# Hi everyone, Alexedishi here with some documentation:
	# There are three distinct values to keep track of:
	# "Pressure" which is the global strength of the Sanusiyyah movement
	# "Entrenchment" which is the strength of the Sanusiyyah movement in a specific state_region
	# "Influence" which is the strength of the Sanusiyyah movement in a specific country
	group = je_group_global_international_situations
	icon = "gfx/interface/icons/event_icons/event_islamism.dds"
#	dlc = dlc_grefm
	weight = 100
	
	activation_description = {
		always = yes
	}
	
	should_be_pinned_by_default_involved = yes
	
	is_shown_in_lobby = {
		any_scope_state = {
			is_in_geographic_region = geographic_region_sanusiyyah
		}
	}
	
	scripted_progress_bar = eqfm_sanusiyyah_phase_progress_bar

	## Major Decisions
	scripted_button = eqfm_embrace_sanusiyyah_button
	scripted_button = eqfm_expell_sanusiyyah_button
	scripted_button = eqfm_send_armed_delegation_button
	scripted_button = eqfm_broker_power_deal_button
	
	## Stance Decisions
	scripted_button = eqfm_adopt_welcoming_stance_button
	scripted_button = eqfm_adopt_aloof_stance_button
	scripted_button = eqfm_adopt_hostile_stance_button
	
	## Hedjaz Buttons
	scripted_button = eqfm_spread_senusiyyah_hedjaz_button
	scripted_button = eqfm_expel_sanusiyyah_hedjaz_button
	scripted_button = eqfm_start_improve_relations_hedjaz_button
	scripted_button = eqfm_stop_improve_relations_hedjaz_button
	scripted_button = eqfm_start_harm_relations_hedjaz_button
	scripted_button = eqfm_stop_harm_relations_hedjaz_button
	scripted_button = eqfm_start_arm_tribes_hedjaz_button
	scripted_button = eqfm_stop_arm_tribes_hedjaz_button
	
	
	scripted_button = eqfm_expel_sanusiyyah_libya_button
	scripted_button = eqfm_expel_sanusiyyah_libyan_desert_button
	scripted_button = eqfm_expel_sanusiyyah_tripoli_button
	scripted_button = eqfm_expel_sanusiyyah_egyptian_desert_button
	scripted_button = eqfm_expel_sanusiyyah_matruh_button
	scripted_button = eqfm_expel_sanusiyyah_chad_button
	
	widget = {
		gui = "gui/com_journal_injects/situation_widgets.gui"
		name = "widget_com_international_situation_illustration"
		container = "custom_widget_container_je_icon"
	}

	widget = {
		gui = "gui/com_journal_injects/situation_widgets.gui"
		name = "widget_com_international_situation_phase_information"
		container = "custom_widget_container_2"
	}
	
	widget = {
		gui = gui/journal_entry_widgets/gbbf_journal_widgets.gui
		name = "widget_gbbf_sanusiyyah_situation"
		container = "custom_widget_container_6"
	}
	
	should_be_involved = {
		has_state_in_state_region = STATE_HEDJAZ # as-Sanussi started here, it will expand automatically in the 1840s
	}
	
	should_show_when_not_involved = {
		OR = {
			any_scope_state = {
				is_in_geographic_region = geographic_region_sanusiyyah
			}
			any_subject_or_below = {
				any_scope_state = {
					is_in_geographic_region = geographic_region_sanusiyyah
				}
				gbbf_country_is_muslim_trigger = yes
			}
			has_strategic_region_interest_tier = {
				strategic_region = sr:region_north_africa
				value >= 2
			}
			has_strategic_region_interest_tier = {
				strategic_region = sr:region_nile_basin
				value >= 2
			}
		}
	}
	
	immediate = {
		com_save_journal_to_global_variable = {
			name = je_eqfm_sanusiyyah
		}
		scope:journal_entry = {
			com_set_situation_illustration_background_effect = {
				texture = gbbf_situation_background_sanusiyyah
			}
			com_set_situation_start_date_effect = {
				month = JUNE
				year = 1835 # Date al-Sanussi established the first lodge in Mecca
			}
			com_set_situation_phase_effect = {
				phase_icon = eqfm_sanusiyyah_tranquility_icon_only
				phase_name = eqfm_sanusiyyah_tranquility_no_icon
			}
			set_variable = com_hide_journal_reason
			set_variable = com_hide_involved_countries
			set_variable = com_hide_scripted_buttons
			set_variable = com_hide_status_desc
		}
		eqfm_initialize_sanusiyyah_variables_effect = yes
		hidden_effect = {
			scope:journal_entry = {
				add_modifier = modifier_eqfm_sanusiyyah_tranquility_phase
			}
			every_state_region_in_sanusiyyah = {
				add_state_trait = gbbf_state_trait_tranquility_phase
			}
		}
		eqfm_update_senussi_character_effect = yes
	}
	
	immediate_all_involved = {
		eqfm_initialize_local_sanusiyyah_variables_effect = yes
	}
	
	on_become_involved_after_activation = {
		eqfm_initialize_local_sanusiyyah_variables_effect = yes
	}
	
	on_no_longer_involved = {
		eqfm_cleanup_sanusiyyah_effect = yes
	}
	
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = gold_positive
		
		flags = { historical }
		
		trigger = {
			any_country_in_sanusiyyah = {
				count = all
				custom_tooltip = {
					text = eqfm_embraced_sanusiyyah_situation
					has_variable = eqfm_embraced_sanusiyyah
				}
			}
		}
		effect = {
			custom_tooltip = {
				text = je_eqfm_sanusiyyah_on_complete_sanusiyyah_survives
				if = {
					limit = {
						any_country_in_sanusiyyah = {
							has_variable = eqfm_embraced_sanusiyyah
						}
					}
					every_country = {
						limit = {
							is_involved_in_journal_entry = je_eqfm_sanusiyyah
						}
						trigger_event = eqfm_sanusiyyah.008
					}
				}
			}
			eqfm_cleanup_global_sanusiyyah_variables_effect = yes
		}
		effect_all_involved = {
			hidden_effect = {
				eqfm_cleanup_sanusiyyah_effect = yes
			}
		}
	}
	
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = blue_negative
		
		trigger = {
			any_country_in_sanusiyyah = {
				count = all
				custom_tooltip = {
					text = eqfm_expelled_sanusiyyah_situation
					any_state_in_sanusiyyah = {
						count = all
						state_region = {
							has_variable = eqfm_expelled_sanusiyyah
						}
					}
					
				}
			}
		}
		effect = {
			custom_tooltip = {
				text = je_eqfm_sanusiyyah_on_complete_sanusiyyah_extinguished
				if = {
					limit = {
						NOT = {
							any_country_in_sanusiyyah = {
								has_variable = eqfm_embraced_sanusiyyah
							}
						}
					}
					every_country = {
						limit = {
							is_involved_in_journal_entry = je_eqfm_sanusiyyah
						}
						trigger_event = eqfm_sanusiyyah.009
					}
				}
			}
			eqfm_cleanup_global_sanusiyyah_variables_effect = yes
		}
		effect_all_involved = {
			hidden_effect = {
				eqfm_cleanup_sanusiyyah_effect = yes
			}
		}
	}
	
	on_weekly_pulse = {
		effect = {
			eqfm_update_senussi_character_effect = yes
			eqfm_update_global_sanusiyyah_variables_effect = yes
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_eqfm_sanusiyyah
				}
				eqfm_update_sanusiyyah_modifier_effect = yes
			}
		}
	}
	
	on_monthly_pulse = {
		effect = {
			eqfm_monthly_entrenchment_effect = yes
			eqfm_validate_sanusiyyah_involvement_effect = yes
			scope:journal_entry = {
				change_variable = {
					name = eqfm_sanusiyyah_pressure_var
					add = var:eqfm_sanusiyyah_pressure_progress_var
				}
				clamp_variable = {
					name = eqfm_sanusiyyah_pressure_var
					max = 100
					min = 0
				}
				set_bar_progress = {
					name = eqfm_sanusiyyah_pressure_progress_bar
					value = var:eqfm_sanusiyyah_pressure_var
				}
			}
			if = {
				limit = {
					NOT = {
						has_global_variable = eqfm_intimidated_sensussi
					}
					global_var:je_eqfm_sanusiyyah.var:eqfm_sanusiyyah_pressure_var >= 50
				}
				set_global_variable = eqfm_intimidated_sensussi
				s:STATE_TRIPOLI = {
					remove_variable = eqfm_sanusiyyah_entrenchment_var
				}
				s:STATE_LIBYA = {
					set_variable = {
						name = eqfm_sanusiyyah_entrenchment_var
						value = 25
					}
				}
				every_country = {
					limit = {
						is_involved_in_journal_entry = je_eqfm_sanusiyyah
					}
					trigger_event = eqfm_sanusiyyah.003
				}
			}
			if = {
				limit = {
					NOT = {
						has_global_variable = eqfm_senussi_brokered_peace
					}
					has_global_variable = eqfm_intimidated_sensussi
					exists = c:ZWY
					exists = c:TBI
					s:STATE_LIBYA = {
						has_variable = eqfm_sanusiyyah_entrenchment_var
						var:eqfm_sanusiyyah_entrenchment_var >= 50
					}
					s:STATE_CHAD = {
						NOR = {
							has_variable = eqfm_sanusiyyah_entrenchment_var
							has_variable = eqfm_sanusiyyah_explusion_var
						}
					}
				}
				set_global_variable = eqfm_senussi_brokered_peace
				s:STATE_CHAD = {
					set_variable = {
						name = eqfm_sanusiyyah_entrenchment_var
						value = 25
					}
				}
				every_country = {
					limit = {
						is_involved_in_journal_entry = je_eqfm_sanusiyyah
					}
					trigger_event = eqfm_sanusiyyah.006
				}
			}
		}
	}
	
	status_desc = {
		triggered_desc = {
			desc = je_eqfm_sanusiyyah_status
			trigger = {
				has_global_variable = je_eqfm_sanusiyyah
			}
		}
	}
}

je_eqfm_sanusiyyah_expedition = {
	group = je_group_expeditions
	icon = "gfx/interface/icons/invention_icons/colonization.dds"
	dlc = dlc_grefm
	weight = 100
	
	activation_description = {
		always = yes
	}
	
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	transferable = no
	can_deactivate = no
	
	scripted_progress_bar = eqfm_sanusiyyah_expedition_progress_bar
	scripted_progress_bar = eqfm_sanusiyyah_expedition_peril_bar

	scripted_button = eqfm_resupply_sanusiyyah_expedition_button
	scripted_button = eqfm_recall_sanusiyyah_expedition_button
	
	immediate = {
		trigger_event = {
			id = eqfm_sanusiyyah.401
			days = { 20 30 }
		}
	}
	
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = gold_positive
		
		trigger = {
			custom_tooltip = {
				text = je_eqfm_sanusiyyah_expedition_complete
				has_variable = eqfm_sanusiyyah_expedition_progress_var
				var:eqfm_sanusiyyah_expedition_progress_var >= 8
			}
		}
		effect = {
			trigger_event = eqfm_sanusiyyah.404
			set_variable = {
				name = eqfm_sanusiyyah_expedition_transit_var
				months = 3
			}
		}
	}
	
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = blue_negative
		
		trigger = {
			OR = {
				eqfm_expedition_peril_greater_than_or_equals_trigger = {
					type = sanusiyyah
					value = 5
				}
				NOT = {
					eqfm_expedition_has_leader_trigger = {
						type = sanusiyyah
					}
				}
			}
		}
		effect = {
			hidden_effect = {
				if = {
					limit = {
						has_variable = eqfm_sanusiyyah_expedition_peril_var
						var:eqfm_sanusiyyah_expedition_peril_var >= 5
					}
					trigger_event = eqfm_sanusiyyah.403
				}
				else = {
					trigger_event = eqfm_sanusiyyah.402
				}
			}
		}
	}
	
	outcome = {
		conclusion_desc = je_standard_failure_conclusion_desc
		flags = { invalid_outcome hidden }
		
		trigger = {
			OR = {
				eqfm_has_sanusiyyah_expedition_trigger = no
				NOT = {
					is_involved_in_journal_entry = je_eqfm_sanusiyyah
				}
			}
		}
		effect = {
			eqfm_clear_sanusiyyah_expedition_effect = yes
		}
	}
	
	on_weekly_pulse = {
		random_events = {
			chance_of_no_event = 50
			5 = eqfm_sanusiyyah.405
			5 = eqfm_sanusiyyah.406
			5 = eqfm_sanusiyyah.407
			5 = eqfm_sanusiyyah.408
			5 = eqfm_sanusiyyah.409
			5 = eqfm_sanusiyyah.410
			5 = eqfm_sanusiyyah.411
			5 = eqfm_sanusiyyah.412
			5 = eqfm_sanusiyyah.413
			5 = eqfm_sanusiyyah.414
			5 = eqfm_sanusiyyah.415
			5 = eqfm_sanusiyyah.416
		}
		effect = {
			eqfm_update_sanusiyyah_expedition_variables_effect = yes
		}
	}
	
	on_monthly_pulse = {
		effect = {
			eqfm_annual_sanusiyyah_expedition_progress_effect = yes
		}
	}
	
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_eqfm_sanusiyyah_expedition_status_transit
				trigger = {
					has_variable = eqfm_sanusiyyah_expedition_transit_var
				}
			}
			triggered_desc = {
				desc = je_eqfm_sanusiyyah_expedition_status_default
				trigger = {
					has_variable = eqfm_sanusiyyah_expedition_var
				}
			}
		}
	}
}
