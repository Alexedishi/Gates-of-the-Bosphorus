je_eqfm_senusiyya = {
	group = je_group_global_international_situations
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	weight = 100
	
	should_be_pinned_by_default = yes
	
	is_shown_in_lobby = {
		any_scope_state = {
			is_in_geographic_region = geographic_region_senusiyya
		}
	}
	
	scripted_progress_bar = eqfm_senusiyya_progressbar
	
	scripted_button = eqfm_embrace_senusiyya_button
	scripted_button = eqfm_expell_senusiyya_button
	
	is_shown_when_inactive = {
		OR = {
			exists = c:TRI
			exists = c:FZN
		}
		any_country_in_senusiyya = {
			NOT = {
				country_has_primary_culture = cu:bedouin
			}
		}
	}
	
	should_be_involved = {
		OR = {
			any_scope_state = {
				is_in_geographic_region = geographic_region_senusiyya
			}
			any_subject_or_below = {
				any_scope_state = {
					is_in_geographic_region = geographic_region_senusiyya
				}
				gbbf_is_muslim_trigger = yes
			}
		}
	}
	
	possible = {
		any_country_in_senusiyya = {
			capital = {
				is_in_geographic_region = geographic_region_senusiyya
			}
			gbbf_is_muslim_trigger = yes
			has_technology_researched = mass_communication
		}
	}
	
	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		eqfm_initialize_global_senusiyya_variables = yes
	}
	
	immediate_all_involved = {
		eqfm_initialize_local_senusiyya_variables_effect = yes
	}
	
	should_show_when_not_involved = {
		OR = {
			has_interest_marker_in_region = region_north_africa
			has_interest_marker_in_region = region_nile_basin
		}
	}
	
	on_become_involved_after_activation = {
		eqfm_initialize_local_senusiyya_variables_effect = yes
	}
	
	on_no_longer_involved = {
		eqfm_cleanup_senusiyya_effect = yes
	}
	
	complete = {
		any_state_in_senusiyya = {
			count = all
			owner = {
				OR = {
					custom_tooltip = {
						text = je_eqfm_senusiyya_complete
						OR = {
							has_variable = eqfm_embraced_senusiyya
							has_variable = eqfm_expelled_senusiyya
						}
					}
					country_has_primary_culture = cu:bedouin
				}
			}
		}
	}
	
	on_complete = {
		remove_global_variable = je_eqfm_senusiyya
		every_state_region = {
			remove_variable = eqfm_senusiyya_entrenchment_var
			remove_variable = eqfm_senusiyya_entrenchment_progress_var
		}
	}
	
	on_complete_all_involved = {
		eqfm_cleanup_senusiyya_effect = yes
	}
	
	on_weekly_pulse = {
		effect = {
			eqfm_update_senussi_character_effect = yes
			eqfm_update_global_senusiyya_variables_effect = yes
			eqfm_update_senusiyya_modifier_effect = yes
		}
	}
	
	on_monthly_pulse = {
		effect = {
			eqfm_update_georegion_senusiyya_variables_effect = yes
			eqfm_validate_senusiyya_involvement_effect = yes
			scope:journal_entry = {
				change_variable = {
					name = eqfm_senusiyya_pressure_var
					add = var:eqfm_senusiyya_pressure_progress_var
				}
				clamp_variable = {
					name = eqfm_senusiyya_pressure_var
					max = 100
					min = 0
				}
				set_bar_progress = {
					name = eqfm_senusiyya_progressbar
					value = owner.var:eqfm_senusiyya_pressure_var
				}
			}
			if = {
				limit = {
					NOT = {
						has_global_variable = eqfm_intimidated_sensussi
					}
					global_var:je_eqfm_senusiyya.var:eqfm_senusiyya_pressure_var >= 50
				}
				every_country = {
					limit = {
						is_involved_in_journal_entry = je_eqfm_senusiyya
					}
					trigger_event = {
						id = eqfm_senusiyya.003
					}
				}
			}
		}
	}
	
	status_desc = {
		triggered_desc = {
			desc = je_eqfm_senusiyya_status
			trigger = {
				has_global_variable = je_eqfm_senusiyya
			}
		}
	}
}
