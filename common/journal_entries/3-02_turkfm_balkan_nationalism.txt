je_turkfm_separatism_riots = {
	icon = "gfx/interface/icons/event_icons/bicorne_hat.dds"

	group = je_group_turkfm_separatism

	should_be_pinned_by_default = yes

	immediate = {
		set_variable = {
			name = turkfm_riot_var
			value = 100
		}
		set_variable = {
			name = turkfm_riot_progress_var
			value = 0
		}
		turkfm_update_riot_variables_effect = yes
	}

	scripted_progress_bar = turkfm_riot_progress_bar

	complete = {
		NOR = {
			any_scope_state = {
				has_modifier = modifier_turkfm_race_riots
			}
			custom_tooltip = {
				text = je_turkfm_separatism_riots_complete
				any_country = {
					is_diplomatic_play_enemy_of = root
					has_modifier = modifier_turkfm_nationalist_uprising
				}
			}
			hidden_trigger = {
				has_variable = turkfm_riot_technical_var
			}
		}
	}

	on_complete = { # Set variables for deescalation and call turkfm_check_phase_effect
		turkfm_add_extremely_large_deescalation = yes
		every_scope_state = {
			limit = {
				has_modifier = modifier_turkfm_race_riots
			}
			remove_modifier = modifier_turkfm_race_riots
		}
		remove_variable = turkfm_riot_var
		remove_variable = turkfm_riot_progress_var
		remove_variable = turkfm_riot_state_1_var
		remove_variable = turkfm_riot_state_2_var
		remove_variable = turkfm_riot_state_3_var
		remove_variable = turkfm_riot_state_1_progress_var
		remove_variable = turkfm_riot_state_2_progress_var
		remove_variable = turkfm_riot_state_3_progress_var
		set_variable = {
			name = turkfm_riot_current_tally_var
			value = 0
		}
	}

	invalid = {
		custom_tooltip = {
			text = je_turkfm_separatism_riots_invalid
			OR = {
				NOT = {
					has_global_variable = turkfm_separatism_phase_var
				}
				global_var:turkfm_separatism_phase_var != 3 # Hostility
			}
		}
	}

	on_invalid = {
		every_scope_state = {
			limit = {
				has_modifier = modifier_turkfm_race_riots
			}
			remove_modifier = modifier_turkfm_race_riots
		}
		remove_variable = turkfm_riot_var
		remove_variable = turkfm_riot_state_1_var
		remove_variable = turkfm_riot_state_2_var
		remove_variable = turkfm_riot_state_3_var
		remove_variable = turkfm_riot_state_1_progress_var
		remove_variable = turkfm_riot_state_2_progress_var
		remove_variable = turkfm_riot_state_3_progress_var
		set_variable = {
			name = turkfm_riot_current_tally_var
			value = 0
		}
	}

	on_weekly_pulse = {
		events = {
			turkfm.206 # Balkan Country Rebellion
			turkfm.207 # Riots in State Suppressed
		}
		effect = {
			turkfm_update_riot_variables_effect = yes
			change_variable = {
				name = turkfm_riot_var
				add = var:turkfm_riot_progress_var
			}
			clamp_variable = {
				name = turkfm_riot_var
				max = 200
				min = 0
			}
			je:je_turkfm_separatism_riots = {
				set_bar_progress = {
					value = owner.var:turkfm_riot_var
					name = turkfm_riot_progress_bar
				}
			}
		}
	}

	on_monthly_pulse = {
		random_events = {
			5 = turkfm.250 # Fortress in State Beseiged
			5 = turkfm.251 # Crowds Burn Administrative Palace
			5 = turkfm.252 # IG Sides with Minorities
			5 = turkfm.253 # Ulema Mobilizes the Faithful
			5 = turkfm.254 # Armed Forces Requests Intervention
			#			5 = turkfm.255 #
			#			5 = turkfm.256 #
			5 = turkfm.257 # Rioting Worsens in State
		}
	}

	weight = 1000

}
