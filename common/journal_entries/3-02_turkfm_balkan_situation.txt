je_turkfm_balkan_situation_riots = {
	icon = "gfx/interface/icons/event_icons/bicorne_hat.dds"

	group = je_group_turkfm_separatism

	should_be_pinned_by_default = yes

	immediate = {
		set_variable = {
			name = turkfm_riot_var
			value = 50
		}
		set_variable = {
			name = turkfm_riot_progress_var
			value = 0
		}
	}

	scripted_progress_bar = turkfm_riot_progress_bar

	complete = {
		OR = {
			custom_tooltip = {
				text = je_turkfm_balkan_situation_riots_invalid
				OR = {
					NOT = {
						has_global_variable = turkfm_balkan_situation_phase_var
					}
					global_var:turkfm_balkan_situation_phase_var != flag:turkfm_hostility
				}
			}
			NOR = {
				any_scope_state = {
					has_modifier = modifier_turkfm_race_riots
				}
				custom_tooltip = {
					text = je_turkfm_balkan_situation_riots_complete
					any_country = {
						is_diplomatic_play_enemy_of = root
						has_modifier = modifier_turkfm_nationalist_uprising
					}
				}
				hidden_trigger = {
					NOT = {
						has_variable = turkfm_recent_revolt_var
					}
				}
			}
		}
	}

	on_complete = { # Set variables for deescalation and call turkfm_check_balkan_situation_phase_effect
		turkfm_add_extremely_large_deescalation = yes
		turkfm_cleanup_riots_effect = yes
	}

	on_weekly_pulse = {
		events = {
			turkfm_balkan_situation.006 # Balkan Country Rebellion
			turkfm_balkan_situation.007 # Riots in State Suppressed
		}
		effect = {
			turkfm_update_riot_variables_effect = yes
			change_variable = {
				name = turkfm_riot_var
				add = var:turkfm_riot_progress_var
			}
			clamp_variable = {
				name = turkfm_riot_var
				max = 100
				min = 0
			}
			je:je_turkfm_balkan_situation_riots = {
				set_bar_progress = {
					value = owner.var:turkfm_riot_var
					name = turkfm_riot_progress_bar
				}
			}
			remove_modifier = modifier_turkfm_nationalist_insurrection
			add_modifier = {
				name = modifier_turkfm_nationalist_insurrection
				multiplier = turkfm_riot_multiplier_value
			}
		}
	}

	on_monthly_pulse = {
		random_events = {
			5 = turkfm_balkan_situation.200 # Fortress in State Beseiged
			5 = turkfm_balkan_situation.201 # Crowds Burn Administrative Palace
			5 = turkfm_balkan_situation.202 # IG Sides with Minorities
			5 = turkfm_balkan_situation.203 # Ulema Mobilizes the Faithful
			5 = turkfm_balkan_situation.204 # Armed Forces Requests Intervention
			5 = turkfm_balkan_situation.207 # Rioting Worsens in State
		}
	}

	weight = 1000

}
