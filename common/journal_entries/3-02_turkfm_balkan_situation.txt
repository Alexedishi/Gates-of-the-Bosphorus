je_turkfm_balkan_situation_riots = {
	icon = "gfx/interface/icons/event_icons/bicorne_hat.dds"
	group = je_group_turkfm_separatism
	should_be_pinned_by_default = yes

	immediate = {
		set_variable = {
			name = turkfm_riot_var
			value = 50
		}
		set_variable = {
			name = turkfm_riot_progress_var
			value = 0
		}
	}

	scripted_progress_bar = turkfm_riot_progress_bar

	complete = {
		OR = {
			custom_tooltip = {
				text = je_turkfm_balkan_situation_riots_invalid
				OR = {
					NOT = {
						has_global_variable = turkfm_balkan_situation_phase_var
					}
					global_var:turkfm_balkan_situation_phase_var != flag:turkfm_hostility
				}
			}
			NOR = {
				any_scope_state = {
					has_modifier = modifier_turkfm_race_riots
				}
				custom_tooltip = {
					text = je_turkfm_balkan_situation_riots_complete
					any_country = {
						is_diplomatic_play_enemy_of = root
						has_modifier = modifier_turkfm_nationalist_uprising
					}
				}
				hidden_trigger = {
					NOT = {
						has_variable = turkfm_recent_revolt_var
					}
				}
			}
		}
	}

	on_complete = { # Set variables for deescalation and call turkfm_check_balkan_situation_phase_effect
		turkfm_add_extremely_large_deescalation = yes
		turkfm_cleanup_riots_effect = yes
	}

	on_weekly_pulse = {
		events = {
			turkfm_balkan_situation.007 # Riots in State Suppressed
		}
		effect = {
			turkfm_update_riot_variables_effect = yes
			turkfm_update_riot_modifiers_effect = yes
			change_variable = {
				name = turkfm_riot_var
				add = var:turkfm_riot_progress_var
			}
			clamp_variable = {
				name = turkfm_riot_var
				max = 100
				min = 0
			}
			je:je_turkfm_balkan_situation_riots = {
				set_bar_progress = {
					value = owner.var:turkfm_riot_var
					name = turkfm_riot_progress_bar
				}
			}
		}
	}

	on_monthly_pulse = {
		random_events = {
			5 = turkfm_balkan_situation.140 # Fortress in State Beseiged
			5 = turkfm_balkan_situation.141 # Crowds Burn Administrative Palace
			5 = turkfm_balkan_situation.142 # IG Sides with Minorities

			5 = turkfm_balkan_situation.144 # Armed Forces Requests Intervention
			5 = turkfm_balkan_situation.147 # Rioting Worsens in State
		}
	}

	weight = 1000

je_great_eastern_crisis = {
	group = je_group_turkfm_separatism
	icon = "gfx/interface/icons/event_icons/event_ottoman.dds"
	weight = 100000

	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no

	scripted_progress_bar = turkfm_crisis_progress_bar

	is_shown_in_lobby = {
		has_dlc_feature = ip3_content #National Awakening
		c:TUR ?= this
	}

	modifiers_while_active = {
	}

	immediate = {
		turkfm_initialize_crisis_effect = yes
	}

	complete = {
		custom_tooltip = {
			text = je_turkfm_balkan_situation_crisis_complete
			global_var:turkfm_crisis_var >= 100
		}
		has_any_secessionists_broken_out = no
		has_any_secessionists_growing = no
		is_at_war = no
	}

	on_complete = {
		show_as_tooltip = {
			trigger_event = {
				id = turkfm_balkan_situation.029 # Order Returns to Rumelia
			}
		}
		turkfm_cleanup_crisis_effect = yes
		turkfm_force_deconfliction_effect = yes
	}

	fail = {
		custom_tooltip = {
			text = je_turkfm_balkan_situation_crisis_complete
			global_var:turkfm_crisis_var <= 0
		}
	}

	on_fail = {
		show_as_tooltip = {
			trigger_event = {
				id = turkfm_balkan_situation.028 # Rumelia in Disarray
			}
		}
		turkfm_cleanup_crisis_effect = yes
		turkfm_force_deconfliction_effect = yes
	}

	invalid = {
		OR = {
			NOT = {
				has_global_variable = turkfm_balkan_situation_phase_var
			}
			AND = {
				has_global_variable = turkfm_balkan_situation_phase_var
				global_var:turkfm_balkan_situation_phase_var != flag:turkfm_crisis
			}
			has_global_variable = turkfm_separatism_complete
		}
	}

	on_invalid = {
		turkfm_cleanup_crisis_effect = yes
		if = {
			limit = {
				NOT = {
					has_global_variable = turkfm_separatism_complete
				}
			}
			turkfm_force_deconfliction_effect = yes
		}
	}

	on_weekly_pulse = {
		effect = {
			set_global_variable = {
				name = turkfm_crisis_progress_var
				value = turkfm_crisis_progress_value
			}
			change_global_variable = {
				name = turkfm_crisis_duration_var
				add = 1
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			change_global_variable = {
				name = turkfm_crisis_var
				add = global_var:turkfm_crisis_progress_var
			}
			clamp_global_variable = {
				name = turkfm_crisis_var
				max = 250
				min = 0
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_global_variable = turkfm_crisis_var
				}
				desc = je_turkfm_balkan_situation_crisis_status
			}
		}
	}

}
