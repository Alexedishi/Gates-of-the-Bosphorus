je_turkfm_tanzimat_main = {
	icon = "gfx/interface/icons/event_icons/event_fez.dds"
	group = je_group_turkfm_tanzimat
	should_be_pinned_by_default = yes

	is_shown_in_lobby = {
		c:TUR ?= this
	}

	immediate = {
		scope:journal_entry = {
			set_variable = {
				name = com_bear_spray_applied
				value = yes
			}
		}
	}

	modifiers_while_active = {
		modifier_turkfm_imperial_household
		modifier_turkfm_decentralized_administration
	}
	
	scripted_progress_bar = turkfm_tanzimat_reforms_progress_bar
	scripted_progress_bar = turkfm_imperial_identity_progress_bar

	scripted_button = turkfm_start_tanzimat_button
	scripted_button = turkfm_enforce_identity_button

	complete = {
		custom_tooltip = {
			text = je_turkfm_tanzimat_main_complete
			hidden_trigger = {
				var:turkfm_tanzimat_reform_var >= 1000
			}
		}
	}

	on_complete = {
		set_global_variable = turkfm_tanzimat_complete
		custom_tooltip = {
			text = BULje_turkfm_tanzimat_main_on_complete_ottomanist
			if = {
				limit = {
					has_variable = turkfm_ottomanist_identity_enforced
				}
				trigger_event = {
					id = turkfm_tanzimat.009 # The Ottoman Motherland
				}
			}
		}
		custom_tooltip = {
			text = BULje_turkfm_tanzimat_main_on_complete_islamist
			if = {
				limit = {
					has_variable = turkfm_islamist_identity_enforced
				}
				trigger_event = {
					id = turkfm_tanzimat.008 # The Eternal Caliphate
				}
			}
		}
		turkfm_cleanup_tanzimat_effect = yes
	}

	fail = {
		OR = {
			custom_tooltip = {
				text = je_turkfm_tanzimat_main_fail
				hidden_trigger = {
					var:turkfm_tanzimat_reform_var <= 0
				}
			}
			trigger_if = {
				limit = {
					has_variable = turkfm_young_turks_started
				}
				custom_tooltip = {
					text = turkfm_young_turks_failed
					has_variable = turkfm_young_turks_failed
				}
			}
		}
	}

	on_fail = {
		set_global_variable = turkfm_tanzimat_failed
		if = {
			limit = {
				has_variable = turkfm_young_turks_started
			}
			trigger_event = {
				id = turkfm_tanzimat.007 # The Three Pashas
			}
		}
		else = {
			trigger_event = {
				id = turkfm_tanzimat.006 # A Stagnant Empire
			}
		}
		turkfm_cleanup_tanzimat_effect = yes
	}

	on_weekly_pulse = {
		effect = {
			turkfm_calculate_tanzimat_variables_effect = yes
		}
	}

	on_monthly_pulse = {
		random_events = {
			chance_of_no_event = 95
			5 = turkfm_tanzimat.003 # The Men of the Tanzimat
		}
		effect = {
			change_variable = {
				name = turkfm_tanzimat_reform_var
				add = var:turkfm_tanzimat_reform_progress_var
			}
			clamp_variable = {
				name = turkfm_tanzimat_reform_var
				max = 1000
				min = 0
			}
			je:je_turkfm_tanzimat_main = {
				set_bar_progress = {
					name = turkfm_tanzimat_reforms_progress_bar
					value = owner.var:turkfm_tanzimat_reform_var
				}
			}
		}
	}

	weight = 4000

}
