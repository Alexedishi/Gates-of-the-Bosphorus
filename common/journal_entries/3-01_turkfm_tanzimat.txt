je_turkfm_tanzimat_main = {
	icon = "gfx/interface/icons/event_icons/event_fez.dds"

	group = je_group_turkfm_tanzimat

	should_be_pinned_by_default = yes

	immediate = {
	}

	modifiers_while_active = {
		modifier_turkfm_imperial_household
		modifier_turkfm_decentralized_administration
	}

	scripted_progress_bar = turkfm_tanzimat_progress_bar

	scripted_button = turkfm_start_tanzimat_button

	complete = {
		custom_tooltip = {
			text = je_turkfm_tanzimat_main_complete
			hidden_trigger = {
				var:turkfm_tanzimat_var >= 1000
			}
		}
	}

	on_complete = {
		trigger_event = {
			id = turkfm_tanzimat.009 # The Ottoman Motherland
		}
	}

	fail = {
		custom_tooltip = {
			text = je_turkfm_tanzimat_main_fail
			hidden_trigger = {
				var:turkfm_tanzimat_var <= 0
			}
		}
	}

	on_fail = {
		show_as_tooltip = {
			trigger_event = {
				id = turkfm_tanzimat.008 # The Eternal Caliphate
			}
		}
		hidden_effect = {
			if = {
				limit = {
					#				has_variable = turkfm_1913_coup_happened
					always = no # TODO remove when young turks ready
				}
				trigger_event = {
					id = turkfm_tanzimat.007 # The Three Pashas
				}
			}
			else = {
				trigger_event = {
					id = turkfm_tanzimat.008 # The Eternal Caliphate
				}
			}
		}
	}

	on_weekly_pulse = {
		effect = {
			set_variable = {
				name = turkfm_tanzimat_progress_var
				value = turkfm_tanzimat_progress_value
			}
			set_variable = {
				name = turkfm_tanzimat_tilt_var
				value = turkfm_tanzimat_tilt_value
			}
		}
	}

	on_monthly_pulse = {
		random_events = {
			chance_of_no_event = 95
			5 = turkfm_tanzimat.003 # The Men of the Tanzimat
		}
		effect = {
			change_variable = {
				name = turkfm_tanzimat_var
				add = var:turkfm_tanzimat_progress_var
			}
			clamp_variable = {
				name = turkfm_tanzimat_var
				max = 1000
				min = 0
			}
			je:je_turkfm_tanzimat_main = {
				set_bar_progress = {
					name = turkfm_tanzimat_progress_bar
					value = owner.var:turkfm_tanzimat_var
				}
			}
		}
	}

	weight = 4000

}

je_turkfm_tanzimat_argiculture = {
	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	group = je_group_turkfm_tanzimat

	is_shown_when_inactive = {
		c:TUR ?= this
		NOT = {
			has_global_variable = turkfm_tanzimat_complete
		}
	}

	possible = {
		custom_tooltip = {
			text = turkfm_tanzimat_started_tt
			has_global_variable = turkfm_tanzimat_started
		}
		has_modifier = modifier_turkfm_feudal_argiculture
	}

	complete = {
		NOR = {
			has_law = law_type:law_traditionalism
			has_law = law_type:law_agrarianism
		}
		NOR = {
			has_law = law_type:law_serfdom
			has_law = law_type:law_tenant_farmers
		}
		any_scope_building = {
			is_building_type = building_wheat_farm
			NOT = {
				has_active_production_method = pm_simple_farming
			}
			level >= 10
			weekly_profit > 5
			occupancy >= 0.9
		}
	}

	on_complete = {
		trigger_event = {
			id = turkfm_tanzimat.010 # Modernized Agriculture
		}
		show_as_tooltip = {
			turkfm_advance_ottomanism1_effect = yes
			turkfm_complete_reform_effect = yes
		}
	}

	invalid = {
		custom_tooltip = {
			text = turkfm_tanzimat_completed_tt
			hidden_trigger = {
				has_global_variable = turkfm_tanzimat_complete
			}
		}
	}

	weight = 3500
}
