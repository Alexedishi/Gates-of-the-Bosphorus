je_balkfm_mon_prince_bishopric = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_montenegro.dds"
	weight = 1000

	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no

	scripted_button = balkfm_visit_tsar_button
	scripted_button = je_montenegrin_visit_moscow
	scripted_button = je_montenegrin_make_saint
	scripted_button = balkfm_mon_principality_button

	is_shown_in_lobby = {
		c:MON ?= this
	}

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		if = {
			limit = {
				has_dlc_feature = ip3_content # National Awakening
			}
			set_variable = {
				name = rus_relations
				value = c:RUS.relations:ROOT
			}
			add_modifier = {
				name = modifier_montenegrin_aid
				multiplier = russian_aid_income
			}
			c:RUS ?= {
				save_scope_as = russia
			}
			c:TUR ?= {
				save_scope_as = ottoman
			}
		}
	}

	modifiers_while_active = {
		modifier_balkfm_prince_bishopric
		modifier_balkfm_spartan_existence
	}

	complete = {
		NOT = {
			has_law_or_variant = law_type:law_theocracy
		}
		government_legitimacy > 50
		trigger_if = {
			limit = {
				has_dlc_feature = ip3_content # National Awakening
			}
			custom_tooltip = {
				text = has_taxation_law_tt
				NOT = {
					has_law_or_variant = law_type:law_consumption_based_taxation
				}
			}
			custom_tooltip = {
				text = has_police_law_tt
				NOT = {
					has_law_or_variant = law_type:law_no_police
				}
			}
			custom_tooltip = {
				text = balkfm_biljarda_built
				any_scope_building = {
					is_building_type = gbbf_building_biljarda
				}
			}
			has_modifier = modifier_balkfm_declared_principality
			NOR = {
				in_default = yes
				has_modifier = declared_bankruptcy
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = je_balkfm_mon_prince_bishopric_tt1
				hidden_trigger = {
					has_variable = balkfm_mon_soc_completed
				}
			}
			custom_tooltip = {
				text = je_balkfm_mon_prince_bishopric_tt2
				hidden_trigger = {
					has_variable = balkfm_mon_gov_completed
				}
			}
			custom_tooltip = {
				text = je_balkfm_mon_prince_bishopric_tt3
				hidden_trigger = {
					has_variable = balkfm_mon_mil_completed
				}
			}
			custom_tooltip = {
				text = je_balkfm_mon_prince_bishopric_tt4
				hidden_trigger = {
					OR = {
						has_modifier = modifier_balkfm_declared_principality
						has_modifier = modifier_balkfm_declared_kingdom
					}
				}
			}
		}
	}

	on_complete = {
		if = {
			limit = {
				has_dlc_feature = ip3_content # National Awakening
				NOT = {
					has_global_variable = eqfm_montenegrin_raiding_settled
				}
			}
			show_as_tooltip = {
				add_modifier = {
					name = modifier_montenegrin_recognition
				}
			}
			custom_tooltip = {
				text = montenegrin_compromise_tt
				trigger_event = {
					id = balkfm_montenegro.018
				}
			}
		}
		else_if = {
			limit = {
				has_dlc_feature = ip3_content # National Awakening
				has_global_variable = eqfm_montenegrin_raiding_settled
			}
			trigger_event = {
				id = balkfm_montenegro.017
			}
			show_as_tooltip = {
				add_modifier = {
					name = modifier_montenegrin_recognition
				}
			}
		}
		else = {
			trigger_event = {
				id = balkfm_montenegro.009 # The Light of Modernity
			}
		}
		set_variable = balkfm_prince_bishopric_completed
	}

	fail = {
		trigger_if = {
			limit = {
				has_dlc_feature = ip3_content # National Awakening
			}
			OR = {
				is_subject_of = c:TUR
				c:RUS ?= {
					relations:root < relations_threshold:cold
				}
			}
		}
	}

	on_fail = {
		if = {
			limit = {
				has_dlc_feature = ip3_content # National Awakening
			}
			if = {
				limit = {
					exists = c:MON
					c:RUS ?= {
						any_scope_treaty = {
							binds = c:MON
							any_scope_article = {
								has_type = guarantee_independence
							}
						}
					}
				}
				c:RUS ?= {
					random_scope_treaty = {
						limit = {
							binds = c:MON
							any_scope_article = {
								has_type = guarantee_independence
							}
						}
						withdraw = {
							country = c:RUS
						}
					}
				}
			}
			custom_tooltip = {
				text = mon_rus_aid_stop_tt
				remove_modifier = modifier_montenegrin_aid
			}
			mon_failure_events = yes
		}
	}

	invalid = {
		has_variable = balkfm_prince_bishopric_completed
	}

	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					has_dlc_feature = ip3_content # National Awakening
				}
				if = {
					limit = {
						var:rus_relations != c:RUS.relations:ROOT
					}
					set_variable = {
						name = rus_relations
						value = c:RUS.relations:ROOT
					}
					remove_modifier = modifier_montenegrin_aid
					add_modifier = {
						name = modifier_montenegrin_aid
						multiplier = russian_aid_income
					}
				}
				if = {
					limit = {
						NOR = {
							has_global_variable = eqfm_montenegrin_raiding_settled
							has_modifier = modifier_eqfm_need_tribal_leader
						}
						eqfm_has_black_mountain_expedition_trigger = no
					}
					add_modifier = modifier_eqfm_need_tribal_leader
				}
				else = {
					remove_modifier = modifier_eqfm_need_tribal_leader
				}
			}
			else = {
				remove_modifier = modifier_montenegrin_aid
				remove_modifier = modifier_eqfm_need_tribal_leader
			}
		}
	}

	status_desc = {
		triggered_desc = {
			trigger = {
				NOT = {
					has_dlc_feature = ip3_content # National Awakening
				}
			}
			desc = je_balkfm_mon_prince_bishopric_status
		}
		triggered_desc = {
			trigger = {
				has_dlc_feature = ip3_content # National Awakening
			}
			desc = je_montenegrin_russian_aid_income
		}
	}
}

		exists = c:AUS

			owner = c:AUS
		}
	}
	
	possible = {
		custom_tooltip = {
			text = je_balkfm_mon_austria_alliance_possible_1
			p:x90A0E0 = {
				owner = c:AUS
			}
		}
		custom_tooltip = {
			text = je_balkfm_mon_austria_alliance_possible_2
			has_global_variable = eqfm_montenegrin_raiding_settled
		}
		is_subject = no
		NOT = {
			has_war_with = c:AUS
		}
	}
	
	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		c:RUS ?= {
			save_scope_as = russia
		}
		c:TUR ?= {
			save_scope_as = ottoman
		}
		c:AUS = {
			save_scope_as = austria
		}
	}
	
	complete = {
		government_legitimacy > 50
		is_at_war = no
		c:RUS ?= {
			relations:root <= relations_threshold:poor
		}
		c:AUS = {
			relations:root >= relations_threshold:friendly
			is_at_war = no
		}
	}
	
	on_complete = {
		s:STATE_MONTENEGRO = {
			set_owner_of_provinces = {
				country = root
				provinces = { x90A0E0 }
			}
		}
		trigger_event = {
			id = mon_state_formation.4
			popup = yes
		}
		set_variable = turkfm_negotiated_kotor
	}
	
	fail = {
		OR = {
			trigger_if = {
				limit = {
					exists = c:TUR
				}
				is_subject_of = c:TUR
			}
			has_war_with = c:AUS
		}
	}
	
	on_fail = {
		c:AUS = {
			change_relations = {
				country = root
				value = -30
			}
			create_diplomatic_catalyst = {
				type = catalyst_event_negative
				target = root
			}
		}
	}

	invalid = {
		NOR = {
			exists = c:AUS
			p:x90A0E0 = {
				owner = c:AUS
			}
		}
	}

	status_desc = {
		desc = je_mon_austria_status_desc
	}
}

je_balkfm_mon_nationalism = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_montenegro2.dds"
	weight = 1000
	
	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no
	can_deactivate = no
	
	is_shown_in_lobby = {
		c:MON ?= this
	}
	
	is_shown_when_inactive = {
		c:MON ?= this
		NOT = {
			has_variable = serbian_montenegro
		}
	}
	
	possible = {
		has_technology_researched = nationalism
		custom_tooltip = {
			text = je_balkfm_mon_nationalism_possible_1
			NOR = {
				has_variable = balkfm_serbenegrotron_var
				has_variable = serbian_montenegro
			}
		}
		custom_tooltip = {
			text = je_balkfm_mon_nationalism_possible_2
			has_global_variable = balkfm_yugoslavism_started
		}
	}
	
	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		trigger_event = {
			id = balkfm_montenegro.200 # The Montenegrin Identity
		}
	}
	
	scripted_button = balkfm_montnegrin_identity_button
	
	complete = {
		OR = {
			has_law_or_variant = law_type:law_ethnostate
			has_law_or_variant = law_type:law_national_supremacy
		}
		owns_entire_state_region = STATE_MONTENEGRO
		custom_tooltip = {
			text = je_balkfm_mon_nationalism_complete_2
			capital ?= {
				state_region = s:STATE_MONTENEGRO
			}
		}
		has_modifier = modifier_balkfm_cultural_schism
		trigger_if = {
			limit = {
				has_dlc_feature = ip3_content
			}
			custom_tooltip = {
				text = je_balkfm_mon_nationalism_complete_1
				NOR = {
					has_variable = balkfm_serbenegrotron_var
					has_variable = serbian_montenegro
				}
			}
		}
	}
	
	on_complete = {
		set_global_variable = balkfm_montegrin_culture_var
		trigger_event = {
			id = balkfm_montenegro.209
		}
		show_as_tooltip = {
			remove_primary_culture = cu:serb
			add_primary_culture = cu:montenegrin
			s:STATE_MONTENEGRO = {
				add_homeland = cu:montenegrin
			}
			custom_tooltip = {
				text = je_balkfm_mon_nationalism_on_complete_1
				s:STATE_MONTENEGRO.region_state:MON = {
					every_scope_pop = {
						limit = {
							has_pop_culture = serb
						}
						change_pop_culture = {
							target = cu:montenegrin
							value = 0.9
						}
					}
				}
			}
			custom_tooltip = {
				text = je_balkfm_mon_nationalism_on_complete_2
				s:STATE_MONTENEGRO.region_state:MON = {
					add_radicals_in_state = {
						culture = cu:serb
						value = 1
					}
				}
			}
		}
	}
	
	fail = {
		OR = {
			custom_tooltip = {
				text = je_balkfm_mon_nationalism_fail_1
				c:YUG ?= ROOT
			}
			custom_tooltip = {
				text = je_balkfm_mon_nationalism_fail_2
				OR = {
					has_variable = balkfm_serbenegrotron_var
					has_variable = serbian_montenegro
					c:SER ?= this
					c:SER_SRP ?= this # I decided to be cheeky and this is totally possible with very careful sequencing
				}
			}
			has_law_or_variant = law_type:law_multicultural
		}
	}
}
