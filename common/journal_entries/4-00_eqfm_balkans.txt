je_eqfm_balkan_situation = { #
	group = je_group_global_international_situations
	icon = "gfx/interface/icons/event_icons/event_ottoman.dds"
	weight = 10000

	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no
	can_deactivate = no

	is_shown_in_lobby = {
		c:TUR ?= this
	}

	scripted_progress_bar = eqfm_balkans_escalation_progress_bar
	scripted_progress_bar = eqfm_balkans_deescalation_progress_bar
	scripted_progress_bar = eqfm_crisis_progress_bar

	scripted_button = eqfm_accelerate_tensions_button
	scripted_button = eqfm_decelerate_tensions_button
	scripted_button = eqfm_declare_disloyalty_button
	scripted_button = eqfm_declare_loyalty_button
	scripted_button = eqfm_greek_gunrunners_button

	is_shown_when_inactive = {
		custom_tooltip = {
			text = eqfm_balkan_situation_resolved_NOT
			NOT = {
				has_global_variable = eqfm_balkan_situation_resolved
			}
		}
	}

	should_be_involved = {
		eqfm_balkans_involved_trigger = yes
	}

	should_show_when_not_involved = {
		OR = {
			eqfm_balkans_interloper_trigger = yes
			is_country_type = recognized
			is_country_type = colonial
		}
	}

	possible = {
		custom_tooltip = {
			text = eqfm_balkan_situation_resolved_NOT
			NOT = {
				has_global_variable = eqfm_balkan_situation_resolved
			}
		}
	}

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		eqfm_initialize_balkan_variables_effect = yes
		c:TUR ?= {
			save_scope_as = ottoman_scope
			add_modifier = modifier_eqfm_stable_phase
		}
	}

	immediate_all_involved = {
		trigger_event = {
			id = eqfm_balkans.011 # The Eastern Question
			days = 5
		}
	}
	
	on_become_involved_after_activation = {
		if = {
			limit = {
				NOT = {
					has_variable = eqfm_eqfm_started
				}
			}
			trigger_event = {
				id = eqfm_balkans.011 # The Eastern Question
			}
		}
		else = {
			post_notification = notitication_eqfm_balkans_involved
		}
	}

	modifiers_while_active = {
	}

	complete = {
		c:TUR ?= {
			OR = {
				custom_tooltip = {
					text = je_eqfm_balkan_situation_complete
					trigger_if = {
						limit = {
							any_scope_state = {
								state_region = {
									is_homeland = cu:greek
								}
							}
						}
						country_average_cultural_acceptance = {
							target = cu:greek
							value >= acceptance_status_5
						}
					}
					trigger_if = {
						limit = {
							any_scope_state = {
								state_region = {
									is_homeland = cu:rhomaios
								}
							}
						}
						country_average_cultural_acceptance = {
							target = cu:rhomaios
							value >= acceptance_status_5
						}
					}
					trigger_if = {
						limit = {
							any_scope_state = {
								state_region = {
									is_homeland = cu:serb
								}
							}
						}
						country_average_cultural_acceptance = {
							target = cu:serb
							value >= acceptance_status_5
						}
					}
					trigger_if = {
						limit = {
							any_scope_state = {
								state_region = {
									is_homeland = cu:bulgarian
								}
							}
						}
						country_average_cultural_acceptance = {
							target = cu:bulgarian
							value >= acceptance_status_5
						}
					}
					trigger_if = {
						limit = {
							any_scope_state = {
								state_region = {
									is_homeland = cu:romanian
								}
							}
						}
						country_average_cultural_acceptance = {
							target = cu:romanian
							value >= acceptance_status_5
						}
					}
					trigger_if = {
						limit = {
							any_scope_state = {
								state_region = {
									is_homeland = cu:bosniak
								}
							}
						}
						country_average_cultural_acceptance = {
							target = cu:bosniak
							value >= acceptance_status_5
						}
					}
					trigger_if = {
						limit = {
							any_scope_state = {
								state_region = {
									is_homeland = cu:albanian
								}
							}
						}
						country_average_cultural_acceptance = {
							target = cu:albanian
							value >= acceptance_status_5
						}
					}
					trigger_if = {
						limit = {
							any_scope_state = {
								state_region = {
									is_homeland = cu:croat
								}
							}
						}
						country_average_cultural_acceptance = {
							target = cu:croat
							value >= acceptance_status_5
						}
					}
					trigger_if = {
						limit = {
							any_scope_state = {
								state_region = {
									is_homeland = cu:slovene
								}
							}
						}
						country_average_cultural_acceptance = {
							target = cu:slovene
							value >= acceptance_status_5
						}
					}
				}
				country_has_primary_culture = cu:ottoman
			}
			legitimacy >= 80
			radical_fraction = {
				value < 0.20
			}
			NOT = {
				any_interest_group = {
					OR = {
						is_insurrectionary = yes
						is_revolutionary = yes
					}
				}
			}
			is_at_war = no
		}
	}

	on_complete = {
		set_global_variable = eqfm_separatism_complete
		eqfm_cleanup_tension_main_effect = yes
	}
	
	on_complete_all_involved = {
		eqfm_cleanup_tension_local_effect = yes
		trigger_event = {
			id = eqfm_balkans.009 # Dominion Over Rumelia
		}
	}
	
	fail = {
		c:TUR ?= {
			OR = {
				NOT = {
					any_scope_state = {
						region = sr:region_balkans
					}
				}
				AND = {
					custom_tooltip = {
						text = turkfm_sick_man_failed
						has_global_variable = turkfm_sick_man_failed
					}
					custom_tooltip = {
						text = turkfm_tanzimat_failed
						has_global_variable = turkfm_tanzimat_failed
					}
				}
			}
			hidden_trigger = {
				# Sanity triggers in case a revolution/secession steals the last state in the Balkans
				has_any_secessionists_broken_out = no
				has_any_secessionists_growing = no
				is_at_war = no
				has_revolution = no
			}
		}
	}

	on_fail = {
		set_global_variable = eqfm_separatism_complete
		eqfm_cleanup_tension_main_effect = yes
	}
	
	on_fail_all_involved = {
		eqfm_cleanup_tension_local_effect = yes
		trigger_event = {
			id = eqfm_balkans.008 # Rumelia Lost
		}
	}

	invalid = {
		custom_tooltip = {
			text = je_eqfm_balkan_situation_invalid
			NOT = {
				exists = c:TUR
			}
		}
	}

	on_invalid = {
		set_global_variable = eqfm_separatism_complete
		eqfm_cleanup_tension_main_effect = yes
	}
	
	on_invalid_all_involved = {
		eqfm_cleanup_tension_local_effect = yes
	}
	
	on_no_longer_involved = {
		eqfm_cleanup_tension_local_effect = yes
		post_notification = notification_eqfm_balkans_not_involved
	}

	on_weekly_pulse = {
		effect = {
			eqfm_calculate_balkan_situation_variables_effect = yes
			eqfm_check_balkan_situation_phase_effect = yes
#			eqfm_update_balkan_situation_lists_effect = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			eqfm_validate_journal_involvement = yes
			eqfm_advance_balkan_situation_variables_effect = yes
		}
	}

	status_desc = {
		triggered_desc = {
			desc = je_eqfm_balkan_situation_status
			trigger = {
				has_global_variable = eqfm_balkans_phase_var
			}
		}
	}
}
