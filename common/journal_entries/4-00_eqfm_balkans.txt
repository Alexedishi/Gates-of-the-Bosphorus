je_eqfm_balkan_situation_riots = {
	group = je_group_internal_affairs

	weight = 500
	
	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no
	
	scripted_progress_bar = eqfm_riot_progress_bar

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		set_variable = {
			name = eqfm_riot_var
			value = 50
		}
		set_variable = {
			name = eqfm_riot_progress_var
			value = 0
		}
	}

	complete = {
		OR = {
			custom_tooltip = {
				text = je_eqfm_balkan_situation_riots_invalid
				OR = {
					NOT = {
						has_global_variable = eqfm_balkans_phase_var
					}
					global_var:eqfm_balkans_phase_var != flag:eqfm_hostility
				}
			}
			NOR = {
				any_scope_state = {
					has_modifier = modifier_eqfm_race_riots
				}
				custom_tooltip = {
					text = je_eqfm_balkan_situation_riots_complete
					any_country = {
						is_diplomatic_play_enemy_of = root
						has_modifier = modifier_eqfm_nationalist_uprising
					}
				}
				hidden_trigger = {
					NOT = {
						has_variable = eqfm_recent_revolt_var
					}
				}
			}
		}
	}

	on_complete = { # Set variables for deescalation and call eqfm_check_balkan_situation_phase_effect
		eqfm_add_extremely_large_deescalation = yes
		eqfm_cleanup_riots_effect = yes
	}

	on_weekly_pulse = {
		events = {
			eqfm_balkans.007 # Riots in State Suppressed
		}
		effect = {
			eqfm_update_riot_variables_effect = yes
			eqfm_update_riot_modifiers_effect = yes
			change_variable = {
				name = eqfm_riot_var
				add = var:eqfm_riot_progress_var
			}
			clamp_variable = {
				name = eqfm_riot_var
				max = 100
				min = 0
			}
			je:je_eqfm_balkan_situation_riots = {
				set_bar_progress = {
					value = owner.var:eqfm_riot_var
					name = eqfm_riot_progress_bar
				}
			}
		}
	}

	on_monthly_pulse = {
		random_events = {
			5 = eqfm_balkans.140 # Fortress in State Beseiged

je_eqfm_eastern_question = {
	group = je_group_global_international_situations
	icon = "gfx/interface/icons/event_icons/event_ottoman.dds"
	weight = 100
	
	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no
	
	scripted_progress_bar = eqfm_balkfm_escalation_progress_bar
	scripted_progress_bar = eqfm_balkfm_deescalation_progress_bar

	scripted_button = eqfm_accelerate_tensions_button
	scripted_button = eqfm_decelerate_tensions_button
	scripted_button = eqfm_declare_disloyalty_button
	scripted_button = eqfm_declare_loyalty_button
	scripted_button = eqfm_greek_gunrunners_button
	
	should_be_involved = {
		always = no
	}
	
	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		eqfm_initialize_eastern_question_effect = yes
		if = {
			limit = {
				NOT = {
					has_variable = eqfm_eqfm_started
				}
			}
			if = {
				limit = {
					c:TUR ?= this
				}
				trigger_event = {
					id = eqfm_balkans.001 # The Peoples of the Empire
					days = 1
					popup = yes
				}
			}
			else = {
				trigger_event = {
					id = eqfm_balkans.011 # The Eastern Question
					days = 1
					popup = yes
				}
			}
		}
		else = {
			set_variable = eqfm_eqfm_started
		}
	}
	
	complete = {
		custom_tooltip = {
			text = eqfm_separatism_completed_third
			has_global_variable = eqfm_separatism_complete
		}
	}
	
	on_complete = {
		eqfm_cleanup_eqfm_effect = yes
	}
	
	fail = {
		custom_tooltip = {
			text = turkfm_separatism_failed_third
			NOR = {
				exists = c:TUR
				c:TUR = {
					has_journal_entry = je_eqfm_balkan_situation_main
				}
			}
		}
	}
	
	on_fail = {
		eqfm_cleanup_eqfm_effect = yes
	}
	
	on_weekly_pulse = {
		events = {
			eqfm_balkans.190 # Ottoman Humiliation Monitor Event
			eqfm_balkans.191 # Ottoman Rivalry Monitor Event
			eqfm_balkans.192 # Ottoman Rivalry Monitor Event
		}
		effect = {
			eqfm_update_balkan_situation_progressbar_effect = yes
		}
	}
	
	on_monthly_pulse = {
		effect = {
			eqfm_update_balkan_situation_lists_effect = yes
		}
		random_events = {
			500 = 0
			5 = eqfm_balkans.182 # Diplomat Offers Mediation
		}
	}
	
	status_desc = {
		triggered_desc = {
			desc = eqfm_tension_status
			trigger = {
				has_global_variable = eqfm_balkans_phase_var
			}
		}
	}
}

je_eqfm_balkan_situation_crisis = {
	group = je_group_crises
	icon = "gfx/interface/icons/event_icons/event_ottoman.dds"
	weight = 100000

	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	transferable = no

	is_shown_in_lobby = {
		has_dlc_feature = ip3_content # National Awakening
		c:TUR ?= this
	}
	
	scripted_progress_bar = eqfm_crisis_progress_bar

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
		eqfm_initialize_crisis_effect = yes
	}

	complete = {
		custom_tooltip = {
			text = je_eqfm_balkan_situation_crisis_complete
			global_var:eqfm_crisis_var >= 100
		}
		has_any_secessionists_broken_out = no
		has_any_secessionists_growing = no
		is_at_war = no
	}

	on_complete = {
		trigger_event = {
			id = eqfm_balkans.029 # Order Returns to Rumelia
		}
		hidden_effect = {
			every_country = {
				limit = {
					has_journal_entry = je_eqfm_eastern_question
				}
				trigger_event = {
					id = eqfm_balkans.029 # Order Returns to Rumelia
				}
			}
		}
		eqfm_cleanup_crisis_effect = yes
		eqfm_force_deconfliction_effect = yes
	}

	fail = {
		custom_tooltip = {
			text = je_eqfm_balkan_situation_crisis_complete
			global_var:eqfm_crisis_var <= 0
		}
	}

	on_fail = {
		trigger_event = {
			id = eqfm_balkans.028 # Rumelia in Disarray
		}
		hidden_effect = {
			every_country = {
				limit = {
					has_journal_entry = je_eqfm_eastern_question
				}
				trigger_event = {
					id = eqfm_balkans.028 # Rumelia in Disarray
				}
			}
		}
		eqfm_cleanup_crisis_effect = yes
		eqfm_force_deconfliction_effect = yes
	}

	invalid = {
		OR = {
			NOT = {
				has_global_variable = eqfm_balkans_phase_var
			}
			AND = {
				has_global_variable = eqfm_balkans_phase_var
				global_var:eqfm_balkans_phase_var != flag:eqfm_crisis
			}
			has_global_variable = eqfm_separatism_complete
		}
	}

	on_invalid = {
		eqfm_cleanup_crisis_effect = yes
		if = {
			limit = {
				NOT = {
					has_global_variable = eqfm_separatism_complete
				}
			}
			eqfm_force_deconfliction_effect = yes
		}
	}

	on_weekly_pulse = {
		effect = {
			set_global_variable = {
				name = eqfm_crisis_progress_var
				value = eqfm_crisis_progress_value
			}
			change_global_variable = {
				name = eqfm_crisis_duration_var
				add = 1
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			change_global_variable = {
				name = eqfm_crisis_var
				add = global_var:eqfm_crisis_progress_var
			}
			clamp_global_variable = {
				name = eqfm_crisis_var
				max = 250
				min = 0
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_global_variable = eqfm_crisis_var
				}
				desc = je_eqfm_balkan_situation_crisis_status
			}
		}
	}
}
