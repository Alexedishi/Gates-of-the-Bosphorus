je_grefm_ion_xenocracy = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	dlc = dlc_grefm
	weight = 1000
	
	activation_description = {
		always = yes
	}

	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	transferable = no

	is_shown_in_lobby = {
		c:ION ?= this
	}

	modifiers_while_active = {
		modifier_grefm_xenocracy
		modifier_grefm_hepantese_administration
	}
	
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = gold_positive
		
		flags = { historical }
		
		trigger = {
			AND = {
				any_scope_building = {
					is_building_type = building_government_administration
					OR = {
						has_active_production_method = pm_vertical_filing_cabinets
						has_active_production_method = pm_switch_boards
					}
				}
				bureaucracy > 0
				approaching_bureaucracy_shortage = no
			}
			institution_investment_level = {
				institution = institution_police
				value >= 3
			}
			institution_investment_level = {
				institution = institution_schools
				value >= 3
			}
		}
		effect = {
			trigger_event = grefm_heptanese.002
		}
	}
	
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = blue_negative
		
		trigger = {
			OR = {
				is_subject = no
				trigger_if = {
					limit = {
						exists = c:GRE
					}
					is_subject_of = c:GRE
				}
			}
		}
		effect = { }
	}

	on_monthly_pulse = {
		random_events = {
			100 = 0
			5 = grefm_heptanese.005 # The Ionian Academy
		}
	}
}
