je_turkfm_the_black_mountain = {
	# This journal integrates je_montenegrin_raiding with GotB content
	group = je_group_turkfm_sick_man
	icon = "gfx/interface/icons/alert_icons/secession.dds"
	weight = 10000

	can_revolution_inherit = yes
	transferable = no
	should_be_pinned_by_default = yes

	is_shown_in_lobby = {
		OR = {
			c:MON ?= this
			c:TUR ?= this
		}
	}

	immediate = {
		add_modifier = modifier_turkfm_practicing_headtaking

	# Montenegrin Buttons
	scripted_button = turkfm_increase_raiding_button

	# Ottoman Buttons
	scripted_button = turkfm_assign_border_guard_button
	scripted_button = turkfm_recall_black_mountain_expedition_button
	
	# Shared Buttons
	scripted_button = turkfm_renounce_headtaking_button

	complete = {
		## MON Completion Trigger
		## IP3's functionality is a sub-set of this journal; separate GotB scripting handles the resolution of raiding if IP3 is active
		trigger_if = {
			limit = {
				c:MON ?= this
			}
			owns_entire_state_region = STATE_MONTENEGRO
			trigger_if = {
				limit = {
					exists = c:TUR
				}
				NOT = {
					is_subject_of = c:TUR
				}
			}
		}
		## TUR Completion Triggers
		trigger_if = { # These are the pre-IP3 GotB triggers
			limit = {
				c:TUR ?= this
				NOT = {
					has_dlc_feature = ip3_content # National Awakening
				}
			}
			c:MON ?= {
				OR = {
					NOT = {
						has_state_in_state_region = STATE_MONTENEGRO
					}
				}
				is_subject_of = root
			}
		}
		trigger_if = { # These are the imported triggers from IP3
			limit = {
				c:TUR ?= this
				has_dlc_feature = ip3_content # National Awakening
			}
			OR = {
				NOT = {
					exists = c:MON
				}
				c:MON ?= {
					OR = {
						is_country_alive = no
						is_subject_of = ROOT
					}
				}
			}
		}
	}

	on_complete = {
		if = { # David Slays Goliath moved to event
			limit = {
				c:MON ?= this
			}
			trigger_event = {
				id = turkfm_expeditions.009 # The Lands of Montenegro
			}
		}
		else_if = {
			limit = {
				c:TUR ?= this
				NOT = {
					has_dlc_feature = ip3_content # National Awakening
				}
			}
			trigger_event = {
				id = turkfm_expeditions.007 # The Black Mountain Subdued
			}
		}
		else_if = {
			limit = {
				c:TUR ?= this
				has_dlc_feature = ip3_content # National Awakening
			}
			trigger_event = {
				id = mon_state_formation.14
				popup = yes
			}
			hidden_effect = {
				add_modifier = {
					name = goliath_slays_david_modifier
				}
			}
			add_modifier = {
				name = put_down_rebels_modifier
				days = very_long_modifier_time
				is_decaying = yes
			}
			add_fervor_target_modifier = {
				culture = cu:serb
				days = stupidly_long_modifier_time
				multiplier = -20
			}
		}
		turkfm_cleanup_black_mountain_effect = yes
		turkfm_update_expedition_costs_effect = yes
	}

	fail = {
		OR = {
			NOT = { # Pre-IP3 fail block, retained in IP3
				has_state_in_state_region = STATE_MONTENEGRO
			}
			trigger_if = { # Imported IP3 fail trigger
				limit = {
					has_dlc_feature = ip3_content # National Awakening
					c:MON ?= this
					exists = c:TUR
				}
				is_subject_of = c:TUR
			}
			trigger_if = { # Alternate fail trigger, present in IP3 only
				limit = {
					c:TUR ?= this
					has_dlc_feature = ip3_content # National Awakening
				}
				NOT = {
					has_claim = s:STATE_MONTENEGRO
				}
			}
		}
	}

	on_fail = {
		if = {
			limit = {
				c:MON ?= this
				NOT = {
					has_dlc_feature = ip3_content # National Awakening
				}
			}
			trigger_event = {
				id = turkfm_expeditions.008 # International Indifference
			}
		}
		else_if = {
			limit = {
				c:MON ?= this
				has_dlc_feature = ip3_content # National Awakening
			}
		}
		else_if = {
			limit = {
				c:TUR ?= this
				NOT = {
					has_dlc_feature = ip3_content # National Awakening
				}
			}
			trigger_event = {
				id = turkfm_expeditions.006 # The Black Mountain Lost
			}
		}
		else_if = {
			limit = {
				c:TUR ?= this
				has_dlc_feature = ip3_content # National Awakening
			}
		}
		turkfm_cleanup_black_mountain_effect = yes
		turkfm_update_expedition_costs_effect = yes
	}

	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					has_dlc_feature = ip3_content # National Awakening
					NOT = {
						has_global_variable = turkfm_montenegrin_raiding_settled
					}
					c:MON ?= this
				}

				turkfm_update_black_mountain_characters_effect = yes
				turkfm_update_expedition_costs_effect = yes
			}
		}
		events = {
			turkfm_expeditions.023 # Expeditionary Force Withdraws from the Border

		}

	}

	immediate = {
		if = {
			limit = {
				NOT = {
					has_global_variable = turkfm_pirates_var

			}
			set_global_variable = {
				name = turkfm_pirates_var

				}
