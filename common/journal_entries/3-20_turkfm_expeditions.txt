je_turkfm_black_mountain = { # This is a temporary journal... je_turkfm_karadag returns to development hell
	icon = "gfx/interface/icons/event_icons/event_montenegro.dds"

	group = je_group_turkfm_sick_man

	should_be_pinned_by_default = yes

	immediate = {
		add_modifier = modifier_turkfm_practicing_headtaking
	}

	scripted_button = turkfm_renounce_headtaking_button
	scripted_button = turkfm_negociate_kotor_button

	complete = {
		trigger_if = {
			limit = {
				c:TUR ?= this
			}
			c:MON ?= {
				NOT = {
					has_state_in_state_region = STATE_MONTENEGRO
				}
			}
		}
		trigger_else = {
			c:TUR ?= {
				NOT = {
					has_state_in_state_region = STATE_MONTENEGRO
				}
			}
		}
	}

	on_complete = {
		if = {
			limit = {
				c:TUR ?= this
			}
			trigger_event = {
				id = turkfm_expeditions.007 # The Black Mountain Subdued
			}
		}
		else_if = {
			limit = {
				c:MON ?= this
			}
			trigger_event = {
				id = turkfm_expeditions.009 # The Lands of Montenegro
			}
		}
		turkfm_cleanup_karadag_effect = yes
	}

	fail = {
		NOT = {
			has_state_in_state_region = STATE_MONTENEGRO
		}
	}

	on_fail = {
		if = {
			limit = {
				c:TUR ?= this
			}
			trigger_event = {
				id = turkfm_expeditions.006 # The Black Mountain Lost
			}
		}
		else_if = {
			limit = {
				c:MON ?= this
			}
			trigger_event = {
				id = turkfm_expeditions.008 # International Indifference
			}
		}
		turkfm_cleanup_karadag_effect = yes
	}

	invalid = {
		NAND = {
			exists = c:TUR
			exists = c:MON
		}
	}

	on_invalid = {
		turkfm_cleanup_karadag_effect = yes
	}

	weight = 1000

}
