	on_monthly_pulse = {
		effect = {
			# Validate journal involvement
			eqfm_update_sanusiyyah_journal_involvement_effect = yes
			
			# Fire on_actions for all involved countries
			scope:journal_entry = {
				every_scope_je_involved = {
					trigger_event = {
						on_action = eqfm_sanusiyyah_involved_monthly_pulse
					}
				}
			}
			
			# Update Desert Scouting Progress so it can contribute to entrenchment math
			scope:journal_entry = {
				eqfm_monthly_reconnaissance_progress_effect = yes
			}
			
			# Calculate and advance state region entrenchment
			eqfm_update_sanusiyya_georegion_variables_effect = yes
			eqfm_monthly_entrenchment_effect = yes
			scope:journal_entry = {
				com_change_and_clamp_variable = {
					name = eqfm_sanusiyyah_pressure_var
					value = var:eqfm_sanusiyyah_pressure_progress_var
					max = 100
					min = 0
				}
				set_bar_progress = {
					name = eqfm_sanusiyyah_pressure_progress_bar
					value = var:eqfm_sanusiyyah_pressure_var
				}
			}
			
			# Evaluate global events
			if = {
				limit = {
					scope:journal_entry = {
						NOT = {
							has_variable = eqfm_intimidated_sensussi
						}
						var:eqfm_sanusiyyah_pressure_var >= 50
					}
					s:STATE_TRIPOLI = {
						has_variable = eqfm_sanusiyyah_entrenchment_var
					}
				}
				s:STATE_TRIPOLI = {
					remove_variable = eqfm_sanusiyyah_entrenchment_var
				}
				s:STATE_LIBYA = {
					set_variable = {
						name = eqfm_sanusiyyah_entrenchment_var
						value = 25
					}
				}
				scope:journal_entry = {
					set_variable = eqfm_intimidated_sensussi
					every_scope_je_involved = {
						trigger_event = eqfm_sanusiyyah.004
					}
				}
			}
			if = {
				limit = {
					scope:journal_entry = {
						NOT = {
							has_variable = eqfm_senussi_brokered_peace
						}
					}
					exists = c:ZWY
					exists = c:TBI
					s:STATE_LIBYA = {
						has_variable = eqfm_sanusiyyah_entrenchment_var
						var:eqfm_sanusiyyah_entrenchment_var >= 50
					}
					s:STATE_CHAD = {
						NOR = {
							has_variable = eqfm_sanusiyyah_entrenchment_var
							has_variable = eqfm_sanusiyyah_explusion_var
						}
					}
				}
				s:STATE_CHAD = {
					set_variable = {
						name = eqfm_sanusiyyah_entrenchment_var
						value = 25
					}
				}
				scope:journal_entry = {
					set_variable = eqfm_senussi_brokered_peace
					every_scope_je_involved = {
						trigger_event = eqfm_sanusiyyah.007
					}
				}
			}
		}
	}
