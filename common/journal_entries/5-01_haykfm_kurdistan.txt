je_haykfm_kur_tribes = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_kurdish.dds"
	dlc = dlc_grefm
	weight = 2000
	
	activation_description = {
		always = yes
	}
	
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	transferable = no
	
	is_shown_in_lobby = {
		country_has_primary_culture = cu:kurdish
	}
	
	scripted_button = haykfm_claim_miri_miran_button
	
	modifiers_while_active = {
		modifier_haykfm_informal_administration
	}
	
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = gold_positive
		
		trigger = {
			army_size >= 20
			NOR = {
				has_law_or_variant = law_type:law_hereditary_bureaucrats
				has_law_or_variant = law_type:law_peasant_levies
				has_law_or_variant = law_type:law_no_police
			}
			any_scope_building = {
				is_building_type = building_university
				occupancy > 0.9
			}
			any_scope_building = {
				is_building_type = building_government_administration
				NOT = {
					has_active_production_method = pm_simple_organization
				}
				occupancy >= 0.9
			}
			bureaucracy >= 0
			approaching_bureaucracy_shortage = no
			ig:ig_intelligentsia = {
				is_marginal = no
			}
		}
		effect = {
			trigger_event = haykfm_kurdistan.002
			remove_modifier = modifier_haykfm_tribal_tradition
		}
	}
}
