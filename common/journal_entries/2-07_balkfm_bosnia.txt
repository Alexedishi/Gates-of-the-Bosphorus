je_balkfm_bos_ali_pasha = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	weight = 1000

	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	transferable = no

	is_shown_in_lobby = {
		c:BOS ?= this
	}

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
	}

	modifiers_while_active = {
		modifier_balkfm_ali_pasha
	}

	complete = {
		trigger_if = {
			limit = {
				exists = c:TUR
			}
			NOT = {
				is_subject_of = c:TUR
			}
		}
		trigger_else = {
			is_subject = no
		}
	}

	on_complete = {
		trigger_event = balkfm_bosnia.002
		c:TUR ?= {
			post_notification = notification_balkfm_ali_rizvanbegovic_ascendant
		}
		c:MON ?= {
			post_notification = notification_balkfm_ali_rizvanbegovic_ascendant
		}
	}

	fail = {
		OR = {
			custom_tooltip = {
				text = je_balkfm_bos_ali_pasha_fail
				ruler = {
					NOR = {
						has_template = balkfm_rizvanbegovic_ali
						has_template = balkfm_rizvanbegovic_mehmed
					}
				}
			}
			NOT = {
				has_claim = s:STATE_MONTENEGRO
			}
		}
		hidden_trigger = {
			NOT = {
				has_global_variable = eqfm_ali_rizvanbegovic_waxed
			}
		}
	}

	on_fail = {
		trigger_event = balkfm_bosnia.003
		c:TUR ?= {
			post_notification = notification_balkfm_ali_rizvanbegovic_waxed
		}
		c:MON ?= {
			post_notification = notification_balkfm_ali_rizvanbegovic_waxed
		}
		set_global_variable = eqfm_ali_rizvanbegovic_waxed
	}
	
	invalid = {
		has_global_variable = eqfm_ali_rizvanbegovic_waxed
	}
}
