	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	weight = 1000

	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	transferable = no

	is_shown_in_lobby = {
		c:BOS ?= this
	}

	immediate = {
		gbbf_assign_dlc_icon_effect = yes
	}

	modifiers_while_active = {
		modifier_balkfm_ali_pasha
	}

	complete = {
		is_subject = no
	}

	on_complete = {
		trigger_event = balkfm_bosnia.009
		hidden_effect = {
			c:TUR ?= {
				trigger_event = balkfm_bosnia.006
			}
			c:MON ?= {
				trigger_event = balkfm_bosnia.006
			}
		}
		set_global_variable = turkfm_ali_pasha_gigachad
	}

	fail = {
		custom_tooltip = {
			text = je_balkfm_bos_ali_pasha_fail
			ruler = {
				NOR = {
					has_template = balkfm_rizvanbegovic_ali
					has_template = balkfm_rizvanbegovic_mehmed
				}
			}
		}
	}

	on_fail = {
		trigger_event = balkfm_bosnia.008
		set_global_variable = balkfm_ali_pasha_waxed
	}
}
