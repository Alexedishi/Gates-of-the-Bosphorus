je_balkfm_bos_ali_pasha = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_balkfm_bosnia

	should_be_pinned_by_default = yes

	modifiers_while_active = {
		modifier_balkfm_ali_pasha
	}

	complete = {
		is_subject = no
	}

	on_complete = {
		trigger_event = {
			id = balkfm_bosnia.009 # House Rizvanbegović of Bosnia
		}
		hidden_effect = {
			c:TUR ?= {
				trigger_event = {
					id = balkfm_bosnia.006 # Ali Pasha Ascendant
				}
			}
			c:MON ?= {
				trigger_event = {
					id = balkfm_bosnia.006 # Ali Pasha Ascendant
				}
			}
		}
		set_global_variable = balkfm_ali_pasha_waxed
	}

	fail = {
		custom_tooltip = {
			text = je_balkfm_bos_ali_pasha_fail
			ruler = {
				NOT = {
					has_template = template_balkfm_rizvanbegovic_ali
				}
			}
		}
	}

	on_fail = {
		trigger_event = {
			id = balkfm_bosnia.008 # Failed Amibitions
		}
		set_global_variable = balkfm_ali_pasha_waxed
	}

	weight = 500

}
