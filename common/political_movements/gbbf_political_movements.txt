		ideology_humanitarian
		ideology_humanitarian_royalist
		ideology_social_democrat

	creation_trigger = {
		exists = c:TUR
		OR = {
			c:TUR = this
			civil_war_origin_country ?= c:TUR
		}
		NOT = {
			has_global_variable = turkfm_tanzimat_complete
		}

	creation_weight = {
		value = 100

		add_movement_enthusiasm_modifier = yes
	}

		has_global_variable = turkfm_tanzimat_complete

		OR = {
			has_ideology = ideology:ideology_reformer

			has_ideology = ideology:ideology_humanitarian_royalist
			has_ideology = ideology:ideology_social_democrat
			has_ideology = ideology:ideology_gbbf_ottomanist_leader
			AND = {
				law_stance = {
					law = law_type:law_autocracy
					value < neutral
				}
				law_stance = {
					law = law_type:law_ethnostate
					value < neutral
				}
				law_stance = {
					law = law_type:law_millet_system

				}

		}

	character_support_weight = {
		value = 200
	}

		NOT = {
			is_pop_type = slaves
		}
	}

		movement_support_capitalists

		movement_support_academics
		movement_support_bureaucrats
		movement_support_clerks
		movement_support_peasants

			value = 3
			desc = "POP_BASE_SUPPORT"
		}

			desc = "URBAN_STATE"
			value = 10
			multiply = state.state_urbanization_rate
		}

		if = {
			limit = {
				strata = upper
			}

					is_pop_type = capitalists

					value = 0

				}
			}

					value = 9
					desc = "UPPER_NO_ICON"
				}
			}
		}

			limit = {
				strata = middle
			}

				limit = {
					is_pop_type = shopkeepers
				}
				add = {
					value = 6
					desc = "POP_SHOPKEEPERS"
				}

			else_if = {
				limit = {
					is_pop_type = academics

movement_gbbf_symphonia = {
	category = movement_category_ideological
	
	ideology = ideology_gbbf_symphonist
	character_ideologies = {
		ideology_gbbf_symphonist_leader
		ideology_royalist
		ideology_theocrat
		ideology_traditionalist
		ideology_integralist
	}
	
	creation_trigger = {
		c:BYZ ?= this
		has_law_or_variant = law_type:law_state_religion
	}
	creation_weight = {
		value = 100
	}
	on_created = {
		owner = {
			if = {
				limit = {
					NOT = {
						has_variable = grefm_byzantine_symphonia_started
					}
				}
#				trigger_event = {
#					id = grefm_byzantium.030
#				}
				set_variable = grefm_byzantine_symphonia_started
			}
		}
	}
	
	disband_trigger = {
		scope:political_movement ?= {
			religion = {
				NOT = { is_state_religion = root }
			}
		}
	}
	
	religion_selection_trigger = {
		is_state_religion = scope:country
	}
	religion_selection_weight = {
		value = 100
	}
	
	character_support_trigger = {
		religion = scope:religion
		OR = {
			is_interest_group_type = ig_devout
			has_ideology = ideology:ideology_traditionalist
			has_ideology = ideology:ideology_theocrat
			has_ideology = ideology:ideology_integralist
		}
	}
	character_support_weight = {
		value = 100
		if = {
			limit = {
				scope:interest_group ?= {
					is_interest_group_type = ig_devout
				}
			}
			multiply = {
				value = 2
			}
		}
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_traditionalist
					has_ideology = ideology:ideology_theocrat
				}
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				has_ideology = ideology:ideology_integralist
			}
			multiply = {
				value = 2
			}
		}
	}
	
	pop_support_trigger = {
		religion ?= scope:religion
	}
	
	pop_support_factors = {
		movement_support_low_urbanization
		movement_support_low_literacy
		movement_support_clergymen
		movement_support_aristocrats
		movement_support_peasants
		movement_support_corporatism_tech
	}
	
	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
					OR = {
						is_being_suppressed = yes
						is_being_bolstered = yes
					}
				}
			}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}
		
		add = {
			desc = "URBAN_STATE" # Increases support in rural states, decreases in urban
			value = 0.5
			subtract = state.state_urbanization_rate
			multiply = 10
		}
		
		add = {
			value = 10
			desc = "POP_BASE_SUPPORT"
		}
		
		if = {
			limit = {
				is_pop_type = clergymen
			}
			add = {
				value = 45
				desc = "POP_CLERGYMEN"
			}
		}
		else_if = {
			limit = {
				is_pop_type = aristocrats
			}
			add = {
				value = 15
				desc = "POP_ARISTOCRATS"
			}
		}
		else_if = {
			limit = {
				is_pop_type = peasants
			}
			add = {
				value = 10
				desc = "POP_PEASANTS"
			}
		}
		
		if = {
			limit = {
				NOT = {
					is_pop_type = clergymen
				}
			}
			add = { # Weakened by increasing literacy rates
				desc = "POP_LITERACY"
				this = {
					value = literacy_rate
					multiply = -2
				}
			}
		}
		
		if = { # Discriminated pops are less inclined to support the state church
			limit = {
				pop_acceptance < acceptance_status_5
			}
			multiply = {
				desc = "POP_ACCEPTANCE"
				value = pop_acceptance
				divide = 100
				add = 0.25
			}
		}
		
		if = {
			limit = {
				owner ?= {
					has_technology_researched = corporatism
				}
			}
			multiply = {
				value = 1.5
				desc = "CORPORATISM_TECH"
			}
		}
		
		if = {
			limit = {
				owner ?= {
					any_political_movement = {
						is_political_movement_type = movement_legitimist
					}
				}
			}
			multiply = {
				value = 0.25
			}
		}
	}
	
	revolution = {
		weight = {
			value = 100
		}
		
		interest_group_can_join = {
			leader = { religion = scope:political_movement.religion }
		}
		
		state_weight = {
			value = 1
			
			if = {
				limit = {
					religion_percent_state = {
						target = scope:political_movement.religion
						value >= 0.75
					}
				}
				add = 75
			}
			else_if = {
				limit = {
					religion_percent_state = {
						target = scope:political_movement.religion
						value >= 0.5
					}
				}
				add = 50
			}
			else_if = {
				limit = {
					religion_percent_state = {
						target = scope:political_movement.religion
						value >= 0.25
					}
				}
				add = 25
			}
			else_if = {
				limit = {
					religion_percent_state = {
						target = scope:political_movement.religion
						value >= 0.1
					}
				}
				add = 10
			}
		}
	}
	
	secession = {
		possible = {
			political_movement_support >= 0.05
			scope:clout <= 0
		}
		
		weight = {
			value = 1000
		}
		
		interest_group_can_join = {
			leader = { religion = scope:political_movement.religion }
		}
		
		state_weight = {
			value = 0
			
			if = {
				limit = {
					religion_percent_state = {
						target = scope:political_movement.religion
						value >= 0.75
					}
				}
				add = 250
			}
			else_if = {
				limit = {
					religion_percent_state = {
						target = scope:political_movement.religion
						value >= 0.5
					}
				}
				add = 150
			}
			else_if = {
				limit = {
					religion_percent_state = {
						target = scope:political_movement.religion
						value >= 0.25
					}
				}
				add = 50
			}
			else_if = {
				limit = {
					religion_percent_state = {
						target = scope:political_movement.religion
						value >= 0.1
					}
				}
				add = 10
			}
		}
	}
	
	additional_radicalism_factors = {
		if = {
			limit = {
				owner.ruler ?= {
					NOT = { religion = root.religion }
				}
			}
			
			add = {
				value = 0.1
				desc = "RULER_WRONG_RELIGION"
			}
		}
		
		add = {
			value = "owner.average_sol"
			subtract = "owner.average_sol_for_religion(scope:religion)"
			multiply = 0.05
			max = 0.5
			min = 0
			desc = "STANDARD_OF_LIVING_FOR_RELIGION"
		}
		
		if = {
			limit = {
				owner = {
					any_neighbouring_state = {
						owner = {
							NOT = { THIS = ROOT.owner }
							any_political_movement = {
								religion ?= ROOT.religion
								is_insurrectionary = yes
							}
						}
					}
				}
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_RELIGIOUS_REVOLUTION"
				value = 0.15
			}
		}
	}
	
}

				}
				NOT = {
					owner ?= {
						any_political_movement = {
							is_political_movement_type = movement_fascist
						}
					}
				}
			}
			multiply = {
				value = 3
			}
		}
		if = {
			limit = {
				is_interest_group_type = ig_devout
				is_interest_group_type = ig_intelligentsia
			}
			multiply = {
				value = 2
			}
		}
	}
	
	pop_support_trigger = {
		NOT = {
			is_pop_type = slaves
		}
	}
	
	pop_support_factors = {
		movement_support_high_literacy
		movement_support_aristocrats
		movement_support_clergymen
		movement_support_shopkeepers
		movement_support_clerks
		movement_support_peasants
	}
	
	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
					OR = {
						is_being_suppressed = yes
						is_being_bolstered = yes
					}
				}
			}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}
		
		if = {
			limit = {
				strata = upper
			}
			if = {
				limit = {
					is_pop_type = capitalists
				}
				add = {
					value = 12
					desc = "POP_CAPITALISTS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = aristocrats
				}
				add = {
					value = 5
					desc = "POP_ARISTOCRATS" # Fascism is too populist
				}
			}
			else = {
				add = {
					value = 5
					desc = "UPPER_NO_ICON"
				}
			}
			
			if = {
				limit = {
					owner ?= {
						any_political_movement = {
							OR = {
								is_political_movement_type = movement_socialist
								is_political_movement_type = movement_anarchist
								is_political_movement_type = movement_communist
								is_political_movement_type = movement_labor
								is_political_movement_type = movement_nihilist
							}
							political_movement_support >= 0.05
						}
					}
				}
				multiply = {
					desc = "STRONG_SOCIALIST_MOVEMENT"
					value = 1.5
					owner ?= {
						every_political_movement = {
							limit = {
								OR = {
									is_political_movement_type = movement_socialist
									is_political_movement_type = movement_anarchist
									is_political_movement_type = movement_communist
									is_political_movement_type = movement_labor
									is_political_movement_type = movement_nihilist
								}
							}
							add = this.political_movement_radicalism
						}
					}
				}
			}
			if = {
				limit = {
					owner = {
						number_of_claims > 0
					}
				}
				
				add = {
					value = culture.culture_current_fervor # Add a base value for having claims at all
					divide = 100 # Culture is 1-100
					multiply = {
						value = owner.number_of_claims
					}
					max = 10
					desc = "NUMBER_OF_CLAIMS"
				}
			}
		}
		else_if = {
			limit = {
				strata = middle
			}
			if = {
				limit = {
					is_pop_type = shopkeepers
				}
				add = {
					value = 15
					desc = "POP_SHOPKEEPERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = bureaucrats
				}
				add = {
					value = 8
					desc = "POP_BUREAUCRATS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = officers
				}
				add = {
					value = 8
					desc = "POP_OFFICERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = farmers
				}
				add = {
					value = 5
					desc = "POP_FARMERS"
				}
			}
			else = {
				add = {
					value = 4
					desc = "MIDDLE_NO_ICON"
				}
			}
			
			if = {
				limit = {
					owner ?= {
						any_political_movement = {
							OR = {
								is_political_movement_type = movement_socialist
								is_political_movement_type = movement_anarchist
								is_political_movement_type = movement_communist
								is_political_movement_type = movement_labor
								is_political_movement_type = movement_nihilist
							}
							political_movement_support >= 0.05
						}
					}
				}
				multiply = {
					desc = "STRONG_SOCIALIST_MOVEMENT"
					value = 1
					owner ?= {
						every_political_movement = {
							limit = {
								OR = {
									is_political_movement_type = movement_socialist
									is_political_movement_type = movement_anarchist
									is_political_movement_type = movement_communist
									is_political_movement_type = movement_labor
									is_political_movement_type = movement_nihilist
								}
							}
							add = this.political_movement_radicalism
						}
					}
				}
			}
			if = {
				limit = {
					owner = {
						number_of_claims > 0
					}
				}
				add = {
					value = culture.culture_current_fervor # Add a base value for having claims at all
					divide = 100 # Culture is 1-100
					multiply = {
						value = owner.number_of_claims
					}
					max = 10
					desc = "NUMBER_OF_CLAIMS"
				}
			}
		}
		else_if = {
			limit = {
				strata = lower
			}
			if = {
				limit = {
					is_pop_type = clerks
				}
				add = {
					value = 5
					desc = "POP_CLERKS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = soldiers
				}
				add = {
					value = 3
					desc = "POP_SOLDIERS"
				}
			}
			else = {
				add = {
					value = 2
					desc = "LOWER_NO_ICON"
				}
			}
			if = {
				limit = {
					OR = {
						is_pop_type = clerks
						is_pop_type = soldiers
					}
					owner = {
						number_of_claims > 0
					}
				}
				add = {
					value = culture.culture_current_fervor # Add a base value for having claims at all
					divide = 125 # Culture is 1-100
					multiply = {
						value = owner.number_of_claims
					}
					max = 5
					desc = "NUMBER_OF_CLAIMS"
				}
			}
		}
		
		if = {
			limit = {
				standard_of_living < state.average_expected_sol
				OR = {
					strata = middle
					strata = upper
				}
			}
			add = { # Increases likelihood of support if impoverished
				desc = "IS_BELOW_EXPECTED_SOL"
				value = state.average_expected_sol
				subtract = standard_of_living
				max = 10
			}
		}
		else_if = {
			limit = {
				standard_of_living > state.average_expected_sol
				OR = {
					strata = middle
					strata = upper
				}
			}
			add = { # Decreases likelihood of support if well-off
				desc = "IS_ABOVE_EXPECTED_SOL"
				value = state.average_expected_sol
				subtract = standard_of_living
				min = -5
			}
		}
		
		if = {
			limit = {
				is_employed = no
			}
			multiply = {
				value = 1.5
				desc = "UNEMPLOYED"
			}
		}
		
		multiply = { # Illiterate pops participate at 1/10 the rate of fully literate pops
			desc = "POP_LITERACY"
			value = literacy_rate
			add = 0.10
		}
		
		multiply = {
			value = 0.011
			multiply = pop_acceptance
			desc = "ACCEPTANCE"
		}
		
		if = {
			limit = {
				OR = {
					strata = upper
					strata = middle
					is_pop_type = clerks
				}
			}
			multiply = { # Multiplied by 2x max
				desc = "POP_RADICALS"
				value = pop_radical_fraction
				multiply = 2
				add = 1
			}
		}
		else = {
			multiply = { # Multiplied by 1.5x max
				desc = "POP_RADICALS"
				value = pop_radical_fraction
				divide = 2
				add = 1
			}
		}
	}
	
	revolution = {
		possible = {
			political_movement_support >= 0.05
		}
		
		weight = {
			value = 100
		}
		
		state_weight = {
			value = 1
		}
	}
	
	secession = {
		possible = {
			political_movement_support >= 0.05
			scope:clout <= 0
		}
		
		weight = {
			value = 1000
		}
	}
	
	additional_radicalism_factors = {
		owner = {
			if = {
				limit = {
					owner.number_of_claims > 0
				}
				
				add = {
					value = 5 # Add a base value for having claims at all
					add = owner.number_of_claims
					multiply = 0.025
					max = 0.3
					desc = "NUMBER_OF_CLAIMS"
				}
			}
		}
		
		if = {
			limit = {
				owner = {
					any_neighbouring_state = {
						owner = {
							NOT = { THIS = ROOT.owner }
							any_political_movement = {
								is_political_movement_type = movement_fascist
								is_insurrectionary = yes
							}
						}
					}
				}
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
				value = 0.15
			}
		}
		
		if = { # Support translates 1:1 to Activism
			limit = {
				owner = {
					NAND = {
						has_law = law_type:law_single_party_state
						ruler.interest_group.party ?= {
							is_party_type = fascist_party
						}
					}
				}
			}
			add = {
				desc = "FASCIST_MOVEMENT_WILL_TO_POWER"
				add = this.political_movement_support
				max = 0.5
			}
		}
	}
