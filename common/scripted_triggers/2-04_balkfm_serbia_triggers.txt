balkfm_has_obrenovic_ruler_trigger = {
	custom_tooltip = {
		text = balkfm_has_obrenovic_ruler_trigger
		ruler = {
			OR = {
				has_template = balkfm_obrenovic_dynamic
				has_template = balkfm_obrenovic_dynamic_heir
				has_template = balkfm_obrenovic_milosh
				has_template = balkfm_obrenovic_milan
				has_template = balkfm_obrenovic_mihailo
				has_template = balkfm_obrenovic_milan_IV
			}
		}
	}
}

balkfm_has_maximum_dynastic_support_trigger = {
	custom_tooltip = {
		text = balkfm_has_maximum_dynasty_support_trigger
		OR = {
			AND = {
				balkfm_has_karadordevic_ruler_trigger = yes
				"variable_map(balkfm_dynastic_rivalry_varmap|flag:balkfm_karadordevic_support_var)" >= 500
			}
			AND = {
				balkfm_has_obrenovic_ruler_trigger = yes
				"variable_map(balkfm_dynastic_rivalry_varmap|flag:balkfm_obrenovic_support_var)" >= 500
			}
		}
		"variable_map(balkfm_dynastic_rivalry_varmap|flag:balkfm_obrenovic_support_var)" != "variable_map(balkfm_dynastic_rivalry_varmap|flag:balkfm_karadordevic_support_var)"
	}
}

balkfm_has_matching_consul_trigger = {
	trigger_if = {
		limit = {
			exists = c:AUS
			balkfm_has_obrenovic_ruler_trigger = yes
		}
		custom_tooltip = {
			text = balkfm_has_austrian_consul
			"variable_map(balkfm_dynastic_rivalry_varmap|flag:balkfm_consul_var)" ?= flag:AUS
		}
	}
	trigger_else_if = {
		limit = {
			exists = c:RUS
			balkfm_has_karadordevic_ruler_trigger = yes
		}
		custom_tooltip = {
			text = balkfm_has_russian_consul
			"variable_map(balkfm_dynastic_rivalry_varmap|flag:balkfm_consul_var)" ?= flag:RUS
		}
	}
	trigger_else = {
		always = no
	}
}
