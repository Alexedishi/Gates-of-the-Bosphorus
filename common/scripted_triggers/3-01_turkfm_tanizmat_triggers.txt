turkfm_has_pro_reform_ideology = {
	OR = {
		has_liberal_ideology = yes
		has_progressive_ideology = yes
		has_ideology = ideology:ideology_gbbf_ottomanist_leader
	}
}

turkfm_has_pro_conversatism_ideology = {
	OR = {
		has_reactionary_ideology = yes
		has_conservative_ideology = yes
		has_paternalist_ideology = yes
		has_ideology = ideology:ideology_gbbf_islamist
	}
}

turkfm_has_tanzimat_agenda_trigger = {
	custom_tooltip = {
		text = turkfm_has_tanzimat_$type$
		has_variable = turkfm_tanzimat_$type$_agenda
	}
}

turkfm_has_tanzimat_agenda_NOT_trigger = {
	custom_tooltip = {
		text = turkfm_has_tanzimat_$type$_NOT
		NOT = {
			has_variable = turkfm_tanzimat_$type$_agenda
		}
	}
}

turkfm_has_completed_tanzimat_agenda_trigger = {
	custom_tooltip = {
		text = turkfm_tanzimat_$type$_complete
		has_variable = turkfm_tanzimat_$type$_complete
	}
}
