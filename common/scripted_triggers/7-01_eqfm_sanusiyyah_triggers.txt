eqfm_sanusiyyah_is_relevant_due_to_subject_trigger = {
	custom_description = {
		text = eqfm_sanusiyyah_is_relevant_due_to_subject_trigger
		subject = this
		gbbf_country_is_muslim_trigger = yes
		any_subject_or_below = {
			any_scope_state = {
				is_in_geographic_region = geographic_region_sanusiyyah
			}
			gbbf_country_is_muslim_trigger = yes
		}
	}
}

eqfm_sanusiyyah_is_relevant_to_overlord_trigger = {
	custom_description = {
		text = eqfm_sanusiyyah_is_relevant_to_overlord_trigger
		subject = this
		is_subject = yes
		gbbf_country_is_muslim_trigger = yes
		overlord = {
			gbbf_country_is_muslim_trigger = yes
		}
	}
}

####---------- Sanusiyyah State Region Triggers ----------####
eqfm_sanusiyyah_entrenchment_trigger = {
	custom_tooltip = {
		text = eqfm_sanusiyyah_entrenchment_trigger
		state_region = {
			has_variable = eqfm_sanusiyyah_entrenchment_var
		}
	}
}

eqfm_state_region_has_sanusiyyah_entrenchment_trigger = {
	custom_tooltip = {
		text = eqfm_sanusiyyah_entrenchment_trigger
		$state_region$ = {
			has_variable = eqfm_sanusiyyah_entrenchment_var
		}
	}
}

eqfm_eligible_for_entrenchment_event_trigger = {
	# Call this trigger in the STATE scope!
	is_in_geographic_region = geographic_region_sanusiyyah
	state_region = {
		has_variable = eqfm_sanusiyyah_entrenchment_var
		NOT = {
			has_variable = eqfm_sanusiyyah_entrenchment_cooldown
		}
	}
}

eqfm_sanusiyyah_entrenchment_less_or_equals_trigger = {
	# Call this trigger in the STATE scope!
	state_region = {
		has_variable = eqfm_sanusiyyah_entrenchment_var
		var:eqfm_sanusiyyah_entrenchment_var <= $value$
	}
}

eqfm_sanusiyyah_entrenchment_greater_or_equals_trigger = {
	# Call this trigger in the STATE scope!
	state_region = {
		has_variable = eqfm_sanusiyyah_entrenchment_var
		var:eqfm_sanusiyyah_entrenchment_var >= $value$
	}
}
