eqfm_sanusiyyah_involvement_trigger = {
	OR = {
		AND = {
			any_scope_state = {
				is_in_geographic_region = geographic_region_sanusiyyah
			}
			NOR = {
				is_country_type = decentralized
				has_variable = eqfm_expelled_sanusiyyah
			}
		}
		any_subject_or_below = {
			is_involved_in_journal_entry = je_eqfm_sanusiyyah
		}
	}
}

	custom_tooltip = {

		$state_region$ = {
			has_variable = eqfm_sanusiyyah_entrenchment_var
		}
	}
}

eqfm_eligible_for_entrenchment_event_trigger = {
	# Call this trigger in the STATE scope!
	is_in_geographic_region = geographic_region_sanusiyyah
	state_region = {
		has_variable = eqfm_sanusiyyah_entrenchment_var
		NOT = {
			has_variable = eqfm_sanusiyyah_entrenchment_cooldown
		}
	}
}

eqfm_sanusiyyah_entrenchment_less_or_equals_trigger = {
	# Call this trigger in the STATE scope!
	state_region = {
		has_variable = eqfm_sanusiyyah_entrenchment_var
		var:eqfm_sanusiyyah_entrenchment_var <= $value$
	}
}

eqfm_sanusiyyah_entrenchment_greater_or_equals_trigger = {
	# Call this trigger in the STATE scope!
	state_region = {
		has_variable = eqfm_sanusiyyah_entrenchment_var
		var:eqfm_sanusiyyah_entrenchment_var >= $value$
	}
}
