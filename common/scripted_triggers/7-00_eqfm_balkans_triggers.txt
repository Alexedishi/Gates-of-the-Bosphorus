		"global_variable_map(eqfm_balkan_powder_keg_varmap|flag:eqfm_balkan_powder_keg_phase_var)" ?= flag:eqfm_unrest
		AND = {
			"global_variable_map(eqfm_balkan_powder_keg_varmap|flag:eqfm_balkan_powder_keg_phase_var)" ?= flag:eqfm_hostility
			eqfm_crisis_possible_trigger = yes

		"global_variable_map(eqfm_balkan_powder_keg_varmap|flag:eqfm_balkan_powder_keg_phase_var)" ?= flag:eqfm_hostility

			"global_variable_map(eqfm_balkan_powder_keg_varmap|flag:eqfm_balkan_powder_keg_phase_var)" ?= flag:eqfm_crisis

		}

eqfm_balkans_should_be_involved_trigger = {
	custom_tooltip = {
		text = eqfm_balkans_should_be_involved_trigger
		OR = {
			has_strategic_region_interest_tier = {
				strategic_region = sr:region_balkans
				value >= 5
			}
			any_scope_state = {
				is_in_geographic_region = geographic_region_rumelia
			}
			any_subject_or_below = {
				is_in_geographic_region = geographic_region_rumelia
			}
		}
	}
}
