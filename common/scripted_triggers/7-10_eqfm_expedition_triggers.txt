## Expedition Triggers
# eqfm_$type$_expedition_var designates the presence of an active expedition
# These are the currently used types:
#	senusiyya
#	adriatic
#	black_mountain
#	desert

eqfm_has_any_expedition_trigger = {
	custom_description = {
		text = eqfm_has_any_expedition
		OR = {
			eqfm_has_senusiyya_expedition_trigger = yes
			eqfm_has_adriatic_expedition_trigger = yes
			eqfm_has_karadag_expedition_trigger = yes
			eqfm_has_desert_expedition_trigger = yes
		}
	}
}

eqfm_expedition_has_leader_trigger = {
	trigger_if = {
		limit = {
			NOT = {
				has_variable = eqfm_$type$_expedition_transit_var
			}
		}
		custom_tooltip = {
			text = eqfm_has_$type$_expedition_leader
			any_scope_character = {
				has_modifier = modifier_eqfm_leading_$type$_expedition
			}
		}
	}
}

eqfm_has_expedition_trigger = {
	custom_description = {
		text = eqfm_has_$type$_expedition
		has_variable = eqfm_$type$_expedition_var
	}
}

eqfm_has_senusiyya_expedition_trigger = {
	custom_description = {
		text = eqfm_has_senusiyya_expedition
		has_variable = eqfm_senusiyya_expedition_var
	}
}

eqfm_has_adriatic_expedition_trigger = {
	custom_description = {
		text = eqfm_has_adriatic_expedition
		has_variable = eqfm_adriatic_expedition_var
	}
}

eqfm_has_karadag_expedition_trigger = {
	trigger_if = {
		limit = {
			c:MON ?= this
		}
		custom_description = {
			text = eqfm_has_tribal_chief
			has_variable = eqfm_karadag_expedition_var
		}
	}
	trigger_else = {
		custom_description = {
			text = eqfm_has_karadag_expedition
			has_variable = eqfm_karadag_expedition_var
		}
	}
}

eqfm_has_desert_expedition_trigger = {
	custom_description = {
		text = eqfm_has_desert_expedition
		has_variable = eqfm_desert_expedition_var
	}
}

eqfm_expedition_progress_equal_trigger = {
	hidden_trigger = {
		save_temporary_scope_value_as = {
			name = eqfm_expedition_loc_scope
			value = $value$
		}
	}
	custom_tooltip = {
		text = eqfm_expedition_progress_equal
		var:eqfm_$type$_expedition_progress_var ?= $value$
	}
}
