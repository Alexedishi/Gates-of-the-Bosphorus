## Expedition Triggers
# eqfm_expedition_var designates the presence of an active expedition
eqfm_has_any_expedition_trigger = {
	custom_description = {
		text = eqfm_has_expedition
		OR = {
			eqfm_has_adriatic_expedition_trigger = yes
			eqfm_has_black_mountain_expedition_trigger = yes
			eqfm_has_desert_expedition_trigger = yes
		}
	}
}

eqfm_has_senusiyya_expedition_trigger = {
	custom_description = {
		text = eqfm_has_senusiyya_expedition
		has_variable = eqfm_senusiyya_expedition_var
	}
}

eqfm_has_adriatic_expedition_trigger = {
	custom_description = {
		text = eqfm_has_adriatic_expedition
		has_variable = eqfm_adriatic_expedition_var
	}
}

eqfm_has_black_mountain_expedition_trigger = {
	trigger_if = {
		limit = {
			c:MON ?= this
		}
		custom_description = {
			text = eqfm_has_tribal_chief
			has_variable = eqfm_black_mountain_expedition_var
		}
	}
	trigger_else = {
		custom_description = {
			text = eqfm_has_black_mountain_expedition
			has_variable = eqfm_black_mountain_expedition_var
		}
	}
}

eqfm_has_desert_expedition_trigger = {
	custom_description = {
		text = eqfm_has_desert_expedition
		has_variable = eqfm_desert_expedition_var
	}
}

eqfm_expedition_progress_equal_trigger = {
	save_temporary_scope_value_as = {
		name = eqfm_expedition_loc_scope
		value = $value$
	}
	custom_tooltip = {
		text = eqfm_expedition_progress_equal
		var:eqfm_$type$_expedition_progress_var ?= $value$
	}
}
