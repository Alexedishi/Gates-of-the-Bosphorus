turkfm_can_escalate_tension_trigger = {
	OR = {
		global_var:turkfm_balkan_situation_phase_var ?= flag:turkfm_stable
		global_var:turkfm_balkan_situation_phase_var ?= flag:turkfm_unrest
		AND = {
			global_var:turkfm_balkan_situation_phase_var ?= flag:turkfm_hostility
			turkfm_crisis_possible_trigger = yes
		}
		global_var:turkfm_balkan_situation_phase_var ?= flag:turkfm_deconfliction
	}
}

turkfm_can_deescalate_tension_trigger = {
	OR = {
		global_var:turkfm_balkan_situation_phase_var ?= flag:turkfm_unrest
		global_var:turkfm_balkan_situation_phase_var ?= flag:turkfm_hostility
		AND = {
			global_var:turkfm_balkan_situation_phase_var ?= flag:turkfm_crisis
			c:TUR ?= {
				has_any_secessionists_broken_out = no
				has_any_secessionists_growing = no
				is_at_war = no
			}
		}
	}
}

turkfm_crisis_deconfliction_trigger = {
	has_variable = turkfm_crisis_attacker
	trigger_if = {
		limit = {
			c:GBR ?= this
		}
		OR = {
			c:TUR ?= {
				owns_entire_state_region = STATE_CYPRUS
			}
			c:CYP ?= {
				is_subject_of = c:TUR
			}
		}
	}
	trigger_if = {
		limit = {
			c:FRA ?= this
		}
		OR = {
			c:TUR ?= {
				owns_entire_state_region = STATE_TUNISIA
			}
			c:TUN ?= {
				is_subject_of = c:TUR
			}
		}
	}
	trigger_if = {
		limit = {
			c:RUS ?= this
		}
		c:TUR ?= {
			owns_entire_state_region = STATE_DOBRUDJA
		}
	}
}
