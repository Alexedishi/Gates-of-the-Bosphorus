eqfm_balkans_involved_trigger = {
	custom_tooltip = {
		text = eqfm_balkans_involved_trigger
		OR = {
			c:TUR ?= this
			is_subject_of = c:TUR
			any_scope_state = {
				region = sr:region_balkans
				is_capital = yes
			}
			gbbf_has_alliance_trigger = {
				country = c:TUR
			}
			gbbf_has_defensive_pact_trigger = {
				country = c:TUR
			}
			gbbf_has_military_assistance_trigger = {
				country = c:TUR
			}
			gbbf_has_rivalry_trigger = {
				country = c:TUR
			}
			gbbf_has_embargo_trigger = {
				country = c:TUR
			}
			AND = {
				country_rank >= rank_value:great_power
				has_interest_marker_in_region = region_balkans
			}
		}
	}
}

		AND = {

eqfm_crisis_deconfliction_trigger = {
	has_variable = eqfm_crisis_attacker
	trigger_if = {
		limit = {
			c:GBR ?= this
		}
		OR = {
			c:TUR ?= {
				owns_entire_state_region = STATE_CYPRUS
			}
			c:CYP ?= {
				is_subject_of = c:TUR
			}
		}
	}
	trigger_if = {
		limit = {
			c:FRA ?= this
		}
		OR = {
			c:TUR ?= {
				owns_entire_state_region = STATE_TUNISIA
			}
			c:TUN ?= {
				is_subject_of = c:TUR
			}
		}
	}
	trigger_if = {
		limit = {
			c:RUS ?= this
		}
		c:TUR ?= {
			owns_entire_state_region = STATE_DOBRUDJA
		}
	}
}
