balkfm_country_has_yugoslav_culture_trigger = {
	OR = {
		country_has_primary_culture = cu:montenegrin
		country_has_primary_culture = cu:serb
		country_has_primary_culture = cu:croat
		country_has_primary_culture = cu:bosniak
		country_has_primary_culture = cu:slovene
		country_has_primary_culture = cu:bulgarian
	}
}

balkfm_country_is_eligible_for_yugoslavism_trigger = {
	OR = {
		country_has_primary_culture = cu:montenegrin
		country_has_primary_culture = cu:serb
		AND = {
			country_has_primary_culture = cu:bulgarian
			custom_tooltip = {
				text = balkfm_pursing_greater_yugoslavism
				has_variable = bul_pursued_greater_yugoslavism
			}
		}
		country_has_primary_culture = cu:bosniak
		country_has_primary_culture = cu:croat
		country_has_primary_culture = cu:slovene
	}
	has_technology_researched = nationalism
	is_secessionist = no
}

	trigger_if = {
		limit = {
			country_has_primary_culture = cu:montenegrin
		}
		any_primary_culture = {
			cu:montenegrin = this
		}

balkfm_yugoslav_radicals_trigger = {
	custom_tooltip = {
		text = balkfm_has_yugoslav_radicals_tt
		OR = {
			trigger_if = {
				limit = {
					any_scope_pop = {
						has_pop_culture = serb
					}
					owner = {
						cultural_acceptance_base = {
							target = cu:serb
							value < acceptance_status_4
						}
					}
				}
				radical_fraction = {
					culture = cu:serb
					value >= 0.33
