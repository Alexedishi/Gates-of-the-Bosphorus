## Expedition Triggers
# turkfm_expedition_var designates the presence of an active expedition
eqfm_has_any_expedition_trigger = {
	custom_tooltip = {
		text = turkfm_has_expedition
		OR = {
			eqfm_has_adriatic_expedition_trigger = yes
			turkfm_has_black_mountain_expedition_trigger = yes
			eqfm_has_desert_expedition_trigger = yes
		}
	}
}

eqfm_no_expedition_trigger = {
	custom_tooltip = {
		text = turkfm_has_expedition_NOT
		NOR = {
			eqfm_has_adriatic_expedition_trigger = yes
			turkfm_has_black_mountain_expedition_trigger = yes
			eqfm_has_desert_expedition_trigger = yes
		}
	}
}
