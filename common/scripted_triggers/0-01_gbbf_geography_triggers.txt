gbbf_owns_entire_georegion_trigger = {
	custom_description = {
		text = gbbf_completely_owns_$georegion$_trigger
		subject = this
		gbbf_georegion_$georegion$_owned_value >= gbbf_georegion_$georegion$_total_value
	}
}
