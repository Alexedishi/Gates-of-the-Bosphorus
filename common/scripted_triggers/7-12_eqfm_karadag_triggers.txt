eqfm_karadag_placid_trigger = {
	NOT = {
		has_global_variable = eqfm_karadag_skirmish_var
	}
	c:MON ?= {
		eqfm_has_karadag_expedition_trigger = no
	}
}

eqfm_karadag_raiding_trigger = {
	NOR = {
		has_global_variable = eqfm_karadag_skirmish_var
		c:TUR ?= this
	}
	eqfm_has_karadag_expedition_trigger = yes
}

eqfm_karadag_raided_trigger = {
	NOT = {
		has_global_variable = eqfm_karadag_skirmish_var
	}
	c:TUR ?= this
	eqfm_has_karadag_expedition_trigger = no
	c:MON ?= {
		eqfm_has_karadag_expedition_trigger = yes
	}
}

eqfm_karadag_patrolling_trigger = {
	NOT = {
		has_global_variable = eqfm_karadag_skirmish_var
	}
	c:TUR ?= this
	eqfm_has_karadag_expedition_trigger = yes
	c:MON ?= {
		eqfm_has_karadag_expedition_trigger = yes
	}
}
