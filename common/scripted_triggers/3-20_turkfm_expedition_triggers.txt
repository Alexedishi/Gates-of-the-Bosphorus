## Expedition Triggers
# turkfm_expedition_var designates the presence of an active expedition
turkfm_has_any_expedition_trigger = {
	custom_tooltip = {
		text = turkfm_has_expedition
		OR = {
			has_variable = turkfm_black_mountain_expedition_var
			has_variable = turkfm_adriatic_expedition_var
		}
	}
}

turkfm_no_expedition_trigger = {
	custom_tooltip = {
		text = turkfm_has_expedition_NOT
		NOR = {
			has_variable = turkfm_black_mountain_expedition_var
			has_variable = turkfm_adriatic_expedition_var
		}
	}
}

## Black Mountain Triggers ##
turkfm_has_black_mountain_expedition_trigger = {
	trigger_if = {
		limit = {
			c:MON ?= this
		}
		custom_tooltip = {
			text = turkfm_has_tribal_chief
			has_variable = turkfm_black_mountain_expedition_var
		}
	}
	trigger_else = {
		custom_tooltip = {
			text = turkfm_has_black_mountain_expedition
			has_variable = turkfm_black_mountain_expedition_var
		}
	}
}

turkfm_black_mountain_placid_trigger = {
	NOT = {
		has_global_variable = turkfm_black_mountain_skirmish_var
	}
	c:MON ?= {
		turkfm_has_black_mountain_expedition_trigger = no
	}
}

turkfm_black_mountain_raiding_trigger = {
	NOT = {
		has_global_variable = turkfm_black_mountain_skirmish_var
	}
	c:MON ?= this
	turkfm_has_black_mountain_expedition_trigger = yes
}

turkfm_black_mountain_raided_trigger = {
	NOT = {
		has_global_variable = turkfm_black_mountain_skirmish_var
	}
	c:TUR ?= this
	turkfm_has_black_mountain_expedition_trigger = no
	c:MON ?= {
		turkfm_has_black_mountain_expedition_trigger = yes
	}
}

turkfm_black_mountain_patrolling_trigger = {
	NOT = {
		has_global_variable = turkfm_black_mountain_skirmish_var
	}
	c:TUR ?= this
	turkfm_has_black_mountain_expedition_trigger = yes
	c:MON ?= {
		turkfm_has_black_mountain_expedition_trigger = yes
	}
}

turkfm_black_mountain_skirmish_trigger = {
	has_global_variable = turkfm_black_mountain_skirmish_var
}

## Pirates of the Adriatic ##
turkfm_has_adriatic_expedition_trigger = {
	custom_tooltip = {
		text = turkfm_has_adriatic_expedition
		has_variable = turkfm_adriatic_expedition_var
	}
}
