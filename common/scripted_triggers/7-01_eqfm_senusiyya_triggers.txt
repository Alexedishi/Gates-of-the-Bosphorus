eqfm_senusiyya_involvement_trigger = {
	OR = {
		AND = {
			any_scope_state = {
				is_in_geographic_region = geographic_region_senusiyya
			}
			NOR = {
				is_country_type = decentralized
				has_variable = eqfm_expelled_senusiyya
			}
		}
		any_subject_or_below = {
			is_involved_in_journal_entry = je_eqfm_senusiyya
		}
	}
}

	OR = {
		has_template = template_eqfm_senussi_dynamic
		has_template = template_eqfm_senussi_hasan
		has_template = template_eqfm_senussi_idris
		has_template = template_eqfm_senussi_ahmed
		has_template = template_eqfm_senussi_muhammad_ii
		has_template = template_eqfm_senussi_muhammad_i
	}
}
