eqfm_senusiyya_involvement_trigger = {
	OR = {
		AND = {
			any_scope_state = {
				is_in_geographic_region = geographic_region_senusiyya
			}
			NOR = {
				is_country_type = decentralized
				has_variable = eqfm_expelled_senusiyya
			}
		}
		any_subject_or_below = {
			is_involved_in_journal_entry = je_eqfm_senusiyya
		}
	}
}

eqfm_character_is_senussi_trigger = {
	OR = {
		has_template = template_eqfm_senussi_dynamic
		has_template = template_eqfm_senussi_hasan
		has_template = template_eqfm_senussi_idris
		has_template = template_eqfm_senussi_ahmed
		has_template = template_eqfm_senussi_muhammad_ii
		has_template = template_eqfm_senussi_muhammad_i
	}
}

eqfm_has_wecloming_stance_trigger = {
	custom_tooltip = {
		text = eqfm_has_wecloming_stance
		var:eqfm_senusiyya_stance_var ?= flag:eqfm_senusiyya_stance_welcoming
	}
}

	}
}

		var:eqfm_senusiyya_stance_var ?= flag:eqfm_senusiyya_stance_hostile
	}

eqfm_eligible_for_entrenchment_event_trigger = {
	is_in_geographic_region = geographic_region_senusiyya
	state_region = {
		has_variable = eqfm_senusiyya_entrenchment_var
		NOT = {
			has_variable = eqfm_senusiyya_entrenchment_cooldown
		}
	}
}

eqfm_senusiyya_entrenchment_less_or_equals_trigger = {
	state_region = {
		has_variable = eqfm_senusiyya_entrenchment_var
		var:eqfm_senusiyya_entrenchment_var <= $value$
	}
}

eqfm_senusiyya_entrenchment_greater_or_equals_trigger = {
	state_region = {
		has_variable = eqfm_senusiyya_entrenchment_var
		var:eqfm_senusiyya_entrenchment_var >= $value$
	}
}
