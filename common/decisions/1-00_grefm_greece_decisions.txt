grefm_corinth_canal_survey_decision = {
	is_shown = {
		NOR = {
			has_global_variable = grefm_corinth_survey_complete
			has_journal_entry = je_grefm_corinth_canal_survey
		}
		owns_entire_state_region = STATE_PELOPONNESE
	}
	possible = {
		owns_entire_state_region = STATE_PELOPONNESE
		has_technology_researched = railways
	}
	when_taken = {
		set_variable = {
			name = grefm_corinth_canal_var
			value = 0
		}
		add_journal_entry = {
			type = je_grefm_corinth_canal_survey
		}
		show_as_tooltip = {
			add_modifier = modifier_grefm_corinth_canal_survey
		}
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				produced_bureaucracy > 500
			}
			add = 100
		}
		if = {
			limit = {
				OR = {
					is_diplomatic_play_committed_participant = yes
					is_at_war = yes
				}
			}
			add = -1000
		}
	}
}

REPLACE:revive_olympic_games_decision = {
	is_shown = {
		country_has_primary_culture = cu:greek
	}
	
	possible = {
		owns_entire_state_region = STATE_ATTICA
		has_technology_researched = organized_sports
		hidden_trigger = {
			NOT = { has_variable = revive_olympics_decision_taken }
		}
	}
	
	when_taken = {
		set_variable = {
			name = revive_olympics_decision_taken
			value = yes
		}
		add_modifier = {
			name = modifier_olympic_games
		}
		add_modifier = {
			name = modifier_olympic_games_bureaucracy
			months = gbbf_2_plus_half_years
		}
	}
	
	ai_chance = {
		value = 0
		if = {
			limit = { bureaucracy > 0 }
			add = 5
		}
		if = {
			limit = {
				OR = {
					is_diplomatic_play_committed_participant = yes
					is_at_war = yes
				}
			}
			multiply = 0
		}
	}
}
