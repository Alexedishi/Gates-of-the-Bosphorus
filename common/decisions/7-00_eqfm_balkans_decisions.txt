eqfm_greek_gunrunners_decision = {
	is_shown = {
		is_ai = yes
		is_involved_in_journal_entry = je_eqfm_balkan_powder_keg
		country_has_primary_culture = cu:greek
		c:TUR ?= {
			OR = {
				owns_entire_state_region = STATE_ALBANIA
				owns_entire_state_region = STATE_MACEDONIA
				owns_entire_state_region = STATE_SKOPIA
				owns_entire_state_region = STATE_CYPRUS
				owns_entire_state_region = STATE_CRETE
				owns_entire_state_region = STATE_AYDIN
			}
		}
	}
	possible = {
		has_technology_researched = nationalism
		any_scope_state = {
			is_in_geographic_region = geographic_region_rumelia
		}
		custom_tooltip = {
			text = eqfm_greek_gunrunners_possible
			NOT = {
				has_variable = eqfm_greek_gunrunners_lockout
			}
		}
	}
	when_taken = {
		if = {
			limit = {
				NOT = {
					has_modifier = modifier_eqfm_supporting_gunrunners
				}
			}
			custom_tooltip = eqfm_greek_gunrunners_effect1
			add_modifier = {
				name = modifier_eqfm_supporting_gunrunners
				multiplier = money_amount_multiplier_medium
			}
		}
		else_if = {
			limit = {
				has_modifier = modifier_eqfm_supporting_gunrunners
			}
			remove_modifier = modifier_eqfm_supporting_gunrunners
		}
		custom_tooltip = {
			text = eqfm_greek_gunrunners_effect2
			set_variable = {
				name = eqfm_greek_gunrunners_lockout
				years = 2
			}
		}
	}
	ai_chance = {
		value = -5
		if = {
			limit = {
				NOT = { has_modifier = modifier_eqfm_supporting_gunrunners }
				c:TUR ?= {
					relations:root < relations_threshold:neutral
				}
				fixed_income > money_amount_multiplier_medium
				gold_reserves > 0
			}
			add = 100
		}
		if = {
			limit = {
				NOT = { has_modifier = modifier_eqfm_supporting_gunrunners }
				fixed_income < money_amount_multiplier_medium
				gold_reserves < 0
			}
			add = -1000
		}
		if = {
			limit = {
				has_modifier = modifier_eqfm_supporting_gunrunners
				c:TUR ?= {
					relations:root > relations_threshold:neutral
				}
			}
			add = 100
		}
		if = {
			limit = {
				has_modifier = modifier_eqfm_supporting_gunrunners
				c:TUR ?= {
					relations:root < relations_threshold:neutral
				}
			}
			add = -100
		}
		if = {
			limit = {
				has_modifier = modifier_eqfm_supporting_gunrunners
				fixed_income < money_amount_multiplier_medium
				gold_reserves < 0
			}
			add = 1000
		}
	}
}

eqfm_revoke_language_ban = {
	is_shown = {
		c:TUR ?= this
		has_variable = eqfm_balkan_powder_keg_language_ban_state_var
		var:eqfm_balkan_powder_keg_language_ban_state_var = {
			owner = root
		}
	}
	possible = {
		legitimacy > 70
		var:eqfm_balkan_powder_keg_language_ban_state_var = {
			radical_fraction = {
				value < 0.20
			}
		}
		NOR = {
			has_law_or_variant = law_type:law_ethnostate
			has_insurrectionary_interest_groups = yes
			has_revolution = yes
		}
	}
	when_taken = {
		var:eqfm_balkan_powder_keg_language_ban_state_var = {
			remove_modifier = modifier_eqfm_languages_banned
			add_modifier = {
				name = modifier_eqfm_languages_restored
				months = gbbf_long_modifier_time
			}
			add_loyalists_in_state = {
				value = large_radicals
				religion = var:eqfm_balkan_powder_keg_language_ban_culture_var
			}
			add_culture_standard_of_living_modifier = {
				culture = var:eqfm_balkan_powder_keg_language_ban_culture_var
				multiplier = -4
			}
		}
		add_modifier = {
			name = modifier_eqfm_restored_languages
			months = gbbf_normal_modifier_time
		}
		every_interest_group = {
			limit = {
				has_conservative_ideology = yes
			}
			add_modifier = {
				name = modifier_eqfm_hates_languages
				months = gbbf_normal_modifier_time
			}
		}
	}
	ai_chance = {  }
}
