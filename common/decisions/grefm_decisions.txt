grefm_corinth_canal_survey_decision = {
	is_shown = {
		NOR = {
			has_global_variable = grefm_corinth_survey_complete
			has_journal_entry = je_grefm_corinth_canal_survey
		}
		owns_entire_state_region = STATE_PELOPONNESE
	}
	possible = {
		owns_entire_state_region = STATE_PELOPONNESE
		has_technology_researched = railways
	}
	when_taken = {
		set_variable = { 
			name = grefm_corinth_canal_var
			value = 0
		}
		add_journal_entry = {
			type = je_grefm_corinth_canal_survey
		}
		show_as_tooltip = {
			add_modifier = modifier_grefm_corinth_canal_survey
		}
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				produced_bureaucracy > 500
			}
			add = 100
		}
		if = {
			limit = {
				OR = {
					is_diplomatic_play_committed_participant = yes
					is_at_war = yes
				}
			}
			add = -1000
		}
	}
}

grefm_annex_hepatanese_decision = {
	is_shown = {
		country_has_primary_culture = cu:greek
		NOT = {
			has_journal_entry = je_grefm_megali_idea
		}
		c:ION ?= {
			this != root
			top_overlord ?= {
				is_player = no
			}
		}
		has_variable = grefm_heptanese_annex_var
	}
	possible = {
		has_technology_researched = nationalism
		c:ION ?= {
			is_at_war = no
			relations:root >= relations_threshold:cordial
			trigger_if = {
				limit = {
					is_subject = yes
				}
				top_overlord ?= {
					custom_tooltip = {
						text = grefm_annex_heptanese_possible_1
						NOT = { has_war_with = ROOT }
					}
					relations:root >= relations_threshold:cordial
				}
			}
		}
		is_at_war = no
		is_subject = no
		custom_tooltip = {
			text = grefm_annex_heptanese_possible_2
			var:grefm_heptanese_annex_var >= 60
		}
	}
	when_taken = {
		custom_tooltip = {
			text = grefm_annex_heptanese_effect
			trigger_event = {
				id = grefm_megali_idea.100 # ION Annexation
				popup = yes
			}
			set_variable = {
				name = grefm_heptanese_annex_var
				value = 0
			}
		}
	}
	ai_chance = {
		value = 100
	}
}
