grefm_annex_hepatanese_decision = {
	is_shown = {
		country_has_primary_culture = cu:greek
		NOT = {
			has_journal_entry = je_grefm_megali_idea
		}
		c:ION ?= {
			this != root
			top_overlord ?= {
				is_player = no
			}
		}
		has_variable = grefm_heptanese_annex_var
	}
	possible = {
		has_technology_researched = nationalism
		c:ION ?= {
			is_at_war = no
			trigger_if = {
				limit = {
					is_subject = yes
				}
				top_overlord ?= {
					custom_tooltip = {
						text = grefm_annex_heptanese_possible_1
						NOT = { has_war_with = ROOT }
					}
					relations:root >= relations_threshold:cordial
				}
			}
			relations:root >= relations_threshold:cordial
		}
		custom_tooltip = {
			text = grefm_annex_heptanese_possible_2
			var:grefm_heptanese_annex_var >= 60
		}
		OR = {
			is_subject = no
			c:ION ?= {
				is_subject_of = root
			}
			trigger_if = {
				limit = {
					is_subject = yes
					c:ION ?= {
						is_subject = yes
					}
				}
				custom_tooltip = {
					text = grefm_has_same_overlord_trigger
					overlord = c:ION.overlord
				}
			}
		}
		is_at_war = no
	}
	when_taken = {
		custom_tooltip = {
			text = grefm_annex_heptanese_effect
			trigger_event = {
				id = grefm_megali_idea.100
				popup = yes
			}
			set_variable = {
				name = grefm_heptanese_annex_var
				value = 0
			}
		}
	}
	ai_chance = {
		value = 100
	}
}
