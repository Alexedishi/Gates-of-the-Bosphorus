gbbf_depose_bey = {
	icon = "gfx/interface/icons/war_goals/gbbf_depose_bey.dds"

	kind = custom

	settings = {
		conflicts_with_make_subject
		validate_subject_relation
		validate_conflicts_war_goals_all
		validate_conflicts_make_subject
	}

	execution_priority = 39

	contestion_type = control_any_target_incorporated_state

	target_type = country

	possible = { # This is temporarily limited to BOS; I will expand this to include *any* Beylik in the future
		scope:target_country = {
			c:BOS ?= this
			has_journal_entry = je_balkfm_bos_ali_pasha
#			is_subject_of = root
#			is_subject_type = subject_type_gbbf_eyalet
		}
	}

	valid = {
		scope:target_country.overlord = root
	}

	maneuvers = {
		value = 0

		add = {
			desc = "MANEUVERS_BASE_VALUE"
			value = 10
		}
	}

	infamy = {
		value = 0

        scope:target_country = {
            add = country_infamy_value
        }

        multiply = {
            desc = "INFAMY_FROM_LOYAL_SUBJECT"
            value = 1

            if = {
                limit = {
                    scope:target_country = {
                        is_subject_of = root
                        liberty_desire < define:NDiplomacy|WAR_GOAL_INFAMY_LOW_LIBERTY_DESIRE_THRESHOLD
                    }
                }
                multiply = define:NDiplomacy|WAR_GOAL_INFAMY_LOW_LIBERTY_DESIRE_SUBJECT_MULT
            }
        }

        multiply = {
            desc = "INFAMY_FROM_REBELLIOUS_SUBJECT" # Rebellious as in they are the aggressor in the play
            value = 1

            if = {
                limit = {
                    scope:target_country = {
                        is_subject_of = root
                        is_diplomatic_play_enemy_of = root
                        scope:diplomatic_play ?= {
                            initiator_is = scope:target_country
                        }
                    }
                }
                multiply = define:NDiplomacy|WAR_GOAL_INFAMY_REBELLING_SUBJECT_MULT
            }
        }
	}

	on_enforced = {
		set_global_variable = eqfm_ali_rizvanbegovic_waxed
		scope:creator_country = {
			trigger_event = {
				id = balkfm_bosnia.004
				popup = yes
			}
		}
		scope:target_country = {
			trigger_event = {
				id = balkfm_bosnia.004
				popup = yes
			}
		}
		if = {
			limit = {
				c:MON ?= {
					is_involved_in_journal_entry = je_eqfm_karadag
				}
			}
			c:MON ?= {
				trigger_event = balkfm_bosnia.004
			}
		}
	}

	ai = {
		is_significant_demand = yes
	}

}
