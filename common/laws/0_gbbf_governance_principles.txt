law_imperial_divan = {
	icon = "gfx/interface/icons/law_icons/neo_absolutism.dds"
	group = lawgroup_distribution_of_power
	parent = law_autocracy
	progressiveness = -100
	
	is_visible = {
		OR = {
			has_law = law_type:law_imperial_divan
			has_law = law_type:law_ministerial_council
		}
	}
	
	disallowing_laws = {
		law_parliamentary_republic
	}
	
	can_impose = {
		always = no
	}
	
	on_activate = {
		scripted_effect_parties_disappearence = yes
		set_variable = com_hide_law_autocracy
		set_variable = com_hide_law_oligarchy
	}
	
	modifier = {
		country_legitimacy_headofstate_add = 30
		country_legitimacy_govt_total_clout_add = 120
		country_legitimacy_ideological_incoherence_mult = 0.25
		country_aristocrats_pol_str_mult = 0.25
		country_bureaucrats_pol_str_mult = 0.25
		country_officers_pol_str_mult = 0.25
		country_authority_add = 200
	}
	
	on_deactivate = {
		if = {
			limit = {
				NOR = {
					has_law = law_type:law_imperial_divan
					has_law = law_type:law_ministerial_council
				}
			}
			remove_variable = com_hide_law_autocracy
			remove_variable = com_hide_law_oligarchy
		}
	}
	
	ai_will_do = {
		ruler ?= {
			has_ideology = ideology:ideology_authoritarian
		}
	}
	
	ai_enact_weight_modifier = {
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
	
	ai_impose_chance = {
		value = 0
	}
}
