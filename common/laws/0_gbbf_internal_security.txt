law_gbbf_palace_service = {
	icon = gfx/interface/icons/law_icons/secret_police.dds
	group = lawgroup_internal_security
	parent = law_secret_police
	progressiveness = -25
	
	is_visible = {
		has_law = law_type:law_gbbf_palace_service
	}
	
	can_impose = {
		always = no
	}
	
	unlocking_technologies = {
		tech_bureaucracy
	}
	
	disallowing_laws = {
		law_protected_speech
		law_guaranteed_liberties
	}
	
	on_activate = {
		set_variable = com_hide_law_secret_police
	}
	
	on_deactivate = {
		if = {
			limit = {
				NOR = {
					has_law = law_type:law_gbbf_palace_service
					has_law = law_type:law_gbbf_teshkilat
				}
			}
			remove_variable = com_hide_law_secret_police
		}
	}
	
	modifier = {
		country_officers_pol_str_mult = 0.10
		country_bureaucrats_pol_str_mult = 0.10
		country_suppression_attraction_factor = -0.10
		political_movement_radicalism_add = -0.05
		country_coup_resistance_add = 0.75
		country_enable_exile_to_libya_bool = yes
	}
	
	ai_enact_weight_modifier = {
		value = 0
	}
}
