turkfm_greek_gunrunners_button = {
	name = "turkfm_greek_gunrunners_button"
	desc = "turkfm_greek_gunrunners_button_desc"

	visible = {
		country_has_primary_culture = cu:greek
		c:TUR ?= {
			OR = {
				owns_entire_state_region = STATE_ALBANIA
				owns_entire_state_region = STATE_MACEDONIA
				owns_entire_state_region = STATE_SKOPIA
				owns_entire_state_region = STATE_CYPRUS
				owns_entire_state_region = STATE_CRETE
				owns_entire_state_region = STATE_AYDIN
			}
		}
	}
	selected = {
		has_modifier = modifier_turkfm_supporting_gunrunners
	}
	possible = {
		has_technology_researched = nationalism
		any_scope_state = {
			region = sr:region_balkans
		}
		custom_tooltip = {
			text = turkfm_greek_gunrunners_possible
			NOT = {
				has_variable = turkfm_greek_gunrunners_lockout
			}
		}
	}
	effect = {
		if = {
			limit = {
				NOT = {
					has_modifier = modifier_turkfm_supporting_gunrunners
				}
			}
			custom_tooltip = turkfm_greek_gunrunners_effect1
			add_modifier = {
				name = modifier_turkfm_supporting_gunrunners
				multiplier = money_amount_multiplier_medium
			}
		}
		else_if = {
			limit = {
				has_modifier = modifier_turkfm_supporting_gunrunners
			}
			remove_modifier = modifier_turkfm_supporting_gunrunners
		}
		custom_tooltip = {
			text = turkfm_greek_gunrunners_effect2
			set_variable = {
				name = turkfm_greek_gunrunners_lockout
				years = 2
			}
		}
	}
	ai_chance = {
		value = -5
		if = {
			limit = {
				NOT = { has_modifier = modifier_turkfm_supporting_gunrunners }
				c:TUR ?= {
					relations:root < relations_threshold:neutral
				}
				weekly_net_fixed_income > money_amount_multiplier_medium
				gold_reserves > 0
			}
			add = 100
		}
		if = {
			limit = {
				NOT = { has_modifier = modifier_turkfm_supporting_gunrunners }
				weekly_net_fixed_income < money_amount_multiplier_medium
				gold_reserves < 0
			}
			add = -1000
		}
		if = {
			limit = {
				has_modifier = modifier_turkfm_supporting_gunrunners
				c:TUR ?= {
					relations:root > relations_threshold:neutral
				}
			}
			add = 100
		}
		if = {
			limit = {
				has_modifier = modifier_turkfm_supporting_gunrunners
				c:TUR ?= {
					relations:root < relations_threshold:neutral
				}
			}
			add = -100
		}
		if = {
			limit = {
				has_modifier = modifier_turkfm_supporting_gunrunners
				weekly_net_fixed_income < money_amount_multiplier_medium
				gold_reserves < 0
			}
			add = 1000
		}
	}
}

turkfm_accelerate_tensions_button = {
	name = "turkfm_accelerate_tensions_button"
	desc = "turkfm_accelerate_tensions_button_desc"

	visible = {
		NOT = {
			is_country_type = decentralized
		}
	}
	selected = {
		has_diplomatic_pact = {
			who = c:TUR
			type = turkfm_accelerate_tensions
		}
	}
	possible = {
		turkfm_balkans_involved_trigger = yes
		NOR = {
			any_scope_treaty = {
				binds = root
				binds = c:TUR
				any_scope_article = {
					has_type = alliance
				}
			}
			any_scope_treaty = {
				binds = root
				binds = c:TUR
				any_scope_article = {
					has_type = defensive_pact
				}
			}
		}
		custom_tooltip = {
			text = turkfm_tension_button_cooldown_trigger
			NOT = {
				has_variable = turkfm_tension_button_cooldown_var
			}
		}
	}
	effect = {
		if = {
			limit = {
				has_diplomatic_pact = {
					who = c:TUR
					type = turkfm_accelerate_tensions
				}
			}
			remove_diplomatic_pact = {
				country = c:TUR
				type = turkfm_accelerate_tensions
			}
		}
		else = {
			create_diplomatic_pact = {
				country = c:TUR
				type = turkfm_accelerate_tensions
			}
			custom_tooltip = {
				text = turkfm_tension_button_effect
				set_variable = {
					name = turkfm_tension_button_cooldown_var
					years = 2
				}
			}
		}
		if = {
			limit = {
				has_diplomatic_pact = {
					who = c:TUR
					type = turkfm_decelerate_tensions
				}
			}
			remove_diplomatic_pact = {
				country = c:TUR
				type = turkfm_decelerate_tensions
			}
		}
	}
}
