eqfm_embrace_senusiyya_button = {
	name = "eqfm_embrace_senusiyya_button"
	desc = "eqfm_embrace_senusiyya_button_desc"
	visible = {
		has_variable = eqfm_senusiyya_influence_var
		NOR = {
			has_variable = eqfm_embraced_senusiyya
			has_variable = eqfm_expelled_senusiyya
		}
	}
	possible = {
		OR = {
			has_law_or_variant = law_type:law_monarchy
			has_law_or_variant = law_type:law_theocracy
		}
		has_law_or_variant = law_type:law_state_religion
		trigger_if = {
			limit = {
				c:TRI ?= this
			}
			NOT = {
				has_journal_entry = je_mgrbfm_libya_centralization
			}
		}
		trigger_if = {
			limit = {
				c:TUN ?= this
			}
			NOT = {
				has_journal_entry = je_mgrbfm_tunis_modernization
			}
		}
		custom_tooltip = {
			text = eqfm_embrace_senusiyya_possible
			var:eqfm_senusiyya_influence_var >= 100
		}
	}
	effect = {
		trigger_event = {
			id = eqfm_senusiyya.009
		}
		set_variable = eqfm_embraced_senusiyya
	}
}

		}
		set_global_variable = eqfm_intimidated_sensussi
	}
	ai_chance = {
		value = 10
	}
