eqfm_embrace_sanusiyyah_button = {
	name = "eqfm_embrace_sanusiyyah_name"
	desc = "eqfm_embrace_sanusiyyah_desc"
	
	tags = { major_decision }
	
	visible = {
		is_involved_in_journal_entry = je_eqfm_sanusiyyah
		has_variable = eqfm_sanusiyyah_influence_var
		NOR = {
			has_variable = eqfm_embraced_sanusiyyah
			has_variable = eqfm_expelled_sanusiyyah
		}
	}
	possible = {
		has_law = law_type:law_people_of_the_book
		OR = {
			has_law_or_variant = law_type:law_monarchy
			has_law_or_variant = law_type:law_theocracy
		}
		trigger_if = {
			limit = {
				c:TRI ?= this
			}
			NOT = {
				has_journal_entry = je_mgrbfm_libya_centralization
			}
		}
		trigger_if = {
			limit = {
				c:TUN ?= this
			}
			NOT = {
				has_journal_entry = je_mgrbfm_tunis_modernization
			}
		}
		trigger_if = {
			limit = {
				c:FZN ?= this
			}
			NOT = {
				has_journal_entry = je_mgrbfm_securing_fezzan
			}
		}
		custom_tooltip = {
			text = eqfm_embrace_sanusiyyah_possible
			var:eqfm_sanusiyyah_influence_var >= 100
		}
	}
	effect = {
		trigger_event =  eqfm_sanusiyyah.304
		active_law:lawgroup_church_and_state = {
			add_amendment = {
				type = amendment_gbbf_state_sanusiyyah
				sponsor = prev.ig:ig_devout

		value = 0
