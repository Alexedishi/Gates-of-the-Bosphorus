# Senussi Movement Scripted Buttons
eqfm_embrace_senussi_button = {
	name = "eqfm_embrace_senussi_button"
	desc = "eqfm_embrace_senussi_button_desc"

	visible = {
		always = yes
	}

	possible = {
		OR = {
			has_law_or_variant = law_type:law_monarchy
			has_law_or_variant = law_type:law_theocracy
		}
		has_law_or_variant = law_type:law_state_religion
		trigger_if = {
			limit = {
				c:TRI ?= this
			}
			NOT = {
				has_journal_entry = je_mgrbfm_libya_centralization
			}
		}
		trigger_if = {
			limit = {
				c:TUN ?= this
			}
			NOT = {
				has_journal_entry = je_mgrbfm_tunis_modernization
			}
		}
		custom_tooltip = {
			text = eqfm_embrace_senussi_possible
			var:eqfm_senusiyya_var >= 100
		}
		hidden_trigger = {
			NOT = {
				has_variable = eqfm_senussi_outcome_var
			}
		}
	}

	effect = {
		custom_tooltip = {
			text = eqfm_embrace_senussi_button_effect
			set_variable = mgrbfm_embraced_senussi
			set_variable = eqfm_senussi_outcome_var
		}
	}

	ai_chance = {
		value = 100
	}
}

eqfm_supress_senusiyya_button = {
	name = "eqfm_supress_senusiyya_button"
	desc = "eqfm_supress_senusiyya_button_desc"

	visible = {
		always = yes
	}

	selected = {
		has_modifier = modifier_eqfm_desert_raids
	}

	possible = {
		has_technology_researched = centralization
		trigger_if = {
			limit = {
				NOT = {
					has_modifier = modifier_eqfm_desert_raids
				}
			}
			authority >= 200
		}
		trigger_if = {
			limit = {
				c:TRI ?= this
			}
			custom_tooltip = {
				text = eqfm_supress_senusiyya_possible_1
				has_global_variable = eqfm_intimidated_sensussi
			}
		}
		custom_tooltip = {
			text = eqfm_supress_senusiyya_possible_2
			NOT = {
				has_variable = mgrbfm_desert_raids_var
			}
		}
	}

	effect = {
		if = {
			limit = {
				has_modifier = modifier_eqfm_desert_raids
			}
			remove_modifier = modifier_eqfm_desert_raids
		}
		else = {
			add_modifier = modifier_eqfm_desert_raids
		}
		custom_tooltip = {
			text = eqfm_supress_senusiyya_effect
			set_variable = {
				name = mgrbfm_desert_raids_var
				years = 2
			}
		}
	}

	ai_chance = {
		value = 0
	}
}

eqfm_armed_delegation_button = {
	name = "eqfm_armed_delegation_button"
	desc = "eqfm_armed_delegation_button_desc"

	visible = {
		NOR = {
			has_global_variable = eqfm_intimidated_sensussi
		}
	}

	possible = {
		has_technology_researched = centralization
		army_size > 25 # Just in case you downsize :p
		OR = {
			eqfm_has_desert_expedition_trigger = yes
			owns_entire_state_region = STATE_LIBYAN_DESERT
		}
	}

	effect = {
		trigger_event = {
			id = eqfm_senusiyya.004 # Armed Diplomacy
		}
		set_global_variable = eqfm_intimidated_sensussi
	}

	ai_chance = {
		value = 10
	}
}
