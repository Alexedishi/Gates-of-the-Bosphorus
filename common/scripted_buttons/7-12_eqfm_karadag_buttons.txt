## Black Mountain Decision Buttons ##
eqfm_renounce_headtaking_button = {
	name = "eqfm_renounce_headtaking_name"
	desc = "eqfm_renounce_headtaking_desc"
	
	visible = {
		OR = {
            has_modifier = modifier_eqfm_practicing_headtaking_mon
            has_modifier = modifier_eqfm_practicing_headtaking_tur
        }
	}
	possible = {
		custom_tooltip = {
			text = eqfm_renounce_headtaking_possible
			NOT = {
				has_global_variable = turkfm_headtaking_cooldown_var
			}
		}
		trigger_if = {
			limit = {
				c:MON ?= this
			}
			c:TUR.relations:root >= relations_threshold:neutral
		}
		trigger_if = {
			limit = {
				c:TUR ?= this
			}
			c:MON.relations:root >= relations_threshold:neutral
		}
	}
	effect = {
		c:MON ?= {
			if = {
				limit = {
					this != root
					has_modifier = modifier_eqfm_practicing_headtaking_mon
				}
				trigger_event = {
					id = eqfm_karadag.010
					days = { 7 14 }
				}
				hidden_effect = {
					trigger_event = {
						id = eqfm_karadag.011
						days = 45
					}
				}
			}
		}
		c:TUR ?= {
			if = {
				limit = {
					this != root
					has_modifier = modifier_eqfm_practicing_headtaking_tur
				}
				trigger_event = {
					id = eqfm_karadag.010
					days = { 7 14 }
				}
				hidden_effect = {
					trigger_event = {
						id = eqfm_karadag.011
						days = 45
					}
				}
			}
		}
		custom_tooltip = {
			text = eqfm_renounce_headtaking_effect
			set_variable = {
				name = turkfm_negociated_headtaking
				months = 3
			}
			trigger_event = {
				id = eqfm_karadag.011
				days = 45
			}
		}
		custom_tooltip = {
			text = eqfm_headtaking_cooldown_GET
			set_global_variable = {
				name = turkfm_headtaking_cooldown_var
				years = 5
			}
		}
	}
	ai_chance = {
		value = 0
	}
}

eqfm_increase_raiding_button = {
	name = "eqfm_increase_raiding_name"
	desc = "eqfm_increase_raiding_desc"
	
	visible = {
		has_dlc_feature = ip3_content # National Awakening
		NOT = {
			has_global_variable = eqfm_montenegrin_raiding_settled
		}
		c:MON ?= this
	}
	possible = {
		custom_tooltip = {
			text = eqfm_raiding_status_cooldown
			NOT = {
				has_variable = eqfm_raiding_status_cooldown_var
			}
		}
		custom_tooltip = {
			text = eqfm_raiding_less_than_max
			var:raiding_intensity < 100
		}
	}
	selected = {
		has_variable = mon_more_raiding
	}
	effect = {
		if = {
			limit = {
				has_variable = mon_more_raiding
			}
			remove_variable = mon_more_raiding
		}
		else = {
			custom_tooltip = {
				text = eqfm_increase_raiding_effect
				set_variable = mon_more_raiding
			}
			custom_tooltip = {
				text = eqfm_raiding_status_cooldown_GET
				set_variable = {
					name = eqfm_raiding_status_cooldown_var
					years = 2
				}
			}
			c:RUS = {
				change_relations = {
					country = root
					value = -15
				}
			}
			c:TUR = {
				change_relations = {
					country = root
					value = -35
				}
			}
		}
	}
}

eqfm_negotiate_kotor_button = {
	name = "eqfm_negotiate_kotor_name"
	desc = "eqfm_negotiate_kotor_desc"
	
	visible = {
		c:MON ?= this
		NOT = {
			has_variable = turkfm_negotiated_kotor
		}
		p:x90A0E0 = { # Kotor
			owner = {
				c:AUS ?= this
			}
		}
	}
	possible = {
		trigger_if = {
			limit = {
				NOT = {
					has_dlc_feature = ip3_content # National Awakening
				}
			}
			OR = {
				is_subject_of = c:AUS
				is_in_same_power_bloc = c:AUS
				gbbf_has_alliance_trigger = {
					country = c:AUS
				}
				gbbf_has_defensive_pact_trigger = {
					country = c:AUS
				}
			}
			c:AUS = {
				relations:root >= relations_threshold:amicable
			}
			NOT = {
				owes_obligation_to = c:AUS
			}
		}
		trigger_if = {
			limit = {
				has_dlc_feature = ip3_content # National Awakening
			}
			government_legitimacy > 50
			c:RUS ?= {
				relations:root <= relations_threshold:poor
			}
			c:AUS = {
				relations:root >= relations_threshold:friendly
			}
			trigger_if = {
				limit = {
					has_journal_entry = je_montenegrin_raiding
				}
				custom_tooltip = {
					text = raiding_income_greater_tt
					eqfm_raiding_income_value < 700
				}
			}
		}
	}
	effect = {
		c:AUS = {
			trigger_event = {
				id = eqfm_karadag.012
				days = { 7 14 }
			}
		}
		set_variable = turkfm_negotiated_kotor
	}
	ai_chance = {
		value = 100
	}
}
