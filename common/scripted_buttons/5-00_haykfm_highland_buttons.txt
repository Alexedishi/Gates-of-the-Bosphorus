haykfm_claim_miri_miran_button = {
	name = "haykfm_claim_miri_miran_name"
	desc = "haykfm_claim_miri_miran_desc"

	visible = {
		NOT = {
			has_global_variable = turkfm_miri_miran_claimed
		}
		any_country = {
			country_has_primary_culture = cu:kurdish
			this != root
		}
	}
	possible = {
		custom_tooltip = {
			text = haykfm_claim_miri_miran_possible
			any_country = {
				count = all
				hidden_trigger = {
					this != root
				}
				country_has_primary_culture = cu:kurdish
				is_subject_of = root
			}
		}
	}
	effect = {
		trigger_event = {
			id = haykfm_kurdistan.003
			popup = yes
		}
		set_global_variable = turkfm_miri_miran_claimed
	}

	ai_chance = {
		value = 1000
	}
}
