gbbf_form_yugoslavia_button = {
	name = "gbbf_form_yugoslavia_name"
	desc = "gbbf_form_yugoslavia_desc"
	
	tags = { debug }
	
	visible = {
		NOT = {
			exists = c:YUG
		}
	}
	possible = {
		always = yes
	}
	effect = {
		random_state_in_yugoslavia = {
			limit = {
				owner = {
					is_player = no
				}
			}
			save_scope_as = yugoslavia_seed_state
			owner = {
				save_scope_as = yugoslavia_seed_country
			}
		}
		create_country = {
			tag = YUG
			origin = scope:yugoslavia_seed_country
			state = scope:yugoslavia_seed_state
			on_created = {
				remove_primary_culture = cu:bosniak
				remove_primary_culture = cu:croat
				remove_primary_culture = cu:slovene
				every_state_in_yugoslavia = {
					limit = {
						owner != c:YUG
					}
					set_state_owner = c:YUG
				}
			}
		}
		hidden_effect = {
			play_as = c:YUG
		}
	}
	ai_chance = {
		value = 0
	}
}
