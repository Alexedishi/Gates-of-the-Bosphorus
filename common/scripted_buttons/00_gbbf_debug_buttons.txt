gbbf_debug_clear_amendments_button = {
	name = "gbbf_debug_clear_amendments_button"
	desc = "gbbf_debug_clear_amendments_button_desc"

	visible = {
		has_variable = gbbf_debug_cheats_var
	}
	possible = {
		any_active_law = {
			OR = {
				has_amendment = amendment_type:amendment_generic_sponsor_polstr_increase
				has_amendment = amendment_type:amendment_generic_sponsor_polstr_approval_increase
				has_amendment = amendment_type:amendment_generic_sponsor_polstr_increase_negotiation_1
				has_amendment = amendment_type:amendment_generic_sponsor_polstr_increase_negotiation_2
				has_amendment = amendment_type:amendment_generic_sponsor_polstr_increase_negotiation_3
			}
		}
	}
	effect = {
		every_active_law = {
			limit = {
				OR = {
					has_amendment = amendment_type:amendment_generic_sponsor_polstr_increase
					has_amendment = amendment_type:amendment_generic_sponsor_polstr_approval_increase
					has_amendment = amendment_type:amendment_generic_sponsor_polstr_increase_negotiation_1
					has_amendment = amendment_type:amendment_generic_sponsor_polstr_increase_negotiation_2
					has_amendment = amendment_type:amendment_generic_sponsor_polstr_increase_negotiation_3
				}
			}
			every_scope_amendment = {
				remove_amendment = yes
			}
		}
	}
}

gbbf_break_ottoman_alliances = {
	name = "gbbf_break_ottoman_alliances"
	desc = "gbbf_break_ottoman_alliances_desc"

	visible = {
		has_variable = gbbf_debug_cheats_var
		exists = c:TUR
	}
	possible = {
		c:TUR = {
			any_scope_treaty = {
				any_scope_article = {
					OR = {
						has_type = alliance
						has_type = defensive_pact
					}
				}
			}
		}
	}
	effect = {
		c:TUR = {
			every_scope_treaty = {
				limit = {
					any_scope_article = {
						OR = {
							has_type = alliance
							has_type = defensive_pact
						}
					}
				}
				withdraw = {
					country = c:TUR
				}
			}
		}
	}
}
