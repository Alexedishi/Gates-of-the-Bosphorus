eqfm_greek_gunrunners_button = {
	name = eqfm_greek_gunrunners_button
	desc = eqfm_greek_gunrunners_button_desc

	visible = {
		balkfm_balkans_involved_trigger = yes
		country_has_primary_culture = cu:greek
		c:TUR ?= {
			OR = {
				owns_entire_state_region = STATE_ALBANIA
				owns_entire_state_region = STATE_MACEDONIA
				owns_entire_state_region = STATE_SKOPIA
				owns_entire_state_region = STATE_CYPRUS
				owns_entire_state_region = STATE_CRETE
				owns_entire_state_region = STATE_AYDIN
			}
		}
	}
	selected = {
		has_modifier = modifier_eqfm_supporting_gunrunners
	}
	possible = {
		has_technology_researched = nationalism
		any_scope_state = {
			region = sr:region_balkans
		}
		custom_tooltip = {
			text = eqfm_greek_gunrunners_possible
			NOT = {
				has_variable = eqfm_greek_gunrunners_lockout
			}
		}
	}
	effect = {
		if = {
			limit = {
				NOT = {
					has_modifier = modifier_eqfm_supporting_gunrunners
				}
			}
			custom_tooltip = eqfm_greek_gunrunners_effect1
			add_modifier = {
				name = modifier_eqfm_supporting_gunrunners
				multiplier = money_amount_multiplier_medium
			}
		}
		else_if = {
			limit = {
				has_modifier = modifier_eqfm_supporting_gunrunners
			}
			remove_modifier = modifier_eqfm_supporting_gunrunners
		}
		custom_tooltip = {
			text = eqfm_greek_gunrunners_effect2
			set_variable = {
				name = eqfm_greek_gunrunners_lockout
				years = 2
			}
		}
	}
	ai_chance = {
		value = -5
		if = {
			limit = {
				NOT = { has_modifier = modifier_eqfm_supporting_gunrunners }
				c:TUR ?= {
					relations:root < relations_threshold:neutral
				}
				fixed_income > money_amount_multiplier_medium
				gold_reserves > 0
			}
			add = 100
		}
		if = {
			limit = {
				NOT = { has_modifier = modifier_eqfm_supporting_gunrunners }
				fixed_income < money_amount_multiplier_medium
				gold_reserves < 0
			}
			add = -1000
		}
		if = {
			limit = {
				has_modifier = modifier_eqfm_supporting_gunrunners
				c:TUR ?= {
					relations:root > relations_threshold:neutral
				}
			}
			add = 100
		}
		if = {
			limit = {
				has_modifier = modifier_eqfm_supporting_gunrunners
				c:TUR ?= {
					relations:root < relations_threshold:neutral
				}
			}
			add = -100
		}
		if = {
			limit = {
				has_modifier = modifier_eqfm_supporting_gunrunners
				fixed_income < money_amount_multiplier_medium
				gold_reserves < 0
			}
			add = 1000
		}
	}
}

eqfm_accelerate_tensions_button = {
	name = eqfm_accelerate_tensions_button
	desc = eqfm_accelerate_tensions_button_desc

	visible = {
		balkfm_balkans_involved_trigger = yes
		NOT = {
			c:TUR ?= this
		}
	}
	selected = {
		has_diplomatic_pact = {
			who = c:TUR
			type = turkfm_accelerate_tensions
		}
	}
	possible = {
		country_rank >= rank_value:great_power
		eqfm_balkans_involved_trigger = yes
		NOR = {
			gbbf_has_alliance_trigger = {
				country = c:TUR
			}
			gbbf_has_defensive_pact_trigger = {
				country = c:TUR
			}
		}
		trigger_if = {
			limit = {
				NOT = {
					has_diplomatic_pact = {
						who = c:TUR
						type = turkfm_accelerate_tensions
					}
				}
			}
			c:TUR.relations:root <= relations_threshold:poor
		}
		custom_tooltip = {
			text = eqfm_tension_button_cooldown_trigger
			NOT = {
				has_variable = eqfm_tension_button_cooldown_var
			}
		}
	}
	effect = {
		if = {
			limit = {
				has_diplomatic_pact = {
					who = c:TUR
					type = turkfm_accelerate_tensions
				}
			}
			remove_diplomatic_pact = {
				country = c:TUR
				type = turkfm_accelerate_tensions
			}
		}
		else = {
			create_diplomatic_pact = {
				country = c:TUR
				type = turkfm_accelerate_tensions
			}
			custom_tooltip = {
				text = eqfm_tension_button_effect
				set_variable = {
					name = eqfm_tension_button_cooldown_var
					years = 2
				}
			}
		}
		if = {
			limit = {
				has_diplomatic_pact = {
					who = c:TUR
					type = turkfm_decelerate_tensions
				}
			}
			remove_diplomatic_pact = {
				country = c:TUR
				type = turkfm_decelerate_tensions
			}
		}
	}
	ai_chance = {
		value = -10

		# Logic to start Accelerating Tensions first
		if = {
			limit = {
				NOT = {
					has_diplomatic_pact = {
						who = c:TUR
						type = turkfm_accelerate_tensions
					}
				}
			}
			if = {
				limit = {
					c:TUR.relations:root >= relations_threshold:hostile
				}
				add = 25
			}
			else_if = {
				limit = {
					c:TUR.relations:root >= relations_threshold:cold
				}
				add = 20
			}
			else_if = {
				limit = {
					c:TUR.relations:root >= relations_threshold:poor
				}
				add = 15
			}
			if = {
				limit = {
					OR = {
						has_secret_goal = {
							who = c:TUR
							secret_goal = antagonize
						}
						has_secret_goal = {
							who = c:TUR
							secret_goal = conquer
						}
						has_secret_goal = {
							who = c:TUR
							secret_goal = dominate
						}
					}
				}
				add = 25
			}
			if = {
				limit = { 
					OR = {
						has_attitude = {
							who = c:TUR
							attitude = antagonistic
						}
						has_attitude = {
							who = c:TUR
							attitude = belligerent
						}
						has_attitude = {
							who = c:TUR
							attitude = domineering
						}
					}
				}
				add = 25
			}
			if = {
				limit = { 
					OR = {
						gbbf_has_rivalry_trigger = {
							country = c:TUR
						}
						gbbf_has_embargo_trigger = {
							country = c:TUR
						}
					}
				}
				add = 100
			}
			if = {
				limit = {
					has_attitude = {
						who = c:TUR
						attitude = conciliatory
					}
				}
				add = -100
			}
			if = {
				limit = {
					has_attitude = {
						who = c:TUR
						attitude = disinterested
					}
				}
				add = -1000
			}
		}
		
		# Logic to stop Accelerating Tensions second
		else_if = { 
			limit = {
				has_diplomatic_pact = {
					who = c:TUR
					type = turkfm_accelerate_tensions
				}
			}
			
			if = {
				limit = { 
					c:TUR.relations:root > relations_threshold:poor
				}
				add = 50
			}
			if = {
				limit = {
					OR = {
						has_secret_goal = {
							who = c:TUR
							secret_goal = befriend
						}
						has_secret_goal = {
							who = c:TUR
							secret_goal = protect
						}
					}
				}
				add = 100
			}
			if = {
				limit = {
					has_attitude = {
						who = c:TUR
						attitude = conciliatory
					}
				}
				add = 50
			}
			if = {
				limit = {
					has_attitude = {
						who = c:TUR
						attitude = disinterested
					}
				}
				add = 1000
			}
		}
	}
}
