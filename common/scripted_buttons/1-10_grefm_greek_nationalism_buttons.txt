## National Identity Buttons ##
grefm_declare_national_language_button = {
	name = "grefm_declare_national_language_button"
	desc = "grefm_declare_national_language_button_desc"
	
	visible = {
		NOT = {
			has_variable = grefm_language_var
		}
	}
	possible = {
		institution_investment_level = {
			institution = institution_schools
			value >= 3
		}
		literacy_rate > 0.40
		custom_tooltip = {
			text = grefm_declare_national_language_button_possible
			hidden_trigger = {
				je:je_grefm_national_language = {
					OR = {
						"scripted_bar_progress(grefm_dimotiki_progress_bar)" >= 100
						"scripted_bar_progress(grefm_katharevousa_progress_bar)" >= 100
						"scripted_bar_progress(grefm_khoine_progress_bar)" >= 100
						"scripted_bar_progress(grefm_rhomaiika_progress_bar)" >= 100
					}
				}
			}
		}
		trigger_if = {
			limit = {
				morgenrote_is_active = yes
			}
			custom_tooltip = {
				text = je_grefm_national_language_complete_mgr
				var:manzoni_literary_tradition_var >= 50
			}
		}
		hidden_trigger = {
			NOT = {
				has_variable = grefm_language_selection_var
			}
		}
	}
	selected = {
		has_variable = grefm_language_selection_var
	}
	effect = {
		trigger_event = grefm_greek_nationalism.007
		set_variable = {
			name = grefm_language_selection_var
			months = 3
		}
	}
}

grefm_embrace_hellenic_identity_button = {
	name = "grefm_embrace_hellenic_identity_button"
	desc = "grefm_embrace_hellenic_identity_button_desc"
	
	visible = {
		NOR = {
			has_law_or_variant = law_type:law_ethnostate
			has_law_or_variant = law_type:law_multicultural
		}
	}
	possible = {
		has_technology_researched = pan-nationalism
		custom_tooltip = {
			text = grefm_embrace_hellenic_identity_button_possible
			hidden_trigger = {
				has_variable = grefm_hellene_culture_var
				var:grefm_hellene_culture_var >= 10
			}
		}
		hidden_trigger = {
			NOT = {
				has_variable = grefm_culture_var
			}
		}
	}
	effect = {
		set_variable = {
			name = grefm_culture_var
			value = 1
		}
		trigger_event = grefm_greek_nationalism.009
		show_as_tooltip = {
			if = {
				limit = {
					NOT = {
						country_has_primary_culture = cu:greek
					}
				}
				add_primary_culture = cu:greek
			}
			if = {
				limit = {
					country_has_primary_culture = cu:rhomaios
				}
				remove_primary_culture = cu:rhomaios
			}
		}
	}
	ai_chance = {
		value = 10
	}
}

grefm_greek_sites_button = {
	name = "grefm_greek_sites_button"
	desc = "grefm_greek_sites_button_desc"
	
	visible = {
		morgenrote_is_active = no
		has_variable = grefm_archeaology_sites_var
		OR = {
			country_has_primary_culture = cu:greek
			has_variable = grefm_hellenic_sites_allowed
		}
	}
	selected = {
		var:grefm_archeaology_sites_var ?= 1
	}
	possible = {
		always = yes
	}
	effect = {
		custom_tooltip = {
			text = grefm_greek_sites_effect
			if = {
				limit = {
					var:grefm_archeaology_sites_var != 1
				}
				set_variable = {
					name = grefm_archeaology_sites_var
					value = 1
				}
			}
			else = {
				set_variable = {
					name = grefm_archeaology_sites_var
					value = 0
				}
			}
		}
	}
}
