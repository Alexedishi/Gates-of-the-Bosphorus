####---------- Unification of Yugoslavia Buttons ----------####
balkfm_form_yugoslavia_button = {
	name = "balkfm_form_yugoslavia"
	desc = "balkfm_form_yugoslavia_desc"
	
	visible = { # Fallback for BUL
		c:BUL ?= this
	}
	possible = { 
	# This are the vanilla requirements to form YUG, just retooled as a button
	# BUL players: if this aligns poorly with previous Bulgarian conquests, remember: you're the one pulling a texas left into Yugoslavism
		has_technology_researched = pan-nationalism
		is_subject = no
		calc_true_if = {
			amount >= 8
			owns_entire_state_region = STATE_WESTERN_SERBIA
			owns_entire_state_region = STATE_EASTERN_SERBIA
			owns_entire_state_region = STATE_MONTENEGRO
			owns_entire_state_region = STATE_BOSNIA
			owns_entire_state_region = STATE_CROATIA
			owns_entire_state_region = STATE_DALMATIA
			owns_entire_state_region = STATE_SLAVONIA
			owns_entire_state_region = STATE_SLOVENIA
			owns_entire_state_region = STATE_ISTRIA
			owns_entire_state_region = STATE_BANAT
			owns_entire_state_region = STATE_DELVIDEK
		}
	}
	effect = {
		change_tag = YUG
		set_global_variable = has_formed_yug
	}
}

balkfm_annex_mon_button = {
	name = "balkfm_annex_mon_button"
	desc = "balkfm_annex_mon_button_desc"
	
	visible = {
		exists = c:MON
		c:MON != this
	}
	possible = {
		has_technology_researched = pan-nationalism
		NOR = {
			has_law_or_variant = law_type:law_ethnostate
			has_law_or_variant = law_type:law_multicultural
		}
		custom_tooltip = {
			text = balkfm_yugoslav_leader_trigger
			global_var:balkfm_yugoslavia_journal_var ?= {
				var:balkfm_yugoslav_leader_var = root
			}
		}
		custom_tooltip = {
			text = balkfm_annex_mon_possible
			has_variable = balkfm_mon_annex_var
			var:balkfm_mon_annex_var >= 60
		}
		c:MON ?= {
			any_neighbouring_state = {
				owner = root
			}
			OR = {
				is_subject = no
				is_subject_of = root
			}
			NOR = {
				has_war_with = root
				has_truce_with = root
			}
		}
	}
	effect = {
		trigger_event = {
			id = balkfm_yugoslavia.020
			popup = yes
		}
		custom_tooltip = balkfm_annex_mon_effect
	}
}

balkfm_promote_montegrin_button = {
	name = "balkfm_promote_montegrin_button"
	desc =  "balkfm_promote_montegrin_button_desc"

	visible = {
		has_global_variable = balkfm_montegrin_culture_var
		NOT = {
			country_has_primary_culture = cu:montenegrin
		}
	}
	possible = {
		government_legitimacy >= 70
		radical_fraction = {
			culture = cu:montenegrin
			value < 0.25
		}
	}
	effect = {
		custom_tooltip = {
			text = balkfm_promote_montenegrin
			set_variable = {
				name = balkfm_yugoslav_promotion_var
				value = 0
			}
			set_variable = {
				name = balkfm_yugoslav_promotion_progress_var
				value = balkfm_yugoslav_promotion_progress_value
			}
			set_variable = {
				name = balkfm_yugoslav_promotion_target_var
				value = cu:montenegrin
			}
		}
	}
}
