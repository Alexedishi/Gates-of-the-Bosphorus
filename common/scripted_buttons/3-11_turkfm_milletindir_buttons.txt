turkfm_embrace_kemalism_button = {
	name = turkfm_embrace_kemalism
	desc = turkfm_embrace_kemalism_desc
	
	visible = { always = yes }
	possible = {
		custom_tooltip = {
			text = turkfm_embrace_kemalism_possible
			je:je_turkfm_milletindir.var:turkfm_kemalism_var >= 10
		}
		hidden_trigger = {
			NOT = {
				has_variable = turkfm_culture_var
			}
		}
	}
	effect = { 
		set_variable = {
			name = turkfm_culture_var
			value = flag:turkfm_kemalism
		}
	}
}
