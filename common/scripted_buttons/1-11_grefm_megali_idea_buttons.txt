grefm_annex_heptanese_button = {
	name = "grefm_annex_heptanese_button"
	desc = "grefm_annex_heptanese_button_desc"
	
	visible = {
		country_has_primary_culture = cu:greek
		c:ION ?= {
			this != root
			top_overlord ?= {
				is_player = no
			}
		}
		has_variable = grefm_heptanese_annex_var
	}
	possible = {
		c:ION ?= {
			is_at_war = no
			trigger_if = {
				limit = {
					is_subject = yes
				}
				top_overlord ?= {
					custom_tooltip = {
						text = grefm_annex_heptanese_possible_1
						NOT = { has_war_with = ROOT }
					}
					relations:root >= relations_threshold:cordial
				}
			}
			relations:root >= relations_threshold:cordial
		}
		custom_tooltip = {
			text = grefm_annex_heptanese_possible_2
			var:grefm_heptanese_annex_var >= 60
		}
		OR = {
			is_subject = no
			c:ION ?= {
				is_subject_of = root
			}
			trigger_if = {
				limit = {
					is_subject = yes
					c:ION ?= {
						is_subject = yes
					}
				}
				custom_tooltip = {
					text = grefm_has_same_overlord_trigger
					overlord = c:ION.overlord
				}
			}
		}
		is_at_war = no
	}
	effect = {
		custom_tooltip = {
			text = grefm_annex_heptanese_effect
			trigger_event = {
				id = grefm_megali_idea.100
				popup = yes
			}
			set_variable = {
				name = grefm_heptanese_annex_var
				value = 0
			}
		}
	}
	ai_chance = {
		value = 100
	}
}

	}

		c:GBR ?= {
			c:GRE = {
				relations:PREV >= relations_threshold:neutral

		}

		}
		NOR = {
			has_truce_with = c:GRE

		}
	}
	
	effect = {
		custom_tooltip = {
			text = grefm_greek_enosis
			hidden_effect = {
				play_as = c:GRE
				c:GRE = {
					trigger_event = grefm_heptanese.021
				}
			}
		}
	}
}
