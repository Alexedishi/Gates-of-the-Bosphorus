## Hellenic Variant Buttons ##
grefm_annex_heptanese_button = {
	name = "grefm_annex_heptanese_button"
	desc = "grefm_annex_heptanese_button_desc"
	
	visible = {
		country_has_primary_culture = cu:greek
		c:ION ?= {
			this != root
			top_overlord ?= {
				is_player = no
			}
		}
		has_variable = grefm_heptanese_annex_var
	}
	possible = {
		c:ION ?= {
			is_at_war = no
			relations:root >= relations_threshold:cordial
			trigger_if = {
				limit = {
					is_subject = yes
				}
				top_overlord ?= {
					custom_tooltip = {
						text = grefm_annex_heptanese_possible_1
						NOT = { has_war_with = ROOT }
					}
					relations:root >= relations_threshold:cordial
				}
			}
		}
		is_at_war = no
		is_subject = no
		custom_tooltip = {
			text = grefm_annex_heptanese_possible_2
			var:grefm_heptanese_annex_var >= 60
		}
	}
	effect = {
		custom_tooltip = { 
			text = grefm_annex_heptanese_effect
			trigger_event = {
				id = grefm_megali_idea.100 # ION Annexation
				popup = yes
			}
			set_variable = { 
				name = grefm_heptanese_annex_var
				value = 0
			}
		}
	}
	ai_chance = {
		value = 100
	}
}

grefm_ionian_enosis_button = {
	name = "grefm_greek_enosis_button"
	desc = "grefm_greek_enosis_button_desc"
	
	visible = {
		c:ION ?= this
		exists = c:GRE
		exists = c:GBR
		is_subject_of = c:GBR
	}
	
	possible = {
		c:GBR ?= {
			c:GRE = {
				relations:PREV >= relations_threshold:neutral
			}
		}
		c:GBR ?= {
			relations:root >= relations_threshold:amicable
		}
		c:GRE ?= {
			relations:root >= relations_threshold:amicable
		}
		NOR = {
			has_truce_with = c:GRE
			has_truce_with = c:GBR
		}
	}
	
	effect = {
		custom_tooltip = {
			text = grefm_greek_enosis
			hidden_effect = {
				play_as = c:GRE
				c:GRE = {
					trigger_event = {
						id = grefm_heptanese.021# Ionian Union with the Greek State
					}
				}
			}
		}
	}
}

grefm_ionian_enosis_alternate_button = {
	# this is in case ION breaks away from GBR
	name = "grefm_greek_enosis_button"
	desc = "grefm_greek_enosis_button_desc"
	
	visible = {
		c:ION ?= this
		exists = c:GRE
		trigger_if = {
			limit = {
				exists = c:GBR
			}
			NOT = {
				is_subject_of = c:GBR
			}
		}
	}
	
	possible = {
		c:GRE ?= {
			relations:root >= relations_threshold:amicable
		}
		OR = {
			is_subject = no
			is_subject_of = c:GRE
		}
		NOT = {
			has_truce_with = c:GRE
		}
	}
	
	effect = {
		custom_tooltip = {
			text = grefm_greek_enosis
			hidden_effect = {
				play_as = c:GRE
				c:GRE = {
					trigger_event = {
						id = grefm_heptanese.021# Ionian Union with the Greek State
					}
				}
			}
		}
	}
}
