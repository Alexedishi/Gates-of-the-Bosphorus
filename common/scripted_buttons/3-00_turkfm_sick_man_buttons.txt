turkfm_arrest_ali_pasha_button = {
	name = "turkfm_arrest_ali_pasha_button"
	desc = "turkfm_arrest_ali_pasha_button_desc"

	visible ={
		c:TUR_BOS ?= {
			is_subject_of = root
			ruler = {
				has_template = template_balkfm_rizvanbegovic_ali
			}
		}
	}
	possible = {
		c:TUR_BOS ?= {
			OR = {
				is_subject_type = subject_type_puppet
				is_subject_type = subject_type_vassal
				liberty_desire < 25
				NOT = { is_diplomatic_play_enemy_of = root }
			}
		}
	}
	effect = {
		trigger_event = {
			id = turkfm_tanzimat.021 # Ali Pasha Captured
			days = { 10 30 }
		}
	}
	ai_chance = {
		value = 100
	}

}
