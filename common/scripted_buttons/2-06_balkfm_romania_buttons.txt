balkfm_unite_principalities_button = {
	name = "balkfm_unite_principalities"
	desc = "balkfm_unite_principalities_desc"
	
	visible = {
		NOT = {
			has_variable = ROM_united_principality
		}
	}
	
	possible = {
		custom_tooltip = {
			text = balkfm_friend_with_all_great_powers_tt
			NOT = {
				any_country = {
					country_rank >= rank_value:great_power
					NOT = { has_war_with = ROOT }
					has_interest_marker_in_region = ROOT.capital.region
					relations:root < 0
				}
			}
		}
		custom_tooltip = {
			text = balkfm_rom_amicable_or_better_relations_tt
			NOR = {
				has_variable = ROM_united_principality
			}
			OR = {
				is_subject_type = subject_type_protectorate
				is_subject_type = subject_type_dominion
				is_subject_type = subject_type_tributary
				is_subject = no
			}
			any_country = {
				NOT = { this = ROOT }
				country_has_primary_culture = cu:romanian
				relations:root >= 25
				OR = {
					AND = {
						exists = top_overlord
						OR = {
							is_subject_type = subject_type_protectorate
							is_subject_type = subject_type_dominion
							is_subject_type = subject_type_tributary
						}
						top_overlord = {
							THIS = root.top_overlord
						}
					}
					is_subject = no
				}
			}
		}
	}
	
	effect = {
		set_variable = ROM_united_principality # this is placed here to reduce variable complexity; the vanilla variable is used for flag definition compatibility
		if = {
			limit = {
				is_subject = yes
				exists = top_overlord
			}
			top_overlord = {
				trigger_event = {
					id = balkfm_romania.790 # The Romanian Declaration of Union (should be from TUR scope)
				}
			}
		}
		trigger_event = {
			id = balkfm_romania.700 # Our Declaration of Union
		}
		set_variable = {
			name = romanian_union_var
			value = 0
		}
	}
}

balkfm_bessarabian_retrocession_button = { # vanilla button, integrated into balkfm
	name = "balkfm_bessarabian_retrocession"
	desc = "balkfm_bessarabian_retrocession_desc"
	
	visible = {
		has_variable = ROM_united_principality
		NOT = { has_variable = balkfm_requested_retrocession }
		NOR = {
			any_scope_state = {
				state_region = s:STATE_BESSARABIA
			}
			any_subject_or_below = {
				any_scope_state = {
					state_region = s:STATE_BESSARABIA
				}
			}
		}
	}
	
	ai_chance = {
		value = 100
	}
	
	possible = {
		custom_tooltip = {
			text = je_south_bessarabian_retrocession_button_tt1
			
			any_country = {
				country_rank >= rank_value:great_power
				NOT = {
					has_war_with = ROOT
					any_scope_state = {
						state_region = s:STATE_BESSARABIA
					}
				}
				has_interest_marker_in_region = ROOT.capital.region
				relations:root > 25
			}
		}
		custom_tooltip = {
			text = je_south_bessarabian_retrocession_button_tt2
			
			OR = {
				any_neighbouring_state = {
					state_region = s:STATE_BESSARABIA
					owner = {
						NOT = { ROOT = this }
						relations:root > 0
						NOT = { has_war_with = ROOT }
					}
					is_split_state = no
					is_capital = no
				}
				any_subject_or_below = {
					any_neighbouring_state = {
						state_region = s:STATE_BESSARABIA
						owner = {
							NOT = { ROOT = this }
							relations:root > 0
							NOT = { has_war_with = ROOT }
						}
						is_split_state = no
						is_capital = no
					}
				}
			}
		}
		hidden_trigger = {
			NOT = {
				has_variable = balkfm_requested_retrocession
			}
		}
	}
	
	effect = {
		set_variable = balkfm_requested_retrocession
		random_country = {
			limit = {
				NOT = { ROOT = this }
				relations:root >= 25
				NOT = { has_war_with = ROOT }

					state_region = s:STATE_BESSARABIA
				}
			}
			trigger_event = {
				id = balkfm_romania.791 # The South Bessarabian Retrocession
			}
		}
	}
}

balkfm_trs_provoke_button = {
	name = "balkfm_trs_provoke_button"
	desc = "balkfm_trs_provoke_button_desc"
	
	visible = {
		NOT = {
			exists = c:TRS
		}
		c:AUS ?= {
			OR = {
				has_state_in_state_region = STATE_NORTHERN_TRANSYLVANIA
				has_state_in_state_region = STATE_SOUTHERN_TRANSYLVANIA
			}
		}
	}
	
	possible = {
		NOT = { has_modifier = modifier_balkfm_nationalist_agitation }
		trigger_if = {
			limit = {
				c:AUS = {
					has_state_in_state_region = STATE_NORTHERN_TRANSYLVANIA
				}
			}
			s:STATE_NORTHERN_TRANSYLVANIA = {
				any_scope_state = {
					owner = {
						c:AUS = this
					}
					radical_fraction = {
						culture = cu:romanian
						value >= 0.50
					}
				}
			}
		}
		trigger_if = {
			limit = {
				c:AUS = {
					has_state_in_state_region = STATE_SOUTHERN_TRANSYLVANIA
				}
			}
			s:STATE_SOUTHERN_TRANSYLVANIA = {
				any_scope_state = {
					owner = {
						c:AUS = this
					}
					radical_fraction = {
						culture = cu:romanian
						value >= 0.50
					}
				}
			}
		}
	}
	
	effect = {
		add_modifier = {
			name = modifier_balkfm_nationalist_agitation
			months = gbbf_duration_3_years
		}
		hidden_effect = {
			c:AUS ?= {
				if = {
					limit = {
						has_state_in_state_region = STATE_NORTHERN_TRANSYLVANIA
						has_state_in_state_region = STATE_SOUTHERN_TRANSYLVANIA
					}
					create_country = {
						tag = TRS
						origin = this
						state = s:STATE_NORTHERN_TRANSYLVANIA.region_state:AUS
						state = s:STATE_SOUTHERN_TRANSYLVANIA.region_state:AUS
					}
				}
				else_if = {
					limit = {
						has_state_in_state_region = STATE_NORTHERN_TRANSYLVANIA
					}
					create_country = {
						tag = TRS
						origin = this
						state = s:STATE_NORTHERN_TRANSYLVANIA.region_state:AUS
					}
				}
				else_if = {
					limit = {
						has_state_in_state_region = STATE_SOUTHERN_TRANSYLVANIA
					}
					create_country = {
						tag = TRS
						origin = this
						state = s:STATE_SOUTHERN_TRANSYLVANIA.region_state:AUS
					}
				}
				trigger_event = {
					id = balkfm_romania.711 # Romanian Provocation in Transylvania (AUS)
				}
			}
			c:TRS ?= {
				every_scope_state = {
					remove_modifier = modifier_balkfm_separatist_tensions
					add_modifier = {
						name = modifier_balkfm_realized_rebellion
						months = gbbf_duration_2_plus_half_years
					}
					add_loyalists_in_state = {
						culture = cu:romanian
						value = 0.50
					}
				}
			}
		}
		trigger_event = {
			id = balkfm_romania.710 # Romanian Provocation in Transylvania (ROM)
		}
	}
	
	ai_chance = {
		value = 0
		if = {
			limit = {
				has_secret_goal = {
					who = c:AUS
					secret_goal = antagonize
				}
			}
			add = 10
		}
	}
}
