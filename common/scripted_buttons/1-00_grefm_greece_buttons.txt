grefm_dismiss_auxilliaries_button = {
	name = "grefm_dismiss_auxilliaries_button"
	desc = "grefm_dismiss_auxilliaries_button_desc"
	
	visible = {
		has_modifier = modifier_grefm_bavarian_auxiliaries
		NOT = {
			has_variable = grefm_dismissed_bavarian_auxiliaries
		}
	}
	
	possible = {
		custom_tooltip = {
			text = grefm_regency_discharged
			NOT = {
				has_modifier = modifier_grefm_armansperg_regency
			}
		}
		has_modifier = modifier_grefm_bavarian_auxiliaries
	}
	
	effect = {
		set_variable = grefm_dismissed_bavarian_auxiliaries
		trigger_event = {
			id = grefm_greece.018 # Bavarian Auxiliaries Dismissed
			popup = yes
		}
		show_as_tooltip = {
			remove_modifier = modifier_grefm_bavarian_auxiliaries
			remove_modifier = modifier_grefm_funding_bavarian_auxiliaries
			add_modifier = {
				name = modifier_grefm_dismissed_bavarians
				months = gbbf_duration_5_years
				is_decaying = yes
			}
			if = {
				limit = {
					exists = c:BAV
				}
				create_incident = {
					country = c:BAV
					value = 5
				}

			}
		}
		custom_tooltip = grefm_dismiss_auxilliaries_effect
		if = {
			limit = {
				has_journal_entry = je_grefm_bavarocracy_main
			}
			custom_tooltip = {
				text = grefm_instability_minus_25
				change_variable = {
					name = grefm_bavarocracy_instability_var
					subtract = 25
				}
			}
		}
		else_if = {
			limit = {
				has_journal_entry = je_grefm_bavarocracy_ip3
			}
			grefm_bavarocracy_medium_support_effect = yes
		}
		grefm_calculate_bavarocracy_variables_effect = yes
	}
	
	ai_chance = {
		value = 10
	}
}

		NOT = {
			has_variable = grefm_ruler_conversion_var

		ruler = {
			religion = rel:catholic
		}
	}
	possible = {
		custom_tooltip = {
			text = grefm_regency_discharged
			NOT = {
				has_modifier = modifier_grefm_armansperg_regency
			}
		}
		custom_tooltip = {
			text = grefm_bavarocracy_convert_button_possible
			ruler.religion = rel:catholic
		}
	}
	effect = {
		trigger_event = {
			id = grefm_greece.093 # The True Faith of Greece
			popup = yes
		}
		show_as_tooltip = {
			custom_tooltip = grefm_bavarocracy_convert_button_effect
			c:BAV ?= {
				change_relations = {
					country = root
					value = -30
				}
				create_diplomatic_catalyst = {
					type = catalyst_event_negative
					target = root
				}
			}
			c:PAP ?= {
				change_relations = {
					country = root
					value = -30
				}
				create_diplomatic_catalyst = {
					type = catalyst_event_negative
					target = root
				}
			}
			c:RUS ?= {
				change_relations = {
					country = root
					value = 30
				}
				create_diplomatic_catalyst = {
					type = catalyst_event_positive
					target = root
				}
			}
			grefm_bavarocracy_medium_support_effect = yes
		}
		set_variable = grefm_ruler_conversion_var
	}

}

grefm_favor_hellene_king_button = {
	name = "grefm_favor_hellene_king_button"
	desc = "grefm_favor_hellene_king_button_desc"
	
	visible = {
		always = yes
	}
	
	selected = {
		has_variable = grefm_favoring_hellene_king
	}
	
	possible = {
		always = yes
	}
	
	effect = {
		custom_tooltip = {
			text = grefm_favor_hellene_king_effect
			hidden_effect = {
				if = {
					limit = {
						has_variable = grefm_favoring_hellene_king
					}
					remove_variable = grefm_favoring_hellene_king
				}
				else = {
					set_variable = grefm_favoring_hellene_king
					remove_variable = grefm_favoring_roman_king
				}
			}
		}
	}
	
}
