grefm_dismiss_auxilliaries_button = {
	name = grefm_dismiss_auxilliaries_name
	desc = grefm_dismiss_auxilliaries_desc
	
	visible = {
		has_modifier = modifier_grefm_bavarian_auxiliaries
		NOT = {
			has_variable = grefm_dismissed_bavarian_auxiliaries
		}
	}
	possible = {
		custom_tooltip = {
			text = grefm_regency_discharged
			NOT = {
				has_modifier = modifier_grefm_armansperg_regency
			}
		}
		has_modifier = modifier_grefm_bavarian_auxiliaries
	}
	effect = {
		set_variable = grefm_dismissed_bavarian_auxiliaries
		trigger_event = {
			id = grefm_greece.013
			popup = yes
		}
		show_as_tooltip = {
			if = {
				limit = {
					has_modifier = modifier_grefm_bavarian_auxiliaries
				}
				remove_modifier = modifier_grefm_bavarian_auxiliaries
			}
			if = {
				limit = {
					has_modifier = modifier_grefm_funding_bavarian_auxiliaries
				}
				remove_modifier = modifier_grefm_funding_bavarian_auxiliaries
			}
			add_modifier = {
				name = modifier_grefm_dismissed_bavarians
				months = gbbf_normal_modifier_time
				is_decaying = yes
			}
			if = {
				limit = {
					exists = c:BAV
				}
				create_incident = {
					country = c:BAV
					value = 5
				}

			}
		}
		custom_tooltip = grefm_dismiss_auxilliaries_effect
		if = {
			limit = {
				has_dlc_feature = ip3_content
			}
			grefm_bavarocracy_medium_support_effect = yes
		}
		else = {
			custom_tooltip = {
				text = grefm_instability_minus_25
				change_variable = {
					name = grefm_bavarocracy_instability_var
					subtract = 25
				}
			}
		}
		grefm_calculate_bavarocracy_variables_effect = yes
	}
	ai_chance = {
		value = 10
	}
}

grefm_philhellenic_loan_button = {
	name = grefm_philhellenic_loan_name
	desc = grefm_philhellenic_loan_desc
	
	visible = {
		always = yes
	}
	possible = {
		NOT = {
			any_scope_treaty = {
				binds = root
				any_scope_article = {
					has_type = gbbf_philhellene_loan
				}
			}
		}
		OR = {
			c:GBR ?= {
				relations:root >= relations_threshold:amicable
			}
			c:FRA ?= {
				relations:root >= relations_threshold:amicable
			}
			c:RUS ?= {
				relations:root >= relations_threshold:amicable
			}
		}
		custom_tooltip = {
			text = grefm_regency_discharged
			NOT = {
				has_modifier = modifier_grefm_armansperg_regency
			}
		}
		custom_tooltip = {
			text = grefm_philhellenic_loan_trigger
			has_variable = grefm_national_bank_var
		}
	}
	effect = {
		add_treasury = 100000
		ordered_country = {
			limit = {
				OR = {
					c:GBR ?= this
					c:FRA ?= this
					c:RUS ?= this
				}
			}
			order_by = relations:root
			position = 0
			check_range_bounds = no
			save_scope_as = grefm_philhellenic_loaner_scope
		}
		create_treaty = {
			first_country = root
			second_country = scope:grefm_philhellenic_loaner_scope
			
			is_draft = no
			binding_period = { years = 5 }
			
			articles_to_create = {
				{
					article = gbbf_philhellene_loan
					source_country = scope:grefm_philhellenic_loaner_scope
					target_country = root
				}
				{
					article = money_transfer
					source_country = root
					target_country = scope:grefm_philhellenic_loaner_scope
					inputs = {
						{ quantity = 400 }
						# Raw math is 384.62, so we round to 400 to simulate interest
						# At 0.04% interest, assuming it compounds monthly, this is wildly better than any rate Greece ever got
					}
				}
			}
		}
	}
}
