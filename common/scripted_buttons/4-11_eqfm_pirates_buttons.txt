## Albanian Piracy Scripted Buttons ##
eqfm_hire_retired_pirates_button = {
	name = "eqfm_hire_retired_pirates_button"
	desc = "eqfm_hire_retired_pirates_button_desc"

	visible = {
		c:TUR ?= this
	}

	possible = {
		NOR = {
			has_modifier = modifier_eqfm_hired_pirates
			hidden_trigger = {
				has_variable = eqfm_selecting_recruits
			}
		}
	}

	effect = {
		trigger_event = {
			id = eqfm_pirates.080 # Treating with Corsairs
			popup = yes
		}
		set_variable = {
			name = eqfm_selecting_recruits
			months = 3
		}
	}

	ai_chance = {
		value = 0
	}
}

	name = "eqfm_adriatic_expedition_button"

		eqfm_has_any_expedition_trigger = no
	}

		eqfm_no_expedition_trigger = yes

			id = eqfm_pirates.002 # Pacifying the Adriatic
			popup = yes

	}

	ai_chance = {
		value = 0
	}
}
