}

## Adriatic Expedition Scripted Buttons ##
eqfm_deploy_adriatic_expedition_button = {
	name = "eqfm_deploy_adriatic_expedition_button"
	desc = "eqfm_deploy_adriatic_expedition_button_desc"

	visible = {
		eqfm_has_adriatic_expedition_trigger = no
	}
	possible = {
		eqfm_has_adriatic_expedition_trigger = no
		NOT = {
			has_modifier = modifier_eqfm_failed_adriatic_expedition_country
		}
	}
	effect = {
		trigger_event = {
			id = eqfm_pirates.002 # Pacifying the Adriatic
			popup = yes
		}
		set_variable = eqfm_adriatic_expedition_var
	}
	ai_chance = {
		value = 5
	}
}

eqfm_recall_adriatic_expedition_button = {
	name = "eqfm_cancel_adriatic_expedition_button"
	desc = "eqfm_cancel_adriatic_expedition_button_desc"

	visible = {
		eqfm_has_adriatic_expedition_trigger = yes
		NOT = {
			hidden_trigger = {
				has_variable = eqfm_expedition_piracy_transit_var
			}
		}
	}
	possible = {
		eqfm_has_adriatic_expedition_trigger = yes
	}
	effect = {
		eqfm_cancel_piracy_expedition_effect = yes
		eqfm_clear_piracy_expedition_variables_effect = yes
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				OR = {
					has_modifier = declared_bankruptcy
					in_default = yes
				}
			}
			add = 1000
		}
	}
}

eqfm_resupply_adriatic_expedition_button = {
	name = "eqfm_resupply_adriatic_expedition_button"
	desc = "eqfm_resupply_adriatic_expedition_button_desc"
	
	visible = {
		eqfm_has_adriatic_expedition_trigger = yes
	}
	possible = {
		NOR = {
			has_modifier = modifier_eqfm_resupplied_adriatic_expedition
			custom_tooltip = {
				text = eqfm_resupply_adriatic_expedition_button_possible_1
				has_variable = eqfm_piracy_chase_var
			}
			hidden_trigger = {
				has_variable = eqfm_expedition_piracy_transit_var
			}
		}
		custom_tooltip = {
			text = eqfm_resupply_adriatic_expedition_button_possible_2
			var:eqfm_adriatic_expedition_peril_var >= 3
		}
		
	}
	effect = {
		add_modifier = {
			name = modifier_eqfm_resupplied_adriatic_expedition
			months = gbbf_duration_2_plus_half_years
			multiplier = money_amount_multiplier_small
		}
		custom_tooltip = {
			text = eqfm_resupply_adriatic_expedition_button_effect
			change_variable = {
				name = eqfm_adriatic_expedition_peril_var
				subtract = 3
			}
		}
	}
	ai_chance = {
		value = 5
		if = {
			limit = {
				var:eqfm_adriatic_expedition_peril_var >= 4
				in_default = no
				NOT = {
					has_modifier = declared_bankruptcy
				}
			}
			add = 25
		}
	}
}
