turkfm_start_tanzimat_button = {
	name = "turkfm_start_tanzimat_button"
	desc = "turkfm_start_tanzimat_button_desc"

	visible = {
		NOT = {
			has_global_variable = turkfm_tanzimat_started
		}
	}
	possible = {
		custom_tooltip = {
			text = turkfm_mahmud_dead_tt
			ruler = {
				NOT = {
					has_template = template_turkfm_osmanoglu_mahmut_ii
				}
			}
		}
	}
	effect = {
		trigger_event = {
			id = turkfm_tanzimat.002 # Edict of Gülhane
		}
		show_as_tooltip = {
			add_modifier = {
				name = modifier_turkfm_reorganizing_spirit
				years = 30
			}
			custom_tooltip = {
				text = turkfm_tanzimat_declaration
				change_variable = {
					name = turkfm_tanzimat_completion_var
					add = 1
				}
			}
			every_interest_group = {
				limit = {
					law_stance = {
						law = law_type:law_state_religion
						value > neutral
					}
					NOR = {
						law_stance = {
							law = law_type:law_landed_voting
							value > neutral
						}
						law_stance = {
							law = law_type:law_wealth_voting
							value > neutral
						}
						law_stance = {
							law = law_type:law_census_voting
							value > neutral
						}
						law_stance = {
							law = law_type:law_universal_suffrage
							value > neutral
						}
					}
				}
				add_modifier = {
					name = modifier_turkfm_influence_questioned
					days = long_modifier_time
					is_decaying = yes
				}
			}
			every_political_movement = {
				limit = {
					OR = {
						is_political_movement_type = movement_religious_majority
						is_political_movement_type = movement_modernizer
					}
				}
				add_modifier = {
					name = modifier_turkfm_declared_reforms
					years = 20
				}
			}
		}
		set_global_variable = turkfm_tanzimat_started
	}
}

turkfm_enforce_identity_button = {
	name = "turkfm_enforce_identity_button"
	desc = "turkfm_enforce_identity_button_desc"

	visible = {
		always = yes
	}
	possible = {
		custom_tooltip = {
			text = turkfm_enforce_identity_button_possible_1
			hidden_trigger = {
				var:turkfm_tanzimat_reform_var >= 1000
			}
		}
		custom_tooltip = {
			text = turkfm_enforce_identity_button_possible_2
			hidden_trigger = {
				OR = {
					var:turkfm_tanzimat_identity_var < 100
					var:turkfm_tanzimat_identity_var > 900
				}
			}
		}
		hidden_trigger = {
			NOT = {
				has_variable = turkfm_selecting_identity_var
			}
		}
	}
	effect = {
		trigger_event = {
			id = turkfm_tanzimat.005 # The Identity of the Empire
		}
		set_variable = turkfm_selecting_identity_var
	}
}
