turkfm_start_tanzimat_button = {
	name = "turkfm_start_tanzimat_name"
	desc = "turkfm_start_tanzimat_desc"

	visible = {
		NOT = {
			has_global_variable = turkfm_tanzimat_started
		}
	}
	possible = {
		custom_tooltip = {
			text = turkfm_mahmud_dead_tt
			ruler = {
				NOT = {
					has_template = template_turkfm_osmanoglu_mahmut_ii
				}
			}
		}
	}
	effect = {
		trigger_event = {
			id = turkfm_tanzimat.002
			popup = yes
		}
		turkfm_add_tanzimat_agenda_effect = {
			type = agriculture
		}
		turkfm_add_tanzimat_agenda_effect = {
			type = military
		}
		turkfm_add_tanzimat_agenda_effect = {
			type = civics
		}
		turkfm_add_tanzimat_agenda_effect = {
			type = economy
		}
		turkfm_add_tanzimat_agenda_effect = {
			type = education
		}
		turkfm_add_tanzimat_agenda_effect = {
			type = security
		}
		turkfm_add_tanzimat_agenda_effect = {
			type = slavery
		}
		add_modifier = {
			name = modifier_turkfm_reorganizing_spirit
			months  = 360
		}
		ig:ig_intelligentsia = {
			add_ideology = ideology_anti_slavery
		}
		every_interest_group = {
			limit = {
				law_stance = {
					law = law_type:law_state_religion
					value > neutral
				}
				NOR = {
					law_stance = {
						law = law_type:law_landed_voting
						value > neutral
					}
					law_stance = {
						law = law_type:law_wealth_voting
						value > neutral
					}
					law_stance = {
						law = law_type:law_census_voting
						value > neutral
					}
					law_stance = {
						law = law_type:law_universal_suffrage
						value > neutral
					}
				}
			}
			add_modifier = {
				name = modifier_turkfm_influence_questioned
				months = gbbf_long_modifier_time
				is_decaying = yes
			}
		}
		every_political_movement = {
			limit = {
				OR = {
					is_political_movement_type = movement_religious_majority
					is_political_movement_type = movement_modernizer
				}
			}
			add_modifier = {
				name = modifier_turkfm_declared_reforms
				months = gbbf_very_long_modifier_time
			}
		}
		every_country = {
			limit = {
				is_involved_in_journal_entry = je_eqfm_balkan_situation
			}
			post_notification = notification_turkfm_tanzimat_started
		}
		set_global_variable = turkfm_tanzimat_started
	}
}

	name = turkfm_abolish_cizye_name
	desc = turkfm_abolish_cizye_desc

		active_law:lawgroup_army_model = {
			has_amendment = amendment_type:amendment_gbbf_cizye_tax
		}

			}
		}
	}

turkfm_enforce_identity_button = {
	name = "turkfm_enforce_identity_name"
	desc = "turkfm_enforce_identity_desc"

	visible = {
		NOR = {
			has_variable = turkfm_ottomanist_identity_enforced
			has_variable = turkfm_islamist_identity_enforced
		}
	}
	possible = {
		custom_tooltip = {
			text = turkfm_enforce_identity_possible_1
			hidden_trigger = {
				var:turkfm_tanzimat_reform_var >= 1000
			}
		}
		custom_tooltip = {
			text = turkfm_enforce_identity_possible_2
			hidden_trigger = {
				OR = {
					var:turkfm_tanzimat_identity_var < 100
					var:turkfm_tanzimat_identity_var > 900
				}
			}
		}
		NOT = {
			has_law_or_variant = law_type:law_subjecthood
		}
	}
	effect = {
		if = {
			limit = {
				var:turkfm_tanzimat_identity_var > 900
			}
			custom_tooltip = {
				text = turkfm_enforce_identity_ottomanist
				set_variable = turkfm_ottomanist_identity_enforced
			}
		}
		else_if = {
			limit = {
				var:turkfm_tanzimat_identity_var < 100
			}
			custom_tooltip = {
				text = turkfm_enforce_identity_button_islamist
				set_variable = turkfm_islamist_identity_enforced
			}
		}
		else = {
			custom_tooltip = turkfm_ottomanist_identity_effect
		}
	}
}
