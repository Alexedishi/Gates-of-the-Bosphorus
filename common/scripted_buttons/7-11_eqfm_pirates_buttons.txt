eqfm_recall_adriatic_expedition_button = {
	name = "eqfm_cancel_adriatic_expedition_name"
	desc = "eqfm_cancel_adriatic_expedition_desc"

	visible = {
		eqfm_has_adriatic_expedition_trigger = yes
		NOT = {
			hidden_trigger = {
				has_variable = eqfm_adriatic_expedition_transit_var
			}
		}
	}
	possible = {
		eqfm_has_adriatic_expedition_trigger = yes
	}
	effect = {
		eqfm_cancel_adriatic_expedition_effect = yes
		eqfm_clear_adriatic_expedition_variables_effect = yes
	}
	ai_chance = {
		value = 0
		if = {
			limit = {
				gbbf_is_broke_trigger = yes
			}
			add = 1000
		}
	}
}

eqfm_resupply_adriatic_expedition_button = {
	name = "eqfm_resupply_adriatic_expedition_name"
	desc = "eqfm_resupply_adriatic_expedition_desc"
	
	visible = {
		eqfm_has_adriatic_expedition_trigger = yes
	}
	possible = {
		NOR = {
			has_modifier = modifier_eqfm_resupplied_adriatic_expedition
			custom_tooltip = {
				text = eqfm_resupply_adriatic_expedition_possible_1
				has_variable = eqfm_piracy_chase_var
			}
			hidden_trigger = {
				has_variable = eqfm_adriatic_expedition_transit_var
			}
		}
		custom_tooltip = {
			text = eqfm_resupply_adriatic_expedition_possible_2
			var:eqfm_adriatic_expedition_peril_var >= 3
		}
		
	}
	effect = {
		add_modifier = {
			name = modifier_eqfm_resupplied_adriatic_expedition
			months = gbbf_short_modifier_time
			multiplier = money_amount_multiplier_small
		}
		custom_tooltip = {
			text = eqfm_resupply_adriatic_expedition_effect
			change_variable = {
				name = eqfm_adriatic_expedition_peril_var
				subtract = 3
			}
		}
	}
	ai_chance = {
		value = 5
		if = {
			limit = {
				var:eqfm_adriatic_expedition_peril_var >= 4
				gbbf_is_broke_trigger = no
			}
			add = 25
		}
	}
}
