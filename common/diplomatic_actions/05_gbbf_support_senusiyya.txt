gbbf_support_senusiyya = {
	groups = {
		general
	}
	
	is_hostile = yes
	requires_approval = no
	show_effect_in_tooltip = yes
	can_use_obligations = no
	should_notify_third_parties = no
	show_confirmation_box = no
	
	unlocking_technologies = {
		nationalism
	}
	
	selectable = {
		is_involved_in_journal_entry = je_eqfm_senusiyya
	}
	
	potential = {
		scope:target_country = {
			has_variable = eqfm_senusiyya_influence_var
		}
	}
	
	possible = {
		eqfm_has_wecloming_stance_trigger = yes
		OR = {
			scope:target_country = {
				is_subject_of = root
			}
			AND = {
				gbbf_is_muslim_trigger = yes
				scope:target_country = {
					is_adjacent_to_country = root
				}
			}
		}
	}
	
	accept_effect = {
		custom_tooltip = gbbf_support_senusiyya_action_effect_breakdown
	}
	
	pact = {
		cost = 75
		is_hostile = no
		is_two_sided_pact = no
		show_in_outliner = yes
		
		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = eqfm_target_has_senusiyya_influence
					scope:target_country = {
						has_variable = eqfm_senusiyya_influence_var
					}
				}
			}
		}
		
		requirement_to_maintain = {
			trigger = {
				OR = {
					scope:target_country = {
						is_subject_of = root
					}
					AND = {
						gbbf_is_muslim_trigger = yes
						scope:target_country = {
							is_adjacent_to_country = root
						}
					}
				}
			}
		}
		
		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = eqfm_both_have_wecloming_stance
					var:eqfm_senusiyya_stance_var ?= flag:eqfm_senusiyya_stance_welcoming
				}
			}
		}
	}
	
	ai = {
		max_influence_spending_fraction = 0.25
		
		evaluation_chance = {
			value = 0.5
			if = {
				limit = {
					var:eqfm_senusiyya_stance_var ?= flag:eqfm_senusiyya_stance_welcoming
				}
				add = 10
			}
			if = {
				limit = {
					var:eqfm_senusiyya_stance_var ?= flag:eqfm_senusiyya_stance_hostile
				}
				add = -1000
			}
		}
		propose_score = {
			value = 10
			if = {
				limit = {
					var:eqfm_senusiyya_stance_var ?= flag:eqfm_senusiyya_stance_welcoming
					scope:target_country = {
						var:eqfm_senusiyya_stance_var ?= flag:eqfm_senusiyya_stance_welcoming
					}
					OR = {
						has_attitude = {
							who = scope:target_country
							attitude = protective
						}
						has_attitude = {
							who = scope:target_country
							attitude = genial
						}
						has_attitude = {
							who = scope:target_country
							attitude = cooperative
						}
					}
				}
				add = 10
			}
			if = {
				limit = {
					is_subject_of = scope:target_country
				}
				multiply = 2.5
			}
		}
		
		will_propose = {
			var:eqfm_senusiyya_stance_var ?= flag:eqfm_senusiyya_stance_welcoming
			scope:target_country = {
				var:eqfm_senusiyya_stance_var ?= flag:eqfm_senusiyya_stance_welcoming
			}
			OR = {
				has_attitude = {
					who = scope:target_country
					attitude = protective
				}
				has_attitude = {
					who = scope:target_country
					attitude = genial
				}
				has_attitude = {
					who = scope:target_country
					attitude = cooperative
				}
			}
		}
		
		will_break = {
			NOT = {
				var:eqfm_senusiyya_stance_var ?= flag:eqfm_senusiyya_stance_welcoming
				scope:target_country = {
					var:eqfm_senusiyya_stance_var ?= flag:eqfm_senusiyya_stance_welcoming
				}
				OR = {
					has_attitude = {
						who = scope:target_country
						attitude = protective
					}
					has_attitude = {
						who = scope:target_country
						attitude = genial
					}
					has_attitude = {
						who = scope:target_country
						attitude = cooperative
					}
				}
			}
		}
	}
}
