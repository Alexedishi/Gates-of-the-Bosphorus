amendment_gbbf_cizye_tax = {
	parent = law_state_religion
	allowed_laws = {
		law_peasant_levies
		law_professional_army
		law_national_militia
		law_mass_conscription
	}
	
	modifier = {
		tax_heathen_add = 0.5
		state_conscription_rate_mult = -0.2
	}
	
	possible = {
		c:TUR ?= this
	}
	would_sponsor = {
		always = no
	}
	can_repeal = {
		legitimacy >= 75
		any_interest_group = {
			filter = {
				law_stance = {
					law = law_type:law_state_religion
					value > neutral
				}
			}
			count = all
			is_in_government = no
			is_powerful = no
		}
	}
	on_deactivate = {
		every_scope_state = {
			limit = {
				has_modifier = modifier_eqfm_overtaxed
			}
			remove_modifier = modifier_eqfm_overtaxed
		}
		every_scope_state = {
			limit = {
				has_modifier = modifier_eqfm_tax_break_given
			}
			remove_modifier = modifier_eqfm_tax_break_given
		}
	}
	
	ai_will_revoke = {
		any_interest_group = {
			is_in_government = yes
			law_stance = {
				law = law_type:law_state_religion
				value < neutral
			}
		}
	}
}
